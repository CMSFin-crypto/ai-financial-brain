    88→// ═══ Technical indicator computations ═══
    89→function computeSMA(closes: number[], period: number): (number | null)[] {
    90→  const result: (number | null)[] = [];
    91→  for (let i = 0; i < closes.length; i++) {
    92→    if (i < period - 1) { result.push(null); continue; }
    93→    let sum = 0;
    94→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
    95→    result.push(sum / period);
    96→  }
    97→  return result;
    98→}
    99→
   100→function computeEMA(closes: number[], period: number): (number | null)[] {
   101→  const k = 2 / (period + 1);
   102→  const result: (number | null)[] = [];
   103→  let ema: number | null = null;
   104→  for (let i = 0; i < closes.length; i++) {
   105→    if (i < period - 1) { result.push(null); continue; }
   106→    if (ema === null) {
   107→      let sum = 0; for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   108→      ema = sum / period;
   109→    } else {
   110→      ema = closes[i] * k + ema * (1 - k);
   111→    }
   112→    result.push(ema);
   113→  }
   114→  return result;
   115→}

   117→function computeRSI(closes: number[], period: number = 14): (number | null)[] {
   118→  const result: (number | null)[] = [];
   119→  const gains: number[] = [];
   120→  const losses: number[] = [];
   121→  for (let i = 1; i < closes.length; i++) {
   122→    const diff = closes[i] - closes[i - 1];
   123→    gains.push(diff > 0 ? diff : 0);
   124→    losses.push(diff < 0 ? -diff : 0);
   125→  }
   126→  let avgGain = 0, avgLoss = 0;
   127→  for (let i = 0; i < Math.min(period, gains.length); i++) { avgGain += gains[i]; avgLoss += losses[i]; }
   128→  avgGain /= period; avgLoss /= period;
   129→  for (let i = 0; i < period - 1; i++) result.push(null);
   130→  const rs = avgLoss === 0 ? 100 : avgGain / avgLoss;
   131→  result.push(100 - 100 / (1 + rs));
   132→  for (let i = period; i < gains.length; i++) {
   133→    avgGain = (avgGain * (period - 1) + gains[i]) / period;
   134→    avgLoss = (avgLoss * (period - 1) + losses[i]) / period;
   135→    const rs2 = avgLoss === 0 ? 100 : avgGain / avgLoss;
   136→    result.push(100 - 100 / (1 + rs2));
   137→  }
   138→  return result;
   139→}

   141→function computeMACD(closes: number[], fastPeriod = 12, slowPeriod = 26, signalPeriod = 9): { macd: (number | null)[]; signal: (number | null)[]; histogram: (number | null)[] } {
   142→  const emaFast = computeEMA(closes, fastPeriod);
   143→  const emaSlow = computeEMA(closes, slowPeriod);
   144→  const macdLine: (number | null)[] = [];
   145→  for (let i = 0; i < closes.length; i++) {
   146→    if (emaFast[i] === null || emaSlow[i] === null) { macdLine.push(null); continue; }
   147→    macdLine.push(emaFast[i]! - emaSlow[i]!);
   148→  }
   149→  // Signal = EMA of MACD line — very short period so signal starts quickly
   150→  const validMACD = macdLine.filter((v): v is number => v !== null);
   151→  const sigPeriod = Math.min(signalPeriod, Math.max(2, Math.floor(validMACD.length * 0.15)));
   152→  const signalEma = computeEMA(validMACD, sigPeriod);
   153→  const signal: (number | null)[] = [];
   154→  let vi = 0;
   155→  for (let i = 0; i < closes.length; i++) {
   156→    if (macdLine[i] === null) { signal.push(null); continue; }
   157→    signal.push(signalEma[vi] ?? null);
   158→    vi++;
   159→  }
   160→  const histogram: (number | null)[] = [];
   161→  for (let i = 0; i < closes.length; i++) {
   162→    if (macdLine[i] === null || signal[i] === null) { histogram.push(null); continue; }
   163→    histogram.push(macdLine[i]! - signal[i]!);
   164→  }
   165→  return { macd: macdLine, signal, histogram };
   166→}

   168→function computeBollingerBands(closes: number[], period: number = 20, mult: number = 2): { upper: (number | null)[]; middle: (number | null)[]; lower: (number | null)[] } {
   169→  const middle = computeSMA(closes, period);
   170→  const upper: (number | null)[] = [];
   171→  const lower: (number | null)[] = [];
   172→  for (let i = 0; i < closes.length; i++) {
   173→    if (middle[i] === null) { upper.push(null); lower.push(null); continue; }
   174→    let sumSq = 0;
   175→    for (let j = i - period + 1; j <= i; j++) sumSq += (closes[j] - middle[i]!) ** 2;
   176→    const std = Math.sqrt(sumSq / period);
   177→    upper.push(middle[i]! + mult * std);
   178→    lower.push(middle[i]! - mult * std);
   179→  }
   180→  return { upper, middle, lower };
   181→}

   183→// ═══ TradingView-Style Technical Chart ═══
   184→function CandlestickChart({ data }: { data: CandleData[] }) {

   208→    const chartW = W - L - R;

   210→    const barW = chartW / n;
   211→    const candleW = Math.max(barW * 0.6, 2);
   212→
   213→    // Layout: price (with volume overlay), separator, RSI, separator, MACD
   214→    const priceH = 280;
   215→    const volRatio = 0.18; // volume takes bottom 18% of price panel
   216→    const volH = priceH * volRatio;
   217→    const candleH = priceH - volH; // candles use top 82%
   218→    const sep = 1;
   219→    const rsiH = 80;

   223→    const priceTop = 0;

   226→    const macdTop = rsiTop + rsiH + sep;

   237→    const bbP       = smaP;
   238→    const emaP      = n < 30 ? 3 : n < 60 ? Math.max(3, Math.round(n * 0.10)) : 12;
   239→    const rsiP      = n < 30 ? 3 : n < 60 ? Math.max(3, Math.round(n * 0.12)) : 14;
   240→    const macdFast  = n < 30 ? 3 : n < 60 ? Math.max(3, Math.round(n * 0.10)) : 12;
   241→    const macdSlow  = n < 30 ? 4 : n < 60 ? Math.max(5, Math.round(n * 0.22)) : 26;
   242→
   243→    // Debug: log data length and periods

   245→
   246→    const sma20 = computeSMA(closes, smaP);
   247→    const sma50 = computeSMA(closes, smaLongP);
   248→    const ema12 = computeEMA(closes, emaP);

   253→    // ═══ Price Y scale (candles area only, not volume) ═══
   254→    const hiVals = [ ...data.map(d => d.high), ...bb.upper.filter((v): v is number => v !== null), ...sma20.filter((v): v is number => v !== null) ];

   256→    const allH = Math.max(...hiVals);
   257→    const allL = Math.min(...loVals);

   259→    const pMin = allL - pad;
   260→    const pMax = allH + pad;

   286→    };
   287→
   288→    // Price grid
   289→    const pGrid: { y: number; lbl: string }[] = [];

   291→      const v = pMin + (pMax - pMin) * (i / 5);

   293→    }
   294→
   295→    // BB fill
   296→    const bbPts: string[] = [];
   297→    for (let i = 0; i < n; i++) if (bb.upper[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.upper[i]!)}`);
   298→    for (let i = n - 1; i >= 0; i--) if (bb.lower[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.lower[i]!)}`);
   299→
   300→    // Last values for legend
   301→    const lastSma20 = sma20[n - 1];
   302→    const lastSma50 = sma50[n - 1];
   303→    const lastEma12 = ema12[n - 1];
   304→    const lastRsi = rsi[n - 1];
   305→    const lastMacd = macdData.macd[n - 1];
   306→    const lastSig = macdData.signal[n - 1];
   307→    const lastClose = data[n - 1].close;
   308→    const lastY = yP(lastClose);
   309→
   310→    return (
   311→      <svg viewBox={`0 0 ${W} ${totalH}`} className="w-full h-full" style={{ background: BG, borderRadius: 0 }}>
   312→        <defs>
   313→          <clipPath id="priceClip"><rect x={L} y={priceTop} width={chartW} height={priceH} /></clipPath>
   314→          <clipPath id="rsiClip"><rect x={L} y={rsiTop} width={chartW} height={rsiH} /></clipPath>
   315→          <clipPath id="macdClip"><rect x={L} y={macdTop} width={chartW} height={macdH} /></clipPath>
   316→        </defs>
   317→        {/* Background fill for each panel */}
   318→        <rect x={L} y={priceTop} width={chartW} height={priceH} fill="#1e222d" opacity={0.3} />
   319→        <rect x={L} y={rsiTop} width={chartW} height={rsiH} fill="#1e222d" opacity={0.2} />
   320→        <rect x={L} y={macdTop} width={chartW} height={macdH} fill="#1e222d" opacity={0.2} />
   321→
   322→        {/* ═══════ PRICE CHART (candles + volume) ═══════ */}
   323→
   324→        {/* Grid lines — dotted, TradingView style */}
   325→        {pGrid.map((g, i) => (
   326→          <g key={"g" + i}>
   327→            <line x1={L} y1={g.y} x2={W - R} y2={g.y} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   328→            <text x={W - R + 6} y={g.y + 3.5} fill={TXT_BRIGHT} fontSize={10} fontFamily="Trebuchet MS, Tahoma, sans-serif">{g.lbl}</text>

   331→
   332→        {/* Vertical grid (time axis) */}
   333→        {data.map((d, i) => {
   334→          if (i % labelN !== 0 || i === 0) return null;

   351→        {/* Bollinger Bands fill */}

   387→        {/* TV-style top-right indicator legend with box */}

   395→        {lastEma12 !== null && <text x={W - R - 130} y={priceTop + 44} fill={TXT_BRIGHT} fontSize={10} fontFamily="Trebuchet MS, sans-serif">{lastEma12.toFixed(2)}</text>}
