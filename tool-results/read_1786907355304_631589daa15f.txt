   151→function formatPct(v: number | null | undefined, decimals = 1): string {
   152→  if (v == null) return "—"
   153→  return v.toFixed(decimals) + "%"
   154→}
   155→
   156→function formatNum(v: number | null | undefined): string {
   157→  if (v == null) return "—"
   158→  return v.toLocaleString("sq-AL")
   159→}

   161→function trendIcon(trend: string) {
   162→  switch (trend) {
   163→    case "improving":
   164→      return <TrendingUp className="size-4 text-emerald-500" />
   165→    case "degrading":
   166→      return <TrendingDown className="size-4 text-red-500" />
   167→    default:
   168→      return <Activity className="size-4 text-muted-foreground" />
   169→  }
   170→}
   171→
   172→function trendBadge(trend: string) {
   173→  const map: Record<string, { label: string; cls: string }> = {
   174→    improving: {
   175→      label: "Duke përmirësuar",
   176→      cls: "bg-emerald-500/15 text-emerald-400 border-emerald-500/30",
   177→    },
   178→    stable: {
   179→      label: "Stabil",
   180→      cls: "bg-muted/50 text-muted-foreground border-border",
   181→    },
   182→    degrading: {
   183→      label: "Duke u dobësuar",
   184→      cls: "bg-red-500/15 text-red-400 border-red-500/30",
   185→    },
   186→  }
   187→  const { label, cls } = map[trend] ?? map.stable
   188→  return (
   189→    <Badge variant="outline" className={cls}>
   190→      {trendIcon(trend)}
   191→      <span className="ml-1">{label}</span>
   192→    </Badge>
   193→  )
   194→}

   196→function driftValue(v: number) {
   197→  const color = v > 0 ? "text-emerald-400" : v < 0 ? "text-red-400" : "text-muted-foreground"
   198→  const sign = v > 0 ? "+" : ""
   199→  return (
   200→    <span className={color}>
   201→      {sign}
   202→      {formatPct(v)}
   203→    </span>
   204→  )
   205→}

   212→      return <AlertTriangle className="size-4 text-amber-500" />

   281→// ─── Custom Tooltip ──────────────────────────────────────────────────────────

   283→interface CustomTooltipProps {
   284→  active?: boolean
   285→  payload?: Array<{ name: string; value: number; color: string }>
   286→  label?: string
   287→}
   288→
   289→function ChartTooltip({ active, payload, label }: CustomTooltipProps) {
   290→  if (!active || !payload || payload.length === 0) return null
   291→  return (
   292→    <div className="rounded-lg border border-border bg-card px-3 py-2 shadow-xl">
   293→      <p className="mb-1 text-xs text-muted-foreground">{label}</p>
   294→      {payload.map((entry, i) => (
   295→        <div key={i} className="flex items-center gap-2 text-sm">
   296→          <span
   297→            className="inline-block size-2 rounded-full"
   298→            style={{ backgroundColor: entry.color }}
   299→          />
   300→          <span className="text-muted-foreground">{entry.name}:</span>
   301→          <span className="font-medium text-foreground">
   302→            {typeof entry.value === "number" ? entry.value.toFixed(4) : entry.value}
   303→          </span>
   304→        </div>
   305→      ))}
   306→    </div>
   307→  )
   308→}
   309→
   310→interface AccuracyTooltipProps {
   311→  active?: boolean
   312→  payload?: Array<{ name: string; value: number; color: string }>
   313→  label?: string
   314→}
   315→
   316→function AccuracyTooltip({ active, payload, label }: AccuracyTooltipProps) {
   317→  if (!active || !payload || payload.length === 0) return null
   318→  return (
   319→    <div className="rounded-lg border border-border bg-card px-3 py-2 shadow-xl">
   320→      <p className="mb-1 text-xs text-muted-foreground">{label}</p>
   321→      {payload.map((entry, i) => (
   322→        <div key={i} className="flex items-center gap-2 text-sm">
   323→          <span
   324→            className="inline-block size-2 rounded-full"
   325→            style={{ backgroundColor: entry.color }}
   326→          />
   327→          <span className="text-muted-foreground">{entry.name}:</span>
   328→          <span className="font-medium text-foreground">
   329→            {typeof entry.value === "number" ? formatPct(entry.value) : entry.value}
   330→          </span>
   331→        </div>
   332→      ))}
   333→    </div>
   334→  )
   335→}
