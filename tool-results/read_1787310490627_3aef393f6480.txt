   105→// ── Helpers ──
   106→function calcEMA(data: number[], period: number): number[] {
   107→  const r: number[] = new Array(data.length).fill(NaN);
   108→  if (data.length < period) return r;
   109→  let s = 0; for (let i = 0; i < period; i++) s += data[i];
   110→  r[period - 1] = s / period;
   111→  const k = 2 / (period + 1);
   112→  for (let i = period; i < data.length; i++) r[i] = data[i] * k + r[i - 1] * (1 - k);
   113→  return r;
   114→}
   115→
   116→function calcATR(data: HistoricalDataPoint[], period = 14): number {
   117→  if (data.length < period + 1) return 0;
   118→  const trs: number[] = [];
   119→  for (let i = 1; i < data.length; i++) {
   120→    trs.push(Math.max(data[i].high - data[i].low, Math.abs(data[i].high - data[i-1].close), Math.abs(data[i].low - data[i-1].close)));
   121→  }
   122→  return trs.slice(-period).reduce((a, b) => a + b, 0) / period;
   123→}
   124→
   125→function pct(data: number[], days: number): number {
   126→  if (data.length < days + 1) return 0;
   127→  const c = data[data.length - 1], p = data[data.length - 1 - days];
   128→  return p > 0 ? ((c - p) / p) * 100 : 0;
   129→}

   131→// Position sizing: risk-based share calculation
   132→function calcPositionSize(entry: number, stop: number, riskBudgetPct: number, accountEquity: number): { shares: number; positionValue: number; riskDollars: number; riskPct: number } {
   133→  const riskPerShare = entry - stop;
   134→  if (riskPerShare <= 0 || entry <= 0) return { shares: 0, positionValue: 0, riskDollars: 0, riskPct: 0 };
   135→  const riskDollars = accountEquity * (riskBudgetPct / 100);
   136→  const shares = Math.max(0, Math.floor(riskDollars / riskPerShare));
   137→  const positionValue = shares * entry;
   138→  const actualRiskPct = positionValue > 0 ? ((shares * riskPerShare) / accountEquity) * 100 : 0;
   139→  return { shares, positionValue: Math.round(positionValue * 100) / 100, riskDollars: Math.round(riskDollars * 100) / 100, riskPct: Math.round(actualRiskPct * 100) / 100 };
   140→}

   142→// IBKR Bracket Order generator
   143→function generateBracketOrder(symbol: string, action: 'BUY' | 'SELL', entry: number, stop: number, target: number, shares: number): object {
   144→  const cOID = `ibkr_${symbol}_${Date.now()}`;
   145→  const parentOID = `${cOID}_parent`;
   146→  const stopChildOID = `${cOID}_stop`;
   147→  const targetChildOID = `${cOID}_target`;
   148→
   149→  return {
   150→    orders: [
   151→      {
   152→        cOID: parentOID,
   153→        symbol,
   154→        orderType: 'LMT',
   155→        side: action,
   156→        quantity: shares,
   157→        price: Math.round(entry * 100) / 100,
   158→        tif: 'GTC',
   159→        transmit: false,
   160→      },
   161→      {
   162→        cOID: stopChildOID,
   163→        symbol,
   164→        orderType: 'STP',
   165→        side: action === 'BUY' ? 'SELL' : 'BUY',
   166→        quantity: shares,
   167→        price: Math.round(stop * 100) / 100,
   168→        tif: 'GTC',
   169→        parentId: parentOID,
   170→        transmit: false,
   171→      },
   172→      {
   173→        cOID: targetChildOID,
   174→        symbol,
   175→        orderType: 'LMT',
   176→        side: action === 'BUY' ? 'SELL' : 'BUY',
   177→        quantity: shares,
   178→        price: Math.round(target * 100) / 100,
   179→        tif: 'GTC',
   180→        parentId: parentOID,
   181→        transmit: true, // last child transmits the bracket
   182→      },
   183→    ],
   184→    summary: `Bracket: ${action} ${shares} ${symbol} @ $${entry.toFixed(2)} | Stop $${stop.toFixed(2)} | Target $${target.toFixed(2)}`,
   185→  };
   186→}

   290→    const sL = spyC.length - 1, qL = qqqC.length - 1;
   291→    const spyA50 = spyC[sL] > (spyS50[sL]||0), spyA200 = spyC[sL] > (spyS200[sL]||0);
   292→    const qqqA50 = qqqC[qL] > (qqqS50[qL]||0), qqqA200 = qqqC[qL] > (qqqS200[qL]||0);
   293→    const regimeOk = spyA50 && spyA200 && qqqA50 && qqqA200;
   294→    const spyRS60 = pct(spyC, 60);
   295→
   296→    // ── 1. Fetch all universe OHLCV ──
   297→    const syms = DEDUPED_UNIVERSE.filter(s => !ETF_SET.has(s));
   298→    const hist: Record<string, HistoricalDataPoint[] | null> = {};

   300→    const BATCH = 8;
   301→    for (let i = 0; i < syms.length; i += BATCH) {
   302→      const batch = syms.slice(i, i + BATCH);
   303→      const res = await Promise.allSettled(batch.map(async s => ({ s, d: await fetchHistoricalData(s, '1y') })));
   304→      for (const r of res) if (r.status === 'fulfilled' && r.value.d) hist[r.value.s] = r.value.d;
   305→      if (i + BATCH < syms.length) await new Promise(r => setTimeout(r, 250));

   336→      const rs60 = pct(closes, 60);

   339→      // NEW: Stacked MA — close > EMA20 > SMA50 > SMA200

   373→        entry: 0, stop: 0, target1R: 0, target2R: 0, target3R: 0,
   374→        riskPct: 0, rewardRiskRatio: 0, swingLow: 0,

   385→    // ── 3. PHASE 2 — Technical analysis on trend-passed stocks ──
   386→    const phase2: FunnelStock[] = [];

   388→    for (const stock of phase1) {
   389→      if (!stock.passedTrend) continue;

   391→      const data = hist[stock.symbol];
   392→      if (!data) continue;
   393→      const closes = data.map(d => d.close), highs = data.map(d => d.high), lows = data.map(d => d.low);

   406→      const sma50 = calculateSMA(closes, 50);
   407→      const sma200 = calculateSMA(closes, 200);

   409→      if (price > (sma50[last]||0)) tScore += 20;

   412→      // NEW: Stacked MA bonus (+15)
   413→      if (stock.stackedMA) tScore += 15;
   414→      // Higher-high structure
   415→      const h20a = Math.max(...highs.slice(-40, -20));
   416→      const h20b = Math.max(...highs.slice(-20));
   417→      if (h20b > h20a) tScore += 15;
   418→      // NEW: ADX bonus (+15 if > 25)
   419→      if (stock.adx > 25) tScore += 15;
   420→      stock.trendScore = Math.min(100, tScore);
   421→
   422→      // ── B) Relative Strength (0-100) — 20% weight ──
   423→      const rsSpy22 = pct(closes, 22) - pct(spyC, 22);
   424→      const rsQqq22 = pct(closes, 22) - pct(qqqC, 22);
   425→      const rsSpy60 = pct(closes, 60) - pct(spyC, 60);
   426→      let rsScore = 50;
   427→      if (rsSpy22 > 0) rsScore += Math.min(25, rsSpy22 * 3);
   428→      else rsScore -= Math.min(25, Math.abs(rsSpy22) * 3);
   429→      if (rsSpy60 > 0) rsScore += Math.min(25, rsSpy60 * 2);
   430→      else rsScore -= Math.min(25, Math.abs(rsSpy60) * 2);
   431→      stock.rsScore = Math.round(Math.max(0, Math.min(100, rsScore)));
   432→      stock.rsVsSPY = Math.round(rsSpy22 * 100) / 100;
   433→      stock.rsVsQQQ = Math.round(rsQqq22 * 100) / 100;
   434→      stock.rsVsSPY60d = Math.round(rsSpy60 * 100) / 100;
   435→
   436→      // ── C) Momentum (0-100) — 15% weight ──
   437→      const mom5 = pct(closes, 5), mom10 = pct(closes, 10), mom22 = pct(closes, 22);
   438→      let mScore = 50;
   439→      if (mom5 > -2) mScore += 10; else mScore -= 10;
   440→      if (mom10 > 0) mScore += 15; else mScore -= 10;
   441→      if (mom22 > 0) mScore += 15; else mScore -= 10;
   442→      if (mom5 < 8) mScore += 10; else mScore -= 15;
   443→      stock.momentumScore = Math.round(Math.max(0, Math.min(100, mScore)));
   444→
   445→      // ── D) Volume Confirmation (0-100) — 15% weight ──
   446→      const avgVol20 = vols.slice(-20).reduce((a,b) => a+b, 0) / 20;
   447→      const recent3 = vols.slice(-3).reduce((a,b) => a+b, 0) / 3;
   448→      const pbVol = vols.slice(-5, -1);
   449→      const priorVol = vols.slice(-10, -5);
   450→      const avgPb = pbVol.length > 0 ? pbVol.reduce((a,b) => a+b, 0)/pbVol.length : 0;
   451→      const avgPrior = priorVol.length > 0 ? priorVol.reduce((a,b) => a+b, 0)/priorVol.length : 0;
   452→      const volDeclining = avgPb < avgPrior * 0.95;
   453→      const lastDaySpike = vols[last] > avgVol20 * 1.1;
   454→      const volRatio = avgVol20 > 0 ? recent3 / avgVol20 : 1;
   455→      let vScore = 50;
   456→      if (volDeclining) vScore += 20;
   457→      if (lastDaySpike) vScore += 15;
   458→      if (volRatio > 0.8 && volRatio < 1.5) vScore += 10;
   459→      if (avgVol20 > 5_000_000) vScore += 5;
   460→      stock.volConfScore = Math.round(Math.max(0, Math.min(100, vScore)));
   461→      stock.volRatio = Math.round(volRatio * 100) / 100;
   462→      stock.volDeclining = volDeclining;
   463→      stock.lastDaySpike = lastDaySpike;
   464→
   465→      // ── E) Setup Quality (0-100) — 10% weight ──
   466→      const ema10Val = ema10[last], ema20Val = ema20[last];
   467→      const dist10 = !isNaN(ema10Val) ? ((price - ema10Val) / ema10Val) * 100 : 99;
   468→      const dist20 = !isNaN(ema20Val) ? ((price - ema20Val) / ema20Val) * 100 : 99;
   469→      stock.distFromEMA10 = Math.round(dist10 * 100) / 100;
   470→      stock.distFromEMA20 = Math.round(dist20 * 100) / 100;
   471→
   472→      // Find pullback
   473→      let highIdx = last;
   474→      for (let i = last; i >= Math.max(0, last - 10); i--) if (closes[i] >= closes[highIdx]) highIdx = i;
   475→      const peakPrice = closes[highIdx];
   476→      const pbPct = peakPrice > 0 ? ((price - peakPrice) / peakPrice) * 100 : 0;
   477→      const swLow = Math.min(...lows.slice(highIdx, last + 1));
   478→      let pbDays = 0;
   479→      for (let i = highIdx + 1; i <= last; i++) if (closes[i] < closes[i-1]) pbDays++;
   480→
   481→      stock.pullbackDays = pbDays;
   482→      stock.pullbackPct = Math.round(pbPct * 100) / 100;
   483→      stock.swingLow = Math.round(swLow * 100) / 100;
   484→
   485→      let setup: 'PULLBACK' | 'BREAKOUT' | 'TREND_CONT' | 'NONE' = 'NONE';
   486→      let sScore = 0;
   487→      const reasons: string[] = [];
   488→
   489→      // Pullback
   490→      if (pbDays >= 2 && pbDays <= 8 && pbPct >= -8 && pbPct <= -0.5) {
   491→        setup = 'PULLBACK';
   492→        sScore += 30;
   493→        if (pbDays >= 3 && pbDays <= 6) { sScore += 20; reasons.push(`Pullback ${pbDays}d (ideal)`); }
   494→        else { sScore += 10; reasons.push(`Pullback ${pbDays}d`); }
   495→        if (Math.abs(dist10) < 3 || Math.abs(dist20) < 3) { sScore += 20; reasons.push('Afer EMA 10/20'); }
   496→        if (volDeclining) { sScore += 15; reasons.push('Volumi ne renie (i mire)'); }
   497→        if (lastDaySpike) { sScore += 15; reasons.push('Volum konfirmim'); }
   498→        if (rsi >= 40 && rsi <= 65) { sScore += 10; reasons.push(`RSI ${rsi.toFixed(0)}`); }
   499→      }
   500→
   501→      // Breakout
   502→      if (setup === 'NONE') {
   503→        const high20 = Math.max(...highs.slice(-20));
   504→        if (price >= high20 * 0.98 && lastDaySpike && rsi >= 45 && rsi <= 70) {
   505→          setup = 'BREAKOUT'; sScore = 55;
   506→          reasons.push('Breakout 20d + volum');
   507→          if (rsi >= 45 && rsi <= 65) { sScore += 15; reasons.push(`RSI ${rsi.toFixed(0)}`); }
   508→        }
   509→      }
   510→
   511→      // Trend continuation
   512→      if (setup === 'NONE' && pbPct > -0.5 && pbPct < 2 && price > (sma50[last]||0)) {
   513→        setup = 'TREND_CONT'; sScore = 40;
   514→        reasons.push('Trend continuation');
   515→        if (rsi >= 50 && rsi <= 65) { sScore += 15; reasons.push(`RSI ${rsi.toFixed(0)}`); }
   516→        if (lastDaySpike) { sScore += 10; reasons.push('Volum konfirmim'); }
   517→      }
   518→
   519→      stock.setup = setup;
   520→      stock.setupScore = Math.min(100, sScore);
   521→      stock.rsi = Math.round(rsi * 10) / 10;
   522→      stock.atr = Math.round(atr * 100) / 100;
   523→      stock.atrPct = Math.round(atrPct * 100) / 100;
   524→      stock.reasons = reasons;
   525→
   526→      // Entry / Stop / Target (BLUEPRINT-ALIGNED)
   527→      const isBreakout = setup === 'BREAKOUT';
   528→      const high20 = Math.max(...highs.slice(-20));
   529→      const entry = isBreakout
   530→        ? Math.round((high20 * 1.002) * 100) / 100
   531→        : Math.round(price * 100) / 100;
   532→
   533→      // NEW: Stop = max(entry - 1.5*ATR, swingLow - 0.2*ATR) — use the tighter (higher) one
   534→      const stopAtr = Math.round((entry - atr * 1.5) * 100) / 100;
   535→      const stopSwing = Math.round((swLow - atr * 0.2) * 100) / 100;
   536→      const stop = Math.max(stopAtr, stopSwing); // tighter stop = higher price
   537→
   538→      const riskPerShare = entry - stop;
   539→      const riskPct = entry > 0 ? (riskPerShare / entry) * 100 : 0;
   540→      const target1R = Math.round((entry + riskPerShare) * 100) / 100;
   541→      const target2R = Math.round((entry + riskPerShare * 2) * 100) / 100;
   542→      const target3R = Math.round((entry + riskPerShare * 3) * 100) / 100;
   543→      const rr = riskPerShare > 0 ? (riskPerShare * 3) / riskPerShare : 0; // based on 3R now
   544→
   545→      stock.entry = entry;
   546→      stock.stop = stop;
   547→      stock.target1R = target1R;
   548→      stock.target2R = target2R;
   549→      stock.target3R = target3R;
   550→      stock.riskPct = Math.round(riskPct * 100) / 100;
   551→      stock.rewardRiskRatio = Math.round(rr * 10) / 10;
   552→
   553→      // NEW: Position sizing
   554→      const pos = calcPositionSize(entry, stop, MAX_RISK_PCT, ACCOUNT_EQUITY);
   555→      stock.positionSize = pos.shares;
   556→      stock.positionValue = pos.positionValue;
   557→      stock.riskDollars = pos.riskDollars;
   558→
   559→      // Horizon based on ATR
   560→      if (atrPct < 1.5) stock.horizon = '10D';
   561→      else if (atrPct < 3) stock.horizon = '5D';
   562→      else stock.horizon = '5D';
   563→
   564→      // ── F) Risk Quality (0-100) — 5% weight ──
   565→      let rScore = 50;
   566→      if (riskPct <= 3) rScore += 20;
   567→      else if (riskPct <= 5) rScore += 10;
   568→      else if (riskPct > 7) rScore -= 20;
   569→      if (rr >= 2) rScore += 15; // 2R+ is good
   570→      else if (rr >= 1.5) rScore += 5;
   571→      else rScore -= 15;
   572→      if (atrPct < 2) rScore += 10;
   573→      else if (atrPct > 4) rScore -= 10;
   574→      // NEW: Event risk penalty
   575→      if (!stock.passedEventRisk) rScore -= 25;
   576→      stock.riskScore = Math.round(Math.max(0, Math.min(100, rScore)));
   577→
   578→      // ── Total Score ──
   579→      stock.totalScore = Math.round(
   580→        stock.trendScore * 0.25 +
   581→        stock.rsScore * 0.20 +
   582→        stock.momentumScore * 0.15 +
   583→        stock.volConfScore * 0.15 +
   584→        sScore * 0.10 +
   585→        50 * 0.10 + // fundamentals placeholder (neutral)
   586→        stock.riskScore * 0.05
   587→      );
   588→
   589→      // Only keep stocks with valid setups
   590→      if (setup !== 'NONE') phase2.push(stock);
   591→    }
   592→
   593→    const passedSetup = phase2.length;
   594→
   595→    // ── 4. PHASE 3 — Risk gate + Decision + Event risk ──
   596→    const phase3: FunnelStock[] = [];
   597→
   598→    for (const stock of phase2) {
   599→      const warnings: string[] = [...stock.warnings];
   600→      let decision: Decision = 'NO_TRADE';
   601→
   602→      // R:R gate (based on 3R now, so 2.0+ is the threshold)
   603→      if (stock.rewardRiskRatio < 2.0) {
   604→        warnings.push(`R:R ${stock.rewardRiskRatio} — i ulet (duhet 1:2 me 3R target)`);
   605→      }
   606→
   607→      // RSI gate
   608→      if (stock.rsi > 75) {
   609→        warnings.push(`RSI ${stock.rsi} — i mbivleresuar`);
   610→      }
   611→
   612→      // Risk % gate
   613→      if (stock.riskPct > 8) {
   614→        warnings.push(`Rreziku ${stock.riskPct}% — shume i larte`);
   615→      }
   616→
   617→      // ADX gate
   618→      if (stock.adx < 20) {
   619→        warnings.push(`ADX ${stock.adx} — trendi i dobet (duhet > 25)`);
   620→      }
   621→
   622→      // Regime gate
   623→      if (!regimeOk) {
   624→        warnings.push('REGJIMI jo OK — vetem watcher');
   625→      }
   626→
   627→      // NEW: Event risk gate
   628→      if (!stock.passedEventRisk) {
   629→        warnings.push(`EVENT RISK: ${stock.eventRisk}`);
   630→      }
   631→
   632→      // Stacked MA bonus/penalty
   633→      if (!stock.stackedMA) {
   634→        warnings.push('MA jo te stackuara (close/EMA20/SMA50/SMA200)');
   635→      }
   636→
   637→      // Extended (chasing)
   638→      if (stock.pullbackPct > -0.5 && stock.setup === 'PULLBACK') {
   639→        warnings.push('Jo te vertete nje pullback — extended');
   640→      }
   641→
   642→      // Determine decision
   643→      const hasRR = stock.rewardRiskRatio >= 2.0;
   644→      const rsiOk = stock.rsi >= 30 && stock.rsi <= 75;
   645→      const riskOk = stock.riskPct <= 8;
   646→      const scoreOk = stock.totalScore >= 45;
   647→      const eventOk = stock.passedEventRisk;
   648→
   649→      if (hasRR && rsiOk && riskOk && scoreOk && regimeOk && eventOk) {
   650→        decision = 'READY';
   651→      } else if (hasRR && rsiOk && riskOk && scoreOk && (eventOk || !regimeOk)) {
   652→        decision = 'WATCHLIST';
   653→        if (!regimeOk) warnings.push('WATCHLIST: Regjimi nuk lejon long tani');
   654→        if (!eventOk) warnings.push('WATCHLIST: Event risk — prit');
   655→      } else if (!hasRR || stock.riskPct > 6) {
   656→        if (stock.setupScore >= 50) decision = 'WATCHLIST';
   657→        else decision = 'NO_TRADE';
   658→      } else if (stock.rsi > 72 || stock.pullbackPct > -0.3) {
   659→        decision = 'EXTENDED';
   660→      } else {
   661→        decision = 'WATCHLIST';
   662→      }
   663→
   664→      // Event risk override
   665→      if (!eventOk && decision === 'READY') {
   666→        decision = 'EVENT_RISK';
   667→        warnings.push('EVENT_RISK: Ngjarje kritike — mos hyr');
   668→      }
   669→
   670→      // RSI overbought override
   671→      if (stock.rsi > 70 && stock.totalScore >= 55) {
   672→        decision = 'EVENT_RISK';
   673→        warnings.push('EVENT_RISK: RSI i larte, rrezik kthimi');
   674→      }
   675→
   676→      stock.decision = decision;
   677→      stock.warnings = warnings;
   678→
   679→      // NEW: Generate bracket order for READY trades
   680→      if (decision === 'READY' && stock.positionSize > 0) {
   681→        stock.bracketOrder = generateBracketOrder(
   682→          stock.symbol, 'BUY', stock.entry, stock.stop, stock.target3R, stock.positionSize
   683→        );
   684→      }
   685→
   686→      if (decision === 'READY' || decision === 'WATCHLIST' || decision === 'EVENT_RISK') {
   687→        phase3.push(stock);
   688→      }
   689→    }
   690→
   691→    const passedRisk = phase3.length;
   692→
   693→    // ── 5. PHASE 4 — Sector exposure limit + Top 10 final ──
   694→    const sectorCount: Record<string, number> = {};
   695→    const afterSectorLimit: FunnelStock[] = [];
   696→
   697→    // Sort by totalScore descending first
   698→    phase3.sort((a, b) => b.totalScore - a.totalScore);
   699→
   700→    for (const stock of phase3) {
   701→      const sec = stock.sector;
   702→      if (!sectorCount[sec]) sectorCount[sec] = 0;
   703→      if (sectorCount[sec] < MAX_PER_SECTOR) {
   704→        afterSectorLimit.push(stock);
   705→        sectorCount[sec]++;
   706→      } else {
   707→        stock.warnings.push(`Sector limit: ${sec} ka tashme ${MAX_PER_SECTOR} aksione`);
   708→      }
   709→    }
   710→
   711→    const passedSectorLimit = afterSectorLimit.length;
   712→    const topStocks = afterSectorLimit.slice(0, 10);
   713→    // READY first, then WATCHLIST, then others
   714→    topStocks.sort((a, b) => {
   715→      const order: Record<Decision, number> = { READY: 0, WATCHLIST: 1, EVENT_RISK: 2, EXTENDED: 3, NO_TRADE: 4 };
   716→      const diff = (order[a.decision] || 5) - (order[b.decision] || 5);
   717→      return diff !== 0 ? diff : b.totalScore - a.totalScore;
   718→    });
   719→
   720→    // Build sector exposure summary
   721→    const sectorExposure: Record<string, number> = {};
   722→    for (const s of topStocks) {
   723→      if (!sectorExposure[s.sector]) sectorExposure[s.sector] = 0;
   724→      sectorExposure[s.sector]++;
   725→    }
   726→
   727→    const elapsed = ((Date.now() - t0) / 1000).toFixed(1);
   728→    console.log(`[IBKR v2] ${syms.length} → ${passedLiquidity} → ${passedTrend} → ${passedSetup} → ${passedRisk} → ${passedSectorLimit} → ${topStocks.length} (${elapsed}s)`);
   729→
   730→    return NextResponse.json({
   731→      scannedAt: new Date().toISOString(),
   732→      regimeOk,
   733→      regimeDetail: {
   734→        spy: { above50: spyA50, above200: spyA200 },
   735→        qqq: { above50: qqqA50, above200: qqqA200 },
   736→      },
   737→      funnel: {
   738→        universe: syms.length,
   739→        passedLiquidity,
   740→        passedTrend,
   741→        passedSetup,
   742→        passedRisk,
   743→        passedEventRisk: passedEventRiskCount,
   744→        passedSectorLimit,
   745→        displayed: topStocks.length,
   746→      },
   747→      results: topStocks,
   748→      sectorExposure,
   749→    } satisfies FunnelResponse);
   750→  } catch (err: any) {
   751→    console.error('[IBKR v2] Error:', err);
   752→    return NextResponse.json({ error: err?.message || 'Gabim skaneri' }, { status: 500 });
   753→  }
   754→}
