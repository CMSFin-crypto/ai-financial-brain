   294→function PredictionScoreGauge({ score, direction }: { score: number; direction: string }) {
   295→  const radius = 54;
   296→  const circumference = 2 * Math.PI * radius;
   297→  const normalizedScore = ((score + 100) / 200) * 100;
   298→  const offset = circumference - (normalizedScore / 100) * circumference;
   299→
   300→  const ringColor = score > 25
   301→    ? 'stroke-emerald-500'
   302→    : score > -25
   303→      ? 'stroke-yellow-500'
   304→      : 'stroke-red-500';
   305→
   306→  return (
   307→    <div className="flex flex-col items-center gap-3">
   308→      <div className="relative w-36 h-36">
   309→        <svg className="w-36 h-36 -rotate-90" viewBox="0 0 120 120">
   310→          <circle
   311→            cx="60" cy="60" r={radius}
   312→            fill="none" className="stroke-muted/30" strokeWidth="8"
   313→          />
   314→          <motion.circle
   315→            cx="60" cy="60" r={radius}
   316→            fill="none" className={ringColor}
   317→            strokeWidth="8" strokeLinecap="round"
   318→            strokeDasharray={circumference}
   319→            initial={{ strokeDashoffset: circumference }}
   320→            animate={{ strokeDashoffset: offset }}
   321→            transition={{ duration: 1.2, ease: 'easeOut' }}
   322→          />
   323→        </svg>
   324→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   325→          <span className={`text-3xl font-bold ${scoreColor(score)}`}>
   326→            {score > 0 ? '+' : ''}{score}
   327→          </span>
   328→          <span className="text-[10px] text-muted-foreground">-100 — +100</span>
   329→        </div>
   330→      </div>
   331→      <Badge
   332→        variant={directionBadgeVariant(direction)}
   333→        className="text-sm px-4 py-1.5 font-semibold"
   334→      >
   335→        {directionLabel(direction)}
   336→      </Badge>
   337→    </div>
   338→  );
   339→}

   421→export function StockPredictor() {
   422→  // Mode
   423→  const [mode, setMode] = useState<'single' | 'scan' | 'ai-hybrid'>('scan');
   424→
   425→  // Single mode
   426→  const [ticker, setTicker] = useState('');
   427→  const [singleResult, setSingleResult] = useState<PredictionResult | null>(null);
   428→  const [singleLoading, setSingleLoading] = useState(false);
   429→  const [singleError, setSingleError] = useState('');
   430→
   431→  // Scan mode
   432→  const [scanResult, setScanResult] = useState<ScanResult | null>(null);
   433→  const [scanLoading, setScanLoading] = useState(false);
   434→  const [scanError, setScanError] = useState('');
   435→  const [scanProgress, setScanProgress] = useState('');
   436→  const [viewTab, setViewTab] = useState<'top-picks' | 'top-shorts' | 'most-confident' | 'all'>('top-picks');
   437→  const [sortField, setSortField] = useState<'score' | 'confidence' | 'symbol' | 'totalScore' | 'hybridConfidence'>('score');
   438→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc');
   439→  const [expandedSymbol, setExpandedSymbol] = useState<string | null>(null);
   440→
   441→  // AI Hybrid scan mode
   442→  const [hybridResult, setHybridResult] = useState<HybridScanResult | null>(null);
   443→  const [hybridLoading, setHybridLoading] = useState(false);
   444→  const [hybridError, setHybridError] = useState('');
   445→  const [hybridViewTab, setHybridViewTab] = useState<'top-picks' | 'top-shorts' | 'most-confident' | 'all'>('top-picks');
   446→  const [hybridSortField, setHybridSortField] = useState<'totalScore' | 'hybridConfidence' | 'symbol'>('totalScore');
   447→  const [hybridSortDir, setHybridSortDir] = useState<'asc' | 'desc'>('desc');
   448→  const [hybridExpandedSymbol, setHybridExpandedSymbol] = useState<string | null>(null);
   449→
   450→  // ─── Handlers ─────────────────────────────────────────────
   451→
   452→  const handleSinglePredict = useCallback(async () => {
   453→    if (!ticker.trim()) return;
   454→    setSingleLoading(true);
   455→    setSingleError('');
   456→    setSingleResult(null);
   457→    try {
   458→      const res = await fetch(`/api/predict/${ticker.trim().toUpperCase()}`);
   459→      const data = await res.json();
   460→      if (!res.ok) throw new Error(data.error || 'Gabim gjatë predikimit');
   461→      setSingleResult(data);
   462→    } catch (err: unknown) {
   463→      setSingleError(err instanceof Error ? err.message : 'Gabim i papritur');
   464→    } finally {
   465→      setSingleLoading(false);
   466→    }
   467→  }, [ticker]);
   468→
   469→  const handleFullScan = useCallback(async () => {
   470→    setScanLoading(true);
   471→    setScanError('');
   472→    setScanResult(null);
   473→    setScanProgress('Duke filluar skanimin...');
   474→    setExpandedSymbol(null);
   475→    try {
   476→      const res = await fetch('/api/predict-scan', {
   477→        signal: AbortSignal.timeout(300000),
   478→      });
   479→      const data = await res.json();
   480→      if (!res.ok) throw new Error(data.error || 'Gabim gjatë skanimit');
   481→      setScanResult(data);
   482→    } catch (err: unknown) {
   483→      if (err instanceof DOMException && err.name === 'TimeoutError') {
   484→        setScanError('Koha i mbaroi. Provoni përsëri.');
   485→      } else {
   486→        setScanError(err instanceof Error ? err.message : 'Gabim i papritur');
   487→      }
   488→    } finally {
   489→      setScanLoading(false);
   490→      setScanProgress('');
   491→    }
   492→  }, []);
   493→
   494→  const handleSort = useCallback((field: 'score' | 'confidence' | 'symbol' | 'totalScore' | 'hybridConfidence') => {
   495→    if (sortField === field) {
   496→      setSortDir((d) => (d === 'desc' ? 'asc' : 'desc'));
   497→    } else {
   498→      setSortField(field);
   499→      setSortDir('desc');
   500→    }
   501→  }, [sortField]);
   502→
   503→  const handleHybridScan = useCallback(async () => {
   504→    setHybridLoading(true);
   505→    setHybridError('');
   506→    setHybridResult(null);
   507→    setHybridExpandedSymbol(null);
   508→    try {
   509→      const res = await fetch('/api/ai-predict-scan', {
   510→        signal: AbortSignal.timeout(300000),
   511→      });
   512→      const data = await res.json();
   513→      if (!res.ok) throw new Error(data.error || 'Gabim gjatë skanimit hibrid');
   514→      setHybridResult(data);
   515→    } catch (err: unknown) {
   516→      if (err instanceof DOMException && err.name === 'TimeoutError') {
   517→        setHybridError('Koha i mbaroi. Provoni përsëri.');
   518→      } else {
   519→        setHybridError(err instanceof Error ? err.message : 'Gabim i papritur');
   520→      }
   521→    } finally {
   522→      setHybridLoading(false);
   523→    }
   524→  }, []);
   525→
   526→  const handleHybridSort = useCallback((field: 'totalScore' | 'hybridConfidence' | 'symbol') => {
   527→    if (hybridSortField === field) {
   528→      setHybridSortDir((d) => (d === 'desc' ? 'asc' : 'desc'));
   529→    } else {
   530→      setHybridSortField(field);
   531→      setHybridSortDir('desc');
   532→    }
   533→  }, [hybridSortField]);
   534→
   535→  // ─── Derived Data ─────────────────────────────────────────
   536→
   537→  // Hybrid derived data
   538→  const currentHybridResults: HybridPredictionResult[] | null = (() => {
   539→    if (!hybridResult) return null;
   540→    const src = (() => {
   541→      switch (hybridViewTab) {
   542→        case 'top-picks': return hybridResult.topPicks;
   543→        case 'top-shorts': return hybridResult.topShorts;
   544→        case 'most-confident': return hybridResult.mostConfident;
   545→        case 'all': return hybridResult.allResults;
   546→        default: return null;
   547→      }
   548→    })();
   549→    if (!src) return null;
   550→    if (hybridViewTab !== 'all') return src;
   551→    return [...src].sort((a, b) => {
   552→      if (hybridSortField === 'symbol') {
   553→        return hybridSortDir === 'asc'
   554→          ? a.symbol.localeCompare(b.symbol)
   555→          : b.symbol.localeCompare(a.symbol);
   556→      }
   557→      const aVal = a[hybridSortField];
   558→      const bVal = b[hybridSortField];
   559→      return hybridSortDir === 'asc' ? (aVal as number) - (bVal as number) : (bVal as number) - (aVal as number);
   560→    });
   561→  })();
   562→
   563→  const currentResults: PredictionResult[] | null = (() => {
   564→    if (!scanResult) return null;
   565→    switch (viewTab) {
   566→      case 'top-picks': return scanResult.topPicks;
   567→      case 'top-shorts': return scanResult.topShorts;
   568→      case 'most-confident': return scanResult.mostConfident;
   569→      case 'all': {
   570→        const sorted = [...scanResult.allResults].sort((a, b) => {
   571→          if (sortField === 'symbol') {
   572→            return sortDir === 'asc'
   573→              ? a.symbol.localeCompare(b.symbol)
   574→              : b.symbol.localeCompare(a.symbol);
   575→          }
   576→          const aVal = sortField === 'score' ? a.score : a.confidence;
   577→          const bVal = sortField === 'score' ? b.score : b.confidence;
   578→          return sortDir === 'asc' ? aVal - bVal : bVal - aVal;
   579→        });
   580→        return sorted;
   581→      }
   582→      default: return null;
   583→    }
   584→  })();
   585→
   586→  // ─── Render ───────────────────────────────────────────────
   587→
   588→  return (
   589→    <div className="space-y-5">
   590→      {/* ═══════ Title ═══════ */}
   591→      <div className="flex items-center gap-2.5">
   592→        <div className="p-2 rounded-lg bg-violet-500/10 border border-violet-500/20">
   593→          <Brain className="w-5 h-5 text-violet-400" />
   594→        </div>
   595→        <div>
   596→          <h2 className="text-lg font-bold text-foreground">Predikues i Stoqeve me AI</h2>
   597→          <p className="text-xs text-muted-foreground">Analizë teknike + fundamentale me sistem mësues</p>
   598→        </div>
   599→      </div>
   600→
   601→      {/* ═══════ Mode Toggle ═══════ */}
   602→      <div className="flex flex-wrap gap-2">
   603→        <Button
   604→          variant={mode === 'single' ? 'default' : 'outline'}
   605→          size="sm"
   606→          onClick={() => setMode('single')}
   607→          className={mode === 'single' ? 'bg-violet-600 hover:bg-violet-700' : ''}
   608→        >
   609→          <Search className="w-3.5 h-3.5 mr-1.5" />
   610→          Predikim Individual
   611→        </Button>
   612→        <Button
   613→          variant={mode === 'scan' ? 'default' : 'outline'}
   614→          size="sm"
   615→          onClick={() => setMode('scan')}
   616→          className={mode === 'scan' ? 'bg-violet-600 hover:bg-violet-700' : ''}
   617→        >
   618→          <BarChart3 className="w-3.5 h-3.5 mr-1.5" />
   619→          Skanim Teknik

   621→        <Button
   622→          variant={mode === 'ai-hybrid' ? 'default' : 'outline'}
   623→          size="sm"
   624→          onClick={() => setMode('ai-hybrid')}
   625→          className={mode === 'ai-hybrid' ? 'bg-violet-600 hover:bg-violet-700' : ''}
   626→        >
   627→          <Sparkles className="w-3.5 h-3.5 mr-1.5" />
   628→          AI Hybrid
   629→        </Button>
   630→      </div>
   631→
   632→      <AnimatePresence mode="wait">
   633→        {/* ═══════════════════════════════════════════════════ */}
   634→        {/* SINGLE STOCK MODE                                 */}
   635→        {/* ═══════════════════════════════════════════════════ */}
   636→        {mode === 'single' && (
   637→          <motion.div
