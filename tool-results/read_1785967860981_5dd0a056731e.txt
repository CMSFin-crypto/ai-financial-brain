    36→
    37→type MetricSnapshot = {
    38→  id: string;
    39→  modelVersion: string;

    42→  accuracy: number | null;

    44→  benchmarkReturn: number | null;

    46→  winRate: number | null;
    47→  brierScore: number | null;
    48→  precisionBuy: number | null;
    49→  recallBuy: number | null;

    54→
    55→// --- Helpers ---
    56→
    57→function pct(v: number | null, decimals = 1): string {
    58→  if (v === null || v === undefined) return 'N/A';
    59→  return `${(v * 100).toFixed(decimals)}%`;
    60→}
    61→
    62→function brierColor(v: number | null): string {
    63→  if (v === null) return 'text-muted-foreground';
    64→  if (v < 0.15) return 'text-emerald-500';
    65→  if (v < 0.25) return 'text-amber-500';
    66→  return 'text-red-500';
    67→}
    68→
    69→function brierLabel(v: number | null): string {
    70→  if (v === null) return 'No data';
    71→  if (v < 0.15) return 'Well calibrated';
    72→  if (v < 0.25) return 'Slightly overconfident';
    73→  return 'Poorly calibrated';
    74→}
    75→
    76→// --- Stat Card ---
    77→
    78→function StatCard({ title, value, subtitle, icon: Icon, color = 'text-foreground' }: {
    79→  title: string;
    80→  value: string;
    81→  subtitle?: string;
    82→  icon: React.ElementType;
    83→  color?: string;
    84→}) {
    85→  return (
    86→    <Card className="border-border/50">
    87→      <CardContent className="p-4">
    88→        <div className="flex items-center justify-between">
    89→          <div className="space-y-1">
    90→            <p className="text-xs text-muted-foreground font-medium">{title}</p>
    91→            <p className={`text-2xl font-bold tabular-nums ${color}`}>{value}</p>
    92→            {subtitle && <p className="text-xs text-muted-foreground">{subtitle}</p>}

    96→          </div>
    97→        </div>
    98→      </CardContent>
    99→    </Card>

   101→}

   116→  brier: number | null;

   130→// --- Main Page ---
   131→

   133→  const [metrics, setMetrics] = useState<ModelMetrics | null>(null);
   134→  const [history, setHistory] = useState<MetricSnapshot[]>([]);
   135→  const [calibration, setCalibration] = useState<CalibrationData | null>(null);
   136→  const [calLoading, setCalLoading] = useState(false);
   137→  const [loading, setLoading] = useState(true);
   138→  const [snapshotting, setSnapshotting] = useState(false);

   140→  const [regimeFilter, setRegimeFilter] = useState<string>('');
   141→  const [error, setError] = useState<string | null>(null);
   142→
   143→  const fetchMetrics = useCallback(async () => {
   144→    setLoading(true);
   145→    setError(null);
   146→    try {
   147→      const params = new URLSearchParams();
   148→      if (horizonFilter !== 'all') params.set('horizonDays', horizonFilter);
   149→      if (regimeFilter) params.set('regime', regimeFilter);
   150→      params.set('modelVersion', 'predict-v3-regime-spillover');
   151→      const [metricsRes, historyRes] = await Promise.all([
   152→        fetch(`/api/model-metrics?${params}`),
   153→        fetch(`/api/model-metrics?history=true&days=90&modelVersion=predict-v3-regime-spillover`),
   154→      ]);
   155→      const metricsData = await metricsRes.json();
   156→      const historyData = await historyRes.json();
   157→      setMetrics(metricsData);
   158→      setHistory(historyData.snapshots || []);
   159→    } catch (e) {
   160→      setError(e instanceof Error ? e.message : 'Failed to load metrics');

   162→      setLoading(false);
   163→    }
   164→  }, [horizonFilter, regimeFilter]);
   165→

   167→    setCalLoading(true);
   168→    try {
   169→      const params = new URLSearchParams({ modelVersion: 'predict-v3-regime-spillover' });
   170→      if (horizonFilter !== 'all') params.set('horizonDays', horizonFilter);
   171→      if (regimeFilter) params.set('regime', regimeFilter);
   172→      const res = await fetch(`/api/model-calibration?${params}`);
   173→      const data = await res.json();
   174→      setCalibration(data);
   175→    } catch {
   176→      // calibration fetch failure is non-critical
   177→    } finally {
   178→      setCalLoading(false);
   179→    }
   180→  }, [horizonFilter, regimeFilter]);
   181→
   182→  useEffect(() => { fetchMetrics(); }, [fetchMetrics]);
   183→  useEffect(() => { fetchCalibration(); }, [fetchCalibration]);
   184→
   185→  const handleSnapshot = async () => {
   186→    setSnapshotting(true);
   187→    try {
   188→      const res = await fetch('/api/model-metrics', {
   189→        method: 'POST',
   190→        headers: { 'Content-Type': 'application/json' },
   191→        body: JSON.stringify({ modelVersion: 'predict-v3-regime-spillover', horizonDays: 1 }),
   192→      });
   193→      const data = await res.json();
   194→      if (data.error) throw new Error(data.error);
   195→      await fetchMetrics();
   196→    } catch (e) {
   197→      setError(e instanceof Error ? e.message : 'Snapshot failed');
   198→    } finally {
   199→      setSnapshotting(false);
   200→    }
   201→  };
   202→
   203→  const historyChartData = history.map(s => ({
   204→    date: new Date(s.createdAt).toLocaleDateString('en-US', { month: 'short', day: 'numeric' }),
   205→    accuracy: s.accuracy ? +(s.accuracy * 100).toFixed(1) : null,
   206→    brierScore: s.brierScore ? +(s.brierScore * 100).toFixed(2) : null,
   207→    alpha: s.alpha != null ? +s.alpha.toFixed(2) : null,
   208→    sampleSize: s.sampleSize,
   209→  }));
   210→
   211→  const gateData = metrics ? [
   212→    { name: 'Traded', value: Math.round((1 - (metrics.noTradeRate ?? 0)) * metrics.sampleSize), fill: '#22c55e' },
   213→    { name: 'No Trade', value: Math.round((metrics.noTradeRate ?? 0) * metrics.sampleSize), fill: '#6b7280' },
   214→  ].filter(d => d.value > 0) : [];
   215→
   216→  if (loading) {
   217→    return (
   218→      <div className="min-h-screen bg-background p-4 md:p-8">
   219→        <div className="max-w-7xl mx-auto space-y-6">
   220→          <Skeleton className="h-10 w-64" />
   221→          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   222→            {Array.from({ length: 10 }).map((_, i) => (
   223→              <Skeleton key={i} className="h-24 rounded-xl" />
   224→            ))}
   225→          </div>
   226→          <Skeleton className="h-96 rounded-xl" />
   227→        </div>
   228→      </div>
   229→    );
   230→  }
   231→
   232→  return (
   233→    <div className="min-h-screen bg-background p-4 md:p-8">
   234→      <div className="max-w-7xl mx-auto space-y-6">
   235→        {/* Header */}
   236→        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   237→          <div>
   238→            <h1 className="text-2xl font-bold tracking-tight">Model Calibration</h1>
   239→            <p className="text-sm text-muted-foreground">
   240→              Brier score, accuracy, alpha, and benchmark-aware metrics
   241→            </p>
   242→          </div>
   243→          <div className="flex items-center gap-2 flex-wrap">
   244→            <Select value={regimeFilter} onValueChange={setRegimeFilter}>
   245→              <SelectTrigger className="w-[160px] h-9">
   246→                <SelectValue placeholder="All Regimes" />
   247→              </SelectTrigger>
   248→              <SelectContent>
   249→                <SelectItem value="">All Regimes</SelectItem>
   250→                <SelectItem value="BULL_LOW_VOL">Bull Low Vol</SelectItem>
   251→                <SelectItem value="BULL_HIGH_VOL">Bull High Vol</SelectItem>
   252→                <SelectItem value="BEAR_LOW_VOL">Bear Low Vol</SelectItem>
   253→                <SelectItem value="BEAR_HIGH_VOL">Bear High Vol</SelectItem>
   254→                <SelectItem value="PANIC_CAPITULATION">Panic</SelectItem>
   255→                <SelectItem value="RELIEF_RALLY">Relief Rally</SelectItem>
   256→                <SelectItem value="RANGE_NEUTRAL">Range Neutral</SelectItem>
   257→              </SelectContent>
   258→            </Select>
   259→            <Select value={horizonFilter} onValueChange={setHorizonFilter}>
   260→              <SelectTrigger className="w-[120px] h-9">
   261→                <SelectValue placeholder="Horizon" />
   262→              </SelectTrigger>
   263→              <SelectContent>
   264→                <SelectItem value="all">All</SelectItem>
   265→                <SelectItem value="1">1D</SelectItem>
   266→                <SelectItem value="5">5D</SelectItem>
   267→                <SelectItem value="20">20D</SelectItem>
   268→              </SelectContent>
   269→            </Select>
   270→            <Button variant="outline" size="sm" onClick={fetchMetrics}>
   271→              <RefreshCw className="h-3.5 w-3.5 mr-1.5" /> Refresh
   272→            </Button>
   273→            <Button size="sm" onClick={handleSnapshot} disabled={snapshotting}>
   274→              {snapshotting ? <RefreshCw className="h-3.5 w-3.5 mr-1.5 animate-spin" /> : <Gauge className="h-3.5 w-3.5 mr-1.5" />}
   275→              Snapshot
   276→            </Button>
   277→          </div>
   278→        </div>
   279→
   280→        {error && (
   281→          <Card className="border-red-500/50 bg-red-500/5">
   282→            <CardContent className="p-3 flex items-center gap-2 text-sm text-red-500">
   283→              <AlertTriangle className="h-4 w-4" /> {error}
   284→            </CardContent>
   285→          </Card>
   286→        )}
   287→
   288→        {/* Primary KPI Cards */}
   289→        <div className="grid grid-cols-2 md:grid-cols-5 gap-4">
   290→          <StatCard
   291→            title="Sample Size"
   292→            value={metrics?.sampleSize?.toLocaleString() ?? '0'}
   293→            subtitle={metrics ? `${Math.round((metrics.noTradeRate ?? 0) * 100)}% no-trade` : undefined}
   294→            icon={BarChart3}
   295→          />
   296→          <StatCard
   297→            title="Accuracy"
   298→            value={pct(metrics?.accuracy ?? null)}
   299→            subtitle={metrics && metrics.accuracy !== null ? `${metrics.sampleSize} evaluated` : 'No data'}
   300→            icon={metrics && metrics.accuracy !== null && metrics.accuracy >= 0.5 ? CheckCircle2 : Target}
   301→            color={metrics && metrics.accuracy !== null && metrics.accuracy >= 0.55 ? 'text-emerald-500' : metrics && metrics.accuracy !== null && metrics.accuracy < 0.45 ? 'text-red-500' : ''}
   302→          />
   303→          <StatCard
   304→            title="Brier Score"
   305→            value={metrics?.brierScore != null ? metrics.brierScore.toFixed(4) : 'N/A'}
   306→            subtitle={brierLabel(metrics?.brierScore ?? null)}
   307→            icon={Gauge}
   308→            color={brierColor(metrics?.brierScore ?? null)}
   309→          />
   310→          <StatCard
   311→            title="Alpha"
   312→            value={metrics?.alpha != null ? `${metrics.alpha > 0 ? '+' : ''}${metrics.alpha.toFixed(2)}%` : 'N/A'}
   313→            subtitle="Excess return vs SPY"
   314→            icon={Zap}
   315→            color={metrics && metrics.alpha != null && metrics.alpha > 0 ? 'text-emerald-500' : metrics && metrics.alpha != null && metrics.alpha < 0 ? 'text-red-500' : ''}
   316→          />
   317→          <StatCard
   318→            title="Avg Return"
   319→            value={metrics?.avgReturn != null ? `${metrics.avgReturn > 0 ? '+' : ''}${metrics.avgReturn.toFixed(2)}%` : 'N/A'}
   320→            subtitle={metrics?.benchmarkReturn != null ? `SPY: ${metrics.benchmarkReturn > 0 ? '+' : ''}${metrics.benchmarkReturn.toFixed(2)}%` : undefined}
   321→            icon={metrics && metrics.avgReturn != null && metrics.avgReturn > 0 ? TrendingUp : TrendingDown}
   322→            color={metrics && metrics.avgReturn != null && metrics.avgReturn > 0 ? 'text-emerald-500' : metrics && metrics.avgReturn != null && metrics.avgReturn < 0 ? 'text-red-500' : ''}
   323→          />
   324→        </div>
   325→
   326→        {/* Secondary KPIs */}
   327→        <div className="grid grid-cols-2 md:grid-cols-5 gap-4">
   328→          <StatCard title="Win Rate" value={pct(metrics?.winRate ?? null)} icon={Activity} />
   329→          <StatCard title="Precision (BUY)" value={pct(metrics?.precisionBuy ?? null)} subtitle="Of BUYs, % correct" icon={Target} />
   330→          <StatCard title="Recall (BUY)" value={pct(metrics?.recallBuy ?? null)} subtitle="Of all ups, % caught" icon={Activity} />
   331→          <StatCard title="No-Trade Rate" value={pct(metrics?.noTradeRate ?? null)} subtitle="Regime-filtered" icon={Shield} />
   332→          <StatCard
   333→            title="Max Drawdown"
   334→            value={metrics?.maxDrawdown != null ? `-${metrics.maxDrawdown.toFixed(2)}%` : 'N/A'}
   335→            subtitle="Cumulative peak-to-trough"
   336→            icon={TrendingDown}
   337→            color={metrics && metrics.maxDrawdown != null && metrics.maxDrawdown > 10 ? 'text-red-500' : ''}
   338→          />
   339→        </div>
   340→
   341→        {/* Charts */}
   342→        <Tabs defaultValue="reliability" className="space-y-4">
   343→          <TabsList>
   344→            <TabsTrigger value="reliability">Model Reliability</TabsTrigger>
   345→            <TabsTrigger value="timeline">Metrics Timeline</TabsTrigger>
   346→            <TabsTrigger value="alpha">Alpha & Returns</TabsTrigger>
   347→            <TabsTrigger value="gates">Gate Analysis</TabsTrigger>
   348→          </TabsList>
   349→
   350→          {/* ── RELIABILITY TAB ── */}
   351→          <TabsContent value="reliability" className="space-y-4">
   352→            {/* Reliability KPIs */}
   353→            <div className="grid grid-cols-2 md:grid-cols-5 gap-4">
   354→              <StatCard
   355→                title="ECE"
   356→                value={calibration?.ece != null ? calibration.ece.toFixed(4) : 'N/A'}
   357→                subtitle={calibration?.ece != null ? (calibration.ece < 0.03 ? 'Well calibrated' : calibration.ece < 0.08 ? 'Acceptable' : 'Needs recalibration') : 'No data'}
   358→                icon={Thermometer}
   359→                color={calibration?.ece != null ? (calibration.ece < 0.03 ? 'text-emerald-500' : calibration.ece < 0.08 ? 'text-amber-500' : 'text-red-500') : ''}
   360→              />
   361→              <StatCard
   362→                title="Brier Score"
   363→                value={calibration?.brier != null ? calibration.brier.toFixed(4) : 'N/A'}
   364→                subtitle={calibration?.calibrator ? `Calibrator: ${calibration.calibrator.method}` : 'No calibrator'}
   365→                icon={Gauge}
   366→                color={calibration?.brier != null ? (calibration.brier < 0.15 ? 'text-emerald-500' : calibration.brier < 0.25 ? 'text-amber-500' : 'text-red-500') : ''}
   367→              />
   368→              <StatCard
   369→                title="MCE"
   370→                value={calibration?.mce != null ? calibration.mce.toFixed(4) : 'N/A'}
   371→                subtitle="Worst bucket error"
   372→                icon={AlertTriangle}
   373→                color={calibration?.mce != null && calibration.mce > 0.20 ? 'text-red-500' : ''}
   374→              />
   375→              <StatCard
   376→                title="Diagnosis"
   377→                value={calibration?.diagnosis ? calibration.diagnosis.replace(/_/g, ' ') : 'N/A'}
   378→                subtitle={calibration ? `${calibration.sampleSize} samples` : 'No data'}
   379→                icon={Crosshair}
   380→                color={calibration?.diagnosis === 'WELL_CALIBRATED' ? 'text-emerald-500' : calibration?.diagnosis === 'OVERCONFIDENT' ? 'text-amber-500' : calibration?.diagnosis === 'UNDERCONFIDENT' ? 'text-blue-500' : ''}
   381→              />
   382→              <StatCard
   383→                title="Calibrator"
   384→                value={calibration?.calibrator?.method?.toUpperCase() ?? 'NONE'}
   385→                subtitle={calibration?.calibrator ? `Trained: ${calibration.calibrator.trainedAt ? new Date(calibration.calibrator.trainedAt).toLocaleDateString() : 'never'}` : 'Run ?train=true'}
   386→                icon={Zap}
   387→              />
   388→            </div>
   389→
   390→            {/* Reliability Diagram + Table */}
   391→            <div className="grid md:grid-cols-2 gap-4">
   392→              {/* Reliability Diagram */}
   393→              <Card>
   394→                <CardHeader>
   395→                  <CardTitle className="text-base">Reliability Diagram</CardTitle>
   396→                  <CardDescription>X = predicted probability, Y = observed frequency. Below diagonal = overconfident. Above = underconfident.</CardDescription>
   397→                </CardHeader>
   398→                <CardContent>
   399→                  {calLoading ? (
   400→                    <Skeleton className="h-72" />
   401→                  ) : !calibration?.reliability?.filter(b => b.count > 0).length ? (
   402→                    <div className="h-72 flex items-center justify-center text-muted-foreground text-sm">
   403→                      <Clock className="h-4 w-4 mr-2" /> Need evaluated predictions
   404→                    </div>
   405→                  ) : (
   406→                    <ResponsiveContainer width="100%" height={300}>
   407→                      <ScatterChart>
   408→                        <CartesianGrid strokeDasharray="3 3" stroke="hsl(var(--border))" />
   409→                        <XAxis
   410→                          type="number"
   411→                          dataKey="avgProbability"
   412→                          name="Predicted"
   413→                          domain={[0, 1]}
   414→                          tickFormatter={(v: number) => v.toFixed(1)}
   415→                          tick={{ fontSize: 11 }}
   416→                          label={{ value: 'Predicted Probability', position: 'insideBottom', offset: -5, fontSize: 11 }}
   417→                        />
   418→                        <YAxis
   419→                          type="number"
   420→                          dataKey="observedFrequency"
   421→                          name="Observed"
   422→                          domain={[0, 1]}
   423→                          tickFormatter={(v: number) => v.toFixed(1)}
   424→                          tick={{ fontSize: 11 }}
   425→                          label={{ value: 'Observed Frequency', angle: -90, position: 'insideLeft', offset: 5, fontSize: 11 }}

   427→                        <Tooltip

   429→                          formatter={(value: number, name: string) => [value.toFixed(3), name]}

   434→                        />
   435→                        <ReferenceLine segment={[{ x: 0, y: 0 }, { x: 1, y: 1 }]} stroke="hsl(var(--muted-foreground))" strokeDasharray="5 5" />
   436→                        <Scatter
   437→                          data={calibration.reliability.filter(b => b.count > 0)}
   438→                          fill={calibration.diagnosis === 'WELL_CALIBRATED' ? '#22c55e' : calibration.diagnosis === 'OVERCONFIDENT' ? '#f97316' : '#3b82f6'}
   439→                        />
   440→                      </ScatterChart>
   441→                    </ResponsiveContainer>
   442→                  )}
   443→                </CardContent>
   444→              </Card>
   445→
   446→              {/* Reliability Table */}
   447→              <Card>
   448→                <CardHeader>
   449→                  <CardTitle className="text-base">Reliability Table</CardTitle>
   450→                  <CardDescription>Per-bucket calibration detail. |Pred - Obs| = calibration error.</CardDescription>
   451→                </CardHeader>
   452→                <CardContent>
   453→                  {calLoading ? (
   454→                    <Skeleton className="h-72" />
   455→                  ) : !calibration?.reliability?.length ? (
   456→                    <div className="h-72 flex items-center justify-center text-muted-foreground text-sm">
   457→                      <Clock className="h-4 w-4 mr-2" /> No data
   458→                    </div>
   459→                  ) : (
   460→                    <div className="overflow-auto max-h-[300px]">
   461→                      <table className="w-full text-xs">
   462→                        <thead className="sticky top-0 bg-muted/80">
   463→                          <tr className="text-left">
   464→                            <th className="pb-2 pr-3 font-medium">Bucket</th>
   465→                            <th className="pb-2 pr-3 font-medium text-right">Predicted</th>

   467→                            <th className="pb-2 pr-3 font-medium text-right">Error</th>
