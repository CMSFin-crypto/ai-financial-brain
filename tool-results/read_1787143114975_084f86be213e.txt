   307→   307→// Simple seeded pseudo-random for consistent sparklines
   308→   308→function seededRandom(seed: number): () => number {
   309→   309→  let s = seed;
   310→   310→  return () => {
   311→   311→    s = (s * 16807 + 0) % 2147483647;
   312→   312→    return (s - 1) / 2147483646;
   313→   313→  };
   314→   314→}

   316→   316→function generateSparklineData(priceChange: number, ticker: string): Array<{ value: number }> {
   317→   317→  const points = 12;
   318→   318→  const data: Array<{ value: number }> = [];
   319→   319→  const magnitude = Math.min(Math.abs(priceChange), 10);
   320→   320→  // Create a seed from ticker hash + priceChange for consistency
   321→   321→  const hash = ticker.split('').reduce((acc, c) => acc + c.charCodeAt(0), 0);
   322→   322→  const rand = seededRandom(hash * 1000 + Math.round(priceChange * 100));
   323→   323→  for (let i = 0; i < points; i++) {
   324→   324→    const progress = i / (points - 1);
   325→   325→    const trend = priceChange >= 0 ? progress * magnitude : -progress * magnitude;
   326→   326→    const noise = (rand() - 0.5) * magnitude * 0.4;
   327→   327→    data.push({ value: 50 + trend + noise });
   328→   328→  }
   329→   329→  return data;
   330→   330→}
