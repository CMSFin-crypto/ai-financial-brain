    88→    88→// ═══ Technical indicator computations ═══
    89→    89→function computeSMA(closes: number[], period: number): (number | null)[] {
    90→    90→  const result: (number | null)[] = [];
    91→    91→  for (let i = 0; i < closes.length; i++) {
    92→    92→    if (i < period - 1) { result.push(null); continue; }
    93→    93→    let sum = 0;
    94→    94→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
    95→    95→    result.push(sum / period);
    96→    96→  }
    97→    97→  return result;
    98→    98→}
    99→    99→
   100→   100→function computeEMA(closes: number[], period: number): (number | null)[] {
   101→   101→  const k = 2 / (period + 1);
   102→   102→  const result: (number | null)[] = [];
   103→   103→  let ema: number | null = null;
   104→   104→  for (let i = 0; i < closes.length; i++) {
   105→   105→    if (i < period - 1) { result.push(null); continue; }
   106→   106→    if (ema === null) {
   107→   107→      let sum = 0; for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   108→   108→      ema = sum / period;
   109→   109→    } else {
   110→   110→      ema = closes[i] * k + ema * (1 - k);
   111→   111→    }
   112→   112→    result.push(ema);
   113→   113→  }
   114→   114→  return result;
   115→   115→}

   117→   117→function computeRSI(closes: number[], period: number = 14): (number | null)[] {
   118→   118→  const result: (number | null)[] = [];
   119→   119→  const gains: number[] = [];
   120→   120→  const losses: number[] = [];
   121→   121→  for (let i = 1; i < closes.length; i++) {
   122→   122→    const diff = closes[i] - closes[i - 1];
   123→   123→    gains.push(diff > 0 ? diff : 0);
   124→   124→    losses.push(diff < 0 ? -diff : 0);
   125→   125→  }
   126→   126→  let avgGain = 0, avgLoss = 0;
   127→   127→  for (let i = 0; i < Math.min(period, gains.length); i++) { avgGain += gains[i]; avgLoss += losses[i]; }
   128→   128→  avgGain /= period; avgLoss /= period;
   129→   129→  for (let i = 0; i < period - 1; i++) result.push(null);
   130→   130→  const rs = avgLoss === 0 ? 100 : avgGain / avgLoss;
   131→   131→  result.push(100 - 100 / (1 + rs));
   132→   132→  for (let i = period; i < gains.length; i++) {
   133→   133→    avgGain = (avgGain * (period - 1) + gains[i]) / period;
   134→   134→    avgLoss = (avgLoss * (period - 1) + losses[i]) / period;
   135→   135→    const rs2 = avgLoss === 0 ? 100 : avgGain / avgLoss;
   136→   136→    result.push(100 - 100 / (1 + rs2));
   137→   137→  }
   138→   138→  return result;
   139→   139→}

   141→   141→function computeMACD(closes: number[], fastPeriod = 12, slowPeriod = 26, signalPeriod = 9): { macd: (number | null)[]; signal: (number | null)[]; histogram: (number | null)[] } {
   142→   142→  const emaFast = computeEMA(closes, fastPeriod);
   143→   143→  const emaSlow = computeEMA(closes, slowPeriod);
   144→   144→  const macdLine: (number | null)[] = [];
   145→   145→  for (let i = 0; i < closes.length; i++) {
   146→   146→    if (emaFast[i] === null || emaSlow[i] === null) { macdLine.push(null); continue; }
   147→   147→    macdLine.push(emaFast[i]! - emaSlow[i]!);
   148→   148→  }
   149→   149→  // Signal = EMA of MACD line — very short period so signal starts quickly
   150→   150→  const validMACD = macdLine.filter((v): v is number => v !== null);
   151→   151→  const sigPeriod = Math.min(signalPeriod, Math.max(2, Math.floor(validMACD.length * 0.15)));
   152→   152→  const signalEma = computeEMA(validMACD, sigPeriod);
   153→   153→  const signal: (number | null)[] = [];
   154→   154→  let vi = 0;
   155→   155→  for (let i = 0; i < closes.length; i++) {
   156→   156→    if (macdLine[i] === null) { signal.push(null); continue; }
   157→   157→    signal.push(signalEma[vi] ?? null);
   158→   158→    vi++;
   159→   159→  }
   160→   160→  const histogram: (number | null)[] = [];
   161→   161→  for (let i = 0; i < closes.length; i++) {
   162→   162→    if (macdLine[i] === null || signal[i] === null) { histogram.push(null); continue; }
   163→   163→    histogram.push(macdLine[i]! - signal[i]!);
   164→   164→  }
   165→   165→  return { macd: macdLine, signal, histogram };
   166→   166→}

   168→   168→function computeBollingerBands(closes: number[], period: number = 20, mult: number = 2): { upper: (number | null)[]; middle: (number | null)[]; lower: (number | null)[] } {
   169→   169→  const middle = computeSMA(closes, period);
   170→   170→  const upper: (number | null)[] = [];
   171→   171→  const lower: (number | null)[] = [];
   172→   172→  for (let i = 0; i < closes.length; i++) {
   173→   173→    if (middle[i] === null) { upper.push(null); lower.push(null); continue; }
   174→   174→    let sumSq = 0;
   175→   175→    for (let j = i - period + 1; j <= i; j++) sumSq += (closes[j] - middle[i]!) ** 2;
   176→   176→    const std = Math.sqrt(sumSq / period);
   177→   177→    upper.push(middle[i]! + mult * std);
   178→   178→    lower.push(middle[i]! - mult * std);
   179→   179→  }
   180→   180→  return { upper, middle, lower };
   181→   181→}

   184→   184→// Renders colored tags at the end of each indicator line (left of Y-axis),
   185→   185→// stacking them vertically if they overlap.
   186→   186→function EdgeLabels({ items, rightEdge, fontSize = 9.5 }: {
   187→   187→  items: { label: string; y: number; color: string }[];
   188→   188→  rightEdge: number;
   189→   189→  fontSize?: number;
   190→   190→}) {
   191→   191→  const LH = fontSize + 3; // line height per label
   192→   192→  const MIN_GAP = 2;
   193→   193→
   194→   194→  // Sort by Y, then resolve overlaps by pushing downward
   195→   195→  const sorted = [...items].sort((a, b) => a.y - b.y);
   196→   196→  const placed: { label: string; y: number; color: string }[] = [];
   197→   197→  for (const item of sorted) {
   198→   198→    let y = item.y - LH / 2; // center vertically on the line
   199→   199→    for (const p of placed) {
   200→   200→      if (y < p.y + LH + MIN_GAP && y + LH > p.y) {
   201→   201→        y = p.y + LH + MIN_GAP; // push below
   202→   202→      }
   203→   203→    }
   204→   204→    placed.push({ ...item, y });
   205→   205→  }
   206→   206→
   207→   207→  return (
   208→   208→    <g>
   209→   209→      {placed.map((p, i) => {
   210→   210→        const textW = p.label.length * fontSize * 0.6 + 8;
   211→   211→        const rx = rightEdge - textW - 4;
   212→   212→        return (
   213→   213→          <g key={i}>
   214→   214→            <rect x={rx} y={p.y} width={textW} height={LH} rx={2} fill={p.color} opacity={0.92} />
   215→   215→            <text
   216→   216→              x={rx + 4}
   217→   217→              y={p.y + fontSize + 1}
   218→   218→              fill="white"
   219→   219→              fontSize={fontSize}
   220→   220→              fontFamily="Trebuchet MS, sans-serif"
   221→   221→              fontWeight="600"
   222→   222→            >{p.label}</text>
   223→   223→          </g>
   224→   224→        );
   225→   225→      })}
   226→   226→    </g>
   227→   227→  );
   228→   228→}

   231→   231→function CandlestickChart({ data }: { data: CandleData[] }) {

   233→   233→    if (!data || data.length < 3) return null;
   234→   234→
   235→   235→    // TradingView dark palette (matching reference)

   240→   240→    const TXT_BRIGHT = '#d1d4dc';
   241→   241→    const BULL = '#26a69a';
   242→   242→    const BEAR = '#ef5350';
   243→   243→    const SMA20_CLR = '#f0b323'; // Gold/Yellow — fast MA
   244→   244→    const SMA50_CLR = '#2962ff'; // Blue — slow MA

   252→   252→    const H = 540;
   253→   253→    const R = 60; // right axis width
   254→   254→    const L = 2;  // left margin
   255→   255→    const chartW = W - L - R;
   256→   256→    const n = data.length;
   257→   257→    const barW = chartW / n;
   258→   258→    const candleW = Math.max(barW * 0.6, 2);
   259→   259→
   260→   260→    // Layout: price (with volume overlay), separator, RSI, separator, MACD
   261→   261→    const priceH = 280;
   262→   262→    const volRatio = 0.18; // volume takes bottom 18% of price panel
   263→   263→    const volH = priceH * volRatio;
   264→   264→    const candleH = priceH - volH; // candles use top 82%
   265→   265→    const sep = 1;
   266→   266→    const rsiH = 80;
   267→   267→    const macdH = 85;
   268→   268→    const bottomLabel = 18;
   269→   269→
   270→   270→    const priceTop = 0;
   271→   271→    const volTop = priceTop + candleH;
   272→   272→    const rsiTop = priceTop + priceH + sep;
   273→   273→    const macdTop = rsiTop + rsiH + sep;
   274→   274→    const totalH = macdTop + macdH + bottomLabel;
   275→   275→
   276→   276→    // ═══ Compute indicators — DUAL STRATEGY ═══
   277→   277→    // 70+ bars → standard TradingView periods
   278→   278→    // <70 bars → scaled-down periods for small datasets
   279→   279→    const closes = data.map(d => d.close);
   280→   280→
   281→   281→    let smaP: number, smaLongP: number, bbP: number, emaP: number, rsiP: number, macdFast: number, macdSlow: number;
   282→   282→
   283→   283→    if (n >= 70) {
   284→   284→      smaP = 20; smaLongP = 50; bbP = 20; emaP = 12; rsiP = 14; macdFast = 12; macdSlow = 26;
   285→   285→    } else {
   286→   286→      const maxPeriod = Math.max(3, Math.floor(n * 0.08) + 1);
   287→   287→      smaLongP  = maxPeriod;
   288→   288→      smaP      = Math.max(2, Math.round(maxPeriod * 0.4));
   289→   289→      bbP       = smaP;
   290→   290→      emaP      = Math.max(2, Math.round(maxPeriod * 0.3));
   291→   291→      rsiP      = Math.max(2, Math.round(maxPeriod * 0.35));
   292→   292→      macdFast  = emaP;
   293→   293→      macdSlow  = Math.max(macdFast + 1, Math.round(maxPeriod * 0.6));
   294→   294→    }
   295→   295→
   296→   296→    const sma20 = computeSMA(closes, smaP);
   297→   297→    const sma50 = computeSMA(closes, smaLongP);
   298→   298→    const ema12 = computeEMA(closes, emaP);
   299→   299→    const bb = computeBollingerBands(closes, bbP);
   300→   300→    const rsi = computeRSI(closes, rsiP);
   301→   301→    const macdData = computeMACD(closes, macdFast, macdSlow);
   302→   302→
   303→   303→    console.log(`[CHART] bars=${n} periods: SMA(${smaP},${smaLongP}) EMA(${emaP}) RSI(${rsiP}) MACD(${macdFast},${macdSlow})`);
   304→   304→
   305→   305→    // ═══ Price Y scale (candles area only, not volume) ═══
   306→   306→    const hiVals = [ ...data.map(d => d.high), ...bb.upper.filter((v): v is number => v !== null), ...sma20.filter((v): v is number => v !== null) ];

   309→   309→    const allL = Math.min(...loVals);

   311→   311→    const pMin = allL - pad;
   312→   312→    const pMax = allH + pad;
   313→   313→    const yP = (v: number) => priceTop + (1 - (v - pMin) / (pMax - pMin)) * candleH;
   314→   314→
   315→   315→    // ═══ Volume Y scale (bottom of price panel) ═══
   316→   316→    const maxVol = Math.max(...data.map(d => d.volume || 0)) || 1;
   317→   317→    const yV = (v: number) => volTop + volH - (v / maxVol) * volH;

   326→   326→    const yM = (v: number) => macdTop + macdH / 2 - (v / mMax) * (macdH / 2);
   327→   327→

   329→   329→    const xOf = (i: number) => L + barW * i + barW / 2;
   330→   330→    const labelN = n <= 15 ? 1 : n <= 30 ? 3 : n <= 60 ? 5 : n <= 130 ? 10 : 20;

   332→   332→    const line = (vals: (number | null)[], color: string, sw = 1, id?: string) => {

   337→   337→          if (firstValidY === null) {
   338→   338→            firstValidY = vals[i];
   339→   339→            // Backfill: extend line to the left edge using first valid value

   341→   341→          }
   342→   342→          pts.push(`${xOf(i)},${vals[i]}`);
   343→   343→        }
   344→   344→      }
   345→   345→      return pts.length > 1 ? <polyline key={id ?? color} points={pts.join(' ')} fill="none" stroke={color} strokeWidth={sw} strokeLinejoin="round" strokeLinecap="round" /> : null;
   346→   346→    };
   347→   347→
   348→   348→    // Price grid
   349→   349→    const pGrid: { y: number; lbl: string }[] = [];
   350→   350→    for (let i = 0; i <= 5; i++) {
   351→   351→      const v = pMin + (pMax - pMin) * (i / 5);
   352→   352→      pGrid.push({ y: yP(v), lbl: '$' + v.toFixed(1) });
   353→   353→    }
   354→   354→
   355→   355→    // BB fill
   356→   356→    const bbPts: string[] = [];
   357→   357→    for (let i = 0; i < n; i++) if (bb.upper[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.upper[i]!)}`);
   358→   358→    for (let i = n - 1; i >= 0; i--) if (bb.lower[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.lower[i]!)}`);
   359→   359→
   360→   360→    // Last values for labels
   361→   361→    const lastSma20 = sma20[n - 1];
   362→   362→    const lastSma50 = sma50[n - 1];
   363→   363→    const lastEma12 = ema12[n - 1];
   364→   364→    const lastRsi = rsi[n - 1];
   365→   365→    const lastMacd = macdData.macd[n - 1];
   366→   366→    const lastSig = macdData.signal[n - 1];
   367→   367→    const lastClose = data[n - 1].close;
   368→   368→    const lastY = yP(lastClose);
   369→   369→
   370→   370→    return (
   371→   371→      <svg viewBox={`0 0 ${W} ${totalH}`} className="w-full h-full" style={{ background: BG, borderRadius: 0 }}>
   372→   372→        <defs>
   373→   373→          <clipPath id="priceClip"><rect x={L} y={priceTop} width={chartW} height={priceH} /></clipPath>
   374→   374→          <clipPath id="rsiClip"><rect x={L} y={rsiTop} width={chartW} height={rsiH} /></clipPath>
   375→   375→          <clipPath id="macdClip"><rect x={L} y={macdTop} width={chartW} height={macdH} /></clipPath>
   376→   376→        </defs>
   377→   377→        {/* Background fill for each panel */}
   378→   378→        <rect x={L} y={priceTop} width={chartW} height={priceH} fill="#1e222d" opacity={0.3} />
   379→   379→        <rect x={L} y={rsiTop} width={chartW} height={rsiH} fill="#1e222d" opacity={0.2} />
   380→   380→        <rect x={L} y={macdTop} width={chartW} height={macdH} fill="#1e222d" opacity={0.2} />
   381→   381→
   382→   382→        {/* ═══════ PRICE CHART (candles + volume) ═══════ */}
   383→   383→
   384→   384→        {/* Grid lines — dotted, TradingView style */}
   385→   385→        {pGrid.map((g, i) => (
   386→   386→          <g key={"g" + i}>
   387→   387→            <line x1={L} y1={g.y} x2={W - R} y2={g.y} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   388→   388→            <text x={W - R + 6} y={g.y + 3.5} fill={TXT_BRIGHT} fontSize={10} fontFamily="Trebuchet MS, Tahoma, sans-serif">{g.lbl}</text>
   389→   389→          </g>
   390→   390→        ))}
   391→   391→
   392→   392→        {/* Vertical grid (time axis) */}
   393→   393→        {data.map((dd, i) => {
   394→   394→          if (i % labelN !== 0 || i === 0) return null;
   395→   395→          return <line key={"vg" + i} x1={xOf(i)} y1={priceTop} x2={xOf(i)} y2={priceTop + priceH} stroke={GRID} strokeOpacity={0.25} strokeDasharray="1 3" />;
   396→   396→        })}
   397→   397→
   398→   398→        {/* Volume bars (behind candles, at bottom of price panel) */}
