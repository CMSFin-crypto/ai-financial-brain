    89→// ═══ Technical indicator computations ═══
    90→function computeSMA(closes: number[], period: number): (number | null)[] {
    91→  const result: (number | null)[] = [];
    92→  for (let i = 0; i < closes.length; i++) {
    93→    if (i < period - 1) { result.push(null); continue; }
    94→    let sum = 0;
    95→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
    96→    result.push(sum / period);
    97→  }
    98→  return result;
    99→}
   100→
   101→function computeEMA(closes: number[], period: number): (number | null)[] {
   102→  const k = 2 / (period + 1);
   103→  const result: (number | null)[] = [];
   104→  let ema: number | null = null;
   105→  for (let i = 0; i < closes.length; i++) {
   106→    if (i < period - 1) { result.push(null); continue; }
   107→    if (ema === null) {
   108→      let sum = 0; for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   109→      ema = sum / period;
   110→    } else {
   111→      ema = closes[i] * k + ema * (1 - k);
   112→    }
   113→    result.push(ema);
   114→  }
   115→  return result;
   116→}

   118→function computeRSI(closes: number[], period: number = 14): (number | null)[] {
   119→  const result: (number | null)[] = [];
   120→  const gains: number[] = [];
   121→  const losses: number[] = [];
   122→  for (let i = 1; i < closes.length; i++) {
   123→    const diff = closes[i] - closes[i - 1];
   124→    gains.push(diff > 0 ? diff : 0);
   125→    losses.push(diff < 0 ? -diff : 0);
   126→  }
   127→  let avgGain = 0, avgLoss = 0;
   128→  for (let i = 0; i < Math.min(period, gains.length); i++) { avgGain += gains[i]; avgLoss += losses[i]; }
   129→  avgGain /= period; avgLoss /= period;
   130→  for (let i = 0; i < period - 1; i++) result.push(null);
   131→  const rs = avgLoss === 0 ? 100 : avgGain / avgLoss;
   132→  result.push(100 - 100 / (1 + rs));
   133→  for (let i = period; i < gains.length; i++) {
   134→    avgGain = (avgGain * (period - 1) + gains[i]) / period;
   135→    avgLoss = (avgLoss * (period - 1) + losses[i]) / period;
   136→    const rs2 = avgLoss === 0 ? 100 : avgGain / avgLoss;
   137→    result.push(100 - 100 / (1 + rs2));
   138→  }
   139→  return result;
   140→}

   142→function computeMACD(closes: number[], fastPeriod = 12, slowPeriod = 26, signalPeriod = 9): { macd: (number | null)[]; signal: (number | null)[]; histogram: (number | null)[] } {
   143→  const emaFast = computeEMA(closes, fastPeriod);
   144→  const emaSlow = computeEMA(closes, slowPeriod);
   145→  const macdLine: (number | null)[] = [];
   146→  for (let i = 0; i < closes.length; i++) {
   147→    if (emaFast[i] === null || emaSlow[i] === null) { macdLine.push(null); continue; }
   148→    macdLine.push(emaFast[i]! - emaSlow[i]!);
   149→  }
   150→  // Signal = EMA of MACD line
   151→  const validMACD = macdLine.filter((v): v is number => v !== null);
   152→  const sigPeriod = Math.min(signalPeriod, Math.max(3, Math.floor(validMACD.length * 0.4)));
   153→  const signalEma = computeEMA(validMACD, sigPeriod);
   154→  const signal: (number | null)[] = [];
   155→  let vi = 0;
   156→  for (let i = 0; i < closes.length; i++) {
   157→    if (macdLine[i] === null) { signal.push(null); continue; }
   158→    signal.push(signalEma[vi] ?? null);
   159→    vi++;
   160→  }
   161→  const histogram: (number | null)[] = [];
   162→  for (let i = 0; i < closes.length; i++) {
   163→    if (macdLine[i] === null || signal[i] === null) { histogram.push(null); continue; }
   164→    histogram.push(macdLine[i]! - signal[i]!);
   165→  }
   166→  return { macd: macdLine, signal, histogram };
   167→}

   169→function computeBollingerBands(closes: number[], period: number = 20, mult: number = 2): { upper: (number | null)[]; middle: (number | null)[]; lower: (number | null)[] } {
   170→  const middle = computeSMA(closes, period);
   171→  const upper: (number | null)[] = [];
   172→  const lower: (number | null)[] = [];
   173→  for (let i = 0; i < closes.length; i++) {
   174→    if (middle[i] === null) { upper.push(null); lower.push(null); continue; }
   175→    let sumSq = 0;
   176→    for (let j = i - period + 1; j <= i; j++) sumSq += (closes[j] - middle[i]!) ** 2;
   177→    const std = Math.sqrt(sumSq / period);
   178→    upper.push(middle[i]! + mult * std);
   179→    lower.push(middle[i]! - mult * std);
   180→  }
   181→  return { upper, middle, lower };
   182→}

   184→// ═══ TradingView-Style Technical Chart ═══
   185→function CandlestickChart({ data }: { data: CandleData[] }) {
   186→  const chart = useMemo(() => {
   187→    if (!data || data.length < 3) return null;
   188→
   189→    // TradingView dark palette
   190→    const BG = 'transparent';
   191→    const GRID = '#2a2e39';
   192→    const GRID_OP = 0.4;
   193→    const TXT = '#787b86';
   194→    const BULL = '#26a69a';
   195→    const BEAR = '#ef5350';
   196→    const SMA20_CLR = '#f0b90b';
   197→    const SMA50_CLR = '#2196f3';
   198→    const EMA12_CLR = '#e040fb';
   199→    const BB_CLR = '#7c4dff';
   200→    const RSI_CLR = '#e040fb';
   201→    const MACD_CLR = '#2196f3';
   202→    const SIG_CLR = '#ff6d00';
   203→
   204→    const W = 880;
   205→    const H = 540;
   206→    const R = 60; // right axis width
   207→    const L = 2;  // left margin
   208→    const chartW = W - L - R;
   209→    const n = data.length;
   210→    const barW = chartW / n;
   211→    const candleW = Math.max(barW * 0.6, 2);
   212→
   213→    // Layout: price (with volume overlay), separator, RSI, separator, MACD
   214→    const priceH = 280;
   215→    const volRatio = 0.18; // volume takes bottom 18% of price panel
   216→    const volH = priceH * volRatio;
   217→    const candleH = priceH - volH; // candles use top 82%
   218→    const sep = 1;
   219→    const rsiH = 80;
   220→    const macdH = 85;
   221→    const bottomLabel = 18;
   222→
   223→    const priceTop = 0;
   224→    const volTop = priceTop + candleH;
   225→    const rsiTop = priceTop + priceH + sep;
   226→    const macdTop = rsiTop + rsiH + sep;
   227→    const totalH = macdTop + macdH + bottomLabel;
   228→
   229→    // ═══ Compute indicators (standard periods — API fetches enough lookback) ═══
   230→    const closes = data.map(d => d.close);
   231→    const smaP = Math.min(20, n);
   232→    const smaLongP = Math.min(50, n);
   233→    const bbP = Math.min(20, n);
   234→    const emaP = Math.min(12, n);
   235→    const rsiP = Math.min(14, n - 2);
   236→    const macdFast = Math.min(12, n);
   237→    const macdSlow = Math.min(26, n);
   238→
   239→    const sma20 = computeSMA(closes, smaP);
   240→    const sma50 = computeSMA(closes, smaLongP);
   241→    const ema12 = computeEMA(closes, emaP);
   242→    const bb = computeBollingerBands(closes, bbP);
   243→    const rsi = computeRSI(closes, rsiP);
   244→    const macdData = computeMACD(closes, macdFast, macdSlow);
   245→
   246→    // ═══ Price Y scale (candles area only, not volume) ═══
   247→    const hiVals = [ ...data.map(d => d.high), ...bb.upper.filter((v): v is number => v !== null), ...sma20.filter((v): v is number => v !== null) ];
   248→    const loVals = [ ...data.map(d => d.low), ...bb.lower.filter((v): v is number => v !== null), ...sma20.filter((v): v is number => v !== null) ];
   249→    const allH = Math.max(...hiVals);
   250→    const allL = Math.min(...loVals);
   251→    const pad = (allH - allL) * 0.06 || 1;
   252→    const pMin = allL - pad;
   253→    const pMax = allH + pad;
   254→    const yP = (v: number) => priceTop + (1 - (v - pMin) / (pMax - pMin)) * candleH;
   255→
   256→    // ═══ Volume Y scale (bottom of price panel) ═══
   257→    const maxVol = Math.max(...data.map(d => d.volume || 0)) || 1;
   258→    const yV = (v: number) => volTop + volH - (v / maxVol) * volH;
   259→
   260→    // ═══ RSI scale 0-100 ═══
   261→    const yR = (v: number) => rsiTop + (1 - v / 100) * rsiH;
   262→
   263→    // ═══ MACD scale ═══
   264→    const macdVals = macdData.macd.filter((v): v is number => v !== null);
   265→    const histVals = macdData.histogram.filter((v): v is number => v !== null);
   266→    const mMax = Math.max(Math.abs(Math.min(...macdVals, ...histVals)), Math.abs(Math.max(...macdVals, ...histVals)), 0.01);
   267→    const yM = (v: number) => macdTop + macdH / 2 - (v / mMax) * (macdH / 2);
   268→
   269→    // ═══ Helpers ═══
   270→    const xOf = (i: number) => L + barW * i + barW / 2;
   271→    const labelN = n <= 15 ? 1 : n <= 30 ? 3 : n <= 60 ? 5 : n <= 130 ? 10 : 20;
   272→
   273→    const line = (vals: (number | null)[], color: string, sw = 1, id?: string) => {
   274→      const pts: string[] = [];
   275→      for (let i = 0; i < vals.length; i++) {
   276→        if (vals[i] !== null) pts.push(`${xOf(i)},${vals[i]}`);
   277→      }
   278→      return pts.length > 1 ? <polyline key={id ?? color} points={pts.join(' ')} fill="none" stroke={color} strokeWidth={sw} strokeLinejoin="round" strokeLinecap="round" /> : null;
   279→    };
   280→
   281→    // Price grid
   282→    const pGrid: { y: number; lbl: string }[] = [];
   283→    for (let i = 0; i <= 5; i++) {
   284→      const v = pMin + (pMax - pMin) * (i / 5);
   285→      pGrid.push({ y: yP(v), lbl: '$' + v.toFixed(1) });
   286→    }
   287→
   288→    // BB fill
   289→    const bbPts: string[] = [];
   290→    for (let i = 0; i < n; i++) if (bb.upper[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.upper[i]!)}`);
   291→    for (let i = n - 1; i >= 0; i--) if (bb.lower[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.lower[i]!)}`);
   292→
   293→    // Last values for legend
   294→    const lastSma20 = sma20[n - 1];
   295→    const lastSma50 = sma50[n - 1];
   296→    const lastEma12 = ema12[n - 1];
   297→    const lastRsi = rsi[n - 1];
   298→    const lastMacd = macdData.macd[n - 1];
   299→    const lastSig = macdData.signal[n - 1];
   300→    const lastClose = data[n - 1].close;
   301→    const lastY = yP(lastClose);
   302→
   303→    return (
   304→      <svg viewBox={`0 0 ${W} ${totalH}`} className="w-full h-full" style={{ background: BG }}>
   305→        <defs>
   306→          <clipPath id="priceClip"><rect x={L} y={priceTop} width={chartW} height={priceH} /></clipPath>
   307→          <clipPath id="rsiClip"><rect x={L} y={rsiTop} width={chartW} height={rsiH} /></clipPath>
   308→          <clipPath id="macdClip"><rect x={L} y={macdTop} width={chartW} height={macdH} /></clipPath>
   309→        </defs>
   310→
   311→        {/* ═══════ PRICE CHART (candles + volume) ═══════ */}
   312→
   313→        {/* Grid lines — dotted, TradingView style */}
   314→        {pGrid.map((g, i) => (
   315→          <g key={"g" + i}>
   316→            <line x1={L} y1={g.y} x2={W - R} y2={g.y} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   317→            <text x={W - R + 6} y={g.y + 3.5} fill={TXT} fontSize={10} fontFamily="Trebuchet MS, Tahoma, sans-serif">{g.lbl}</text>
   318→          </g>
   319→        ))}
   320→
   321→        {/* Vertical grid (time axis) */}
   322→        {data.map((d, i) => {
   323→          if (i % labelN !== 0 || i === 0) return null;
   324→          return <line key={"vg" + i} x1={xOf(i)} y1={priceTop} x2={xOf(i)} y2={priceTop + priceH} stroke={GRID} strokeOpacity={0.25} strokeDasharray="1 3" />;
   325→        })}
   326→
   327→        {/* Volume bars (behind candles, at bottom of price panel) */}
   328→        <g clipPath="url(#priceClip)">
   329→          {data.map((d, i) => {
   330→            const cx = xOf(i);
   331→            const isUp = d.close >= d.open;
   332→            const vy = yV(d.volume || 0);
   333→            return (
   334→              <rect key={"vol" + i} x={cx - candleW / 2} y={vy} width={candleW} height={Math.max(volTop + volH - vy, 0)}
   335→                fill={isUp ? 'rgba(38,166,154,0.25)' : 'rgba(239,83,80,0.25)'} />
   336→            );
   337→          })}
   338→        </g>
   339→
   340→        {/* Bollinger Bands fill */}
   341→        {bbPts.length > 2 && <polygon points={bbPts.join(' ')} fill="rgba(124,77,255,0.06)" stroke="none" />}
   342→
   343→        {/* Bollinger Band lines */}
   344→        {line(bb.upper.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-u')}
   345→        {line(bb.middle.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-m')}
   346→        {line(bb.lower.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-l')}
   347→
   348→        {/* Moving Averages */}
   349→        {line(sma20.map(v => v !== null ? yP(v) : null), SMA20_CLR, 1.5, 's20')}
   350→        {line(sma50.map(v => v !== null ? yP(v) : null), SMA50_CLR, 1.5, 's50')}
   351→        {line(ema12.map(v => v !== null ? yP(v) : null), EMA12_CLR, 1.2, 'e12')}
   352→
   353→        {/* Candlesticks */}
   354→        <g clipPath="url(#priceClip)">
   355→          {data.map((d, i) => {
   356→            const cx = xOf(i);
   357→            const up = d.close >= d.open;
   358→            const c = up ? BULL : BEAR;
   359→            const bT = yP(Math.max(d.open, d.close));
   360→            const bB = yP(Math.min(d.open, d.close));
   361→            const bH = Math.max(bB - bT, 1);
   362→            return (
   363→              <g key={"k" + i}>
   364→                <line x1={cx} y1={yP(d.high)} x2={cx} y2={yP(d.low)} stroke={c} strokeWidth={1} />
   365→                <rect x={cx - candleW / 2} y={bT} width={candleW} height={bH} fill={c} stroke={c} strokeWidth={0.5} />
   366→              </g>
   367→            );
   368→          })}
   369→        </g>
   370→
   371→        {/* Current price line + tag */}
   372→        <line x1={L} y1={lastY} x2={W - R} y2={lastY} stroke={data[n-1].close >= data[n-1].open ? BULL : BEAR} strokeDasharray="4 2" strokeOpacity={0.7} strokeWidth={1} />
   373→        <rect x={W - R + 1} y={lastY - 10} width={R - 2} height={20} rx={2} fill={data[n-1].close >= data[n-1].open ? BULL : BEAR} />
   374→        <text x={W - R + 8} y={lastY + 4} fill="white" fontSize={10.5} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">{lastClose.toFixed(2)}</text>
   375→
   376→        {/* TV-style top-left indicator legend */}
   377→        {lastSma20 !== null && <text x={L + 8} y={priceTop + 14} fill={SMA20_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">SMA(20) {lastSma20.toFixed(2)}</text>}
   378→        {lastSma50 !== null && <text x={L + 8} y={priceTop + 26} fill={SMA50_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">SMA(50) {lastSma50.toFixed(2)}</text>}
   379→        {lastEma12 !== null && <text x={L + 8} y={priceTop + 38} fill={EMA12_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">EMA(12) {lastEma12.toFixed(2)}</text>}
   380→        <text x={L + 8} y={priceTop + 50} fill={BB_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">BB (20, 2)</text>
   381→
   382→        {/* Volume label */}
   383→        <text x={L + 8} y={volTop + 12} fill={TXT} fontSize={9} fontFamily="Trebuchet MS, sans-serif" opacity={0.6}>Vol</text>
   384→        <text x={W - R + 6} y={volTop + 12} fill={TXT} fontSize={9} fontFamily="Trebuchet MS, sans-serif" opacity={0.5}>{(maxVol / 1e6).toFixed(1)}M</text>
   385→
   386→        {/* ═══════ PANEL SEPARATOR ═══════ */}
   387→        <line x1={L} y1={rsiTop - 1} x2={W - R} y2={rsiTop - 1} stroke={GRID} strokeOpacity={0.5} />
   388→
   389→        {/* ═══════ RSI (14) ═══════ */}
   390→        {/* RSI grid */}
   391→        <line x1={L} y1={yR(70)} x2={W - R} y2={yR(70)} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   392→        <line x1={L} y1={yR(50)} x2={W - R} y2={yR(50)} stroke={GRID} strokeOpacity={GRID_OP * 0.6} strokeDasharray="1 3" />
   393→        <line x1={L} y1={yR(30)} x2={W - R} y2={yR(30)} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   394→        {/* RSI Y labels */}
   395→        <text x={W - R + 6} y={yR(70) + 3.5} fill={TXT} fontSize={9.5} fontFamily="Trebuchet MS, sans-serif">70.00</text>
   396→        <text x={W - R + 6} y={yR(50) + 3.5} fill={TXT} fontSize={9.5} fontFamily="Trebuchet MS, sans-serif" opacity={0.6}>50.00</text>
   397→        <text x={W - R + 6} y={yR(30) + 3.5} fill={TXT} fontSize={9.5} fontFamily="Trebuchet MS, sans-serif">30.00</text>
   398→        {/* RSI line */}
   399→        <g clipPath="url(#rsiClip)">
   400→          {line(rsi.map(v => v !== null ? yR(v) : null), RSI_CLR, 1.5, 'rsi')}
   401→        </g>
   402→        {/* TV-style RSI label top-left */}
   403→        <text x={L + 8} y={rsiTop + 14} fill={RSI_CLR} fontSize={10.5} fontFamily="Trebuchet MS, sans-serif" fontWeight="700">RSI (14)</text>
   404→        {lastRsi !== null && (
   405→          <text x={L + 72} y={rsiTop + 14} fill={lastRsi > 70 ? BEAR : lastRsi < 30 ? BULL : TXT} fontSize={10.5} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">{lastRsi.toFixed(2)}</text>
   406→        )}
   407→
   408→        {/* ═══════ PANEL SEPARATOR ═══════ */}
   409→        <line x1={L} y1={macdTop - 1} x2={W - R} y2={macdTop - 1} stroke={GRID} strokeOpacity={0.5} />
   410→
   411→        {/* ═══════ MACD ═══════ */}

   421→            const barY = pos ? yM(v) : macdTop + macdH / 2;
   422→            const barH = Math.abs(yM(v) - (macdTop + macdH / 2));

   432→        {/* TV-style MACD label top-left */}
   433→        <text x={L + 8} y={macdTop + 14} fill={MACD_CLR} fontSize={10.5} fontFamily="Trebuchet MS, sans-serif" fontWeight="700">MACD</text>

   437→        {lastSig !== null && (
   438→          <text x={L + 110} y={macdTop + 14} fill={SIG_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif">Signal {lastSig.toFixed(2)}</text>
   439→        )}
   440→
   441→        {/* ═══════ X-AXIS DATE LABELS ═══════ */}
   442→        {data.map((d, i) => {

   446→              {d.date.substring(5)}
   447→            </text>
   448→          );
   449→        })}
   450→      </svg>
   451→    );
   452→  }, [data]);
   453→
   454→  return chart || <div className="flex items-center justify-center h-full text-muted-foreground text-sm">Asnjë të dhënë grafiku</div>;
   455→}
