    87→    87→function mapRange(
    88→    88→  value: number,
    89→    89→  inMin: number,
    90→    90→  inMax: number,
    91→    91→  outMin: number,
    92→    92→  outMax: number,
    93→    93→): number {
    94→    94→  if (inMax === inMin) return (outMin + outMax) / 2;
    95→    95→  const result = outMin + ((value - inMin) / (inMax - inMin)) * (outMax - outMin);
    96→    96→  return clamp(result, Math.min(outMin, outMax), Math.max(outMin, outMax));
    97→    97→}

    99→    99→/** Simple Moving Average — full-length array with NaN padding */
   100→   100→function sma(data: number[], period: number): number[] {
   101→   101→  const result: number[] = new Array(data.length).fill(NaN);
   102→   102→  if (data.length < period) return result;
   103→   103→
   104→   104→  let sum = 0;
   105→   105→  for (let i = 0; i < period; i++) sum += data[i];
   106→   106→  result[period - 1] = sum / period;
   107→   107→
   108→   108→  for (let i = period; i < data.length; i++) {
   109→   109→    sum += data[i] - data[i - period];
   110→   110→    result[i] = sum / period;
   111→   111→  }
   112→   112→  return result;
   113→   113→}
   114→   114→
   115→   115→/** Exponential Moving Average — full-length array with NaN padding */
   116→   116→function ema(data: number[], period: number): number[] {
   117→   117→  const result: number[] = new Array(data.length).fill(NaN);
   118→   118→  if (data.length < period) return result;
   119→   119→
   120→   120→  const multiplier = 2 / (period + 1);
   121→   121→
   122→   122→  let sum = 0;
   123→   123→  for (let i = 0; i < period; i++) sum += data[i];
   124→   124→  result[period - 1] = sum / period;
   125→   125→
   126→   126→  for (let i = period; i < data.length; i++) {
   127→   127→    result[i] = (data[i] - result[i - 1]) * multiplier + result[i - 1];
   128→   128→  }
   129→   129→  return result;
   130→   130→}

   132→   132→/** Rolling standard deviation */
   133→   133→function rollingStdDev(data: number[], period: number): number[] {
   134→   134→  const result: number[] = new Array(data.length).fill(NaN);
   135→   135→  if (data.length < period) return result;
   136→   136→
   137→   137→  for (let i = period - 1; i < data.length; i++) {
   138→   138→    const slice = data.slice(i - period + 1, i + 1);
   139→   139→    const mean = slice.reduce((a, b) => a + b, 0) / period;
   140→   140→    const variance = slice.reduce((a, b) => a + (b - mean) ** 2, 0) / period;
   141→   141→    result[i] = Math.sqrt(variance);
   142→   142→  }
   143→   143→  return result;
   144→   144→}

   146→   146→/** Simple linear regression → { slope, intercept } */
   147→   147→function linearRegression(values: number[]): { slope: number; intercept: number } {
   148→   148→  const n = values.length;
   149→   149→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0 };
   150→   150→
   151→   151→  let sumX = 0;
   152→   152→  let sumY = 0;
   153→   153→  let sumXY = 0;
   154→   154→  let sumX2 = 0;
   155→   155→  for (let i = 0; i < n; i++) {
   156→   156→    sumX += i;
   157→   157→    sumY += values[i];
   158→   158→    sumXY += i * values[i];
   159→   159→    sumX2 += i * i;
   160→   160→  }
   161→   161→
   162→   162→  const denom = n * sumX2 - sumX * sumX;
   163→   163→  if (denom === 0) return { slope: 0, intercept: sumY / n };
   164→   164→
   165→   165→  return {
   166→   166→    slope: (n * sumXY - sumX * sumY) / denom,
   167→   167→    intercept: (sumY - ((n * sumXY - sumX * sumY) / denom) * sumX) / n,
   168→   168→  };
   169→   169→}

   171→   171→/** Percentile of a numeric array (0–100) */
   172→   172→function percentile(values: number[], p: number): number {
   173→   173→  const sorted = [...values].filter((v) => !isNaN(v)).sort((a, b) => a - b);
   174→   174→  if (sorted.length === 0) return 0;
   175→   175→  const idx = (p / 100) * (sorted.length - 1);
   176→   176→  const lo = Math.floor(idx);
   177→   177→  const hi = Math.ceil(idx);
   178→   178→  if (lo === hi) return sorted[lo];
   179→   179→  return sorted[lo] + (idx - lo) * (sorted[hi] - sorted[lo]);
   180→   180→}

   183→   183→function lastValid(arr: number[]): number | undefined {
   184→   184→  for (let i = arr.length - 1; i >= 0; i--) {
   185→   185→    if (!isNaN(arr[i])) return arr[i];
   186→   186→  }
   187→   187→  return undefined;
   188→   188→}

   192→   192→/** RSI using Wilder's smoothing */
   193→   193→function calculateRSI(closes: number[], period: number = 14): number[] {
   194→   194→  const result: number[] = new Array(closes.length).fill(NaN);
   195→   195→  if (closes.length < period + 1) return result;
   196→   196→
   197→   197→  let avgGain = 0;
   198→   198→  let avgLoss = 0;
   199→   199→
   200→   200→  for (let i = 1; i <= period; i++) {
   201→   201→    const change = closes[i] - closes[i - 1];
   202→   202→    if (change > 0) avgGain += change;
   203→   203→    else avgLoss += Math.abs(change);
   204→   204→  }
   205→   205→  avgGain /= period;
   206→   206→  avgLoss /= period;
   207→   207→
   208→   208→  result[period] = avgLoss === 0 ? 100 : 100 - 100 / (1 + avgGain / avgLoss);
   209→   209→
   210→   210→  for (let i = period + 1; i < closes.length; i++) {
   211→   211→    const change = closes[i] - closes[i - 1];
   212→   212→    const gain = change > 0 ? change : 0;
   213→   213→    const loss = change < 0 ? Math.abs(change) : 0;
   214→   214→
   215→   215→    avgGain = (avgGain * (period - 1) + gain) / period;
   216→   216→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   217→   217→
   218→   218→    const rs = avgLoss === 0 ? Infinity : avgGain / avgLoss;
   219→   219→    result[i] = clamp(100 - 100 / (1 + rs), 0, 100);
   220→   220→  }
   221→   221→  return result;
   222→   222→}

   224→   224→/** MACD: { macdLine, signalLine, histogram } */
   225→   225→function calculateMACD(
   226→   226→  closes: number[],
   227→   227→): { macdLine: number[]; signalLine: number[]; histogram: number[] } {
   228→   228→  const ema12 = ema(closes, 12);
   229→   229→  const ema26 = ema(closes, 26);
   230→   230→
   231→   231→  const macdLine: number[] = new Array(closes.length).fill(NaN);
   232→   232→  for (let i = 0; i < closes.length; i++) {
   233→   233→    if (!isNaN(ema12[i]) && !isNaN(ema26[i])) {
   234→   234→      macdLine[i] = ema12[i] - ema26[i];
   235→   235→    }
   236→   236→  }
   237→   237→
   238→   238→  // Build valid-only MACD array for signal-line EMA
   239→   239→  const validMacd: number[] = [];
   240→   240→  for (let i = 0; i < macdLine.length; i++) {
   241→   241→    if (!isNaN(macdLine[i])) validMacd.push(macdLine[i]);
   242→   242→  }
   243→   243→
   244→   244→  const sigEma = ema(validMacd, 9);
   245→   245→
   246→   246→  const signalLine: number[] = new Array(closes.length).fill(NaN);
   247→   247→  const histogram: number[] = new Array(closes.length).fill(NaN);
   248→   248→
   249→   249→  let vi = 0;
   250→   250→  for (let i = 0; i < closes.length; i++) {
   251→   251→    if (!isNaN(macdLine[i])) {
   252→   252→      if (vi < sigEma.length && !isNaN(sigEma[vi])) {
   253→   253→        signalLine[i] = sigEma[vi];
   254→   254→        histogram[i] = macdLine[i] - sigEma[vi];
   255→   255→      }
   256→   256→      vi++;
   257→   257→    }
   258→   258→  }
   259→   259→
   260→   260→  return { macdLine, signalLine, histogram };
   261→   261→}

   263→   263→/** Bollinger Bands: { upper, middle, lower, bandwidth } */
   264→   264→function calculateBollingerBands(
   265→   265→  closes: number[],
   266→   266→  period: number = 20,
   267→   267→  stdDevMult: number = 2,
   268→   268→): { upper: number[]; middle: number[]; lower: number[]; bandwidth: number[] } {
   269→   269→  const middle = sma(closes, period);
   270→   270→  const sd = rollingStdDev(closes, period);
   271→   271→
   272→   272→  const upper: number[] = new Array(closes.length).fill(NaN);
   273→   273→  const lower: number[] = new Array(closes.length).fill(NaN);
   274→   274→  const bandwidth: number[] = new Array(closes.length).fill(NaN);
   275→   275→
   276→   276→  for (let i = 0; i < closes.length; i++) {
   277→   277→    if (!isNaN(middle[i]) && !isNaN(sd[i])) {
   278→   278→      upper[i] = middle[i] + stdDevMult * sd[i];
   279→   279→      lower[i] = middle[i] - stdDevMult * sd[i];
   280→   280→      bandwidth[i] = middle[i] === 0 ? 0 : ((upper[i] - lower[i]) / middle[i]) * 100;
   281→   281→    }
   282→   282→  }
   283→   283→
   284→   284→  return { upper, middle, lower, bandwidth };
   285→   285→}

   287→   287→/** Stochastic Oscillator: { k, d } */
   288→   288→function calculateStochastic(
   289→   289→  highs: number[],
   290→   290→  lows: number[],
   291→   291→  closes: number[],
   292→   292→  kPeriod: number = 14,
   293→   293→  dPeriod: number = 3,
   294→   294→): { k: number[]; d: number[] } {
   295→   295→  const k: number[] = new Array(closes.length).fill(NaN);
   296→   296→
   297→   297→  for (let i = kPeriod - 1; i < closes.length; i++) {
   298→   298→    let lowest = lows[i];
   299→   299→    let highest = highs[i];
   300→   300→    for (let j = i - kPeriod + 1; j < i; j++) {
   301→   301→      if (lows[j] < lowest) lowest = lows[j];
   302→   302→      if (highs[j] > highest) highest = highs[j];
   303→   303→    }
   304→   304→    const range = highest - lowest;
   305→   305→    k[i] = range === 0 ? 50 : ((closes[i] - lowest) / range) * 100;
   306→   306→  }
   307→   307→
   308→   308→  const d = sma(k, dPeriod);
   309→   309→  return { k, d };
   310→   310→}

   312→   312→/** ADX with +DI and -DI */
   313→   313→function calculateADX(
   314→   314→  highs: number[],
   315→   315→  lows: number[],
   316→   316→  closes: number[],
   317→   317→  period: number = 14,
   318→   318→): { adx: number[]; plusDI: number[]; minusDI: number[] } {
   319→   319→  const adx: number[] = new Array(closes.length).fill(NaN);
   320→   320→  const plusDI: number[] = new Array(closes.length).fill(NaN);
   321→   321→  const minusDI: number[] = new Array(closes.length).fill(NaN);
   322→   322→
   323→   323→  if (closes.length < period * 2) return { adx, plusDI, minusDI };
   324→   324→
   325→   325→  // Compute raw TR, +DM, -DM
   326→   326→  const trRaw: number[] = new Array(closes.length).fill(0);
   327→   327→  const plusDMRaw: number[] = new Array(closes.length).fill(0);
   328→   328→  const minusDMRaw: number[] = new Array(closes.length).fill(0);
   329→   329→
   330→   330→  for (let i = 1; i < closes.length; i++) {
   331→   331→    trRaw[i] = Math.max(
   332→   332→      highs[i] - lows[i],
   333→   333→      Math.abs(highs[i] - closes[i - 1]),
   334→   334→      Math.abs(lows[i] - closes[i - 1]),
   335→   335→    );
   336→   336→
   337→   337→    const upMove = highs[i] - highs[i - 1];
   338→   338→    const downMove = lows[i - 1] - lows[i];
   339→   339→
   340→   340→    plusDMRaw[i] = upMove > downMove && upMove > 0 ? upMove : 0;
   341→   341→    minusDMRaw[i] = downMove > upMove && downMove > 0 ? downMove : 0;
   342→   342→  }
   343→   343→
   344→   344→  // Wilder's smoothing for first period
   345→   345→  let smoothTR = 0;
   346→   346→  let smoothPlusDM = 0;
   347→   347→  let smoothMinusDM = 0;
   348→   348→  for (let i = 1; i <= period; i++) {
   349→   349→    smoothTR += trRaw[i];
   350→   350→    smoothPlusDM += plusDMRaw[i];
   351→   351→    smoothMinusDM += minusDMRaw[i];
   352→   352→  }
   353→   353→
   354→   354→  const dxValues: number[] = [];
   355→   355→
   356→   356→  const computeDI = (idx: number) => {
   357→   357→    if (smoothTR === 0) {
   358→   358→      plusDI[idx] = 0;
   359→   359→      minusDI[idx] = 0;
   360→   360→      dxValues.push(0);
   361→   361→      return;
   362→   362→    }
   363→   363→    plusDI[idx] = (100 * smoothPlusDM) / smoothTR;
   364→   364→    minusDI[idx] = (100 * smoothMinusDM) / smoothTR;
   365→   365→    const diSum = plusDI[idx] + minusDI[idx];
   366→   366→    dxValues.push(diSum === 0 ? 0 : (100 * Math.abs(plusDI[idx] - minusDI[idx])) / diSum);
   367→   367→  };
   368→   368→
   369→   369→  computeDI(period);
   370→   370→
   371→   371→  for (let i = period + 1; i < closes.length; i++) {
   372→   372→    smoothTR = smoothTR - smoothTR / period + trRaw[i];
   373→   373→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMRaw[i];
   374→   374→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMRaw[i];
   375→   375→    computeDI(i);
   376→   376→  }
   377→   377→
   378→   378→  // ADX = Wilder's smoothing of DX
   379→   379→  if (dxValues.length >= period) {
   380→   380→    let adxVal =
   381→   381→      dxValues.slice(0, period).reduce((a, b) => a + b, 0) / period;
   382→   382→    const adxStartIdx = period - 1; // index in closes where first ADX goes
   383→   383→    const baseIdx = period; // index in closes where first DX was computed
   384→   384→    adx[baseIdx + adxStartIdx] = adxVal;
   385→   385→
   386→   386→    for (let i = period; i < dxValues.length; i++) {
   387→   387→      adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   388→   388→      adx[baseIdx + i] = adxVal;
   389→   389→    }
   390→   390→  }
   391→   391→
   392→   392→  return { adx, plusDI, minusDI };
   393→   393→}

   395→   395→/** Average True Range */
   396→   396→function calculateATR(
   397→   397→  highs: number[],
   398→   398→  lows: number[],
   399→   399→  closes: number[],
   400→   400→  period: number = 14,
   401→   401→): number[] {
   402→   402→  const result: number[] = new Array(closes.length).fill(NaN);
   403→   403→  if (closes.length < period + 1) return result;
   404→   404→
   405→   405→  const tr: number[] = [highs[0] - lows[0]];
   406→   406→  for (let i = 1; i < closes.length; i++) {
   407→   407→    tr.push(
   408→   408→      Math.max(
   409→   409→        highs[i] - lows[i],
   410→   410→        Math.abs(highs[i] - closes[i - 1]),
   411→   411→        Math.abs(lows[i] - closes[i - 1]),
   412→   412→      ),
   413→   413→    );
   414→   414→  }
   415→   415→
   416→   416→  // First ATR = SMA of first `period` TRs (TR starts at index 0, but Wilder uses first `period` starting from index 1)
   417→   417→  let atr = 0;
   418→   418→  for (let i = 1; i <= period; i++) atr += tr[i];
   419→   419→  atr /= period;
   420→   420→  result[period] = atr;
   421→   421→
   422→   422→  for (let i = period + 1; i < closes.length; i++) {
   423→   423→    atr = (atr * (period - 1) + tr[i]) / period;
   424→   424→    result[i] = atr;
   425→   425→  }
   426→   426→  return result;
   427→   427→}

   429→   429→/** On-Balance Volume */
   430→   430→function calculateOBV(closes: number[], volumes: number[]): number[] {
   431→   431→  const obv: number[] = new Array(closes.length).fill(0);
   432→   432→  for (let i = 1; i < closes.length; i++) {
   433→   433→    if (closes[i] > closes[i - 1]) obv[i] = obv[i - 1] + volumes[i];
   434→   434→    else if (closes[i] < closes[i - 1]) obv[i] = obv[i - 1] - volumes[i];
   435→   435→    else obv[i] = obv[i - 1];
   436→   436→  }
   437→   437→  return obv;
   438→   438→}

   443→   443→function scoreRSI(rsiValues: number[]): IndicatorScore {
   444→   444→  const rsiVal = lastValid(rsiValues);
   445→   445→  if (rsiVal === undefined) return noData('RSI');
   446→   446→
   447→   447→  let score: number;
   448→   448→  let name: string;
   449→   449→  let description: string;
   450→   450→
   451→   451→  if (rsiVal < 20) {
   452→   452→    score = mapRange(20 - rsiVal, 0, 20, 70, 100);
   453→   453→    name = 'RSI Mbushje e Shitjes';
   454→   454→    description = `RSI në ${rsiVal.toFixed(1)} — kushtet e thella të mbushjes së shitjes, sinjal i fortë për blerje`;
   455→   455→  } else if (rsiVal < 30) {
   456→   456→    score = mapRange(30 - rsiVal, 0, 10, 50, 70);

   536→   536→function scoreBollinger(
   537→   537→  closes: number[],
   538→   538→  bb: { upper: number[]; middle: number[]; lower: number[]; bandwidth: number[] },
   539→   539→): IndicatorScore {
   540→   540→  const last = closes.length - 1;
   541→   541→  const price = closes[last];
   542→   542→  const upper = bb.upper[last];
   543→   543→  const lower = bb.lower[last];
   544→   544→  const bw = bb.bandwidth[last];
   545→   545→
   546→   546→  if (isNaN(upper) || isNaN(lower) || isNaN(bw) || upper === lower) {
   547→   547→    return noData('Bollinger Bands');
   548→   548→  }
   549→   549→
   550→   550→  const position = (price - lower) / (upper - lower); // 0 to 1 (may go outside)
   551→   551→
   552→   552→  // Detect squeeze: bandwidth narrowing over last 5 valid values
   553→   553→  let squeeze = false;
   554→   554→  const recentBW: number[] = [];
   555→   555→  for (let i = last; i >= 0 && recentBW.length < 5; i--) {
   556→   556→    if (!isNaN(bb.bandwidth[i])) recentBW.unshift(bb.bandwidth[i]);
   557→   557→  }
   558→   558→  if (recentBW.length >= 4) {
   559→   559→    squeeze = recentBW[recentBW.length - 1] < recentBW[0] * 0.85;
   560→   560→  }
   561→   561→
   562→   562→  let score: number;
   563→   563→  let name: string;
   564→   564→  let description: string;
   565→   565→
   566→   566→  if (position < 0.1) {
   567→   567→    score = mapRange(position, 0, 0.1, 100, 70);
   568→   568→    name = squeeze ? 'Bollinger Squeeze & Mbushje' : 'Bollinger Nën Band';
   569→   569→    description = squeeze
   570→   570→      ? 'Çmimi nën Bollinger Band të ulët me ngushtim, shpërthim i madh pozitiv i pritur'
   571→   571→      : 'Çmimi nën Bollinger Band të ulët, potencial i lartë për kthim lart';
   572→   572→  } else if (position < 0.3) {
   573→   573→    score = mapRange(position, 0.1, 0.3, 70, 50);
   574→   574→    name = 'Bollinger Pozicion i Ulët';
   575→   575→    description = 'Çmimi në pjesën e poshtme të Bollinger Bands, lehtësisht bulliz';
   576→   576→  } else if (position < 0.7) {
   577→   577→    score = mapRange(position, 0.3, 0.7, 50, -50);
   578→   578→    name = 'Bollinger Neutral';
   579→   579→    description = 'Çmimi në mes të Bollinger Bands, asnjë sinjal i qartë';
   580→   580→  } else if (position < 0.9) {
   581→   581→    score = mapRange(position, 0.7, 0.9, -50, -70);
   582→   582→    name = 'Bollinger Pozicion i Lartë';
   583→   583→    description = 'Çmimi në pjesën e sipërme të Bollinger Bands, lehtësisht bearish';
   584→   584→  } else {
   585→   585→    score = mapRange(position, 0.9, 1.2, -70, -100);
   586→   586→    name = squeeze ? 'Bollinger Squeeze & Mbushje' : 'Bollinger Mbi Band';
   587→   587→    description = squeeze
   588→   588→      ? 'Çmimi mbi Bollinger Band të sipërme me ngushtim, shpërthim i madh negativ i pritur'
   589→   589→      : 'Çmimi mbi Bollinger Band të sipërme, rreziku i kthimit poshtë';
   590→   590→  }
   591→   591→
   592→   592→  // Squeeze amplification
   593→   593→  if (squeeze) {
   594→   594→    score = clamp(score * 1.1, -100, 100);
   595→   595→  }
   596→   596→
   597→   597→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: Math.abs(score) > 60 ? 'HIGH' : 'MEDIUM', description };
   598→   598→}

   786→   786→function scoreROC(closes: number[]): IndicatorScore {
   787→   787→  const last = closes.length - 1;
   788→   788→  if (last < 20) return noData('ROC');
   789→   789→
   790→   790→  const roc5 = ((closes[last] - closes[last - 5]) / closes[last - 5]) * 100;
   791→   791→  const roc10 = ((closes[last] - closes[last - 10]) / closes[last - 10]) * 100;
   792→   792→  const roc20 = ((closes[last] - closes[last - 20]) / closes[last - 20]) * 100;
   793→   793→
   794→   794→  // Weighted combination
   795→   795→  let rawScore = roc5 * 0.5 + roc10 * 0.3 + roc20 * 0.2;
   796→   796→
   797→   797→  // Extreme positive → mean reversion penalty
   798→   798→  if (roc5 > 10) {
   799→   799→    rawScore -= (roc5 - 10) * 2;
   800→   800→  }
   801→   801→  // Extreme negative → mean reversion penalty
   802→   802→  if (roc5 < -10) {
   803→   803→    rawScore += (-10 - roc5) * 2;
   804→   804→  }
   805→   805→
   806→   806→  const score = clamp(rawScore * 5, -100, 100); // Scale percentage to score
   807→   807→
   808→   808→  let name: string;
   809→   809→  let description: string;
   810→   810→
   811→   811→  if (roc5 > 5) {
   812→   812→    name = 'Momentum i Fortë Pozitiv';
   813→   813→    description = `ROC 5-ditor ${roc5.toFixed(2)}%, momentumi i shkurtër shumë pozitiv`;
   814→   814→  } else if (roc5 < -5) {
   815→   815→    name = 'Momentum i Fortë Negativ';
   816→   816→    description = `ROC 5-ditor ${roc5.toFixed(2)}%, momentumi i shkurtër shumë negativ`;
   817→   817→  } else if (rawScore > 1) {
   818→   818→    name = 'Momentum Pozitiv';
   819→   819→    description = `ROC i përgjithshëm pozitiv (5d: ${roc5.toFixed(2)}%, 10d: ${roc10.toFixed(2)}%, 20d: ${roc20.toFixed(2)}%)`;
   820→   820→  } else if (rawScore < -1) {
   821→   821→    name = 'Momentum Negativ';
   822→   822→    description = `ROC i përgjithshëm negativ (5d: ${roc5.toFixed(2)}%, 10d: ${roc10.toFixed(2)}%, 20d: ${roc20.toFixed(2)}%)`;
   823→   823→  } else {
   824→   824→    name = 'Momentum Neutral';
   825→   825→    description = 'Shkalla e ndryshimit afatshkurtër afër zero';

  1201→  1201→  }

  1203→  1203→  // 4. Shooting star: small body, long upper shadow (at least 2x body), body in lower 1/3
  1204→  1204→  {
  1205→  1205→    const body = Math.abs(c[n - 1] - o[n - 1]);
  1206→  1206→    const totalRange = h[n - 1] - l[n - 1];
  1207→  1207→    const lowerShadow = Math.min(o[n - 1], c[n - 1]) - l[n - 1];
  1208→  1208→    const upperShadow = h[n - 1] - Math.max(o[n - 1], c[n - 1]);
  1209→  1209→
  1210→  1210→    if (totalRange > 0 && body > 0 && upperShadow >= 2 * body && lowerShadow <= body * 0.3) {
  1211→  1211→      const advancing = c[n - 2] > o[n - 2] || c[n - 3] > o[n - 3];
  1212→  1212→      if (advancing) {
  1213→  1213→        const score = -65;
  1214→  1214→        if (Math.abs(score) > Math.abs(bestScore)) {
  1215→  1215→          bestScore = score;
  1216→  1216→          bestName = 'Yllë e Qitjes';
  1217→  1217→          bestSignal = 'BEARISH';
  1218→  1218→          bestImpact = 'MEDIUM';
  1219→  1219→          bestDescription = 'Yllë e qitjes pas një prirjeje të lartë, sinjal i shmangies negative';
  1220→  1220→        }
  1221→  1221→      }
  1222→  1222→    }
  1223→  1223→  }
  1224→  1224→
  1225→  1225→  // 5. Bullish engulfing: red candle followed by larger green candle
  1226→  1226→  if (n >= 2) {
  1227→  1227→    const prevRed = c[n - 2] < o[n - 2];
  1228→  1228→    const currGreen = c[n - 1] > o[n - 1];
  1229→  1229→    const prevBody = Math.abs(c[n - 2] - o[n - 2]);
  1230→  1230→    const currBody = Math.abs(c[n - 1] - o[n - 1]);
  1231→  1231→    const engulfsBody = c[n - 1] > o[n - 2] && o[n - 1] < c[n - 2];
  1232→  1232→
  1233→  1233→    if (prevRed && currGreen && engulfsBody && currBody > prevBody) {
  1234→  1234→      const score = 70;
  1235→  1235→      if (Math.abs(score) > Math.abs(bestScore)) {
  1236→  1236→        bestScore = score;
  1237→  1237→        bestName = 'Mbështjellje Bulliz';
  1238→  1238→        bestSignal = 'BULLISH';
  1239→  1239→        bestImpact = 'HIGH';
  1240→  1240→        bestDescription = 'Modeli i mbështjelljes bullize, sinjal i fortë për ndryshim të drejtimit';
  1241→  1241→      }
  1242→  1242→    }
  1243→  1243→  }
  1244→  1244→
  1245→  1245→  // 6. Bearish engulfing: green candle followed by larger red candle
  1246→  1246→  if (n >= 2) {
  1247→  1247→    const prevGreen = c[n - 2] > o[n - 2];
  1248→  1248→    const currRed = c[n - 1] < o[n - 1];
  1249→  1249→    const prevBody = Math.abs(c[n - 2] - o[n - 2]);
  1250→  1250→    const currBody = Math.abs(c[n - 1] - o[n - 1]);
  1251→  1251→    const engulfsBody = o[n - 1] > c[n - 2] && c[n - 1] < o[n - 2];
  1252→  1252→
  1253→  1253→    if (prevGreen && currRed && engulfsBody && currBody > prevBody) {
  1254→  1254→      const score = -70;
  1255→  1255→      if (Math.abs(score) > Math.abs(bestScore)) {
  1256→  1256→        bestScore = score;
  1257→  1257→        bestName = 'Mbështjellje Bearish';
  1258→  1258→        bestSignal = 'BEARISH';
  1259→  1259→        bestImpact = 'HIGH';
  1260→  1260→        bestDescription = 'Modeli i mbështjelljes bearish, sinjal i fortë për ndryshim të drejtimit';
  1261→  1261→      }
  1262→  1262→    }
  1263→  1263→  }
  1264→  1264→
  1265→  1265→  return { score: bestScore, name: bestName, signal: bestSignal, impact: bestImpact, description: bestDescription };
  1266→  1266→}

  1280→  1280→// ============== MAIN PREDICTION FUNCTION =====================
  1281→  1281→
  1282→  1282→export function predictStock(symbol: string, data: PricePoint[]): PredictionResult {
  1283→  1283→  const emptyResult = (): PredictionResult => ({
  1284→  1284→    symbol,
  1285→  1285→    score: 0,
  1286→  1286→    direction: 'NEUTRAL',
  1287→  1287→    confidence: 0,
  1288→  1288→    shortTerm: { prediction: 'SIDEWAYS', probability: 50, expectedMove: 0 },
  1289→  1289→    mediumTerm: { prediction: 'SIDEWAYS', probability: 50, expectedMove: 0 },
  1290→  1290→    riskLevel: 'HIGH',
  1291→  1291→    volatility: 'MEDIUM',
  1292→  1292→    keyFactors: [
  1293→  1293→      {
  1294→  1294→        name: 'Të Dhëna',
  1295→  1295→        signal: 'NEUTRAL' as SignalType,
  1296→  1296→        impact: 'HIGH' as ImpactLevel,
  1297→  1297→        score: 0,
  1298→  1298→        description: 'Të dhëna të pamjaftueshme për analizë teknike',
  1299→  1299→      },
  1300→  1300→    ],
  1301→  1301→    indicatorScores: {},
  1302→  1302→    timestamp: new Date().toISOString(),
  1303→  1303→  });
  1304→  1304→
  1305→  1305→  if (!data || data.length < MIN_DATA_POINTS) return emptyResult();
  1306→  1306→
  1307→  1307→  // ---------- Extract price arrays ----------
  1308→  1308→  const closes = data.map((d) => d.close);
  1309→  1309→  const highs = data.map((d) => d.high);
  1310→  1310→  const lows = data.map((d) => d.low);
  1311→  1311→  const volumes = data.map((d) => d.volume);
  1312→  1312→  const opens = data.map((d) => d.open);
  1313→  1313→
  1314→  1314→  // Validate
  1315→  1315→  if (closes.some((c) => c <= 0) || volumes.some((v) => v < 0)) return emptyResult();
  1316→  1316→
  1317→  1317→  // ---------- Calculate raw indicators ----------
  1318→  1318→  const rsiValues = calculateRSI(closes, 14);
  1319→  1319→  const macd = calculateMACD(closes);
  1320→  1320→  const bb = calculateBollingerBands(closes, 20, 2);
  1321→  1321→  const sma20 = sma(closes, 20);
  1322→  1322→  const sma50 = sma(closes, 50);
  1323→  1323→  const sma200 = sma(closes, 200);
  1324→  1324→  const stoch = calculateStochastic(highs, lows, closes, 14, 3);
  1325→  1325→  const adxResult = calculateADX(highs, lows, closes, 14);
  1326→  1326→  const atrValues = calculateATR(highs, lows, closes, 14);
  1327→  1327→  const atr20Values = calculateATR(highs, lows, closes, 20);
  1328→  1328→
  1329→  1329→  // ---------- Score each indicator ----------
  1330→  1330→  const scores: Record<string, IndicatorScore> = {};
  1331→  1331→
  1332→  1332→  scores.rsi = scoreRSI(rsiValues);
  1333→  1333→  scores.macdHistogram = scoreMACDHistogram(macd.histogram);
  1334→  1334→  scores.bollinger = scoreBollinger(closes, bb);
  1335→  1335→  scores.maTrend = scoreMATrend(closes, sma20, sma50, sma200);
  1336→  1336→  scores.stochastic = scoreStochastic(stoch.k, stoch.d);
  1337→  1337→  scores.adx = scoreADX(adxResult.adx, adxResult.plusDI, adxResult.minusDI);
  1338→  1338→  scores.atr = scoreATR(atrValues, closes);
  1339→  1339→  scores.roc = scoreROC(closes);
  1340→  1340→  scores.obv = scoreOBV(closes, volumes);
  1341→  1341→  scores.volumeConfirm = scoreVolumeConfirmation(closes, volumes);
  1342→  1342→  scores.macdCrossover = scoreMACDCrossover(macd.macdLine, macd.signalLine);
  1343→  1343→  scores.priceChannel = scorePriceChannel(highs, lows, closes);
  1344→  1344→  scores.divergence = scoreDivergence(closes, rsiValues);
  1345→  1345→  scores.vwap = scoreVWAP(closes, highs, lows, volumes);
  1346→  1346→  scores.pattern = scorePattern(opens, highs, lows, closes, volumes);
  1347→  1347→
  1348→  1348→  // ---------- Weighted final score ----------
  1349→  1349→  let finalScore = 0;
  1350→  1350→  const indicatorScores: Record<string, number> = {};
  1351→  1351→
  1352→  1352→  for (const [key, weight] of Object.entries(WEIGHTS)) {
  1353→  1353→    const s = scores[key]?.score ?? 0;
  1354→  1354→    finalScore += s * weight;
  1355→  1355→    indicatorScores[key] = Math.round(s * 100) / 100;
  1356→  1356→  }
  1357→  1357→  finalScore = clamp(Math.round(finalScore * 100) / 100, -100, 100);
  1358→  1358→
  1359→  1359→  // ---------- Confidence ----------
  1360→  1360→  const allIndicatorScores = Object.values(scores).map((s) => s.score);
  1361→  1361→  const mean = allIndicatorScores.reduce((a, b) => a + b, 0) / allIndicatorScores.length;
  1362→  1362→  const variance = allIndicatorScores.reduce((a, b) => a + (b - mean) ** 2, 0) / allIndicatorScores.length;
  1363→  1363→  const stdDev = Math.sqrt(variance);
  1364→  1364→  const confidence = clamp(Math.round((100 - stdDev * 1.5) * 100) / 100, 0, 100);
  1365→  1365→
  1366→  1366→  // ---------- Direction ----------
  1367→  1367→  const direction: Direction =
  1368→  1368→    finalScore > 60 ? 'STRONG_BUY' : finalScore > 25 ? 'BUY' : finalScore > -25 ? 'NEUTRAL' : finalScore > -60 ? 'SELL' : 'STRONG_SELL';
  1369→  1369→
  1370→  1370→  // ---------- Short-term prediction (1-3 days) ----------
  1371→  1371→  const shortTermWeights = { rsi: 0.25, stochastic: 0.20, macdHistogram: 0.25, volumeConfirm: 0.15, pattern: 0.15 };
  1372→  1372→  let shortScore = 0;
  1373→  1373→  for (const [k, w] of Object.entries(shortTermWeights)) {
  1374→  1374→    shortScore += (scores[k]?.score ?? 0) * w;
  1375→  1375→  }
  1376→  1376→  const shortPrediction: TrendPrediction = shortScore > 20 ? 'UP' : shortScore < -20 ? 'DOWN' : 'SIDEWAYS';
  1377→  1377→  const shortProbability = clamp(Math.round(Math.abs(shortScore) * 0.8 + 20), 0, 100);
  1378→  1378→  const currentATR = lastValid(atrValues) ?? 0;
  1379→  1379→  const lastClose = closes[closes.length - 1];
  1380→  1380→  const shortExpectedMove = lastClose > 0 ? Math.round(((currentATR / lastClose) * 100) * 100) / 100 : 0;
  1381→  1381→
  1382→  1382→  // ---------- Medium-term prediction (1-2 weeks) ----------
  1383→  1383→  const medTermWeights = { maTrend: 0.30, macdHistogram: 0.20, bollinger: 0.20, obv: 0.15, roc: 0.15 };
  1384→  1384→  let medScore = 0;
  1385→  1385→  for (const [k, w] of Object.entries(medTermWeights)) {
  1386→  1386→    medScore += (scores[k]?.score ?? 0) * w;
  1387→  1387→  }
  1388→  1388→  const medPrediction: TrendPrediction = medScore > 20 ? 'UP' : medScore < -20 ? 'DOWN' : 'SIDEWAYS';
  1389→  1389→  const medProbability = clamp(Math.round(Math.abs(medScore) * 0.8 + 20), 0, 100);
  1390→  1390→  const currentATR20 = lastValid(atr20Values) ?? currentATR;
  1391→  1391→  const medExpectedMove = lastClose > 0 ? Math.round(((currentATR20 / lastClose) * 100) * 100) / 100 : 0;
  1392→  1392→
  1393→  1393→  // ---------- Risk level & volatility ----------
  1394→  1394→  const atrHistory: number[] = [];
  1395→  1395→  const atrLast = atrValues.length - 1;
  1396→  1396→  for (let i = atrLast; i >= 0 && atrHistory.length < 60; i--) {
  1397→  1397→    if (!isNaN(atrValues[i]) && atrValues[i] > 0) atrHistory.unshift(atrValues[i]);
  1398→  1398→  }
  1399→  1399→
  1400→  1400→  const p30 = atrHistory.length > 5 ? percentile(atrHistory, 30) : 0;
  1401→  1401→  const p70 = atrHistory.length > 5 ? percentile(atrHistory, 70) : Infinity;
  1402→  1402→
  1403→  1403→  const atrHigh = currentATR > p70;
  1404→  1404→  const atrLow = currentATR < p30;
  1405→  1405→
  1406→  1406→  const volatility: RiskOrVolatility = atrHigh ? 'HIGH' : atrLow ? 'LOW' : 'MEDIUM';
  1407→  1407→
  1408→  1408→  const riskLevel: RiskOrVolatility =
  1409→  1409→    atrHigh || confidence < 30 ? 'HIGH' : atrLow && confidence > 70 ? 'LOW' : 'MEDIUM';
  1410→  1410→
  1411→  1411→  // ---------- Top 5 key factors ----------
  1412→  1412→  const sortedFactors = Object.values(scores)
  1413→  1413→    .sort((a, b) => Math.abs(b.score) - Math.abs(a.score))
  1414→  1414→    .slice(0, 5);
  1415→  1415→
  1416→  1416→  const keyFactors: KeyFactor[] = sortedFactors.map((f) => ({
  1417→  1417→    name: f.name,
  1418→  1418→    signal: f.signal,
  1419→  1419→    impact: f.impact,
  1420→  1420→    score: Math.round(f.score * 100) / 100,
  1421→  1421→    description: f.description,
  1422→  1422→  }));
  1423→  1423→
  1424→  1424→  // ---------- Build result ----------
  1425→  1425→  return {
  1426→  1426→    symbol,
  1427→  1427→    score: finalScore,
  1428→  1428→    direction,
  1429→  1429→    confidence,
  1430→  1430→    shortTerm: {
  1431→  1431→      prediction: shortPrediction,
  1432→  1432→      probability: shortProbability,
  1433→  1433→      expectedMove: shortExpectedMove,
  1434→  1434→    },
  1435→  1435→    mediumTerm: {
  1436→  1436→      prediction: medPrediction,
  1437→  1437→      probability: medProbability,
  1438→  1438→      expectedMove: medExpectedMove,
  1439→  1439→    },
  1440→  1440→    riskLevel,
  1441→  1441→    volatility,
  1442→  1442→    keyFactors,
  1443→  1443→    indicatorScores,
  1444→  1444→    timestamp: new Date().toISOString(),
  1445→  1445→  };
  1446→  1446→}
