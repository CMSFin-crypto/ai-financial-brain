   123→   123→   123→function mapRange(
   124→   124→   124→  value: number,
   125→   125→   125→  inMin: number,
   126→   126→   126→  inMax: number,
   127→   127→   127→  outMin: number,
   128→   128→   128→  outMax: number,
   129→   129→   129→): number {
   130→   130→   130→  if (inMax === inMin) return (outMin + outMax) / 2;
   131→   131→   131→  const result = outMin + ((value - inMin) / (inMax - inMin)) * (outMax - outMin);
   132→   132→   132→  return clamp(result, Math.min(outMin, outMax), Math.max(outMin, outMax));
   133→   133→   133→}

   135→   135→   135→/** Simple Moving Average — full-length array with NaN padding */
   136→   136→   136→function sma(data: number[], period: number): number[] {
   137→   137→   137→  const result: number[] = new Array(data.length).fill(NaN);
   138→   138→   138→  if (data.length < period) return result;
   139→   139→   139→
   140→   140→   140→  let sum = 0;
   141→   141→   141→  for (let i = 0; i < period; i++) sum += data[i];
   142→   142→   142→  result[period - 1] = sum / period;
   143→   143→   143→
   144→   144→   144→  for (let i = period; i < data.length; i++) {
   145→   145→   145→    sum += data[i] - data[i - period];
   146→   146→   146→    result[i] = sum / period;
   147→   147→   147→  }
   148→   148→   148→  return result;
   149→   149→   149→}

   151→   151→   151→/** Exponential Moving Average — full-length array with NaN padding */
   152→   152→   152→function ema(data: number[], period: number): number[] {
   153→   153→   153→  const result: number[] = new Array(data.length).fill(NaN);
   154→   154→   154→  if (data.length < period) return result;
   155→   155→   155→
   156→   156→   156→  const multiplier = 2 / (period + 1);
   157→   157→   157→
   158→   158→   158→  let sum = 0;
   159→   159→   159→  for (let i = 0; i < period; i++) sum += data[i];
   160→   160→   160→  result[period - 1] = sum / period;
   161→   161→   161→
   162→   162→   162→  for (let i = period; i < data.length; i++) {
   163→   163→   163→    result[i] = (data[i] - result[i - 1]) * multiplier + result[i - 1];
   164→   164→   164→  }
   165→   165→   165→  return result;
   166→   166→   166→}

   168→   168→   168→/** Rolling standard deviation */
   169→   169→   169→function rollingStdDev(data: number[], period: number): number[] {
   170→   170→   170→  const result: number[] = new Array(data.length).fill(NaN);
   171→   171→   171→  if (data.length < period) return result;
   172→   172→   172→
   173→   173→   173→  for (let i = period - 1; i < data.length; i++) {
   174→   174→   174→    const slice = data.slice(i - period + 1, i + 1);
   175→   175→   175→    const mean = slice.reduce((a, b) => a + b, 0) / period;
   176→   176→   176→    const variance = slice.reduce((a, b) => a + (b - mean) ** 2, 0) / period;
   177→   177→   177→    result[i] = Math.sqrt(variance);
   178→   178→   178→  }
   179→   179→   179→  return result;
   180→   180→   180→}

   182→   182→   182→/** Simple linear regression → { slope, intercept } */
   183→   183→   183→function linearRegression(values: number[]): { slope: number; intercept: number } {
   184→   184→   184→  const n = values.length;
   185→   185→   185→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0 };
   186→   186→   186→
   187→   187→   187→  let sumX = 0;
   188→   188→   188→  let sumY = 0;
   189→   189→   189→  let sumXY = 0;
   190→   190→   190→  let sumX2 = 0;
   191→   191→   191→  for (let i = 0; i < n; i++) {
   192→   192→   192→    sumX += i;
   193→   193→   193→    sumY += values[i];
   194→   194→   194→    sumXY += i * values[i];
   195→   195→   195→    sumX2 += i * i;
   196→   196→   196→  }
   197→   197→   197→
   198→   198→   198→  const denom = n * sumX2 - sumX * sumX;
   199→   199→   199→  if (denom === 0) return { slope: 0, intercept: sumY / n };
   200→   200→   200→
   201→   201→   201→  return {
   202→   202→   202→    slope: (n * sumXY - sumX * sumY) / denom,
   203→   203→   203→    intercept: (sumY - ((n * sumXY - sumX * sumY) / denom) * sumX) / n,
   204→   204→   204→  };
   205→   205→   205→}

   207→   207→   207→/** Percentile of a numeric array (0–100) */
   208→   208→   208→function percentile(values: number[], p: number): number {
   209→   209→   209→  const sorted = [...values].filter((v) => !isNaN(v)).sort((a, b) => a - b);
   210→   210→   210→  if (sorted.length === 0) return 0;
   211→   211→   211→  const idx = (p / 100) * (sorted.length - 1);
   212→   212→   212→  const lo = Math.floor(idx);
   213→   213→   213→  const hi = Math.ceil(idx);
   214→   214→   214→  if (lo === hi) return sorted[lo];
   215→   215→   215→  return sorted[lo] + (idx - lo) * (sorted[hi] - sorted[lo]);
   216→   216→   216→}

   228→   228→   228→/** RSI using Wilder's smoothing */
   229→   229→   229→function calculateRSI(closes: number[], period: number = 14): number[] {
   230→   230→   230→  const result: number[] = new Array(closes.length).fill(NaN);
   231→   231→   231→  if (closes.length < period + 1) return result;
   232→   232→   232→
   233→   233→   233→  let avgGain = 0;
   234→   234→   234→  let avgLoss = 0;
   235→   235→   235→
   236→   236→   236→  for (let i = 1; i <= period; i++) {
   237→   237→   237→    const change = closes[i] - closes[i - 1];
   238→   238→   238→    if (change > 0) avgGain += change;
   239→   239→   239→    else avgLoss += Math.abs(change);
   240→   240→   240→  }
   241→   241→   241→  avgGain /= period;
   242→   242→   242→  avgLoss /= period;
   243→   243→   243→
   244→   244→   244→  result[period] = avgLoss === 0 ? 100 : 100 - 100 / (1 + avgGain / avgLoss);
   245→   245→   245→
   246→   246→   246→  for (let i = period + 1; i < closes.length; i++) {
   247→   247→   247→    const change = closes[i] - closes[i - 1];
   248→   248→   248→    const gain = change > 0 ? change : 0;
   249→   249→   249→    const loss = change < 0 ? Math.abs(change) : 0;
   250→   250→   250→
   251→   251→   251→    avgGain = (avgGain * (period - 1) + gain) / period;
   252→   252→   252→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   253→   253→   253→
   254→   254→   254→    const rs = avgLoss === 0 ? Infinity : avgGain / avgLoss;
   255→   255→   255→    result[i] = clamp(100 - 100 / (1 + rs), 0, 100);
   256→   256→   256→  }
   257→   257→   257→  return result;
   258→   258→   258→}

   260→   260→   260→/** MACD: { macdLine, signalLine, histogram } */
   261→   261→   261→function calculateMACD(
   262→   262→   262→  closes: number[],
   263→   263→   263→): { macdLine: number[]; signalLine: number[]; histogram: number[] } {
   264→   264→   264→  const ema12 = ema(closes, 12);
   265→   265→   265→  const ema26 = ema(closes, 26);
   266→   266→   266→
   267→   267→   267→  const macdLine: number[] = new Array(closes.length).fill(NaN);
   268→   268→   268→  for (let i = 0; i < closes.length; i++) {
   269→   269→   269→    if (!isNaN(ema12[i]) && !isNaN(ema26[i])) {
   270→   270→   270→      macdLine[i] = ema12[i] - ema26[i];
   271→   271→   271→    }
   272→   272→   272→  }
   273→   273→   273→
   274→   274→   274→  // Build valid-only MACD array for signal-line EMA
   275→   275→   275→  const validMacd: number[] = [];
   276→   276→   276→  for (let i = 0; i < macdLine.length; i++) {
   277→   277→   277→    if (!isNaN(macdLine[i])) validMacd.push(macdLine[i]);
   278→   278→   278→  }
   279→   279→   279→
   280→   280→   280→  const sigEma = ema(validMacd, 9);
   281→   281→   281→
   282→   282→   282→  const signalLine: number[] = new Array(closes.length).fill(NaN);
   283→   283→   283→  const histogram: number[] = new Array(closes.length).fill(NaN);
   284→   284→   284→
   285→   285→   285→  let vi = 0;
   286→   286→   286→  for (let i = 0; i < closes.length; i++) {
   287→   287→   287→    if (!isNaN(macdLine[i])) {
   288→   288→   288→      if (vi < sigEma.length && !isNaN(sigEma[vi])) {
   289→   289→   289→        signalLine[i] = sigEma[vi];
   290→   290→   290→        histogram[i] = macdLine[i] - sigEma[vi];
   291→   291→   291→      }
   292→   292→   292→      vi++;
   293→   293→   293→    }
   294→   294→   294→  }
   295→   295→   295→
   296→   296→   296→  return { macdLine, signalLine, histogram };
   297→   297→   297→}

   299→   299→   299→/** Bollinger Bands: { upper, middle, lower, bandwidth } */
   300→   300→   300→function calculateBollingerBands(
   301→   301→   301→  closes: number[],
   302→   302→   302→  period: number = 20,
   303→   303→   303→  stdDevMult: number = 2,
   304→   304→   304→): { upper: number[]; middle: number[]; lower: number[]; bandwidth: number[] } {
   305→   305→   305→  const middle = sma(closes, period);
   306→   306→   306→  const sd = rollingStdDev(closes, period);
   307→   307→   307→
   308→   308→   308→  const upper: number[] = new Array(closes.length).fill(NaN);
   309→   309→   309→  const lower: number[] = new Array(closes.length).fill(NaN);
   310→   310→   310→  const bandwidth: number[] = new Array(closes.length).fill(NaN);
   311→   311→   311→
   312→   312→   312→  for (let i = 0; i < closes.length; i++) {
   313→   313→   313→    if (!isNaN(middle[i]) && !isNaN(sd[i])) {
   314→   314→   314→      upper[i] = middle[i] + stdDevMult * sd[i];
   315→   315→   315→      lower[i] = middle[i] - stdDevMult * sd[i];
   316→   316→   316→      bandwidth[i] = middle[i] === 0 ? 0 : ((upper[i] - lower[i]) / middle[i]) * 100;
   317→   317→   317→    }
   318→   318→   318→  }
   319→   319→   319→
   320→   320→   320→  return { upper, middle, lower, bandwidth };
   321→   321→   321→}

   324→   324→   324→function calculateStochastic(
   325→   325→   325→  highs: number[],
   326→   326→   326→  lows: number[],
   327→   327→   327→  closes: number[],
   328→   328→   328→  kPeriod: number = 14,
   329→   329→   329→  dPeriod: number = 3,
   330→   330→   330→): { k: number[]; d: number[] } {
   331→   331→   331→  const k: number[] = new Array(closes.length).fill(NaN);
   332→   332→   332→
   333→   333→   333→  for (let i = kPeriod - 1; i < closes.length; i++) {
   334→   334→   334→    let lowest = lows[i];
   335→   335→   335→    let highest = highs[i];
   336→   336→   336→    for (let j = i - kPeriod + 1; j < i; j++) {
   337→   337→   337→      if (lows[j] < lowest) lowest = lows[j];
   338→   338→   338→      if (highs[j] > highest) highest = highs[j];
   339→   339→   339→    }
   340→   340→   340→    const range = highest - lowest;
   341→   341→   341→    k[i] = range === 0 ? 50 : ((closes[i] - lowest) / range) * 100;
   342→   342→   342→  }
   343→   343→   343→
   344→   344→   344→  const d = sma(k, dPeriod);
   345→   345→   345→  return { k, d };
   346→   346→   346→}

   348→   348→   348→/** ADX with +DI and -DI */
   349→   349→   349→function calculateADX(
   350→   350→   350→  highs: number[],
   351→   351→   351→  lows: number[],
   352→   352→   352→  closes: number[],
   353→   353→   353→  period: number = 14,
   354→   354→   354→): { adx: number[]; plusDI: number[]; minusDI: number[] } {
   355→   355→   355→  const adx: number[] = new Array(closes.length).fill(NaN);
   356→   356→   356→  const plusDI: number[] = new Array(closes.length).fill(NaN);
   357→   357→   357→  const minusDI: number[] = new Array(closes.length).fill(NaN);
   358→   358→   358→
   359→   359→   359→  if (closes.length < period * 2) return { adx, plusDI, minusDI };
   360→   360→   360→
   361→   361→   361→  // Compute raw TR, +DM, -DM
   362→   362→   362→  const trRaw: number[] = new Array(closes.length).fill(0);
   363→   363→   363→  const plusDMRaw: number[] = new Array(closes.length).fill(0);
   364→   364→   364→  const minusDMRaw: number[] = new Array(closes.length).fill(0);
   365→   365→   365→
   366→   366→   366→  for (let i = 1; i < closes.length; i++) {
   367→   367→   367→    trRaw[i] = Math.max(
   368→   368→   368→      highs[i] - lows[i],
   369→   369→   369→      Math.abs(highs[i] - closes[i - 1]),
   370→   370→   370→      Math.abs(lows[i] - closes[i - 1]),
   371→   371→   371→    );
   372→   372→   372→
   373→   373→   373→    const upMove = highs[i] - highs[i - 1];
   374→   374→   374→    const downMove = lows[i - 1] - lows[i];
   375→   375→   375→
   376→   376→   376→    plusDMRaw[i] = upMove > downMove && upMove > 0 ? upMove : 0;
   377→   377→   377→    minusDMRaw[i] = downMove > upMove && downMove > 0 ? downMove : 0;
   378→   378→   378→  }
   379→   379→   379→
   380→   380→   380→  // Wilder's smoothing for first period
   381→   381→   381→  let smoothTR = 0;
   382→   382→   382→  let smoothPlusDM = 0;
   383→   383→   383→  let smoothMinusDM = 0;
   384→   384→   384→  for (let i = 1; i <= period; i++) {
   385→   385→   385→    smoothTR += trRaw[i];
   386→   386→   386→    smoothPlusDM += plusDMRaw[i];
   387→   387→   387→    smoothMinusDM += minusDMRaw[i];
   388→   388→   388→  }
   389→   389→   389→
   390→   390→   390→  const dxValues: number[] = [];
   391→   391→   391→
   392→   392→   392→  const computeDI = (idx: number) => {
   393→   393→   393→    if (smoothTR === 0) {
   394→   394→   394→      plusDI[idx] = 0;
   395→   395→   395→      minusDI[idx] = 0;
   396→   396→   396→      dxValues.push(0);
   397→   397→   397→      return;
   398→   398→   398→    }
   399→   399→   399→    plusDI[idx] = (100 * smoothPlusDM) / smoothTR;
   400→   400→   400→    minusDI[idx] = (100 * smoothMinusDM) / smoothTR;
   401→   401→   401→    const diSum = plusDI[idx] + minusDI[idx];
   402→   402→   402→    dxValues.push(diSum === 0 ? 0 : (100 * Math.abs(plusDI[idx] - minusDI[idx])) / diSum);
   403→   403→   403→  };
   404→   404→   404→
   405→   405→   405→  computeDI(period);
   406→   406→   406→
   407→   407→   407→  for (let i = period + 1; i < closes.length; i++) {
   408→   408→   408→    smoothTR = smoothTR - smoothTR / period + trRaw[i];
   409→   409→   409→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMRaw[i];
   410→   410→   410→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMRaw[i];
   411→   411→   411→    computeDI(i);
   412→   412→   412→  }
   413→   413→   413→
   414→   414→   414→  // ADX = Wilder's smoothing of DX
   415→   415→   415→  if (dxValues.length >= period) {
   416→   416→   416→    let adxVal =
   417→   417→   417→      dxValues.slice(0, period).reduce((a, b) => a + b, 0) / period;
   418→   418→   418→    const adxStartIdx = period - 1; // index in closes where first ADX goes
   419→   419→   419→    const baseIdx = period; // index in closes where first DX was computed
   420→   420→   420→    adx[baseIdx + adxStartIdx] = adxVal;
   421→   421→   421→
   422→   422→   422→    for (let i = period; i < dxValues.length; i++) {
   423→   423→   423→      adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   424→   424→   424→      adx[baseIdx + i] = adxVal;
   425→   425→   425→    }
   426→   426→   426→  }
   427→   427→   427→
   428→   428→   428→  return { adx, plusDI, minusDI };
   429→   429→   429→}
   430→   430→   430→
   431→   431→   431→/** Average True Range */
   432→   432→   432→function calculateATR(
   433→   433→   433→  highs: number[],
   434→   434→   434→  lows: number[],
   435→   435→   435→  closes: number[],
   436→   436→   436→  period: number = 14,
   437→   437→   437→): number[] {
   438→   438→   438→  const result: number[] = new Array(closes.length).fill(NaN);
   439→   439→   439→  if (closes.length < period + 1) return result;
   440→   440→   440→
   441→   441→   441→  const tr: number[] = [highs[0] - lows[0]];
   442→   442→   442→  for (let i = 1; i < closes.length; i++) {
   443→   443→   443→    tr.push(
   444→   444→   444→      Math.max(
   445→   445→   445→        highs[i] - lows[i],
   446→   446→   446→        Math.abs(highs[i] - closes[i - 1]),
   447→   447→   447→        Math.abs(lows[i] - closes[i - 1]),
   448→   448→   448→      ),
   449→   449→   449→    );
   450→   450→   450→  }
   451→   451→   451→
   452→   452→   452→  // First ATR = SMA of first `period` TRs (TR starts at index 0, but Wilder uses first `period` starting from index 1)
   453→   453→   453→  let atr = 0;
   454→   454→   454→  for (let i = 1; i <= period; i++) atr += tr[i];
   455→   455→   455→  atr /= period;
   456→   456→   456→  result[period] = atr;
   457→   457→   457→
   458→   458→   458→  for (let i = period + 1; i < closes.length; i++) {
   459→   459→   459→    atr = (atr * (period - 1) + tr[i]) / period;
   460→   460→   460→    result[i] = atr;
   461→   461→   461→  }
   462→   462→   462→  return result;
   463→   463→   463→}

   465→   465→   465→/** On-Balance Volume */
   466→   466→   466→function calculateOBV(closes: number[], volumes: number[]): number[] {
   467→   467→   467→  const obv: number[] = new Array(closes.length).fill(0);
   468→   468→   468→  for (let i = 1; i < closes.length; i++) {
   469→   469→   469→    if (closes[i] > closes[i - 1]) obv[i] = obv[i - 1] + volumes[i];
   470→   470→   470→    else if (closes[i] < closes[i - 1]) obv[i] = obv[i - 1] - volumes[i];
   471→   471→   471→    else obv[i] = obv[i - 1];
   472→   472→   472→  }
   473→   473→   473→  return obv;
   474→   474→   474→}

   572→   572→   572→function scoreBollinger(
   573→   573→   573→  closes: number[],
   574→   574→   574→  bb: { upper: number[]; middle: number[]; lower: number[]; bandwidth: number[] },
   575→   575→   575→): IndicatorScore {
   576→   576→   576→  const last = closes.length - 1;
   577→   577→   577→  const price = closes[last];
   578→   578→   578→  const upper = bb.upper[last];
   579→   579→   579→  const lower = bb.lower[last];
   580→   580→   580→  const bw = bb.bandwidth[last];
   581→   581→   581→
   582→   582→   582→  if (isNaN(upper) || isNaN(lower) || isNaN(bw) || upper === lower) {
   583→   583→   583→    return noData('Bollinger Bands');
   584→   584→   584→  }
   585→   585→   585→
   586→   586→   586→  const position = (price - lower) / (upper - lower); // 0 to 1 (may go outside)
   587→   587→   587→

   589→   589→   589→  let squeeze = false;

   591→   591→   591→  for (let i = last; i >= 0 && recentBW.length < 5; i--) {

   593→   593→   593→  }

   595→   595→   595→    squeeze = recentBW[recentBW.length - 1] < recentBW[0] * 0.85;
   596→   596→   596→  }
   597→   597→   597→
   598→   598→   598→  let score: number;
   599→   599→   599→  let name: string;
   600→   600→   600→  let description: string;
   601→   601→   601→
   602→   602→   602→  if (position < 0.1) {

   604→   604→   604→    name = squeeze ? 'Bollinger Squeeze & Mbushje' : 'Bollinger Nën Band';
   605→   605→   605→    description = squeeze
   606→   606→   606→      ? 'Çmimi nën Bollinger Band të ulët me ngushtim, shpërthim i madh pozitiv i pritur'
   607→   607→   607→      : 'Çmimi nën Bollinger Band të ulët, potencial i lartë për kthim lart';
   608→   608→   608→  } else if (position < 0.3) {
   609→   609→   609→    score = mapRange(position, 0.1, 0.3, 70, 50);
   610→   610→   610→    name = 'Bollinger Pozicion i Ulët';
   611→   611→   611→    description = 'Çmimi në pjesën e poshtme të Bollinger Bands, lehtësisht bulliz';
   612→   612→   612→  } else if (position < 0.7) {
   613→   613→   613→    score = mapRange(position, 0.3, 0.7, 50, -50);
   614→   614→   614→    name = 'Bollinger Neutral';
   615→   615→   615→    description = 'Çmimi në mes të Bollinger Bands, asnjë sinjal i qartë';
   616→   616→   616→  } else if (position < 0.9) {
   617→   617→   617→    score = mapRange(position, 0.7, 0.9, -50, -70);
   618→   618→   618→    name = 'Bollinger Pozicion i Lartë';
   619→   619→   619→    description = 'Çmimi në pjesën e sipërme të Bollinger Bands, lehtësisht bearish';
   620→   620→   620→  } else {
   621→   621→   621→    score = mapRange(position, 0.9, 1.2, -70, -100);
   622→   622→   622→    name = squeeze ? 'Bollinger Squeeze & Mbushje' : 'Bollinger Mbi Band';
   623→   623→   623→    description = squeeze

   625→   625→   625→      : 'Çmimi mbi Bollinger Band të sipërme, rreziku i kthimit poshtë';
   626→   626→   626→  }
   627→   627→   627→
   628→   628→   628→  // Squeeze amplification
   629→   629→   629→  if (squeeze) {
   630→   630→   630→    score = clamp(score * 1.1, -100, 100);
   631→   631→   631→  }
   632→   632→   632→
   633→   633→   633→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: Math.abs(score) > 60 ? 'HIGH' : 'MEDIUM', description };

  1127→  1127→  1127→function scoreVWAP(closes: number[], highs: number[], lows: number[], volumes: number[]): IndicatorScore {
  1128→  1128→  1128→  const last = closes.length - 1;
  1129→  1129→  1129→  if (last < 20) return noData('VWAP');
  1130→  1130→  1130→
  1131→  1131→  1131→  // Approximate VWAP over 20 days: cumulative (H+L+C)/3 * volume / cumulative volume
  1132→  1132→  1132→  let cumVP = 0;
  1133→  1133→  1133→  let cumVol = 0;
  1134→  1134→  1134→  for (let i = last - 19; i <= last; i++) {
  1135→  1135→  1135→    const tp = (highs[i] + lows[i] + closes[i]) / 3;
  1136→  1136→  1136→    cumVP += tp * volumes[i];
  1137→  1137→  1137→    cumVol += volumes[i];
  1138→  1138→  1138→  }
  1139→  1139→  1139→
  1140→  1140→  1140→  const vwap = cumVol === 0 ? closes[last] : cumVP / cumVol;
  1141→  1141→  1141→  const price = closes[last];
  1142→  1142→  1142→  const deviation = ((price - vwap) / vwap) * 100; // percentage deviation
  1143→  1143→  1143→
  1144→  1144→  1144→  let score: number;
  1145→  1145→  1145→  let name: string;
  1146→  1146→  1146→  let description: string;
  1147→  1147→  1147→
  1148→  1148→  1148→  if (price > vwap) {
  1149→  1149→  1149→    score = clamp(mapRange(deviation, 0, 5, 50, 65), 50, 65);

  1151→  1151→  1151→    description = `Çmimi ${deviation.toFixed(2)}% mbi VWAP, blerësit në kontroll`;
  1152→  1152→  1152→  } else {

  1155→  1155→  1155→    description = `Çmimi ${(-deviation).toFixed(2)}% poshtë VWAP, shitësit në kontroll`;
  1156→  1156→  1156→  }
  1157→  1157→  1157→
  1158→  1158→  1158→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: 'LOW', description };

  1403→  1403→  1403→  const variance = allIndicatorScores.reduce((a, b) => a + (b - mean) ** 2, 0) / allIndicatorScores.length;
  1404→  1404→  1404→  const stdDev = Math.sqrt(variance);
  1405→  1405→  1405→  const confidence = clamp(Math.round((100 - stdDev * 1.5) * 100) / 100, 0, 100);
  1406→  1406→  1406→
  1407→  1407→  1407→  // ---------- Direction ----------
  1408→  1408→  1408→  const direction: Direction =
  1409→  1409→  1409→    finalScore > 60 ? 'STRONG_BUY' : finalScore > 25 ? 'BUY' : finalScore > -25 ? 'NEUTRAL' : finalScore > -60 ? 'SELL' : 'STRONG_SELL';
  1410→  1410→  1410→
  1411→  1411→  1411→  // ---------- Short-term prediction (1-3 days) ----------
  1412→  1412→  1412→  const shortTermWeights = { rsi: 0.25, stochastic: 0.20, macdHistogram: 0.25, volumeConfirm: 0.15, pattern: 0.15 };
  1413→  1413→  1413→  let shortScore = 0;
  1414→  1414→  1414→  for (const [k, w] of Object.entries(shortTermWeights)) {
  1415→  1415→  1415→    shortScore += (scores[k]?.score ?? 0) * w;
  1416→  1416→  1416→  }
  1417→  1417→  1417→  const shortPrediction: TrendPrediction = shortScore > 20 ? 'UP' : shortScore < -20 ? 'DOWN' : 'SIDEWAYS';
  1418→  1418→  1418→  const shortProbability = clamp(Math.round(Math.abs(shortScore) * 0.8 + 20), 0, 100);
  1419→  1419→  1419→  const currentATR = lastValid(atrValues) ?? 0;
  1420→  1420→  1420→  const lastClose = closes[closes.length - 1];
  1421→  1421→  1421→  const shortExpectedMove = lastClose > 0 ? Math.round(((currentATR / lastClose) * 100) * 100) / 100 : 0;
  1422→  1422→  1422→
  1423→  1423→  1423→  // ---------- Medium-term prediction (1-2 weeks) ----------
  1424→  1424→  1424→  const medTermWeights = { maTrend: 0.30, macdHistogram: 0.20, bollinger: 0.20, obv: 0.15, roc: 0.15 };
  1425→  1425→  1425→  let medScore = 0;
  1426→  1426→  1426→  for (const [k, w] of Object.entries(medTermWeights)) {
  1427→  1427→  1427→    medScore += (scores[k]?.score ?? 0) * w;
  1428→  1428→  1428→  }
  1429→  1429→  1429→  const medPrediction: TrendPrediction = medScore > 20 ? 'UP' : medScore < -20 ? 'DOWN' : 'SIDEWAYS';
  1430→  1430→  1430→  const medProbability = clamp(Math.round(Math.abs(medScore) * 0.8 + 20), 0, 100);
  1431→  1431→  1431→  const currentATR20 = lastValid(atr20Values) ?? currentATR;
  1432→  1432→  1432→  const medExpectedMove = lastClose > 0 ? Math.round(((currentATR20 / lastClose) * 100) * 100) / 100 : 0;
  1433→  1433→  1433→
  1434→  1434→  1434→  // ---------- Risk level & volatility ----------
  1435→  1435→  1435→  const atrHistory: number[] = [];
  1436→  1436→  1436→  const atrLast = atrValues.length - 1;
  1437→  1437→  1437→  for (let i = atrLast; i >= 0 && atrHistory.length < 60; i--) {
  1438→  1438→  1438→    if (!isNaN(atrValues[i]) && atrValues[i] > 0) atrHistory.unshift(atrValues[i]);
  1439→  1439→  1439→  }
  1440→  1440→  1440→
  1441→  1441→  1441→  const p30 = atrHistory.length > 5 ? percentile(atrHistory, 30) : 0;
  1442→  1442→  1442→  const p70 = atrHistory.length > 5 ? percentile(atrHistory, 70) : Infinity;
  1443→  1443→  1443→
  1444→  1444→  1444→  const atrHigh = currentATR > p70;
  1445→  1445→  1445→  const atrLow = currentATR < p30;
  1446→  1446→  1446→
  1447→  1447→  1447→  const volatility: RiskOrVolatility = atrHigh ? 'HIGH' : atrLow ? 'LOW' : 'MEDIUM';
  1448→  1448→  1448→
  1449→  1449→  1449→  const riskLevel: RiskOrVolatility =
  1450→  1450→  1450→    atrHigh || confidence < 30 ? 'HIGH' : atrLow && confidence > 70 ? 'LOW' : 'MEDIUM';
  1451→  1451→  1451→
  1452→  1452→  1452→  // ---------- Top 5 key factors ----------
  1453→  1453→  1453→  const sortedFactors = Object.values(scores)
  1454→  1454→  1454→    .sort((a, b) => Math.abs(b.score) - Math.abs(a.score))
  1455→  1455→  1455→    .slice(0, 5);
  1456→  1456→  1456→
  1457→  1457→  1457→  const keyFactors: KeyFactor[] = sortedFactors.map((f) => ({
  1458→  1458→  1458→    name: f.name,
  1459→  1459→  1459→    signal: f.signal,
  1460→  1460→  1460→    impact: f.impact,
  1461→  1461→  1461→    score: Math.round(f.score * 100) / 100,
  1462→  1462→  1462→    description: f.description,
  1463→  1463→  1463→  }));
  1464→  1464→  1464→
  1465→  1465→  1465→  // ---------- Build result ----------
  1466→  1466→  1466→  return {
  1467→  1467→  1467→    symbol,
  1468→  1468→  1468→    score: finalScore,
  1469→  1469→  1469→    direction,
  1470→  1470→  1470→    confidence,
  1471→  1471→  1471→    shortTerm: {
  1472→  1472→  1472→      prediction: shortPrediction,
  1473→  1473→  1473→      probability: shortProbability,
  1474→  1474→  1474→      expectedMove: shortExpectedMove,
  1475→  1475→  1475→    },
  1476→  1476→  1476→    mediumTerm: {
  1477→  1477→  1477→      prediction: medPrediction,
  1478→  1478→  1478→      probability: medProbability,
  1479→  1479→  1479→      expectedMove: medExpectedMove,
  1480→  1480→  1480→    },
  1481→  1481→  1481→    riskLevel,
  1482→  1482→  1482→    volatility,
  1483→  1483→  1483→    keyFactors,
  1484→  1484→  1484→    indicatorScores,
  1485→  1485→  1485→    timestamp: new Date().toISOString(),
  1486→  1486→  1486→    technicalScore: finalScore,
  1487→  1487→  1487→    fundamentalData: null,
  1488→  1488→  1488→    learningData: null,
  1489→  1489→  1489→    combinedScore: finalScore,
  1490→  1490→  1490→  };
  1491→  1491→  1491→}

  1493→  1493→  1493→// ============= ENHANCED PREDICTION WITH FUNDAMENTALS + LEARNING =============
  1494→  1494→  1494→
  1495→  1495→  1495→interface EnhancedPredictionOptions {
  1496→  1496→  1496→  fundamentalScore?: number | null;
  1497→  1497→  1497→  fundamentalSummary?: string;
  1498→  1498→  1498→  fundamentalScores?: Record<string, number>;
  1499→  1499→  1499→  learningStats?: {
  1500→  1500→  1500→    totalPredictions: number;
  1501→  1501→  1501→    directionAccuracy: number;
  1502→  1502→  1502→    shortTermAccuracy: number;
  1503→  1503→  1503→    mediumTermAccuracy: number;
  1504→  1504→  1504→    bestIndicators: string[];
  1505→  1505→  1505→    worstIndicators: string[];
  1506→  1506→  1506→    recentAccuracy: number;
  1507→  1507→  1507→  } | null;
  1508→  1508→  1508→  /** Weight for fundamentals (0-1, default 0.25) */
  1509→  1509→  1509→  fundamentalWeight?: number;
  1510→  1510→  1510→}
  1511→  1511→  1511→
  1512→  1512→  1512→/**
  1513→  1513→  1513→ * Enhanced prediction that combines technical + fundamental scores.
  1514→  1514→  1514→ * Uses learning weights if available.
  1515→  1515→  1515→ */
  1516→  1516→  1516→export function predictStockEnhanced(
  1517→  1517→  1517→  baseResult: PredictionResult,
  1518→  1518→  1518→  options: EnhancedPredictionOptions = {},
  1519→  1519→  1519→): PredictionResult {
  1520→  1520→  1520→  const fundWeight = options.fundamentalWeight ?? 0.25;
  1521→  1521→  1521→  const techWeight = 1 - fundWeight;
  1522→  1522→  1522→  const techScore = baseResult.technicalScore;
  1523→  1523→  1523→  const fundScore = options.fundamentalScore ?? 0;
  1524→  1524→  1524→
  1525→  1525→  1525→  // Adjust fund weight based on learning data
  1526→  1526→  1526→  let adjustedFundWeight = fundWeight;
  1527→  1527→  1527→  if (options.learningStats && options.learningStats.totalPredictions > 10) {
  1528→  1528→  1528→    // If fundamentals have been more accurate, increase their weight
  1529→  1529→  1529→    const avgAcc = (options.learningStats.directionAccuracy + options.learningStats.recentAccuracy) / 2;
  1530→  1530→  1530→    if (avgAcc > 55) {
  1531→  1531→  1531→      adjustedFundWeight = Math.min(fundWeight * 1.3, 0.40);
  1532→  1532→  1532→    } else if (avgAcc < 45) {
  1533→  1533→  1533→      adjustedFundWeight = fundWeight * 0.7;
  1534→  1534→  1534→    }
  1535→  1535→  1535→  }
  1536→  1536→  1536→
  1537→  1537→  1537→  const adjustedTechWeight = 1 - adjustedFundWeight;
  1538→  1538→  1538→  const combinedScore = clamp(
  1539→  1539→  1539→    Math.round((techScore * adjustedTechWeight + fundScore * adjustedFundWeight) * 100) / 100,
  1540→  1540→  1540→    -100, 100
  1541→  1541→  1541→  );
  1542→  1542→  1542→
  1543→  1543→  1543→  // Recalculate direction based on combined score
  1544→  1544→  1544→  const direction: Direction =
  1545→  1545→  1545→    combinedScore > 60 ? 'STRONG_BUY' : combinedScore > 25 ? 'BUY' : combinedScore > -25 ? 'NEUTRAL' : combinedScore > -60 ? 'SELL' : 'STRONG_SELL';
  1546→  1546→  1546→
  1547→  1547→  1547→  // Add fundamental key factors if available
  1548→  1548→  1548→  const keyFactors = [...baseResult.keyFactors];
  1549→  1549→  1549→  if (options.fundamentalSummary) {
  1550→  1550→  1550→    // Insert fundamental as first factor if significant
  1551→  1551→  1551→    if (Math.abs(fundScore) > 15) {
  1552→  1552→  1552→      keyFactors.unshift({
  1553→  1553→  1553→        name: 'Analiza Fondamentale',
  1554→  1554→  1554→        signal: fundScore > 0 ? 'BULLISH' as SignalType : 'BEARISH' as SignalType,
  1555→  1555→  1555→        impact: Math.abs(fundScore) > 40 ? 'HIGH' as ImpactLevel : 'MEDIUM' as ImpactLevel,
  1556→  1556→  1556→        score: fundScore,
  1557→  1557→  1557→        description: options.fundamentalSummary,
  1558→  1558→  1558→      });
  1559→  1559→  1559→    }
  1560→  1560→  1560→  }
  1561→  1561→  1561→
  1562→  1562→  1562→  // Trim to top 5
  1563→  1563→  1563→  const topFactors = keyFactors
  1564→  1564→  1564→    .sort((a, b) => Math.abs(b.score) - Math.abs(a.score))
  1565→  1565→  1565→    .slice(0, 5);
  1566→  1566→  1566→
  1567→  1567→  1567→  // Boost confidence if both technical and fundamental agree
  1568→  1568→  1568→  let confidence = baseResult.confidence;
  1569→  1569→  1569→  if (techScore > 15 && fundScore > 15) {
  1570→  1570→  1570→    confidence = clamp(confidence + 8, 0, 100);
  1571→  1571→  1571→  } else if (techScore < -15 && fundScore < -15) {
  1572→  1572→  1572→    confidence = clamp(confidence + 8, 0, 100);
  1573→  1573→  1573→  } else if ((techScore > 15 && fundScore < -15) || (techScore < -15 && fundScore > 15)) {
  1574→  1574→  1574→    confidence = clamp(confidence - 10, 0, 100);
  1575→  1575→  1575→  }
  1576→  1576→  1576→  confidence = Math.round(confidence * 100) / 100;
  1577→  1577→  1577→
  1578→  1578→  1578→  // Update short/medium term predictions with fundamental influence
  1579→  1579→  1579→  const shortTerm = { ...baseResult.shortTerm };
  1580→  1580→  1580→  const mediumTerm = { ...baseResult.mediumTerm };
  1581→  1581→  1581→
  1582→  1582→  1582→  // Fundamental influence on medium term (stronger effect)
  1583→  1583→  1583→  if (Math.abs(fundScore) > 20) {
  1584→  1584→  1584→    const fundBias = fundScore > 0 ? 5 : -5;
  1585→  1585→  1585→    mediumTerm.probability = clamp(mediumTerm.probability + Math.abs(fundBias), 0, 100);
  1586→  1586→  1586→    if (fundScore > 40 && mediumTerm.prediction === 'SIDEWAYS') {
  1587→  1587→  1587→      mediumTerm.prediction = 'UP';
  1588→  1588→  1588→    } else if (fundScore < -40 && mediumTerm.prediction === 'SIDEWAYS') {
  1589→  1589→  1589→      mediumTerm.prediction = 'DOWN';
  1590→  1590→  1590→    }
  1591→  1591→  1591→  }
  1592→  1592→  1592→
  1593→  1593→  1593→  return {
  1594→  1594→  1594→    ...baseResult,
  1595→  1595→  1595→    score: combinedScore,
  1596→  1596→  1596→    direction,
  1597→  1597→  1597→    confidence,
  1598→  1598→  1598→    keyFactors: topFactors,
  1599→  1599→  1599→    fundamentalData: options.fundamentalScore !== null && options.fundamentalScore !== undefined
  1600→  1600→  1600→      ? {
  1601→  1601→  1601→          score: fundScore,
  1602→  1602→  1602→          summary: options.fundamentalSummary || '',
  1603→  1603→  1603→          scores: options.fundamentalScores || {},
  1604→  1604→  1604→        }
  1605→  1605→  1605→      : null,
  1606→  1606→  1606→    learningData: options.learningStats ?? null,
  1607→  1607→  1607→    combinedScore,
  1608→  1608→  1608→    shortTerm,
  1609→  1609→  1609→    mediumTerm,
  1610→  1610→  1610→  };
  1611→  1611→  1611→}
