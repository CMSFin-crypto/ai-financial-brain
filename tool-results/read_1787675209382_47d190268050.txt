   584→          reasons.push('Breakout 20d + volum');
   585→          if (rsi >= 45 && rsi <= 65) { sScore += 15; reasons.push(`RSI ${rsi.toFixed(0)}`); }
   586→        }
   587→      }
   588→
   589→      // Trend continuation
   590→      if (setup === 'NONE' && pbPct > -0.5 && pbPct < 2 && price > (sma50[last]||0)) {
   591→        setup = 'TREND_CONT'; sScore = 40;
   592→        reasons.push('Trend continuation');
   593→        if (rsi >= 50 && rsi <= 65) { sScore += 15; reasons.push(`RSI ${rsi.toFixed(0)}`); }
   594→        if (lastDaySpike) { sScore += 10; reasons.push('Volum konfirmim'); }
   595→      }
   596→
   597→      stock.setup = setup;
   598→      stock.setupScore = Math.min(100, sScore);
   599→      stock.rsi = Math.round(rsi * 10) / 10;
   600→      stock.atr = Math.round(atr * 100) / 100;
   601→      stock.atrPct = Math.round(atrPct * 100) / 100;
   602→      stock.reasons = reasons;
   603→
   604→      // Entry / Stop / Target (BLUEPRINT-ALIGNED)
   605→      const isBreakout = setup === 'BREAKOUT';
   606→      const high20 = Math.max(...highs.slice(-20));
   607→      const entry = isBreakout
   608→        ? Math.round((high20 * 1.002) * 100) / 100
   609→        : Math.round(price * 100) / 100;
   610→
   611→      // NEW: Stop = max(entry - 1.5*ATR, swingLow - 0.2*ATR) — use the tighter (higher) one
   612→      const stopAtr = Math.round((entry - atr * 1.5) * 100) / 100;
   613→      const stopSwing = Math.round((swLow - atr * 0.2) * 100) / 100;
   614→      const stop = Math.max(stopAtr, stopSwing); // tighter stop = higher price
   615→
   616→      const riskPerShare = entry - stop;
   617→      const riskPct = entry > 0 ? (riskPerShare / entry) * 100 : 0;
   618→      const target1R = Math.round((entry + riskPerShare) * 100) / 100;
   619→      const target2R = Math.round((entry + riskPerShare * 2) * 100) / 100;
   620→      const target3R = Math.round((entry + riskPerShare * 3) * 100) / 100;
   621→      const rr = riskPerShare > 0 ? (riskPerShare * 3) / riskPerShare : 0; // based on 3R now
   622→
   623→      stock.entry = entry;
   624→      stock.stop = stop;
   625→      stock.target1R = target1R;
   626→      stock.target2R = target2R;
   627→      stock.target3R = target3R;
   628→      stock.riskPct = Math.round(riskPct * 100) / 100;
   629→      stock.rewardRiskRatio = Math.round(rr * 10) / 10;
   630→
   631→      // NEW: Position sizing (apply catalyst multiplier)
   632→      const multiplier = stock.positionSizeMultiplier > 0 ? stock.positionSizeMultiplier : 1;
   633→      const pos = calcPositionSize(entry, stop, MAX_RISK_PCT * multiplier, ACCOUNT_EQUITY);
   634→      stock.positionSize = pos.shares;
   635→      stock.positionValue = pos.positionValue;
   636→      stock.riskDollars = pos.riskDollars;
   637→
   638→      // Horizon based on ATR
   639→      if (atrPct < 1.5) stock.horizon = '10D';
   640→      else if (atrPct < 3) stock.horizon = '5D';
   641→      else stock.horizon = '5D';
   642→
   643→      // ── F) Risk Quality (0-100) — 5% weight ──
   644→      let rScore = 50;
   645→      if (riskPct <= 3) rScore += 20;
   646→      else if (riskPct <= 5) rScore += 10;
   647→      else if (riskPct > 7) rScore -= 20;
   648→      if (rr >= 2) rScore += 15; // 2R+ is good
   649→      else if (rr >= 1.5) rScore += 5;
   650→      else rScore -= 15;
   651→      if (atrPct < 2) rScore += 10;
   652→      else if (atrPct > 4) rScore -= 10;
   653→      // NEW: Event risk penalty
   654→      if (!stock.passedEventRisk) rScore -= 25;
   655→      stock.riskScore = Math.round(Math.max(0, Math.min(100, rScore)));
   656→
   657→      // ── Total Score ──
   658→      stock.totalScore = Math.round(
   659→        stock.trendScore * 0.25 +
   660→        stock.rsScore * 0.20 +
   661→        stock.momentumScore * 0.15 +
   662→        stock.volConfScore * 0.15 +
   663→        sScore * 0.10 +
   664→        50 * 0.10 + // fundamentals placeholder (neutral)
   665→        stock.riskScore * 0.05
   666→      );
   667→
   668→      // Only keep stocks with valid setups
   669→      if (setup !== 'NONE') phase2.push(stock);
   670→    }
   671→
   672→    const passedSetup = phase2.length;
   673→
   674→    // ── 4. PHASE 3 — Risk gate + Decision + Event risk ──
   675→    const phase3: FunnelStock[] = [];
   676→
   677→    for (const stock of phase2) {
   678→      const warnings: string[] = [...stock.warnings];
   679→      let decision: Decision = 'NO_TRADE';
   680→
   681→      // R:R gate (based on 3R now, so 2.0+ is the threshold)
   682→      if (stock.rewardRiskRatio < 2.0) {
   683→        warnings.push(`R:R ${stock.rewardRiskRatio} — i ulet (duhet 1:2 me 3R target)`);
   684→      }
   685→
   686→      // RSI gate
   687→      if (stock.rsi > 75) {
   688→        warnings.push(`RSI ${stock.rsi} — i mbivleresuar`);
   689→      }
   690→
   691→      // Risk % gate
   692→      if (stock.riskPct > 8) {
   693→        warnings.push(`Rreziku ${stock.riskPct}% — shume i larte`);
   694→      }
   695→
   696→      // ADX gate
   697→      if (stock.adx < 20) {
   698→        warnings.push(`ADX ${stock.adx} — trendi i dobet (duhet > 25)`);
   699→      }
   700→
   701→      // Regime gate
   702→      if (!regimeOk) {
   703→        warnings.push('REGJIMI jo OK — vetem watcher');
   704→      }
   705→
   706→      // NEW: Event risk gate (enhanced with catalyst status)
   707→      if (!stock.passedEventRisk) {
   708→        warnings.push(`EVENT RISK: ${stock.eventRisk}`);
   709→      }
   710→      if (!stock.allowNewEntry) {
   711→        warnings.push(`CATALYST GATE: ${stock.catalystStatus} — mos hap long të ri`);
   712→      } else if (stock.positionSizeMultiplier < 1) {
   713→        warnings.push(`CATALYST GATE: ${stock.catalystStatus} — pozicion ${Math.round(stock.positionSizeMultiplier * 100)}%`);
   714→      }
   715→
   716→      // Stacked MA bonus/penalty
   717→      if (!stock.stackedMA) {
   718→        warnings.push('MA jo te stackuara (close/EMA20/SMA50/SMA200)');
   719→      }
   720→
   721→      // Sector RS gate
   722→      if (stock.sectorRsStatus === 'LAGGING' && !stock.sectorAboveSma50) {
   723→        warnings.push(`SECTOR RS: ${stock.rsVsSector20d}% vs ${stock.sectorEtf} — sektori i dobet`);
   724→      }
   725→
   726→      // Extended (chasing)
   727→      if (stock.pullbackPct > -0.5 && stock.setup === 'PULLBACK') {
   728→        warnings.push('Jo te vertete nje pullback — extended');
   729→      }
   730→
   731→      // Determine decision (catalyst gate + sector RS affects READY)
   732→      const hasRR = stock.rewardRiskRatio >= 2.0;
   733→      const rsiOk = stock.rsi >= 30 && stock.rsi <= 75;
   734→      const riskOk = stock.riskPct <= 8;
   735→      const scoreOk = stock.totalScore >= 45;
   736→      const eventOk = stock.passedEventRisk;
   737→      const catalystOk = stock.allowNewEntry;
   738→      const sectorRsOk = !(stock.sectorRsStatus === 'LAGGING' && !stock.sectorAboveSma50);
   739→
   740→      if (hasRR && rsiOk && riskOk && scoreOk && regimeOk && eventOk && catalystOk && sectorRsOk) {
   741→        decision = 'READY';
   742→      } else if (hasRR && rsiOk && riskOk && scoreOk && (eventOk || !regimeOk)) {
   743→        decision = 'WATCHLIST';
   744→        if (!regimeOk) warnings.push('WATCHLIST: Regjimi nuk lejon long tani');
   745→        if (!eventOk) warnings.push('WATCHLIST: Event risk — prit');
   746→        if (!catalystOk) warnings.push('WATCHLIST: Catalyst gate — prit');
   747→        if (!sectorRsOk) warnings.push('WATCHLIST: Sector RS i dobet — prit');
   748→      } else if (!hasRR || stock.riskPct > 6) {
   749→        if (stock.setupScore >= 50) decision = 'WATCHLIST';
   750→        else decision = 'NO_TRADE';
   751→      } else if (stock.rsi > 72 || stock.pullbackPct > -0.3) {
   752→        decision = 'EXTENDED';
   753→      } else {
   754→        decision = 'WATCHLIST';
   755→      }
   756→
   757→      // Event risk override
   758→      if (!eventOk && decision === 'READY') {
   759→        decision = 'EVENT_RISK';
   760→        warnings.push('EVENT_RISK: Ngjarje kritike — mos hyr');
   761→      }
   762→
   763→      // RSI overbought override
   764→      if (stock.rsi > 70 && stock.totalScore >= 55) {
   765→        decision = 'EVENT_RISK';
   766→        warnings.push('EVENT_RISK: RSI i larte, rrezik kthimi');
   767→      }
   768→
   769→      stock.decision = decision;
   770→      stock.warnings = warnings;
   771→
   772→      // NEW: Generate bracket order for READY trades
   773→      if (decision === 'READY' && stock.positionSize > 0) {
   774→        stock.bracketOrder = generateBracketOrder(
   775→          stock.symbol, 'BUY', stock.entry, stock.stop, stock.target3R, stock.positionSize
   776→        );
   777→      }
   778→
   779→      if (decision === 'READY' || decision === 'WATCHLIST' || decision === 'EVENT_RISK') {
   780→        phase3.push(stock);
   781→      }
   782→    }
   783→
   784→    const passedRisk = phase3.length;
   785→
   786→    // ── 5. PHASE 4 — Sector exposure limit + Top 10 final ──
   787→    const sectorCount: Record<string, number> = {};
   788→    const afterSectorLimit: FunnelStock[] = [];
   789→
   790→    // Sort by totalScore descending first
   791→    phase3.sort((a, b) => b.totalScore - a.totalScore);
   792→
   793→    for (const stock of phase3) {
   794→      const sec = stock.sector;
   795→      if (!sectorCount[sec]) sectorCount[sec] = 0;
   796→      if (sectorCount[sec] < MAX_PER_SECTOR) {
   797→        afterSectorLimit.push(stock);
   798→        sectorCount[sec]++;
   799→      } else {
   800→        stock.warnings.push(`Sector limit: ${sec} ka tashme ${MAX_PER_SECTOR} aksione`);
   801→      }
   802→    }
   803→
   804→    const passedSectorLimit = afterSectorLimit.length;
   805→    const topStocks = afterSectorLimit.slice(0, 10);
   806→    // READY first, then WATCHLIST, then others
   807→    topStocks.sort((a, b) => {
   808→      const order: Record<Decision, number> = { READY: 0, WATCHLIST: 1, EVENT_RISK: 2, EXTENDED: 3, NO_TRADE: 4 };
   809→      const diff = (order[a.decision] || 5) - (order[b.decision] || 5);
   810→      return diff !== 0 ? diff : b.totalScore - a.totalScore;
   811→    });
   812→
   813→    // Build sector exposure summary
   814→    const sectorExposure: Record<string, number> = {};
   815→    for (const s of topStocks) {
   816→      if (!sectorExposure[s.sector]) sectorExposure[s.sector] = 0;
   817→      sectorExposure[s.sector]++;
   818→    }
   819→
   820→    const elapsed = ((Date.now() - t0) / 1000).toFixed(1);
   821→    console.log(`[IBKR v2] ${syms.length} → ${passedLiquidity} → ${passedTrend} → ${passedSetup} → ${passedRisk} → ${passedSectorLimit} → ${topStocks.length} (${elapsed}s)`);
   822→
   823→    return NextResponse.json({
   824→      scannedAt: new Date().toISOString(),
   825→      regimeOk,
   826→      regimeDetail: {
   827→        spy: { above50: spyA50, above200: spyA200 },
   828→        qqq: { above50: qqqA50, above200: qqqA200 },
   829→      },
   830→      funnel: {
   831→        universe: syms.length,
   832→        passedLiquidity,
   833→        passedTrend,
   834→        passedSetup,
   835→        passedRisk,
   836→        passedEventRisk: passedEventRiskCount,
   837→        passedSectorLimit,
   838→        displayed: topStocks.length,
   839→      },
   840→      results: topStocks,
   841→      sectorExposure,
   842→    } satisfies FunnelResponse);
   843→  } catch (err: any) {
   844→    console.error('[IBKR v2] Error:', err);
   845→    return NextResponse.json({ error: err?.message || 'Gabim skaneri' }, { status: 500 });
   846→  }
   847→}
