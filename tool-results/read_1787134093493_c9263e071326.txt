    64→    64→async function fetchFromYahooFinance(ticker: string, endpointIndex = 0): Promise<LivePrice | null> {
    65→    65→  const base = YAHOO_ENDPOINTS[endpointIndex] || YAHOO_ENDPOINTS[0];

    67→    67→    const url = `${base}/v8/finance/chart/${ticker}?range=1d&interval=1d&includePrePost=false`;
    68→    68→    const res = await fetch(url, {
    69→    69→      signal: AbortSignal.timeout(8000), // 8 second timeout
    70→    70→      headers: BROWSER_HEADERS,
    71→    71→    });
    72→    72→
    73→    73→    if (!res.ok) {
    74→    74→      console.log(`[PRICE] ${ticker}: Yahoo ${base} returned ${res.status}`);
    75→    75→      return null;
    76→    76→    }
    77→    77→
    78→    78→    const data = await res.json();
    79→    79→    const result = data?.chart?.result?.[0];
    80→    80→    if (!result) {
    81→    81→      console.log(`[PRICE] ${ticker}: Yahoo ${base} returned no chart data`);
    82→    82→      return null;
    83→    83→    }
    84→    84→
    85→    85→    const meta = result.meta;
    86→    86→    const price = meta?.regularMarketPrice;
    87→    87→    const prevClose = meta?.chartPreviousClose || meta?.previousClose;
    88→    88→    const volume = meta?.regularMarketVolume || 0;
    89→    89→
    90→    90→    if (!price || price <= 0) {
    91→    91→      console.log(`[PRICE] ${ticker}: Yahoo ${base} returned invalid price: ${price}`);
    92→    92→      return null;
    93→    93→    }
    94→    94→
    95→    95→    const change = prevClose ? ((price - prevClose) / prevClose) * 100 : 0;
    96→    96→
    97→    97→    console.log(`[PRICE] ${ticker}: $${price.toFixed(2)} (${change >= 0 ? '+' : ''}${change.toFixed(2)}%) [Yahoo ${endpointIndex + 1}]`);
    98→    98→    return {
    99→    99→      price: parseFloat(price.toFixed(2)),
   100→   100→      change: parseFloat(change.toFixed(2)),
   101→   101→      previousClose: prevClose ? parseFloat(prevClose.toFixed(2)) : price,
   102→   102→      volume,
   103→   103→      source: 'yahoo_finance',
   104→   104→      timestamp: new Date().toISOString(),
   105→   105→    };
   106→   106→  } catch (err: any) {
   107→   107→    const msg = err?.name === 'TimeoutError' ? 'timeout' : err?.message || 'unknown';
   108→   108→    console.log(`[PRICE] ${ticker}: Yahoo ${base} failed: ${msg}`);
   109→   109→    return null;
   110→   110→  }

   161→   161→function chunkArray<T>(arr: T[], size: number): T[][] {
   162→   162→  const chunks: T[][] = [];
   163→   163→  for (let i = 0; i < arr.length; i += size) {
   164→   164→    chunks.push(arr.slice(i, i + size));
   165→   165→  }
   166→   166→  return chunks;
   167→   167→}

   169→   169→function delay(ms: number): Promise<void> {
   170→   170→  return new Promise(resolve => setTimeout(resolve, ms));

   179→   179→export async function getRealPrice(
   180→   180→  ticker: string,
   181→   181→  options: FetchOptions = {},
   182→   182→): Promise<LivePrice | null> {
   183→   183→  const t = ticker.toUpperCase().trim();
   184→   184→
   185→   185→  // Check cache first
   186→   186→  const cached = priceCache.get(t);
   187→   187→  if (!options.forceRefresh && cached && Date.now() - cached.fetchedAt < CACHE_TTL_MS) {
   188→   188→    return cached.data;
   189→   189→  }
   190→   190→
   191→   191→  // Try Yahoo Finance (with retry on secondary endpoint)
   192→   192→  const yfPrice = await fetchFromYahooWithRetry(t);
   193→   193→  if (yfPrice) {
   194→   194→    priceCache.set(t, { data: yfPrice, fetchedAt: Date.now() });
   195→   195→    return yfPrice;
   196→   196→  }
   197→   197→
   198→   198→  // Try Alpha Vantage as fallback (if key is configured)
   199→   199→  const avPrice = await fetchFromAlphaVantage(t);
   200→   200→  if (avPrice) {
   201→   201→    priceCache.set(t, { data: avPrice, fetchedAt: Date.now() });
   202→   202→    return avPrice;
   203→   203→  }
   204→   204→
   205→   205→  console.warn(`[PRICE] Could not fetch real price for ${ticker} from any source`);
   206→   206→  return null;
   207→   207→}

   209→   209→/**
   210→   210→ * Fetch real-time prices for multiple tickers (batched).
   211→   211→ * Processes in chunks of 6 with 800ms delay between chunks to avoid rate limiting.
   212→   212→ * Returns a map of ticker -> LivePrice (only successfully fetched).
   213→   213→ */
   214→   214→export async function getRealPrices(
   215→   215→  tickers: string[],
   216→   216→  options: FetchOptions = {},
   217→   217→): Promise<Record<string, LivePrice>> {
   218→   218→  const prices: Record<string, LivePrice> = {};
   219→   219→  const toFetch: string[] = [];
   220→   220→
   221→   221→  // First, check which tickers have valid cached data
   222→   222→  for (const ticker of tickers) {
   223→   223→    const t = ticker.toUpperCase().trim();
   224→   224→    const cached = priceCache.get(t);
   225→   225→    if (!options.forceRefresh && cached && Date.now() - cached.fetchedAt < CACHE_TTL_MS) {
   226→   226→      prices[t] = cached.data;
   227→   227→    } else {
   228→   228→      toFetch.push(t);
   229→   229→    }
   230→   230→  }
   231→   231→
   232→   232→  if (toFetch.length === 0) {
   233→   233→    console.log(`[PRICE] All ${tickers.length} prices from cache`);
   234→   234→    return prices;
   235→   235→  }
   236→   236→
   237→   237→  console.log(`[PRICE] Cache hits: ${Object.keys(prices).length}, fetching: ${toFetch.length} tickers...`);
   238→   238→
   239→   239→  // Process in batches of 6 with delay between batches
   240→   240→  const BATCH_SIZE = 6;
   241→   241→  const BATCH_DELAY_MS = 800;
   242→   242→  const chunks = chunkArray(toFetch, BATCH_SIZE);
   243→   243→
   244→   244→  for (let i = 0; i < chunks.length; i++) {
   245→   245→    const chunk = chunks[i];
   246→   246→
   247→   247→    // Fetch all tickers in this chunk in parallel
   248→   248→    const results = await Promise.allSettled(
   249→   249→      chunk.map(t => getRealPrice(t, options))
   250→   250→    );
   251→   251→
   252→   252→    // Collect results
   253→   253→    chunk.forEach((ticker, j) => {
   254→   254→      const result = results[j];
   255→   255→      if (result.status === 'fulfilled' && result.value) {
   256→   256→        prices[ticker] = result.value;
   257→   257→      }
   258→   258→    });
   259→   259→
   260→   260→    // Delay between batches (except after last batch)
   261→   261→    if (i < chunks.length - 1) {
   262→   262→      await delay(BATCH_DELAY_MS);
   263→   263→    }
   264→   264→  }
   265→   265→
   266→   266→  const successCount = Object.keys(prices).length;
   267→   267→  const failCount = toFetch.length - (successCount - tickers.length + toFetch.length);
   268→   268→  console.log(`[PRICE] Batch complete: ${successCount}/${tickers.length} prices obtained`);
   269→   269→
   270→   270→  return prices;
   271→   271→}

   289→   289→export async function fetchHistoricalData(

   293→   293→): Promise<HistoricalDataPoint[] | null> {

   295→   295→  const r = range || '6mo';
   296→   296→
   297→   297→  // Determine interval based on range
   298→   298→  const interval = r === '1d' ? '5m' : '1d';
   299→   299→
   300→   300→  // Check cache first (include range in cache key)
   301→   301→  const cacheKey = `${t}_${r}`;
   302→   302→  const cached = chartCache.get(cacheKey);
   303→   303→  if (!options.forceRefresh && cached && Date.now() - cached.fetchedAt < CHART_CACHE_TTL_MS) {
   304→   304→    return cached.data;
   305→   305→  }
   306→   306→
   307→   307→  // Try Yahoo Finance v8 chart API with specified range
   308→   308→  const endpoints = ['https://query1.finance.yahoo.com', 'https://query2.finance.yahoo.com'];
   309→   309→
   310→   310→  for (const base of endpoints) {
   311→   311→    try {
   312→   312→      const url = `${base}/v8/finance/chart/${t}?range=${r}&interval=${interval}&includePrePost=false`;
   313→   313→      const res = await fetch(url, {
   314→   314→        signal: AbortSignal.timeout(10000),
   315→   315→        headers: BROWSER_HEADERS,
   316→   316→      });
   317→   317→
   318→   318→      if (!res.ok) continue;
   319→   319→
   320→   320→      const data = await res.json();
   321→   321→      const result = data?.chart?.result?.[0];
   322→   322→      if (!result) continue;
   323→   323→
   324→   324→      const timestamps: number[] = result.timestamp || [];

   326→   326→      if (!quote || !timestamps.length) continue;
   327→   327→
   328→   328→      const points: HistoricalDataPoint[] = [];
   329→   329→      for (let i = 0; i < timestamps.length; i++) {
   330→   330→        const open = quote.open?.[i];
   331→   331→        const high = quote.high?.[i];
   332→   332→        const low = quote.low?.[i];
   333→   333→        const close = quote.close?.[i];

   343→   343→            high: +high.toFixed(2),
   344→   344→            low: +low.toFixed(2),

   355→   355→      }
   356→   356→    } catch (err: any) {
   357→   357→      const msg = err?.name === 'TimeoutError' ? 'timeout' : err?.message || 'unknown';

   359→   359→      continue;
   360→   360→    }
   361→   361→  }

   420→   420→async function fetchQuoteSummary(ticker: string, endpointIndex = 0): Promise<YahooFundamentals | null> {
   421→   421→  const base = YAHOO_ENDPOINTS[endpointIndex] || YAHOO_ENDPOINTS[0];
   422→   422→  const modules = 'financialData,defaultKeyStatistics,earningsTrend,earningsHistory';
   423→   423→  try {
   424→   424→    const url = `${base}/v10/finance/quoteSummary/${ticker}?modules=${modules}`;
   425→   425→    const res = await fetch(url, {
   426→   426→      signal: AbortSignal.timeout(10000),
   427→   427→      headers: BROWSER_HEADERS,
   428→   428→    });
   429→   429→
   430→   430→    if (!res.ok) {
   431→   431→      console.log(`[FUND] ${ticker}: quoteSummary ${base} returned ${res.status}`);
   432→   432→      return null;
   433→   433→    }
   434→   434→
   435→   435→    const data = await res.json();
   436→   436→    const body = data?.quoteSummary?.result?.[0];
   437→   437→    if (!body) {
   438→   438→      console.log(`[FUND] ${ticker}: No quoteSummary result from ${base}`);
   439→   439→      return null;
   440→   440→    }
   441→   441→
   442→   442→    const fd = body.financialData || {};
   443→   443→    const dks = body.defaultKeyStatistics || {};
   444→   444→    const et = body.earningsTrend || {};
   445→   445→    const eh = body.earningsHistory || {};
   446→   446→
   447→   447→    // Extract earnings trend for forward EPS
   448→   448→    const trendData = et.trend || [];
   449→   449→    let epsForward = extractNum(fd.forwardEps);
   450→   450→    if (!epsForward && trendData.length > 0) {
   451→   451→      epsForward = extractNum(trendData[0]?.earningsEstimate?.avgForecast);
   452→   452→    }
   453→   453→
   454→   454→    // Extract quarterly growth from earnings history
   455→   455→    const historyData = eh.history || [];
   456→   456→    let earningsQuarterlyGrowth = extractNum(fd.earningsQuarterlyGrowth);
   457→   457→    let revenueQuarterlyGrowth = extractNum(fd.revenueQuarterlyGrowth);
   458→   458→    if (!earningsQuarterlyGrowth && historyData.length > 0) {
   459→   459→      const latest = historyData[0];
   460→   460→      const epsSurprise = extractNum(latest?.epsSurprisePercent);
   461→   461→      if (epsSurprise) earningsQuarterlyGrowth = epsSurprise;
   462→   462→    }

   549→   549→/**
   550→   550→ * Fetch real fundamental data for multiple tickers (batched).
   551→   551→ * Returns a map of ticker -> YahooFundamentals (only successfully fetched).
   552→   552→ * Falls back to local JSON database when Yahoo v10/quoteSummary returns 401.
   553→   553→ */
   554→   554→export async function getRealFundamentalsBatch(tickers: string[]): Promise<Record<string, YahooFundamentals>> {
   555→   555→  const results: Record<string, YahooFundamentals> = {};
   556→   556→  const toFetch: string[] = [];
   557→   557→
   558→   558→  for (const ticker of tickers) {
   559→   559→    const t = ticker.toUpperCase().trim();
   560→   560→    const cached = fundCache.get(t);
   561→   561→    if (cached && Date.now() - cached.fetchedAt < FUND_CACHE_TTL_MS) {
   562→   562→      results[t] = cached.data;
   563→   563→    } else {
   564→   564→      toFetch.push(t);
   565→   565→    }
   566→   566→  }
   567→   567→
   568→   568→  if (toFetch.length === 0) {
   569→   569→    console.log(`[FUND] All ${tickers.length} fundamentals from cache`);
   570→   570→    return results;
   571→   571→  }
   572→   572→
   573→   573→  console.log(`[FUND] Cache hits: ${Object.keys(results).length}, fetching: ${toFetch.length} fundamentals...`);
   574→   574→
   575→   575→  // Try Yahoo v10/quoteSummary first (may return 401 now)
   576→   576→  const BATCH_SIZE = 3;
   577→   577→  const BATCH_DELAY_MS = 1200;
   578→   578→  const chunks = chunkArray(toFetch, BATCH_SIZE);
   579→   579→
   580→   580→  for (let i = 0; i < chunks.length; i++) {
   581→   581→    const chunk = chunks[i];
   582→   582→    const fetchResults = await Promise.allSettled(
   583→   583→      chunk.map(t => getRealFundamentals(t))
   584→   584→    );
   585→   585→
   586→   586→    chunk.forEach((ticker, j) => {
   587→   587→      const result = fetchResults[j];
   588→   588→      if (result.status === 'fulfilled' && result.value) {
   589→   589→        results[ticker] = result.value;
   590→   590→      }
   591→   591→    });
   592→   592→
   593→   593→    if (i < chunks.length - 1) {
   594→   594→      await delay(BATCH_DELAY_MS);
   595→   595→    }
   596→   596→  }
   597→   597→
   598→   598→  // Fall back to local JSON for any stocks that failed from Yahoo
   599→   599→  const missing = toFetch.filter(t => !results[t]);
   600→   600→  if (missing.length > 0) {
   601→   601→    console.log(`[FUND] ${missing.length} stocks missing from Yahoo, falling back to local JSON...`);
   602→   602→    const localData = loadLocalFundamentals();
   603→   603→    for (const t of missing) {
   604→   604→      const d = localData[t];
   605→   605→      if (!d) continue;
   606→   606→      const fund = localToYahooFundamentals(t, d);
   607→   607→      if (!fund) continue;
   608→   608→      results[t] = fund;
   609→   609→      fundCache.set(t, { data: fund, fetchedAt: Date.now() });
   610→   610→    }
   611→   611→  }
   612→   612→
   613→   613→  console.log(`[FUND] Batch complete: ${Object.keys(results).length}/${tickers.length} fundamentals obtained`);
   614→   614→  return results;
   615→   615→}

   626→   626→// Helper to parse percentage strings like "85.2%" to decimals like 0.852
   627→   627→function parsePercent(val: unknown): number {
   628→   628→  if (val === null || val === undefined || val === '') return 0;
   629→   629→  const str = String(val).trim();
   630→   630→  if (str === 'N/A') return 0;
   631→   631→  const num = parseFloat(str.replace('%', '').replace(',', ''));
   632→   632→  return isNaN(num) ? 0 : num / 100;
   633→   633→}
   634→   634→
   635→   635→// Helper to parse numeric strings like "$298.42" or "22.42" to numbers
   636→   636→function parseNum(val: unknown): number {
   637→   637→  if (val === null || val === undefined || val === '') return 0;
   638→   638→  if (typeof val === 'number') return isNaN(val) ? 0 : val;
   639→   639→  const str = String(val).replace(/[^0-9.\-]/g, '');
   640→   640→  const num = parseFloat(str);
   641→   641→  return isNaN(num) ? 0 : num;
   642→   642→}

   656→   656→// Convert local JSON data to YahooFundamentals interface format
   657→   657→function localToYahooFundamentals(sym: string, d: Record<string, any>): YahooFundamentals | null {
   658→   658→  const price = typeof d.price === 'number' ? d.price : parseFloat(String(d.price));
   659→   659→  if (!price || price <= 0) return null;
   660→   660→
   661→   661→  const targetMean = parseNum(d.targetPrice);
   662→   662→  const targetHigh = parseNum(d.highTarget);
   663→   663→  const targetLow = parseNum(d.lowTarget);
   664→   664→
   665→   665→  const result: YahooFundamentals = {
   666→   666→    currentPrice: price,
   667→   667→    previousClose: price - (typeof d.change === 'number' ? d.change : 0),
   668→   668→    trailingPE: parseNum(d.pe),
   669→   669→    forwardPE: parseNum(d.fwdPE),
   670→   670→    pegRatio: parseNum(d.peg),
   671→   671→    priceToBook: parseNum(d.pb),
   672→   672→    enterpriseToEbitda: parseNum(d.evEbitda),
   673→   673→    grossMargins: parsePercent(d.grossMargin),
   674→   674→    operatingMargins: parsePercent(d.opMargin),
   675→   675→    profitMargins: parsePercent(d.netMargin),
   676→   676→    revenueGrowth: parsePercent(d.revGrowth),
   677→   677→    earningsGrowth: parsePercent(d.epsGrowth),
   678→   678→    revenueQuarterlyGrowth: parsePercent(d.qRevGrowth),
   679→   679→    earningsQuarterlyGrowth: parsePercent(d.qEpsGrowth),
   680→   680→    targetMeanPrice: targetMean,
   681→   681→    targetHighPrice: targetHigh,
   682→   682→    targetLowPrice: targetLow,
   683→   683→    recommendationKey: d.recommendation ? String(d.recommendation) : '',
   684→   684→    numberOfAnalystOpinions: parseNum(d.analysts),
   685→   685→    totalRevenue: 0, // not in local JSON
   686→   686→    ebitda: 0,     // not in local JSON
   687→   687→    totalDebt: 0,   // not in local JSON
   688→   688→    totalCash: 0,   // not in local JSON
   689→   689→    debtToEquity: parseNum(d.debtEq),
   690→   690→    returnOnEquity: parsePercent(d.roe),
   691→   691→    freeCashflow: 0, // not in local JSON
   692→   692→    epsForward: parseNum(d.fwdEps),
   693→   693→    source: 'local_json',
   694→   694→    fetchedAt: new Date().toISOString(),
   695→   695→  };
   696→   696→
   697→   697→  return result;

   702→   702→  options: FetchOptions = {},

   704→   704→  // Check cache first

   730→   730→    }

   732→   732→    // Also store in individual fund cache

   734→   734→  }
   735→   735→
   736→   736→  // Cache the batch result
   737→   737→  batchQuoteCache.set(cacheKey, { data: results, fetchedAt: Date.now() });
   738→   738→  console.log(`[BATCH-QUOTE] Total: ${Object.keys(results).length}/${tickers.length} fundamentals obtained (local JSON)`);
   739→   739→
   740→   740→  return results;
