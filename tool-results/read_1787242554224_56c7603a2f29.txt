   190→  metricKey: string;

   192→}) {
   193→  const info = METRIC_INFO[metricKey];
   194→  const popupRef = useRef<HTMLDivElement>(null);
   195→
   196→  useEffect(() => {
   197→    const handleClick = (e: MouseEvent) => {
   198→      if (popupRef.current && !popupRef.current.contains(e.target as Node)) {
   199→        onClose();
   200→      }
   201→    };
   202→    const handleEsc = (e: KeyboardEvent) => {
   203→      if (e.key === 'Escape') onClose();

   205→    setTimeout(() => {
   206→      document.addEventListener('mousedown', handleClick);
   207→      document.addEventListener('keydown', handleEsc);
   208→    }, 10);

   210→      document.removeEventListener('mousedown', handleClick);
   211→      document.removeEventListener('keydown', handleEsc);
   212→    };

   217→  return (

   327→// Simple seeded pseudo-random for consistent sparklines
   328→function seededRandom(seed: number): () => number {
   329→  let s = seed;
   330→  return () => {
   331→    s = (s * 16807 + 0) % 2147483647;
   332→    return (s - 1) / 2147483646;
   333→  };
   334→}

   336→function generateSparklineData(priceChange: number, ticker: string): Array<{ value: number }> {
   337→  const points = 12;
   338→  const data: Array<{ value: number }> = [];
   339→  const magnitude = Math.min(Math.abs(priceChange), 10);
   340→  // Create a seed from ticker hash + priceChange for consistency
   341→  const hash = ticker.split('').reduce((acc, c) => acc + c.charCodeAt(0), 0);
   342→  const rand = seededRandom(hash * 1000 + Math.round(priceChange * 100));
   343→  for (let i = 0; i < points; i++) {
   344→    const progress = i / (points - 1);
   345→    const trend = priceChange >= 0 ? progress * magnitude : -progress * magnitude;
   346→    const noise = (rand() - 0.5) * magnitude * 0.4;
   347→    data.push({ value: 50 + trend + noise });
   348→  }
   349→  return data;
   350→}
