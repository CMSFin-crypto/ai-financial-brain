    89→// ═══ Technical indicator computations ═══
    90→function computeSMA(closes: number[], period: number): (number | null)[] {
    91→  const result: (number | null)[] = [];
    92→  for (let i = 0; i < closes.length; i++) {
    93→    if (i < period - 1) { result.push(null); continue; }
    94→    let sum = 0;
    95→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
    96→    result.push(sum / period);
    97→  }
    98→  return result;
    99→}
   100→
   101→function computeEMA(closes: number[], period: number): (number | null)[] {
   102→  const k = 2 / (period + 1);
   103→  const result: (number | null)[] = [];
   104→  let ema: number | null = null;
   105→  for (let i = 0; i < closes.length; i++) {
   106→    if (i < period - 1) { result.push(null); continue; }
   107→    if (ema === null) {
   108→      let sum = 0; for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   109→      ema = sum / period;
   110→    } else {
   111→      ema = closes[i] * k + ema * (1 - k);
   112→    }
   113→    result.push(ema);
   114→  }
   115→  return result;
   116→}

   118→function computeRSI(closes: number[], period: number = 14): (number | null)[] {
   119→  const result: (number | null)[] = [];
   120→  const gains: number[] = [];
   121→  const losses: number[] = [];
   122→  for (let i = 1; i < closes.length; i++) {
   123→    const diff = closes[i] - closes[i - 1];
   124→    gains.push(diff > 0 ? diff : 0);
   125→    losses.push(diff < 0 ? -diff : 0);
   126→  }
   127→  let avgGain = 0, avgLoss = 0;
   128→  for (let i = 0; i < Math.min(period, gains.length); i++) { avgGain += gains[i]; avgLoss += losses[i]; }
   129→  avgGain /= period; avgLoss /= period;
   130→  for (let i = 0; i < period - 1; i++) result.push(null);
   131→  const rs = avgLoss === 0 ? 100 : avgGain / avgLoss;
   132→  result.push(100 - 100 / (1 + rs));
   133→  for (let i = period; i < gains.length; i++) {
   134→    avgGain = (avgGain * (period - 1) + gains[i]) / period;
   135→    avgLoss = (avgLoss * (period - 1) + losses[i]) / period;
   136→    const rs2 = avgLoss === 0 ? 100 : avgGain / avgLoss;
   137→    result.push(100 - 100 / (1 + rs2));
   138→  }
   139→  return result;
   140→}

   142→function computeMACD(closes: number[], fastPeriod = 12, slowPeriod = 26, signalPeriod = 9): { macd: (number | null)[]; signal: (number | null)[]; histogram: (number | null)[] } {
   143→  const emaFast = computeEMA(closes, fastPeriod);
   144→  const emaSlow = computeEMA(closes, slowPeriod);
   145→  const macdLine: (number | null)[] = [];
   146→  for (let i = 0; i < closes.length; i++) {
   147→    if (emaFast[i] === null || emaSlow[i] === null) { macdLine.push(null); continue; }
   148→    macdLine.push(emaFast[i]! - emaSlow[i]!);
   149→  }
   150→  // Signal = EMA of MACD line
   151→  const validMACD = macdLine.filter((v): v is number => v !== null);
   152→  const sigPeriod = Math.min(signalPeriod, Math.max(3, Math.floor(validMACD.length * 0.4)));
   153→  const signalEma = computeEMA(validMACD, sigPeriod);
   154→  const signal: (number | null)[] = [];
   155→  let vi = 0;
   156→  for (let i = 0; i < closes.length; i++) {
   157→    if (macdLine[i] === null) { signal.push(null); continue; }
   158→    signal.push(signalEma[vi] ?? null);
   159→    vi++;
   160→  }
   161→  const histogram: (number | null)[] = [];
   162→  for (let i = 0; i < closes.length; i++) {
   163→    if (macdLine[i] === null || signal[i] === null) { histogram.push(null); continue; }
   164→    histogram.push(macdLine[i]! - signal[i]!);
   165→  }
   166→  return { macd: macdLine, signal, histogram };
   167→}

   169→function computeBollingerBands(closes: number[], period: number = 20, mult: number = 2): { upper: (number | null)[]; middle: (number | null)[]; lower: (number | null)[] } {
   170→  const middle = computeSMA(closes, period);
   171→  const upper: (number | null)[] = [];
   172→  const lower: (number | null)[] = [];
   173→  for (let i = 0; i < closes.length; i++) {
   174→    if (middle[i] === null) { upper.push(null); lower.push(null); continue; }
   175→    let sumSq = 0;
   176→    for (let j = i - period + 1; j <= i; j++) sumSq += (closes[j] - middle[i]!) ** 2;
   177→    const std = Math.sqrt(sumSq / period);
   178→    upper.push(middle[i]! + mult * std);
   179→    lower.push(middle[i]! - mult * std);
   180→  }
   181→  return { upper, middle, lower };
   182→}

   184→// ═══ TradingView-Style Technical Chart ═══
   185→function CandlestickChart({ data }: { data: CandleData[] }) {
   186→  const chart = useMemo(() => {
   187→    if (!data || data.length < 3) return null;
   188→
   189→    // TradingView dark palette (matching reference)
   190→    const BG = '#131722';
   191→    const GRID = '#2a2e39';
   192→    const GRID_OP = 0.3;
   193→    const TXT = '#787b86';
   194→    const TXT_BRIGHT = '#d1d4dc';
   195→    const BULL = '#26a69a';
   196→    const BEAR = '#ef5350';
   197→    const SMA20_CLR = '#f0b323'; // Gold/Yellow — fast MA
   198→    const SMA50_CLR = '#2962ff'; // Blue — slow MA
   199→    const EMA12_CLR = '#ff6d00'; // Orange — EMA
   200→    const BB_CLR = '#7c4dff';   // Purple — Bollinger Bands
   201→    const RSI_CLR = '#2962ff';  // Blue
   202→    const MACD_CLR = '#2962ff'; // Blue
   203→    const SIG_CLR = '#ff6d00';  // Orange
   204→
   205→    const W = 880;
   206→    const H = 540;
   207→    const R = 60; // right axis width
   208→    const L = 2;  // left margin
   209→    const chartW = W - L - R;
   210→    const n = data.length;
   211→    const barW = chartW / n;
   212→    const candleW = Math.max(barW * 0.6, 2);
   213→
   214→    // Layout: price (with volume overlay), separator, RSI, separator, MACD
   215→    const priceH = 280;
   216→    const volRatio = 0.18; // volume takes bottom 18% of price panel
   217→    const volH = priceH * volRatio;
   218→    const candleH = priceH - volH; // candles use top 82%
   219→    const sep = 1;
   220→    const rsiH = 80;
   221→    const macdH = 85;
   222→    const bottomLabel = 18;
   223→
   224→    const priceTop = 0;
   225→    const volTop = priceTop + candleH;
   226→    const rsiTop = priceTop + priceH + sep;
   227→    const macdTop = rsiTop + rsiH + sep;
   228→    const totalH = macdTop + macdH + bottomLabel;
   229→
   230→    // ═══ Compute indicators — adaptive periods based on data length ═══
   231→    // Scales periods proportionally so lines span ~85%+ of chart width
   232→    const closes = data.map(d => d.close);
   233→    const smaP = Math.max(3, Math.min(20, Math.round(n * 0.16)));
   234→    const smaLongP = Math.max(5, Math.min(50, Math.round(n * 0.40)));
   235→    const bbP = Math.max(3, Math.min(20, Math.round(n * 0.16)));
   236→    const emaP = Math.max(3, Math.min(12, Math.round(n * 0.10)));
   237→    const rsiP = Math.max(3, Math.min(14, Math.round(n * 0.11)));
   238→    const macdFast = Math.max(3, Math.min(12, Math.round(n * 0.10)));
   239→    const macdSlow = Math.max(5, Math.min(26, Math.round(n * 0.21)));
   240→
   241→    const sma20 = computeSMA(closes, smaP);
   242→    const sma50 = computeSMA(closes, smaLongP);
   243→    const ema12 = computeEMA(closes, emaP);
   244→    const bb = computeBollingerBands(closes, bbP);
   245→    const rsi = computeRSI(closes, rsiP);
   246→    const macdData = computeMACD(closes, macdFast, macdSlow);
   247→
   248→    // ═══ Price Y scale (candles area only, not volume) ═══
   249→    const hiVals = [ ...data.map(d => d.high), ...bb.upper.filter((v): v is number => v !== null), ...sma20.filter((v): v is number => v !== null) ];
   250→    const loVals = [ ...data.map(d => d.low), ...bb.lower.filter((v): v is number => v !== null), ...sma20.filter((v): v is number => v !== null) ];
   251→    const allH = Math.max(...hiVals);
   252→    const allL = Math.min(...loVals);
   253→    const pad = (allH - allL) * 0.06 || 1;
   254→    const pMin = allL - pad;
   255→    const pMax = allH + pad;
   256→    const yP = (v: number) => priceTop + (1 - (v - pMin) / (pMax - pMin)) * candleH;
   257→
   258→    // ═══ Volume Y scale (bottom of price panel) ═══
   259→    const maxVol = Math.max(...data.map(d => d.volume || 0)) || 1;
   260→    const yV = (v: number) => volTop + volH - (v / maxVol) * volH;
   261→
   262→    // ═══ RSI scale 0-100 ═══
   263→    const yR = (v: number) => rsiTop + (1 - v / 100) * rsiH;
   264→
   265→    // ═══ MACD scale ═══
   266→    const macdVals = macdData.macd.filter((v): v is number => v !== null);
   267→    const histVals = macdData.histogram.filter((v): v is number => v !== null);
   268→    const mMax = Math.max(Math.abs(Math.min(...macdVals, ...histVals)), Math.abs(Math.max(...macdVals, ...histVals)), 0.01);
   269→    const yM = (v: number) => macdTop + macdH / 2 - (v / mMax) * (macdH / 2);
   270→
   271→    // ═══ Helpers ═══
   272→    const xOf = (i: number) => L + barW * i + barW / 2;
   273→    const labelN = n <= 15 ? 1 : n <= 30 ? 3 : n <= 60 ? 5 : n <= 130 ? 10 : 20;
   274→
   275→    const line = (vals: (number | null)[], color: string, sw = 1, id?: string) => {
   276→      const pts: string[] = [];
   277→      for (let i = 0; i < vals.length; i++) {
   278→        if (vals[i] !== null) pts.push(`${xOf(i)},${vals[i]}`);
   279→      }
   280→      return pts.length > 1 ? <polyline key={id ?? color} points={pts.join(' ')} fill="none" stroke={color} strokeWidth={sw} strokeLinejoin="round" strokeLinecap="round" /> : null;
   281→    };
   282→
   283→    // Price grid
   284→    const pGrid: { y: number; lbl: string }[] = [];
   285→    for (let i = 0; i <= 5; i++) {
   286→      const v = pMin + (pMax - pMin) * (i / 5);
   287→      pGrid.push({ y: yP(v), lbl: '$' + v.toFixed(1) });
   288→    }
   289→
   290→    // BB fill
   291→    const bbPts: string[] = [];
   292→    for (let i = 0; i < n; i++) if (bb.upper[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.upper[i]!)}`);
   293→    for (let i = n - 1; i >= 0; i--) if (bb.lower[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.lower[i]!)}`);
   294→
   295→    // Last values for legend
   296→    const lastSma20 = sma20[n - 1];
   297→    const lastSma50 = sma50[n - 1];
   298→    const lastEma12 = ema12[n - 1];
   299→    const lastRsi = rsi[n - 1];
   300→    const lastMacd = macdData.macd[n - 1];
   301→    const lastSig = macdData.signal[n - 1];
   302→    const lastClose = data[n - 1].close;
   303→    const lastY = yP(lastClose);
   304→
   305→    return (
   306→      <svg viewBox={`0 0 ${W} ${totalH}`} className="w-full h-full" style={{ background: BG, borderRadius: 0 }}>
   307→        <defs>
   308→          <clipPath id="priceClip"><rect x={L} y={priceTop} width={chartW} height={priceH} /></clipPath>
   309→          <clipPath id="rsiClip"><rect x={L} y={rsiTop} width={chartW} height={rsiH} /></clipPath>
   310→          <clipPath id="macdClip"><rect x={L} y={macdTop} width={chartW} height={macdH} /></clipPath>
   311→        </defs>
   312→        {/* Background fill for each panel */
   313→        <rect x={L} y={priceTop} width={chartW} height={priceH} fill="#1e222d" opacity={0.3} />
   314→        <rect x={L} y={rsiTop} width={chartW} height={rsiH} fill="#1e222d" opacity={0.2} />
   315→        <rect x={L} y={macdTop} width={chartW} height={macdH} fill="#1e222d" opacity={0.2} />
   316→
   317→        {/* ═══════ PRICE CHART (candles + volume) ═══════ */}
   318→
   319→        {/* Grid lines — dotted, TradingView style */}
   320→        {pGrid.map((g, i) => (
   321→          <g key={"g" + i}>
   322→            <line x1={L} y1={g.y} x2={W - R} y2={g.y} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   323→            <text x={W - R + 6} y={g.y + 3.5} fill={TXT_BRIGHT} fontSize={10} fontFamily="Trebuchet MS, Tahoma, sans-serif">{g.lbl}</text>
   324→          </g>
   325→        ))}
   326→
   327→        {/* Vertical grid (time axis) */}
   328→        {data.map((d, i) => {
   329→          if (i % labelN !== 0 || i === 0) return null;
   330→          return <line key={"vg" + i} x1={xOf(i)} y1={priceTop} x2={xOf(i)} y2={priceTop + priceH} stroke={GRID} strokeOpacity={0.25} strokeDasharray="1 3" />;
   331→        })}
   332→
   333→        {/* Volume bars (behind candles, at bottom of price panel) */}
   334→        <g clipPath="url(#priceClip)">
   335→          {data.map((d, i) => {
   336→            const cx = xOf(i);
   337→            const isUp = d.close >= d.open;
   338→            const vy = yV(d.volume || 0);
   339→            return (
   340→              <rect key={"vol" + i} x={cx - candleW / 2} y={vy} width={candleW} height={Math.max(volTop + volH - vy, 0)}
   341→                fill={isUp ? 'rgba(38,166,154,0.4)' : 'rgba(239,83,80,0.4)'} />
   342→            );
   343→          })}
   344→        </g>
   345→
   346→        {/* Bollinger Bands fill */}
   347→        {bbPts.length > 2 && <polygon points={bbPts.join(' ')} fill="rgba(124,77,255,0.08)" stroke="none" />}
   348→
   349→        {/* Bollinger Band lines */}
   350→        {line(bb.upper.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-u')}
   351→        {line(bb.middle.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-m')}
   352→        {line(bb.lower.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-l')}
   353→
   354→        {/* Moving Averages */}
   355→        {line(sma20.map(v => v !== null ? yP(v) : null), SMA20_CLR, 1.5, 's20')}
   356→        {line(sma50.map(v => v !== null ? yP(v) : null), SMA50_CLR, 1.5, 's50')}
   357→        {line(ema12.map(v => v !== null ? yP(v) : null), EMA12_CLR, 1.2, 'e12')}
   358→
   359→        {/* Candlesticks */}
   360→        <g clipPath="url(#priceClip)">
   361→          {data.map((d, i) => {
   362→            const cx = xOf(i);
   363→            const up = d.close >= d.open;
   364→            const c = up ? BULL : BEAR;
   365→            const bT = yP(Math.max(d.open, d.close));
   366→            const bB = yP(Math.min(d.open, d.close));
   367→            const bH = Math.max(bB - bT, 1);
   368→            return (
   369→              <g key={"k" + i}>
   370→                <line x1={cx} y1={yP(d.high)} x2={cx} y2={yP(d.low)} stroke={c} strokeWidth={1} />
   371→                <rect x={cx - candleW / 2} y={bT} width={candleW} height={bH} fill={c} stroke={c} strokeWidth={0.5} />
   372→              </g>
   373→            );
   374→          })}
   375→        </g>
   376→
   377→        {/* Current price line + tag */}
   378→        <line x1={L} y1={lastY} x2={W - R} y2={lastY} stroke={data[n-1].close >= data[n-1].open ? BULL : BEAR} strokeDasharray="4 2" strokeOpacity={0.7} strokeWidth={1} />
   379→        <rect x={W - R + 1} y={lastY - 10} width={R - 2} height={20} rx={2} fill={data[n-1].close >= data[n-1].open ? BULL : BEAR} />
   380→        <text x={W - R + 8} y={lastY + 4} fill="white" fontSize={10.5} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">{lastClose.toFixed(2)}</text>
   381→
   382→        {/* TV-style top-right indicator legend with box */}
   383→        <rect x={W - R - 210} y={priceTop + 4} width={206} height={52} rx={4} fill="rgba(19,23,34,0.85)" stroke="#2a2e39" strokeWidth={0.5} />
   384→        <text x={W - R - 204} y={priceTop + 17} fill={TXT_BRIGHT} fontSize={9} fontFamily="Trebuchet MS, sans-serif" fontWeight="600" opacity={0.7}>Indicators</text>
   385→        {lastSma20 !== null && <text x={W - R - 204} y={priceTop + 30} fill={SMA20_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">SMA({smaP})</text>}
   386→        {lastSma20 !== null && <text x={W - R - 130} y={priceTop + 30} fill={TXT_BRIGHT} fontSize={10} fontFamily="Trebuchet MS, sans-serif">{lastSma20.toFixed(2)}</text>}
   387→        {lastSma50 !== null && <text x={W - R - 60} y={priceTop + 30} fill={SMA50_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">SMA({smaLongP})</text>}
   388→        {lastSma50 !== null && <text x={W - R - 4} y={priceTop + 30} fill={TXT_BRIGHT} fontSize={10} fontFamily="Trebuchet MS, sans-serif" textAnchor="end">{lastSma50.toFixed(2)}</text>}
   389→        {lastEma12 !== null && <text x={W - R - 204} y={priceTop + 44} fill={EMA12_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">EMA({emaP})</text>}
   390→        {lastEma12 !== null && <text x={W - R - 130} y={priceTop + 44} fill={TXT_BRIGHT} fontSize={10} fontFamily="Trebuchet MS, sans-serif">{lastEma12.toFixed(2)}</text>}
   391→        <text x={W - R - 60} y={priceTop + 44} fill={BB_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">BB({bbP})</text>
   392→
   393→        {/* Volume label */}
   394→        <text x={L + 8} y={volTop + 12} fill={TXT} fontSize={9} fontFamily="Trebuchet MS, sans-serif" opacity={0.5}>Vol</text>
   395→
   396→        {/* ═══════ PANEL SEPARATOR ═══════ */}
   397→        <line x1={L} y1={rsiTop - 1} x2={W - R} y2={rsiTop - 1} stroke={GRID} strokeOpacity={0.5} />
   398→
   399→        {/* ═══════ RSI (14) ═══════ */}
   400→        {/* RSI grid */}
   401→        <line x1={L} y1={yR(70)} x2={W - R} y2={yR(70)} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   402→        <line x1={L} y1={yR(50)} x2={W - R} y2={yR(50)} stroke={GRID} strokeOpacity={GRID_OP * 0.6} strokeDasharray="1 3" />
