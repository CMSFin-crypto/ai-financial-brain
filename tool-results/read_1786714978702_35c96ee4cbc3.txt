    88→// ═══ Technical indicator computations ═══
    89→function computeSMA(closes: number[], period: number): (number | null)[] {
    90→  const result: (number | null)[] = [];
    91→  for (let i = 0; i < closes.length; i++) {
    92→    if (i < period - 1) { result.push(null); continue; }
    93→    let sum = 0;
    94→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
    95→    result.push(sum / period);
    96→  }
    97→  return result;
    98→}
    99→
   100→function computeEMA(closes: number[], period: number): (number | null)[] {
   101→  const k = 2 / (period + 1);
   102→  const result: (number | null)[] = [];
   103→  let ema: number | null = null;
   104→  for (let i = 0; i < closes.length; i++) {
   105→    if (i < period - 1) { result.push(null); continue; }
   106→    if (ema === null) {
   107→      let sum = 0; for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   108→      ema = sum / period;
   109→    } else {
   110→      ema = closes[i] * k + ema * (1 - k);
   111→    }
   112→    result.push(ema);
   113→  }
   114→  return result;
   115→}

   117→function computeRSI(closes: number[], period: number = 14): (number | null)[] {
   118→  const result: (number | null)[] = [];
   119→  const gains: number[] = [];
   120→  const losses: number[] = [];
   121→  for (let i = 1; i < closes.length; i++) {
   122→    const diff = closes[i] - closes[i - 1];
   123→    gains.push(diff > 0 ? diff : 0);
   124→    losses.push(diff < 0 ? -diff : 0);
   125→  }
   126→  let avgGain = 0, avgLoss = 0;
   127→  for (let i = 0; i < Math.min(period, gains.length); i++) { avgGain += gains[i]; avgLoss += losses[i]; }
   128→  avgGain /= period; avgLoss /= period;
   129→  for (let i = 0; i < period - 1; i++) result.push(null);
   130→  const rs = avgLoss === 0 ? 100 : avgGain / avgLoss;
   131→  result.push(100 - 100 / (1 + rs));
   132→  for (let i = period; i < gains.length; i++) {
   133→    avgGain = (avgGain * (period - 1) + gains[i]) / period;
   134→    avgLoss = (avgLoss * (period - 1) + losses[i]) / period;
   135→    const rs2 = avgLoss === 0 ? 100 : avgGain / avgLoss;
   136→    result.push(100 - 100 / (1 + rs2));
   137→  }
   138→  return result;
   139→}

   141→function computeMACD(closes: number[], fastPeriod = 12, slowPeriod = 26, signalPeriod = 9): { macd: (number | null)[]; signal: (number | null)[]; histogram: (number | null)[] } {
   142→  const emaFast = computeEMA(closes, fastPeriod);
   143→  const emaSlow = computeEMA(closes, slowPeriod);
   144→  const macdLine: (number | null)[] = [];
   145→  for (let i = 0; i < closes.length; i++) {
   146→    if (emaFast[i] === null || emaSlow[i] === null) { macdLine.push(null); continue; }
   147→    macdLine.push(emaFast[i]! - emaSlow[i]!);
   148→  }
   149→  // Signal = EMA of MACD line — very short period so signal starts quickly
   150→  const validMACD = macdLine.filter((v): v is number => v !== null);
   151→  const sigPeriod = Math.min(signalPeriod, Math.max(2, Math.floor(validMACD.length * 0.15)));
   152→  const signalEma = computeEMA(validMACD, sigPeriod);
   153→  const signal: (number | null)[] = [];
   154→  let vi = 0;
   155→  for (let i = 0; i < closes.length; i++) {
   156→    if (macdLine[i] === null) { signal.push(null); continue; }
   157→    signal.push(signalEma[vi] ?? null);
   158→    vi++;
   159→  }
   160→  const histogram: (number | null)[] = [];
   161→  for (let i = 0; i < closes.length; i++) {
   162→    if (macdLine[i] === null || signal[i] === null) { histogram.push(null); continue; }
   163→    histogram.push(macdLine[i]! - signal[i]!);
   164→  }
   165→  return { macd: macdLine, signal, histogram };
   166→}

   168→function computeBollingerBands(closes: number[], period: number = 20, mult: number = 2): { upper: (number | null)[]; middle: (number | null)[]; lower: (number | null)[] } {
   169→  const middle = computeSMA(closes, period);
   170→  const upper: (number | null)[] = [];
   171→  const lower: (number | null)[] = [];
   172→  for (let i = 0; i < closes.length; i++) {
   173→    if (middle[i] === null) { upper.push(null); lower.push(null); continue; }
   174→    let sumSq = 0;
   175→    for (let j = i - period + 1; j <= i; j++) sumSq += (closes[j] - middle[i]!) ** 2;
   176→    const std = Math.sqrt(sumSq / period);
   177→    upper.push(middle[i]! + mult * std);
   178→    lower.push(middle[i]! - mult * std);
   179→  }
   180→  return { upper, middle, lower };
   181→}

   183→// ═══ TradingView-style right-edge label group ═══
   184→// Renders colored tags at the end of each indicator line (left of Y-axis),
   185→// stacking them vertically if they overlap.
   186→function EdgeLabels({ items, rightEdge, fontSize = 9.5 }: {
   187→  items: { label: string; y: number; color: string }[];
   188→  rightEdge: number;
   189→  fontSize?: number;
   190→}) {
   191→  const LH = fontSize + 3; // line height per label
   192→  const MIN_GAP = 2;
   193→
   194→  // Sort by Y, then resolve overlaps by pushing downward
   195→  const sorted = [...items].sort((a, b) => a.y - b.y);
   196→  const placed: { label: string; y: number; color: string }[] = [];
   197→  for (const item of sorted) {
   198→    let y = item.y - LH / 2; // center vertically on the line
   199→    for (const p of placed) {
   200→      if (y < p.y + LH + MIN_GAP && y + LH > p.y) {
   201→        y = p.y + LH + MIN_GAP; // push below
   202→      }
   203→    }
   204→    placed.push({ ...item, y });
   205→  }
   206→
   207→  return (
   208→    <g>
   209→      {placed.map((p, i) => {
   210→        const textW = p.label.length * fontSize * 0.6 + 8;
   211→        const rx = rightEdge - textW - 4;
   212→        return (
   213→          <g key={i}>
   214→            <rect x={rx} y={p.y} width={textW} height={LH} rx={2} fill={p.color} opacity={0.92} />
   215→            <text
   216→              x={rx + 4}
   217→              y={p.y + fontSize + 1}
   218→              fill="white"
   219→              fontSize={fontSize}
   220→              fontFamily="Trebuchet MS, sans-serif"
   221→              fontWeight="600"
   222→            >{p.label}</text>
   223→          </g>
   224→        );
   225→      })}
   226→    </g>
   227→  );
   228→}

   230→// ═══ TradingView-Style Technical Chart ═══
   231→function CandlestickChart({ data }: { data: CandleData[] }) {
   232→  const chart = useMemo(() => {
   233→    if (!data || data.length < 3) return null;
   234→
   235→    // TradingView dark palette (matching reference)
   236→    const BG = '#131722';
   237→    const GRID = '#2a2e39';
   238→    const GRID_OP = 0.3;
   239→    const TXT = '#787b86';
   240→    const TXT_BRIGHT = '#d1d4dc';
   241→    const BULL = '#26a69a';
   242→    const BEAR = '#ef5350';
   243→    const SMA20_CLR = '#f0b323'; // Gold/Yellow — fast MA
   244→    const SMA50_CLR = '#2962ff'; // Blue — slow MA
   245→    const EMA12_CLR = '#ff6d00'; // Orange — EMA
   246→    const BB_CLR = '#7c4dff';   // Purple — Bollinger Bands
   247→    const RSI_CLR = '#2962ff';  // Blue
   248→    const MACD_CLR = '#2962ff'; // Blue
   249→    const SIG_CLR = '#ff6d00';  // Orange
   250→
   251→    const W = 880;
   252→    const H = 540;
   253→    const R = 60; // right axis width
   254→    const L = 2;  // left margin
   255→    const chartW = W - L - R;
   256→    const n = data.length;
   257→    const barW = chartW / n;
   258→    const candleW = Math.max(barW * 0.6, 2);
   259→
   260→    // Layout: price (with volume overlay), separator, RSI, separator, MACD
   261→    const priceH = 280;
   262→    const volRatio = 0.18; // volume takes bottom 18% of price panel
   263→    const volH = priceH * volRatio;
   264→    const candleH = priceH - volH; // candles use top 82%
   265→    const sep = 1;
   266→    const rsiH = 80;
   267→    const macdH = 85;
   268→    const bottomLabel = 18;
   269→
   270→    const priceTop = 0;
   271→    const volTop = priceTop + candleH;
   272→    const rsiTop = priceTop + priceH + sep;
   273→    const macdTop = rsiTop + rsiH + sep;
   274→    const totalH = macdTop + macdH + bottomLabel;
   275→
   276→    // ═══ Compute indicators — DUAL STRATEGY ═══
   277→    // 70+ bars → standard TradingView periods
   278→    // <70 bars → scaled-down periods for small datasets
   279→    const closes = data.map(d => d.close);
   280→
   281→    let smaP: number, smaLongP: number, bbP: number, emaP: number, rsiP: number, macdFast: number, macdSlow: number;
   282→
   283→    if (n >= 70) {
   284→      smaP = 20; smaLongP = 50; bbP = 20; emaP = 12; rsiP = 14; macdFast = 12; macdSlow = 26;
   285→    } else {
   286→      const maxPeriod = Math.max(3, Math.floor(n * 0.08) + 1);
   287→      smaLongP  = maxPeriod;
   288→      smaP      = Math.max(2, Math.round(maxPeriod * 0.4));
   289→      bbP       = smaP;
   290→      emaP      = Math.max(2, Math.round(maxPeriod * 0.3));
   291→      rsiP      = Math.max(2, Math.round(maxPeriod * 0.35));
   292→      macdFast  = emaP;
   293→      macdSlow  = Math.max(macdFast + 1, Math.round(maxPeriod * 0.6));
   294→    }
   295→
   296→    const sma20 = computeSMA(closes, smaP);
   297→    const sma50 = computeSMA(closes, smaLongP);
   298→    const ema12 = computeEMA(closes, emaP);
   299→    const bb = computeBollingerBands(closes, bbP);
   300→    const rsi = computeRSI(closes, rsiP);
   301→    const macdData = computeMACD(closes, macdFast, macdSlow);
   302→
   303→    console.log(`[CHART] bars=${n} periods: SMA(${smaP},${smaLongP}) EMA(${emaP}) RSI(${rsiP}) MACD(${macdFast},${macdSlow})`);
   304→
   305→    // ═══ Price Y scale (candles area only, not volume) ═══
   306→    const hiVals = [ ...data.map(d => d.high), ...bb.upper.filter((v): v is number => v !== null), ...sma20.filter((v): v is number => v !== null) ];
   307→    const loVals = [ ...data.map(d => d.low), ...bb.lower.filter((v): v is number => v !== null), ...sma20.filter((v): v is number => v !== null) ];
   308→    const allH = Math.max(...hiVals);
   309→    const allL = Math.min(...loVals);
   310→    const pad = (allH - allL) * 0.06 || 1;
   311→    const pMin = allL - pad;
   312→    const pMax = allH + pad;
   313→    const yP = (v: number) => priceTop + (1 - (v - pMin) / (pMax - pMin)) * candleH;
   314→
   315→    // ═══ Volume Y scale (bottom of price panel) ═══
   316→    const maxVol = Math.max(...data.map(d => d.volume || 0)) || 1;
   317→    const yV = (v: number) => volTop + volH - (v / maxVol) * volH;
   318→
   319→    // ═══ RSI scale 0-100 ═══
   320→    const yR = (v: number) => rsiTop + (1 - v / 100) * rsiH;
   321→
   322→    // ═══ MACD scale ═══
   323→    const macdVals = macdData.macd.filter((v): v is number => v !== null);
   324→    const histVals = macdData.histogram.filter((v): v is number => v !== null);
   325→    const mMax = Math.max(Math.abs(Math.min(...macdVals, ...histVals)), Math.abs(Math.max(...macdVals, ...histVals)), 0.01);
   326→    const yM = (v: number) => macdTop + macdH / 2 - (v / mMax) * (macdH / 2);
   327→
   328→    // ═══ Helpers ═══
   329→    const xOf = (i: number) => L + barW * i + barW / 2;
   330→    const labelN = n <= 15 ? 1 : n <= 30 ? 3 : n <= 60 ? 5 : n <= 130 ? 10 : 20;
   331→
   332→    const line = (vals: (number | null)[], color: string, sw = 1, id?: string) => {
   333→      const pts: string[] = [];
   334→      let firstValidY: number | null = null;
   335→      for (let i = 0; i < vals.length; i++) {
   336→        if (vals[i] !== null) {
   337→          if (firstValidY === null) {
   338→            firstValidY = vals[i];
   339→            // Backfill: extend line to the left edge using first valid value
   340→            if (i > 0) pts.push(`${xOf(0)},${vals[i]}`);
   341→          }
   342→          pts.push(`${xOf(i)},${vals[i]}`);
   343→        }
   344→      }
   345→      return pts.length > 1 ? <polyline key={id ?? color} points={pts.join(' ')} fill="none" stroke={color} strokeWidth={sw} strokeLinejoin="round" strokeLinecap="round" /> : null;
   346→    };
   347→
   348→    // Price grid
   349→    const pGrid: { y: number; lbl: string }[] = [];
   350→    for (let i = 0; i <= 5; i++) {
   351→      const v = pMin + (pMax - pMin) * (i / 5);
   352→      pGrid.push({ y: yP(v), lbl: '$' + v.toFixed(1) });
   353→    }
   354→
   355→    // BB fill
   356→    const bbPts: string[] = [];
   357→    for (let i = 0; i < n; i++) if (bb.upper[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.upper[i]!)}`);
   358→    for (let i = n - 1; i >= 0; i--) if (bb.lower[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.lower[i]!)}`);
   359→
   360→    // Last values for labels
   361→    const lastSma20 = sma20[n - 1];
   362→    const lastSma50 = sma50[n - 1];
   363→    const lastEma12 = ema12[n - 1];
   364→    const lastRsi = rsi[n - 1];
   365→    const lastMacd = macdData.macd[n - 1];
   366→    const lastSig = macdData.signal[n - 1];
   367→    const lastClose = data[n - 1].close;
   368→    const lastY = yP(lastClose);
   369→
   370→    return (
   371→      <svg viewBox={`0 0 ${W} ${totalH}`} className="w-full h-full" style={{ background: BG, borderRadius: 0 }}>
   372→        <defs>
   373→          <clipPath id="priceClip"><rect x={L} y={priceTop} width={chartW} height={priceH} /></clipPath>
   374→          <clipPath id="rsiClip"><rect x={L} y={rsiTop} width={chartW} height={rsiH} /></clipPath>
   375→          <clipPath id="macdClip"><rect x={L} y={macdTop} width={chartW} height={macdH} /></clipPath>
   376→        </defs>
   377→        {/* Background fill for each panel */}
   378→        <rect x={L} y={priceTop} width={chartW} height={priceH} fill="#1e222d" opacity={0.3} />
   379→        <rect x={L} y={rsiTop} width={chartW} height={rsiH} fill="#1e222d" opacity={0.2} />
   380→        <rect x={L} y={macdTop} width={chartW} height={macdH} fill="#1e222d" opacity={0.2} />
   381→
   382→        {/* ═══════ PRICE CHART (candles + volume) ═══════ */}
   383→
   384→        {/* Grid lines — dotted, TradingView style */}
   385→        {pGrid.map((g, i) => (
   386→          <g key={"g" + i}>
   387→            <line x1={L} y1={g.y} x2={W - R} y2={g.y} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   388→            <text x={W - R + 6} y={g.y + 3.5} fill={TXT_BRIGHT} fontSize={10} fontFamily="Trebuchet MS, Tahoma, sans-serif">{g.lbl}</text>
   389→          </g>
   390→        ))}
   391→
   392→        {/* Vertical grid (time axis) */}
   393→        {data.map((dd, i) => {
   394→          if (i % labelN !== 0 || i === 0) return null;
   395→          return <line key={"vg" + i} x1={xOf(i)} y1={priceTop} x2={xOf(i)} y2={priceTop + priceH} stroke={GRID} strokeOpacity={0.25} strokeDasharray="1 3" />;
   396→        })}
   397→
   398→        {/* Volume bars (behind candles, at bottom of price panel) */}
   399→        <g clipPath="url(#priceClip)">
   400→          {data.map((dd, i) => {
   401→            const cx = xOf(i);
   402→            const isUp = dd.close >= dd.open;
   403→            const vy = yV(dd.volume || 0);
   404→            return (
   405→              <rect key={"vol" + i} x={cx - candleW / 2} y={vy} width={candleW} height={Math.max(volTop + volH - vy, 0)}
   406→                fill={isUp ? 'rgba(38,166,154,0.4)' : 'rgba(239,83,80,0.4)'} />
   407→            );
   408→          })}
   409→        </g>
   410→
   411→        {/* Bollinger Bands fill */}
   412→        {bbPts.length > 2 && <polygon points={bbPts.join(' ')} fill="rgba(124,77,255,0.08)" stroke="none" />}
   413→
   414→        {/* Bollinger Band lines */}
   415→        {line(bb.upper.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-u')}
   416→        {line(bb.middle.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-m')}
   417→        {line(bb.lower.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-l')}
   418→
   419→        {/* Moving Averages */}
   420→        {line(sma20.map(v => v !== null ? yP(v) : null), SMA20_CLR, 1.5, 's20')}
   421→        {line(sma50.map(v => v !== null ? yP(v) : null), SMA50_CLR, 1.5, 's50')}
   422→        {line(ema12.map(v => v !== null ? yP(v) : null), EMA12_CLR, 1.2, 'e12')}
   423→
   424→        {/* Candlesticks */}
   425→        <g clipPath="url(#priceClip)">
   426→          {data.map((dd, i) => {
   427→            const cx = xOf(i);
   428→            const up = dd.close >= dd.open;
   429→            const c = up ? BULL : BEAR;
   430→            const bT = yP(Math.max(dd.open, dd.close));
   431→            const bB = yP(Math.min(dd.open, dd.close));
   432→            const bH = Math.max(bB - bT, 1);
   433→            return (
   434→              <g key={"k" + i}>
   435→                <line x1={cx} y1={yP(dd.high)} x2={cx} y2={yP(dd.low)} stroke={c} strokeWidth={1} />
   436→                <rect x={cx - candleW / 2} y={bT} width={candleW} height={bH} fill={c} stroke={c} strokeWidth={0.5} />
   437→              </g>
   438→            );
   439→          })}
   440→        </g>
   441→
   442→        {/* Current price line + tag */}
   443→        <line x1={L} y1={lastY} x2={W - R} y2={lastY} stroke={data[n-1].close >= data[n-1].open ? BULL : BEAR} strokeDasharray="4 2" strokeOpacity={0.7} strokeWidth={1} />
   444→        <rect x={W - R + 1} y={lastY - 10} width={R - 2} height={20} rx={2} fill={data[n-1].close >= data[n-1].open ? BULL : BEAR} />
   445→        <text x={W - R + 8} y={lastY + 4} fill="white" fontSize={10.5} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">{lastClose.toFixed(2)}</text>
   446→
   447→        {/* ═══ TradingView-style RIGHT-EDGE line labels ═══ */}
