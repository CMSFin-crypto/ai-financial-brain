   151→function hitRateColor(rate: number): string {
   152→  if (rate > 60) return 'text-emerald-400';
   153→  if (rate >= 40) return 'text-amber-400';
   154→  return 'text-red-400';
   155→}
   156→
   157→function hitRateBg(rate: number): string {
   158→  if (rate > 60) return 'bg-emerald-500/15 text-emerald-400 border-emerald-500/25';
   159→  if (rate >= 40) return 'bg-amber-500/15 text-amber-400 border-amber-500/25';
   160→  return 'bg-red-500/15 text-red-400 border-red-500/25';
   161→}
   162→
   163→function outcomeBadge(outcome: string) {
   164→  switch (outcome) {
   165→    case 'CORRECT_OVERRIDE':
   166→      return (
   167→        <Badge className="bg-emerald-500/15 text-emerald-400 border-emerald-500/25 gap-1">
   168→          <CheckCircle2 className="size-3" />
   169→          Korrekt
   170→        </Badge>
   171→      );
   172→    case 'WRONG_OVERRIDE':
   173→      return (
   174→        <Badge className="bg-red-500/15 text-red-400 border-red-500/25 gap-1">
   175→          <XCircle className="size-3" />
   176→          Gabim
   177→        </Badge>
   178→      );
   179→    default:
   180→      return (
   181→        <Badge className="bg-muted text-muted-foreground border-border gap-1">
   182→          <HelpCircle className="size-3" />
   183→          Pending
   184→        </Badge>
   185→      );
   186→  }
   187→}

   189→function overrideDecisionBadge(decision: string) {
   190→  switch (decision) {
   191→    case 'ACCEPTED':
   192→      return (
   193→        <Badge className="bg-blue-500/15 text-blue-400 border-blue-500/25 gap-1">
   194→          <ThumbsUp className="size-3" />

   198→    case 'REJECTED':
   199→      return (
   200→        <Badge className="bg-amber-500/15 text-amber-400 border-amber-500/25 gap-1">
   201→          <Ban className="size-3" />
   202→          {decision}
   203→        </Badge>
   204→      );

   208→          <Edit className="size-3" />

   210→        </Badge>
   211→      );
   212→    default:
   213→      return <Badge variant="outline">{decision}</Badge>;
   214→  }

   217→function returnCell(value: number | null | undefined) {
   218→  if (value == null) return <span className="text-muted-foreground">—</span>;
   219→  const color = value >= 0 ? 'text-emerald-400' : 'text-red-400';
   220→  const sign = value >= 0 ? '+' : '';
   221→  return <span className={`font-medium ${color}`}>{sign}{value.toFixed(2)}%</span>;
   222→}
   223→
   224→function formatDate(iso: string) {
   225→  const d = new Date(iso);
   226→  return d.toLocaleDateString('sq-AL', {
   227→    day: '2-digit',
   228→    month: '2-digit',
   229→    year: 'numeric',
   230→    hour: '2-digit',
   231→    minute: '2-digit',
   232→  });
   233→}
