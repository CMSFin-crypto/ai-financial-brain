   507→interface SearchResult { ticker: string; company: string; exchange: string; price: number; change: number; volume: number; }
   508→
   509→// ── Single Stock Search Component ──
   510→function StockSearchBox() {
   511→  const [query, setQuery] = useState('');
   512→  const [results, setResults] = useState<SearchResult[]>([]);
   513→  const [showDropdown, setShowDropdown] = useState(false);
   514→  const [analyzing, setAnalyzing] = useState<string | null>(null);
   515→  const [analyzedStock, setAnalyzedStock] = useState<FunnelStock | null>(null);
   516→  const [analyzeError, setAnalyzeError] = useState<string | null>(null);
   517→  const [regimeOk, setRegimeOk] = useState(true);
   518→  const [funnelPhases, setFunnelPhases] = useState({ passedLiquidity: 0, passedTrend: 0, passedSetup: 0, passedRisk: 0 });
   519→  const timerRef = useRef<ReturnType<typeof setTimeout>>();
   520→  const inputRef = useRef<HTMLInputElement>(null);
   521→
   522→  // Search with debounce
   523→  const handleInput = (val: string) => {
   524→    setQuery(val);
   525→    if (timerRef.current) clearTimeout(timerRef.current);
   526→    if (val.length < 2) { setResults([]); setShowDropdown(false); return; }
   527→    timerRef.current = setTimeout(async () => {
   528→      try {
   529→        const res = await fetch(`/api/ticker-search?q=${encodeURIComponent(val)}`);
   530→        const json = await res.json();
   531→        setResults(json.results || []);
   532→        setShowDropdown(true);
   533→      } catch { setResults([]); }
   534→    }, 350);
   535→  };
   536→
   537→  const selectTicker = async (ticker: string) => {
   538→    setShowDropdown(false); setQuery(ticker);
   539→    setAnalyzedStock(null); setAnalyzeError(null);
   540→    setAnalyzing(ticker);
   541→    try {
   542→      const res = await fetch(`/api/ibkr-analyze/${ticker}?_t=${Date.now()}`, { cache: 'no-store' });
   543→      const json = await res.json();
   544→      setRegimeOk(json.regimeOk);
   545→      setFunnelPhases(json.funnel || { passedLiquidity: 0, passedTrend: 0, passedSetup: 0, passedRisk: 0 });
   546→      if (json.error && !json.stock) {
   547→        setAnalyzeError(json.error);
   548→      } else if (json.stock) {
   549→        setAnalyzedStock(json.stock);
   550→      }
   551→    } catch { setAnalyzeError('Gabim rrjeti'); }
   552→    setAnalyzing(null);
   553→  };
   554→
   555→  return (
   556→    <div className="space-y-3">
   557→      <div className="relative">
   558→        <div className="flex items-center gap-2">
   559→          <div className="relative flex-1">
   560→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground/50" />
   561→            <input
   562→              ref={inputRef}
   563→              value={query}
   564→              onChange={e => handleInput(e.target.value)}
   565→              onFocus={() => results.length > 0 && setShowDropdown(true)}
   566→              placeholder="Kerko nje aksion (p.sh. AAPL, TSLA, COIN...)"
   567→              className="w-full pl-9 pr-9 py-2.5 rounded-lg bg-muted/10 border border-border/50 text-[13px] text-foreground placeholder:text-muted-foreground/40 focus:outline-none focus:border-blue-500/50 focus:ring-1 focus:ring-blue-500/20 transition-all"
   568→            />
   569→            {query && (
   570→              <button onClick={() => { setQuery(''); setResults([]); setShowDropdown(false); setAnalyzedStock(null); setAnalyzeError(null); }} className="absolute right-3 top-1/2 -translate-y-1/2"><X className="w-3.5 h-3.5 text-muted-foreground/50 hover:text-muted-foreground" /></button>
   571→            )}
   572→          </div>
