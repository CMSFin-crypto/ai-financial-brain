   166→function MetricInfoPopup({

   169→}: {
   170→  metricKey: string;
   171→  onClose: () => void;
   172→}) {
   173→  const info = METRIC_INFO[metricKey];
   174→  const popupRef = useRef<HTMLDivElement>(null);
   175→
   176→  useEffect(() => {
   177→    const handleClick = (e: MouseEvent) => {
   178→      if (popupRef.current && !popupRef.current.contains(e.target as Node)) {
   179→        onClose();
   180→      }
   181→    };
   182→    const handleEsc = (e: KeyboardEvent) => {
   183→      if (e.key === 'Escape') onClose();
   184→    };
   185→    setTimeout(() => {
   186→      document.addEventListener('mousedown', handleClick);
   187→      document.addEventListener('keydown', handleEsc);
   188→    }, 10);
   189→    return () => {
   190→      document.removeEventListener('mousedown', handleClick);
   191→      document.removeEventListener('keydown', handleEsc);
   192→    };
   193→  }, [onClose]);

   195→  if (!info) return null;
   196→
   197→  return (
   198→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/60 backdrop-blur-sm p-4" onClick={(e) => { if (e.target === e.currentTarget) onClose(); }}>

   307→// Simple seeded pseudo-random for consistent sparklines
   308→function seededRandom(seed: number): () => number {
   309→  let s = seed;
   310→  return () => {
   311→    s = (s * 16807 + 0) % 2147483647;
   312→    return (s - 1) / 2147483646;
   313→  };
   314→}

   316→function generateSparklineData(priceChange: number, ticker: string): Array<{ value: number }> {
   317→  const points = 12;
   318→  const data: Array<{ value: number }> = [];
   319→  const magnitude = Math.min(Math.abs(priceChange), 10);
   320→  // Create a seed from ticker hash + priceChange for consistency
   321→  const hash = ticker.split('').reduce((acc, c) => acc + c.charCodeAt(0), 0);
   322→  const rand = seededRandom(hash * 1000 + Math.round(priceChange * 100));
   323→  for (let i = 0; i < points; i++) {
   324→    const progress = i / (points - 1);
   325→    const trend = priceChange >= 0 ? progress * magnitude : -progress * magnitude;
   326→    const noise = (rand() - 0.5) * magnitude * 0.4;
   327→    data.push({ value: 50 + trend + noise });
   328→  }
   329→  return data;
   330→}
