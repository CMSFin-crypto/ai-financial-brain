     1→// ============================================================
     2→// Feature Definitions — centralized schema for all model features.
     3→//
     4→// This is the SINGLE SOURCE OF TRUTH for what features exist,
     5→// how they're computed, and what their expected types/ranges are.
     6→// Both training (research) and inference (production) MUST use
     7→// the same definitions — this prevents the #1 cause of
     8→// "offline works, live fails" bugs.
     9→//
    10→// Features are versioned. When a feature definition changes,
    11→// bump its version. The feature store will detect mismatches.
    12→// ============================================================
    13→
    14→// ─── Types ────────────────────────────────────────────────────
    15→
    16→export type FeatureType = 'numeric' | 'categorical' | 'boolean' | 'array_numeric';
    17→
    18→export type FeatureSource = 'technical' | 'fundamental' | 'regime' | 'market' | 'spillover' | 'event' | 'derived';
    19→
    20→export type FeatureDefinition = {
    21→  name: string;              // unique key, e.g. "rsi_14"
    22→  displayName: string;       // human-readable, e.g. "RSI (14-period)"
    23→  type: FeatureType;
    24→  source: FeatureSource;
    25→  version: number;           // bump on any logic change
    26→  description: string;
    27→  
    28→  // Constraints (for validation)
    29→  expectedRange?: [number, number];  // min, max for numeric
    30→  allowedValues?: string[];          // for categorical
    31→  defaultValue?: unknown;
    32→  nullable?: boolean;
    33→  
    34→  // Computation reference
    35→  computeFn?: string;        // reference to the function/module that computes it
    36→  dependsOn?: string[];      // names of other features this depends on
    37→  
    38→  // Metadata
    39→  tags?: string[];
    40→  enabled: boolean;          // can be disabled without deleting
    41→  addedAt: string;
    42→  updatedAt: string;
    43→};
    44→
    45→export type FeatureVersionInfo = {
    46→  featureName: string;
    47→  version: number;
    48→  changedAt: string;
    49→  changeDescription: string;
    50→};

   651→/**
   652→ * Get all feature definitions, optionally filtered.
   653→ */
   654→export function getAllFeatures(opts?: {
   655→  source?: FeatureSource;
   656→  enabledOnly?: boolean;
   657→  tag?: string;
   658→}): FeatureDefinition[] {
   659→  let result = ALL_FEATURES;
   660→  if (opts?.source) result = result.filter(f => f.source === opts.source);
   661→  if (opts?.enabledOnly) result = result.filter(f => f.enabled);
   662→  if (opts?.tag) result = result.filter(f => f.tags?.includes(opts.tag!));
   663→  return result;

   689→/**
   690→ * Validate a feature value against its definition.
   691→ * Returns { valid, errors }.
   692→ */
   693→export function validateFeature(
   694→  name: string,
   695→  value: unknown,
   696→): { valid: boolean; errors: string[] } {
   697→  const def = FEATURE_INDEX.get(name);
   698→  const errors: string[] = [];
   699→
   700→  if (!def) {
   701→    errors.push(`Unknown feature: ${name}`);
   702→    return { valid: false, errors };
   703→  }
   704→
   705→  if (!def.enabled) {
   706→    errors.push(`Feature ${name} is disabled`);
   707→    return { valid: false, errors };
   708→  }
   709→
   710→  if (value === null || value === undefined) {
   711→    if (def.nullable) return { valid: true, errors: [] };
   712→    errors.push(`Feature ${name} is not nullable but got null`);
   713→    return { valid: false, errors };
   714→  }
   715→
   716→  if (def.type === 'numeric') {
   717→    const num = Number(value);
   718→    if (isNaN(num)) {
   719→      errors.push(`Feature ${name} expects numeric, got ${typeof value}`);
   720→    } else if (def.expectedRange) {
   721→      const [min, max] = def.expectedRange;
   722→      // Allow 10% tolerance for range violations (data can be slightly out of range)
   723→      const tolerance = (max - min) * 0.1;
   724→      if (num < min - tolerance) {
   725→        errors.push(`Feature ${name} value ${num} below range [${min}, ${max}]`);
   726→      } else if (num > max + tolerance) {
   727→        errors.push(`Feature ${name} value ${num} above range [${min}, ${max}]`);
   728→      }
   729→    }
   730→  }
   731→
   732→  if (def.type === 'categorical' && def.allowedValues) {
   733→    if (!def.allowedValues.includes(String(value))) {
   734→      errors.push(`Feature ${name} value "${value}" not in allowed values: ${def.allowedValues.join(', ')}`);
   735→    }
   736→  }
   737→
   738→  return { valid: errors.length === 0, errors };
   739→}

   741→/**
   742→ * Validate an entire feature vector at once.
   743→ * Returns { valid, featureErrors, versionHash }.
   744→ */
   745→export function validateFeatureVector(
   746→  features: Record<string, unknown>,
   747→): { valid: boolean; featureErrors: Record<string, string[]>; versionHash: string } {
   748→  const featureErrors: Record<string, string[]> = {};
   749→  let allValid = true;
   750→
   751→  for (const [name, value] of Object.entries(features)) {
   752→    const result = validateFeature(name, value);
   753→    if (!result.valid) {
   754→      featureErrors[name] = result.errors;
   755→      allValid = false;
   756→    }
   757→  }
   758→
   759→  // Check for missing required features
   760→  const required = getModelFeatureSet().filter(n => {
   761→    const def = FEATURE_INDEX.get(n);
   762→    return def && !def.nullable && def.source !== 'derived';
   763→  });
   764→  const missing = required.filter(n => !(n in features));
   765→  // Don't fail on missing optional features, just warn
   766→  for (const m of missing) {
   767→    const def = FEATURE_INDEX.get(m);
   768→    if (def && (def.source === 'fundamental' || def.source === 'spillover' || def.source === 'market')) {
   769→      // These are nullable in practice — fundamentals/spillover might not be available
   770→      continue;
   771→    }
   772→  }
   773→
   774→  return {
   775→    valid: allValid,
   776→    featureErrors,
   777→    versionHash: computeVersionHash(),
   778→  };
   779→}

   781→/**
   782→ * Compute a hash representing the current feature schema version.
   783→ * Training and inference should compare hashes to detect mismatches.
   784→ */
   785→export function computeVersionHash(): string {
   786→  const parts = ALL_FEATURES
   787→    .filter(f => f.enabled)
   788→    .sort((a, b) => a.name.localeCompare(b.name))
   789→    .map(f => `${f.name}:v${f.version}`);
   790→
   791→  // Simple deterministic hash (not crypto, just fingerprint)
   792→  let hash = 0;
   793→  const str = parts.join('|');
   794→  for (let i = 0; i < str.length; i++) {
   795→    const char = str.charCodeAt(i);
   796→    hash = ((hash << 5) - hash) + char;
   797→    hash |= 0; // Convert to 32bit integer
   798→  }
   799→  return `fs-${Math.abs(hash).toString(16).padStart(8, '0')}`;
   800→}
