   140→function directionColor(direction: string): string {
   141→  switch (direction) {
   142→    case 'STRONG_BUY': return 'text-emerald-400';
   143→    case 'BUY': return 'text-emerald-500';
   144→    case 'NEUTRAL': return 'text-yellow-500';
   145→    case 'SELL': return 'text-red-500';
   146→    case 'STRONG_SELL': return 'text-red-400';
   147→    default: return 'text-muted-foreground';
   148→  }
   149→}
   150→
   151→function directionBg(direction: string): string {
   152→  switch (direction) {
   153→    case 'STRONG_BUY': return 'bg-emerald-500/15 border-emerald-500/30';
   154→    case 'BUY': return 'bg-emerald-500/10 border-emerald-500/20';
   155→    case 'NEUTRAL': return 'bg-yellow-500/10 border-yellow-500/20';
   156→    case 'SELL': return 'bg-red-500/10 border-red-500/20';
   157→    case 'STRONG_SELL': return 'bg-red-500/15 border-red-500/30';
   158→    default: return 'bg-muted/50 border-border/50';
   159→  }
   160→}

   162→function directionLabel(direction: string): string {
   163→  switch (direction) {

   173→function directionBadgeVariant(direction: string): 'default' | 'secondary' | 'destructive' | 'outline' {
   174→  if (direction === 'STRONG_BUY' || direction === 'BUY') return 'default';
   175→  if (direction === 'SELL' || direction === 'STRONG_SELL') return 'destructive';
   176→  return 'secondary';

   179→function scoreColor(score: number): string {
   180→  if (score > 60) return 'text-emerald-400';
   181→  if (score > 25) return 'text-emerald-500';
   182→  if (score > -25) return 'text-yellow-500';
   183→  if (score > -60) return 'text-red-500';
   184→  return 'text-red-400';
   185→}
   186→
   187→function confidenceColor(conf: number): string {
   188→  if (conf > 70) return 'text-emerald-400';
   189→  if (conf > 40) return 'text-yellow-400';
   190→  return 'text-red-400';
   191→}

   240→function PredictionScoreGauge({ score, direction }: { score: number; direction: string }) {
   241→  const radius = 54;
   242→  const circumference = 2 * Math.PI * radius;
   243→  const normalizedScore = ((score + 100) / 200) * 100;
   244→  const offset = circumference - (normalizedScore / 100) * circumference;
   245→
   246→  const ringColor = score > 25
   247→    ? 'stroke-emerald-500'
   248→    : score > -25
   249→      ? 'stroke-yellow-500'
   250→      : 'stroke-red-500';
   251→
   252→  return (
   253→    <div className="flex flex-col items-center gap-3">
   254→      <div className="relative w-36 h-36">
   255→        <svg className="w-36 h-36 -rotate-90" viewBox="0 0 120 120">
   256→          <circle
   257→            cx="60" cy="60" r={radius}
   258→            fill="none" className="stroke-muted/30" strokeWidth="8"
   259→          />
   260→          <motion.circle
   261→            cx="60" cy="60" r={radius}
   262→            fill="none" className={ringColor}
   263→            strokeWidth="8" strokeLinecap="round"
   264→            strokeDasharray={circumference}
   265→            initial={{ strokeDashoffset: circumference }}
   266→            animate={{ strokeDashoffset: offset }}
   267→            transition={{ duration: 1.2, ease: 'easeOut' }}
   268→          />
   269→        </svg>
   270→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   271→          <span className={`text-3xl font-bold ${scoreColor(score)}`}>
   272→            {score > 0 ? '+' : ''}{score}
   273→          </span>
   274→          <span className="text-[10px] text-muted-foreground">-100 — +100</span>
   275→        </div>
   276→      </div>
   277→      <Badge
   278→        variant={directionBadgeVariant(direction)}
   279→        className="text-sm px-4 py-1.5 font-semibold"
   280→      >
   281→        {directionLabel(direction)}
   282→      </Badge>
   283→    </div>
   284→  );
   285→}

   383→  const [sortField, setSortField] = useState<'score' | 'confidence' | 'symbol' | 'totalScore' | 'hybridConfidence'>('score');
   384→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc');
   385→  const [expandedSymbol, setExpandedSymbol] = useState<string | null>(null);
   386→
   387→  // AI Hybrid scan mode
   388→  const [hybridResult, setHybridResult] = useState<HybridScanResult | null>(null);
   389→  const [hybridLoading, setHybridLoading] = useState(false);
   390→  const [hybridError, setHybridError] = useState('');
   391→  const [hybridViewTab, setHybridViewTab] = useState<'top-picks' | 'top-shorts' | 'most-confident' | 'all'>('top-picks');
   392→  const [hybridSortField, setHybridSortField] = useState<'totalScore' | 'hybridConfidence' | 'symbol'>('totalScore');
   393→  const [hybridSortDir, setHybridSortDir] = useState<'asc' | 'desc'>('desc');
   394→  const [hybridExpandedSymbol, setHybridExpandedSymbol] = useState<string | null>(null);
   395→
   396→  // ─── Handlers ─────────────────────────────────────────────
   397→
   398→  const handleSinglePredict = useCallback(async () => {
   399→    if (!ticker.trim()) return;
   400→    setSingleLoading(true);
   401→    setSingleError('');
   402→    setSingleResult(null);
   403→    try {
   404→      const res = await fetch(`/api/predict/${ticker.trim().toUpperCase()}`);
   405→      const data = await res.json();
   406→      if (!res.ok) throw new Error(data.error || 'Gabim gjatë predikimit');
   407→      setSingleResult(data);
   408→    } catch (err: unknown) {
   409→      setSingleError(err instanceof Error ? err.message : 'Gabim i papritur');
   410→    } finally {
   411→      setSingleLoading(false);
   412→    }
   413→  }, [ticker]);
   414→
   415→  const handleFullScan = useCallback(async () => {
   416→    setScanLoading(true);
   417→    setScanError('');
   418→    setScanResult(null);
   419→    setScanProgress('Duke filluar skanimin...');
   420→    setExpandedSymbol(null);
   421→    try {
   422→      const res = await fetch('/api/predict-scan', {
   423→        signal: AbortSignal.timeout(300000),
   424→      });
   425→      const data = await res.json();
   426→      if (!res.ok) throw new Error(data.error || 'Gabim gjatë skanimit');

   428→    } catch (err: unknown) {
   429→      if (err instanceof DOMException && err.name === 'TimeoutError') {
   430→        setScanError('Koha i mbaroi. Provoni përsëri.');
   431→      } else {
   432→        setScanError(err instanceof Error ? err.message : 'Gabim i papritur');
   433→      }
   434→    } finally {
   435→      setScanLoading(false);
   436→      setScanProgress('');
   437→    }
   438→  }, []);

   440→  const handleSort = useCallback((field: 'score' | 'confidence' | 'symbol' | 'totalScore' | 'hybridConfidence') => {
   441→    if (sortField === field) {
   442→      setSortDir((d) => (d === 'desc' ? 'asc' : 'desc'));
   443→    } else {
   444→      setSortField(field);
   445→      setSortDir('desc');
   446→    }
   447→  }, [sortField]);
   448→
   449→  const handleHybridScan = useCallback(async () => {
   450→    setHybridLoading(true);
   451→    setHybridError('');
   452→    setHybridResult(null);
   453→    setHybridExpandedSymbol(null);
   454→    try {
   455→      const res = await fetch('/api/ai-predict-scan', {
   456→        signal: AbortSignal.timeout(300000),
   457→      });
   458→      const data = await res.json();
   459→      if (!res.ok) throw new Error(data.error || 'Gabim gjatë skanimit hibrid');

   461→    } catch (err: unknown) {

   464→      } else {
   465→        setHybridError(err instanceof Error ? err.message : 'Gabim i papritur');
   466→      }
   467→    } finally {
   468→      setHybridLoading(false);
   469→    }
   470→  }, []);
   471→
   472→  const handleHybridSort = useCallback((field: 'totalScore' | 'hybridConfidence' | 'symbol') => {
   473→    if (hybridSortField === field) {
   474→      setHybridSortDir((d) => (d === 'desc' ? 'asc' : 'desc'));
   475→    } else {
   476→      setHybridSortField(field);
   477→      setHybridSortDir('desc');
   478→    }
   479→  }, [hybridSortField]);
   480→
   481→  // ─── Derived Data ─────────────────────────────────────────
   482→
   483→  // Hybrid derived data
   484→  const currentHybridResults: HybridPredictionResult[] | null = (() => {
   485→    if (!hybridResult) return null;
   486→    const src = (() => {
   487→      switch (hybridViewTab) {
   488→        case 'top-picks': return hybridResult.topPicks;
   489→        case 'top-shorts': return hybridResult.topShorts;
   490→        case 'most-confident': return hybridResult.mostConfident;
   491→        case 'all': return hybridResult.allResults;
   492→        default: return null;
   493→      }
   494→    })();
   495→    if (!src) return null;
   496→    if (hybridViewTab !== 'all') return src;
   497→    return [...src].sort((a, b) => {
   498→      if (hybridSortField === 'symbol') {
   499→        return hybridSortDir === 'asc'
   500→          ? a.symbol.localeCompare(b.symbol)
   501→          : b.symbol.localeCompare(a.symbol);
   502→      }
   503→      const aVal = a[hybridSortField];
   504→      const bVal = b[hybridSortField];
   505→      return hybridSortDir === 'asc' ? (aVal as number) - (bVal as number) : (bVal as number) - (aVal as number);
   506→    });
   507→  })();
   508→
   509→  const currentResults: PredictionResult[] | null = (() => {
   510→    if (!scanResult) return null;
   511→    switch (viewTab) {
   512→      case 'top-picks': return scanResult.topPicks;
   513→      case 'top-shorts': return scanResult.topShorts;
   514→      case 'most-confident': return scanResult.mostConfident;
   515→      case 'all': {
   516→        const sorted = [...scanResult.allResults].sort((a, b) => {
   517→          if (sortField === 'symbol') {
   518→            return sortDir === 'asc'
   519→              ? a.symbol.localeCompare(b.symbol)
   520→              : b.symbol.localeCompare(a.symbol);
   521→          }
   522→          const aVal = sortField === 'score' ? a.score : a.confidence;
   523→          const bVal = sortField === 'score' ? b.score : b.confidence;
   524→          return sortDir === 'asc' ? aVal - bVal : bVal - aVal;
   525→        });
   526→        return sorted;
   527→      }
   528→      default: return null;
   529→    }
   530→  })();
   531→

   536→      {/* ═══════ Title ═══════ */}
   537→      <div className="flex items-center gap-2.5">
   538→        <div className="p-2 rounded-lg bg-violet-500/10 border border-violet-500/20">
   539→          <Brain className="w-5 h-5 text-violet-400" />
   540→        </div>
   541→        <div>
   542→          <h2 className="text-lg font-bold text-foreground">Predikues i Stoqeve me AI</h2>
   543→          <p className="text-xs text-muted-foreground">Analizë teknike e avancuar me më shumë se 15 indikatorë</p>
   544→        </div>
   545→      </div>
   546→
   547→      {/* ═══════ Mode Toggle ═══════ */}
   548→      <div className="flex flex-wrap gap-2">
   549→        <Button
