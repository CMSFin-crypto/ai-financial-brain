    89→// ═══ Self-contained Candlestick SVG Chart ═══
    90→// Avoids recharts Customized internal API which breaks across versions
    91→function CandlestickChart({ data }: { data: CandleData[] }) {
    92→  const chart = useMemo(() => {
    93→    if (!data || data.length === 0) return null;
    94→
    95→    const W = 800;
    96→    const H = 360;
    97→    const margin = { top: 15, right: 60, bottom: 30, left: 5 };
    98→    const priceH = H * 0.72;
    99→    const volH = H * 0.18;
   100→    const gap = H * 0.05;
   101→    const chartW = W - margin.left - margin.right;
   102→    const n = data.length;
   103→    const barSpace = chartW / n;
   104→    const bodyW = Math.max(barSpace * 0.55, 2);
   105→
   106→    // Price range
   107→    const allHigh = Math.max(...data.map(d => d.high));
   108→    const allLow = Math.min(...data.map(d => d.low));
   109→    const pricePad = (allHigh - allLow) * 0.08 || 1;
   110→    const pMin = allLow - pricePad;
   111→    const pMax = allHigh + pricePad;
   112→
   113→    const yPrice = (v: number) => margin.top + (1 - (v - pMin) / (pMax - pMin)) * priceH;
   114→
   115→    // Volume range
   116→    const maxVol = Math.max(...data.map(d => d.volume || 0)) || 1;
   117→    const volTop = margin.top + priceH + gap;
   118→    const yVol = (v: number) => volTop + volH - (v / maxVol) * volH;
   119→
   120→    // Grid lines (price)
   121→    const gridLines: { y: number; label: string }[] = [];
   122→    const steps = 5;
   123→    for (let i = 0; i <= steps; i++) {
   124→      const v = pMin + (pMax - pMin) * (i / steps);
   125→      gridLines.push({ y: yPrice(v), label: '$' + v.toFixed(1) });
   126→    }
   127→
   128→    // X-axis labels (show every N-th)
   129→    const labelEvery = n <= 10 ? 1 : n <= 20 ? 3 : 5;
   130→
   131→    const borderClr = 'hsl(240 6% 30%)';
   132→    const textClr = 'hsl(240 5% 65%)';
   133→
   134→    return (
   135→      <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-full" style={{ background: 'transparent' }}>
   136→        {/* Grid lines */}
   137→        {gridLines.map((g, i) => (
   138→          <g key={i}>
   139→            <line x1={margin.left} y1={g.y} x2={W - margin.right} y2={g.y} stroke={borderClr} strokeDasharray="3 3" strokeOpacity={0.4} />
   140→            <text x={W - margin.right + 5} y={g.y + 3} fill={textClr} fontSize={9} fontFamily="monospace">{g.label}</text>
   141→          </g>
   142→        ))}
   143→
   144→        {/* Volume separator */}
   145→        <line x1={margin.left} y1={volTop} x2={W - margin.right} y2={volTop} stroke={borderClr} strokeOpacity={0.3} />
   146→
   147→        {/* Candlesticks */}
   148→        {data.map((d, i) => {
   149→          const cx = margin.left + barSpace * i + barSpace / 2;
   150→          const isUp = d.close >= d.open;
   151→          const color = isUp ? '#10b981' : '#ef4444';
   152→          const volColor = isUp ? 'rgba(16,185,129,0.25)' : 'rgba(239,68,68,0.25)';
   153→
   154→          const bodyTop = yPrice(Math.max(d.open, d.close));
   155→          const bodyBot = yPrice(Math.min(d.open, d.close));
   156→          const bodyH = Math.max(Math.abs(bodyBot - bodyTop), 1);
   157→
   158→          const wickTop = yPrice(d.high);
   159→          const wickBot = yPrice(d.low);
   160→
   161→          const vy = yVol(d.volume || 0);
   162→          const vh = Math.max(volTop + volH - vy, 0);
   163→
   164→          return (
   165→            <g key={i}>
   166→              {/* Volume bar */}
   167→              <rect x={cx - bodyW / 2} y={vy} width={bodyW} height={vh} fill={volColor} rx={1} />
   168→              {/* Wick */}
   169→              <line x1={cx} y1={wickTop} x2={cx} y2={wickBot} stroke={color} strokeWidth={1} />
   170→              {/* Body */}
   171→              <rect x={cx - bodyW / 2} y={bodyTop} width={bodyW} height={bodyH} fill={color} stroke={color} strokeWidth={0.5} rx={1} />
   172→              {/* X label */}
   173→              {i % labelEvery === 0 && (
   174→                <text x={cx} y={H - 5} fill={textClr} fontSize={8} textAnchor="middle" fontFamily="monospace">
   175→                  {d.date.substring(5)}
   176→                </text>
   177→              )}
   178→            </g>
   179→          );
   180→        })}
   181→      </svg>
   182→    );
   183→  }, [data]);
   184→
   185→  return chart || <div className="flex items-center justify-center h-full text-muted-foreground text-sm">Asnjë të dhënë grafiku</div>;
   186→}
