    89→// ═══ Technical indicator computations ═══
    90→function computeSMA(closes: number[], period: number): (number | null)[] {
    91→  const result: (number | null)[] = [];
    92→  for (let i = 0; i < closes.length; i++) {
    93→    if (i < period - 1) { result.push(null); continue; }
    94→    let sum = 0;
    95→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
    96→    result.push(sum / period);
    97→  }
    98→  return result;
    99→}
   100→
   101→function computeEMA(closes: number[], period: number): (number | null)[] {
   102→  const k = 2 / (period + 1);
   103→  const result: (number | null)[] = [];
   104→  let ema: number | null = null;
   105→  for (let i = 0; i < closes.length; i++) {
   106→    if (i < period - 1) { result.push(null); continue; }
   107→    if (ema === null) {
   108→      let sum = 0; for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   109→      ema = sum / period;
   110→    } else {
   111→      ema = closes[i] * k + ema * (1 - k);
   112→    }
   113→    result.push(ema);
   114→  }
   115→  return result;
   116→}
   117→
   118→function computeRSI(closes: number[], period: number = 14): (number | null)[] {
   119→  const result: (number | null)[] = [];
   120→  const gains: number[] = [];
   121→  const losses: number[] = [];
   122→  for (let i = 1; i < closes.length; i++) {
   123→    const diff = closes[i] - closes[i - 1];
   124→    gains.push(diff > 0 ? diff : 0);
   125→    losses.push(diff < 0 ? -diff : 0);
   126→  }
   127→  let avgGain = 0, avgLoss = 0;
   128→  for (let i = 0; i < Math.min(period, gains.length); i++) { avgGain += gains[i]; avgLoss += losses[i]; }
   129→  avgGain /= period; avgLoss /= period;
   130→  for (let i = 0; i < period - 1; i++) result.push(null);
   131→  const rs = avgLoss === 0 ? 100 : avgGain / avgLoss;
   132→  result.push(100 - 100 / (1 + rs));
   133→  for (let i = period; i < gains.length; i++) {
   134→    avgGain = (avgGain * (period - 1) + gains[i]) / period;
   135→    avgLoss = (avgLoss * (period - 1) + losses[i]) / period;
   136→    const rs2 = avgLoss === 0 ? 100 : avgGain / avgLoss;
   137→    result.push(100 - 100 / (1 + rs2));
   138→  }
   139→  return result;
   140→}

   142→function computeMACD(closes: number[]): { macd: (number | null)[]; signal: (number | null)[]; histogram: (number | null)[] } {
   143→  const ema12 = computeEMA(closes, 12);
   144→  const ema26 = computeEMA(closes, 26);
   145→  const macdLine: (number | null)[] = [];
   146→  for (let i = 0; i < closes.length; i++) {
   147→    if (ema12[i] === null || ema26[i] === null) { macdLine.push(null); continue; }
   148→    macdLine.push(ema12[i]! - ema26[i]!);
   149→  }
   150→  // Signal = 9-period EMA of MACD line
   151→  const validMACD = macdLine.filter((v): v is number => v !== null);
   152→  const signalEma = computeEMA(validMACD, 9);
   153→  const signal: (number | null)[] = [];
   154→  let vi = 0;
   155→  for (let i = 0; i < closes.length; i++) {
   156→    if (macdLine[i] === null) { signal.push(null); continue; }
   157→    signal.push(signalEma[vi] ?? null);
   158→    vi++;
   159→  }
   160→  const histogram: (number | null)[] = [];
   161→  for (let i = 0; i < closes.length; i++) {
   162→    if (macdLine[i] === null || signal[i] === null) { histogram.push(null); continue; }
   163→    histogram.push(macdLine[i]! - signal[i]!);
   164→  }
   165→  return { macd: macdLine, signal, histogram };
   166→}

   168→function computeBollingerBands(closes: number[], period: number = 20, mult: number = 2): { upper: (number | null)[]; middle: (number | null)[]; lower: (number | null)[] } {
   169→  const middle = computeSMA(closes, period);
   170→  const upper: (number | null)[] = [];
   171→  const lower: (number | null)[] = [];
   172→  for (let i = 0; i < closes.length; i++) {
   173→    if (middle[i] === null) { upper.push(null); lower.push(null); continue; }
   174→    let sumSq = 0;
   175→    for (let j = i - period + 1; j <= i; j++) sumSq += (closes[j] - middle[i]!) ** 2;
   176→    const std = Math.sqrt(sumSq / period);
   177→    upper.push(middle[i]! + mult * std);
   178→    lower.push(middle[i]! - mult * std);
   179→  }
   180→  return { upper, middle, lower };
   181→}

   183→// ═══ Full Technical Chart with Indicators ═══
   184→function CandlestickChart({ data }: { data: CandleData[] }) {
   185→  const chart = useMemo(() => {
   186→    if (!data || data.length < 3) return null;
   187→
   188→    const W = 800;
   189→    const H = 620;
   190→    const margin = { top: 12, right: 58, bottom: 22, left: 5 };
   191→    const chartW = W - margin.left - margin.right;
   192→    const n = data.length;
   193→    const barSpace = chartW / n;
   194→    const bodyW = Math.max(barSpace * 0.55, 2);
   195→
   196→    // Panel heights
   197→    const priceH = H * 0.46;
   198→    const volH = H * 0.10;
   199→    const rsiH = H * 0.12;
   200→    const macdH = H * 0.14;
   201→    const gapH = H * 0.02;
   202→    
   203→    let yCursor = margin.top;
   204→    const priceTop = yCursor; yCursor += priceH;
   205→    const volTop = yCursor; yCursor += volH + gapH;
   206→    const rsiTop = yCursor; yCursor += rsiH;
   207→    const macdTop = yCursor; yCursor += macdH;
   208→    const xLabelY = H - 4;
   209→
   210→    // ═══ Compute indicators ═══
   211→    const closes = data.map(d => d.close);
   212→    const sma20 = computeSMA(closes, Math.min(20, n));
   213→    const sma50 = computeSMA(closes, Math.min(50, n));
   214→    const ema12 = computeEMA(closes, Math.min(12, n));
   215→    const bb = computeBollingerBands(closes, Math.min(20, n));
   216→    const rsi = computeRSI(closes, 14);
   217→    const macdData = computeMACD(closes);
   218→
   219→    // ═══ Price scale ═══
   220→    const allH = Math.max(...data.map(d => d.high), ...sma20.filter((v): v is number => v !== null) ?? [], ...bb.upper.filter((v): v is number => v !== null) ?? []);
   221→    const allL = Math.min(...data.map(d => d.low), ...sma20.filter((v): v is number => v !== null) ?? [], ...bb.lower.filter((v): v is number => v !== null) ?? []);
   222→    const pPad = (allH - allL) * 0.06 || 1;
   223→    const pMin = allL - pPad;
   224→    const pMax = allH + pPad;
   225→    const yP = (v: number) => priceTop + (1 - (v - pMin) / (pMax - pMin)) * priceH;
   226→
   227→    // ═══ Volume scale ═══
   228→    const maxVol = Math.max(...data.map(d => d.volume || 0)) || 1;
   229→    const yV = (v: number) => volTop + volH - (v / maxVol) * volH;
   230→
   231→    // ═══ RSI scale (0-100) ═══
   232→    const yR = (v: number) => rsiTop + (1 - v / 100) * rsiH;
   233→
   234→    // ═══ MACD scale ═══
   235→    const allMacd = macdData.macd.filter((v): v is number => v !== null);
   236→    const allHist = macdData.histogram.filter((v): v is number => v !== null);
   237→    const macdMax = Math.max(Math.abs(Math.min(...allMacd, ...allHist)), Math.abs(Math.max(...allMacd, ...allHist)), 0.01);
   238→    const yM = (v: number) => macdTop + macdH / 2 - (v / macdMax) * (macdH / 2);
   239→
   240→    // ═══ Helpers ═══
   241→    const xOf = (i: number) => margin.left + barSpace * i + barSpace / 2;
   242→    const labelEvery = n <= 10 ? 1 : n <= 20 ? 3 : 5;
   243→    const gridClr = 'hsl(240 6% 30%)';
   244→    const txtClr = 'hsl(240 5% 60%)';
   245→
   246→    // ═══ Polyline helper ═══
   247→    const polyline = (vals: (number | null)[], color: string, width = 1.2) => {
   248→      const pts: string[] = [];
   249→      for (let i = 0; i < vals.length; i++) {
   250→        if (vals[i] !== null) pts.push(`${xOf(i)},${vals[i]}`);
   251→      }
   252→      return pts.length > 1 ? <polyline key={color} points={pts.join(' ')} fill="none" stroke={color} strokeWidth={width} strokeOpacity={0.85} /> : null;
   253→    };
   254→
   255→    // ═══ Grid lines (price) ═══
   256→    const priceGrid: { y: number; label: string }[] = [];
   257→    for (let i = 0; i <= 4; i++) {
   258→      const v = pMin + (pMax - pMin) * (i / 4);
   259→      priceGrid.push({ y: yP(v), label: '$' + v.toFixed(1) });
   260→    }
   261→
   262→    // ═══ Bollinger Band fill ═══
   263→    const bbFill: string[] = [];
   264→    for (let i = 0; i < n; i++) {
   265→      if (bb.upper[i] !== null) bbFill.push(`${xOf(i)},${yP(bb.upper[i]!)}`);
   266→    }
   267→    for (let i = n - 1; i >= 0; i--) {
   268→      if (bb.lower[i] !== null) bbFill.push(`${xOf(i)},${yP(bb.lower[i]!)}`);
   269→    }
   270→
   271→    // ═══ RSI zones ═══
   272→    const rsiOverbought = yR(70);
   273→    const rsiOversold = yR(30);
   274→    const rsiMid = yR(50);
   275→
   276→    return (
   277→      <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-full" style={{ background: 'transparent' }}>
   278→        {/* ── PRICE PANEL ── */}
   279→        {/* Grid */}
   280→        {priceGrid.map((g, i) => (
   281→          <g key={i}>
   282→            <line x1={margin.left} y1={g.y} x2={W - margin.right} y2={g.y} stroke={gridClr} strokeDasharray="3 3" strokeOpacity={0.35} />
   283→            <text x={W - margin.right + 4} y={g.y + 3} fill={txtClr} fontSize={8.5} fontFamily="monospace">{g.label}</text>
   284→          </g>
   285→        ))}
   286→
   287→        {/* Bollinger Bands shaded area */}
   288→        {bbFill.length > 2 && (
   289→          <polygon points={bbFill.join(' ')} fill="rgba(139,92,246,0.07)" stroke="none" />
   290→        )}
   291→
   292→        {/* Bollinger Band lines */}
   293→        {polyline(bb.upper.map(v => v !== null ? yP(v) : null), '#8b5cf6', 0.8)}
   294→        {polyline(bb.middle.map(v => v !== null ? yP(v) : null), '#8b5cf6', 0.8)}
   295→        {polyline(bb.lower.map(v => v !== null ? yP(v) : null), '#8b5cf6', 0.8)}
   296→
   297→        {/* Moving Average lines */}
   298→        {polyline(sma20.map(v => v !== null ? yP(v) : null), '#f59e0b', 1.2)}
   299→        {polyline(sma50.map(v => v !== null ? yP(v) : null), '#3b82f6', 1.2)}
   300→        {polyline(ema12.map(v => v !== null ? yP(v) : null), '#ec4899', 1)}
   301→
   302→        {/* Candlesticks */}
   303→        {data.map((d, i) => {
   304→          const cx = xOf(i);
   305→          const isUp = d.close >= d.open;
   306→          const color = isUp ? '#10b981' : '#ef4444';
   307→          const bodyTop = yP(Math.max(d.open, d.close));
   308→          const bodyBot = yP(Math.min(d.open, d.close));
   309→          const bodyH = Math.max(Math.abs(bodyBot - bodyTop), 1);
   310→          return (
   311→            <g key={i}>
   312→              <line x1={cx} y1={yP(d.high)} x2={cx} y2={yP(d.low)} stroke={color} strokeWidth={1} />
   313→              <rect x={cx - bodyW / 2} y={bodyTop} width={bodyW} height={bodyH} fill={color} stroke={color} strokeWidth={0.5} rx={1} />
   314→            </g>
   315→          );
   316→        })}
   317→
   318→        {/* Panel separator */}
   319→        <line x1={margin.left} y1={volTop - 2} x2={W - margin.right} y2={volTop - 2} stroke={gridClr} strokeOpacity={0.3} />
   320→
   321→        {/* ── VOLUME PANEL ── */}
   322→        {data.map((d, i) => {
   323→          const cx = xOf(i);
   324→          const isUp = d.close >= d.open;
   325→          const vy = yV(d.volume || 0);
   326→          const vh = Math.max(volTop + volH - vy, 0);
   327→          return (
   328→            <rect key={i} x={cx - bodyW / 2} y={vy} width={bodyW} height={vh}
   329→              fill={isUp ? 'rgba(16,185,129,0.3)' : 'rgba(239,68,68,0.3)'} rx={1} />
   330→          );
   331→        })}
   332→        <text x={margin.left + 2} y={volTop + 10} fill={txtClr} fontSize={8} opacity={0.6}>VOL</text>
   333→
   334→        {/* Panel separator */}
   335→        <line x1={margin.left} y1={rsiTop - 2} x2={W - margin.right} y2={rsiTop - 2} stroke={gridClr} strokeOpacity={0.3} />
   336→
   337→        {/* ── RSI PANEL ── */}
   338→        {/* Overbought / oversold zones */}
   339→        <rect x={margin.left} y={rsiTop} width={chartW} height={yR(70) - rsiTop} fill="rgba(239,68,68,0.06)" />
   340→        <rect x={margin.left} y={yR(30)} width={chartW} height={rsiTop + rsiH - yR(30)} fill="rgba(16,185,129,0.06)" />
   341→        <line x1={margin.left} y1={rsiOverbought} x2={W - margin.right} y2={rsiOverbought} stroke={gridClr} strokeDasharray="3 3" strokeOpacity={0.4} />
   342→        <line x1={margin.left} y1={rsiOversold} x2={W - margin.right} y2={rsiOversold} stroke={gridClr} strokeDasharray="3 3" strokeOpacity={0.4} />
   343→        <line x1={margin.left} y1={rsiMid} x2={W - margin.right} y2={rsiMid} stroke={gridClr} strokeDasharray="2 4" strokeOpacity={0.2} />
   344→        <text x={W - margin.right + 4} y={rsiOverbought + 3} fill={txtClr} fontSize={7.5}>70</text>
   345→        <text x={W - margin.right + 4} y={rsiOversold + 3} fill={txtClr} fontSize={7.5}>30</text>
   346→        {polyline(rsi.map(v => v !== null ? yR(v) : null), '#a855f7', 1.5)}
   347→        <text x={margin.left + 2} y={rsiTop + 10} fill={txtClr} fontSize={8} opacity={0.6}>RSI(14)</text>
   348→
   349→        {/* Panel separator */}
   350→        <line x1={margin.left} y1={macdTop - 2} x2={W - margin.right} y2={macdTop - 2} stroke={gridClr} strokeOpacity={0.3} />
   351→
   352→        {/* ── MACD PANEL ── */}
   353→        <line x1={margin.left} y1={macdTop + macdH / 2} x2={W - margin.right} y2={macdTop + macdH / 2} stroke={gridClr} strokeDasharray="2 4" strokeOpacity={0.25} />
   354→        {/* MACD histogram */}
   355→        {macdData.histogram.map((v, i) => {
   356→          if (v === null) return null;
   357→          const cx = xOf(i);
   358→          const isPos = v >= 0;
   359→          const barY = isPos ? yM(v) : macdTop + macdH / 2;
   360→          const barH = Math.abs(yM(v) - (macdTop + macdH / 2));
   361→          return (
   362→            <rect key={i} x={cx - bodyW * 0.4} y={barY} width={bodyW * 0.8} height={Math.max(barH, 0.5)}
   363→              fill={isPos ? 'rgba(16,185,129,0.5)' : 'rgba(239,68,68,0.5)'} rx={0.5} />
   364→          );
   365→        })}
   366→        {/* MACD line & signal line */}
   367→        {polyline(macdData.macd.map(v => v !== null ? yM(v) : null), '#06b6d4', 1.3)}
   368→        {polyline(macdData.signal.map(v => v !== null ? yM(v) : null), '#f97316', 1)}
   369→        <text x={margin.left + 2} y={macdTop + 10} fill={txtClr} fontSize={8} opacity={0.6}>MACD</text>
   370→        <text x={W - margin.right + 4} y={macdTop + 10} fill="#06b6d4" fontSize={7}>MACD</text>
   371→        <text x={W - margin.right + 4} y={macdTop + 20} fill="#f97316" fontSize={7}>Signal</text>
   372→
   373→        {/* ── X-AXIS LABELS ── */}
   374→        {data.map((d, i) => {
   375→          if (i % labelEvery !== 0) return null;
   376→          return (
   377→            <text key={i} x={xOf(i)} y={xLabelY} fill={txtClr} fontSize={7.5} textAnchor="middle" fontFamily="monospace">
   378→              {d.date.substring(5)}
   379→            </text>
   380→          );
   381→        })}
   382→      </svg>
   383→    );
   384→  }, [data]);
   385→
   386→  return chart || <div className="flex items-center justify-center h-full text-muted-foreground text-sm">Asnjë të dhënë grafiku</div>;
   387→}
