   138→   138→function ensureDataDir(): void {

   140→   140→    if (!fs.existsSync(DATA_DIR)) {

   148→   148→function loadFromDisk(): void {
   149→   149→  try {
   150→   150→    ensureDataDir();
   151→   151→    if (fs.existsSync(DATA_FILE)) {
   152→   152→      const raw = fs.readFileSync(DATA_FILE, 'utf-8');
   153→   153→      const data = JSON.parse(raw);
   154→   154→      if (data.snapshot) {
   155→   155→        snapshot = data.snapshot;
   156→   156→      }
   157→   157→      if (data.records && Array.isArray(data.records)) {
   158→   158→        records = data.records.slice(-MAX_RECORDS);
   159→   159→      }
   160→   160→      console.log(`[LEARN] Loaded ${records.length} records and ${Object.keys(snapshot.indicatorAccuracies).length} indicator accuracies from disk`);
   161→   161→    }
   162→   162→  } catch (err: any) {
   163→   163→    console.log(`[LEARN] Could not load from disk: ${err.message}`);
   164→   164→    snapshot = createEmptySnapshot();
   165→   165→    records = [];
   166→   166→  }
   167→   167→  isLoaded = true;
   168→   168→}

   179→   179→    console.log(`[LEARN] Could not save to disk: ${err.message}`);
   180→   180→  }
   181→   181→}

   191→   191→
   192→   192→export function recordPrediction(data: {
   193→   193→  ticker: string;
   194→   194→  priceAtPrediction: number;
   195→   195→  technicalScores: Record<string, number>;
   196→   196→  technicalDirection: string;
   197→   197→  technicalScore: number;
   198→   198→  fundamentalScores?: Record<string, number> | null;
   199→   199→  fundamentalDirection?: string;
   200→   200→  fundamentalScore?: number;
   201→   201→  totalScore: number;
   202→   202→  direction: string;
   203→   203→  confidence: number;
   204→   204→}): void {
   205→   205→  ensureLoaded();
   206→   206→
   207→   207→  const record: IndicatorPredictionRecord = {
   208→   208→    id: `${data.ticker}_${Date.now()}`,
   209→   209→    ticker: data.ticker,
   210→   210→    timestamp: new Date().toISOString(),
   211→   211→    priceAtPrediction: data.priceAtPrediction,
   212→   212→    technicalScores: data.technicalScores,
   213→   213→    technicalDirection: data.technicalDirection,
   214→   214→    technicalScore: data.technicalScore,
   215→   215→    fundamentalScores: data.fundamentalScores || null,
   216→   216→    fundamentalDirection: data.fundamentalDirection || 'NEUTRAL',
   217→   217→    fundamentalScore: data.fundamentalScore || 0,
   218→   218→    totalScore: data.totalScore,
   219→   219→    direction: data.direction,
   220→   220→    confidence: data.confidence,
   221→   221→  };
   222→   222→
   223→   223→  records.push(record);
   224→   224→
   225→   225→  // Trim to max
   226→   226→  if (records.length > MAX_RECORDS) {
   227→   227→    records = records.slice(-MAX_RECORDS);
   228→   228→  }
   229→   229→
   230→   230→  snapshot.totalRecorded++;
   231→   231→  snapshot.lastUpdatedAt = new Date().toISOString();
   232→   232→
   233→   233→  // Save periodically (every 10 records)
   234→   234→  if (snapshot.totalRecorded % 10 === 0) {
   235→   235→    saveToDisk();
   236→   236→  }
   237→   237→
   238→   238→  console.log(`[LEARN] Recorded prediction: ${data.ticker} ${data.direction} (score: ${data.totalScore}, price: $${data.priceAtPrediction})`);
   239→   239→}

   243→   243→export function recordFromHybridResult(
   244→   244→  ticker: string,
   245→   245→  result: {
   246→   246→    score: number;
   247→   247→    direction: string;
   248→   248→    confidence: number;
   249→   249→    indicatorScores: Record<string, number>;
   250→   250→    fundamentalScore?: { score: number; signal: string; factors: Record<string, { score: number }> } | null;
   251→   251→    fundamentalAvailable?: boolean;
   252→   252→    totalScore?: number;
   253→   253→    shortTerm?: { prediction: string };
   254→   254→    mediumTerm?: { prediction: string };
   255→   255→  }
   256→   256→): void {
   257→   257→  const predPrice = 0; // Will be filled during evaluation
   258→   258→
   259→   259→  // Determine technical direction
   260→   260→  let techDir = 'SIDEWAYS';
   261→   261→  if (result.shortTerm?.prediction === 'UP' || result.mediumTerm?.prediction === 'UP') techDir = 'UP';
   262→   262→  else if (result.shortTerm?.prediction === 'DOWN' || result.mediumTerm?.prediction === 'DOWN') techDir = 'DOWN';
   263→   263→
   264→   264→  // Extract fundamental factor scores
   265→   265→  let fundScores: Record<string, number> | null = null;
   266→   266→  let fundDir = 'NEUTRAL';
   267→   267→  let fundTotalScore = 0;
   268→   268→
   269→   269→  if (result.fundamentalScore && result.fundamentalAvailable) {
   270→   270→    fundScores = {};
   271→   271→    fundDir = result.fundamentalScore.signal;
   272→   272→    fundTotalScore = result.fundamentalScore.score;
   273→   273→    for (const [key, factor] of Object.entries(result.fundamentalScore.factors)) {
   274→   274→      fundScores[key] = factor.score;
   275→   275→    }
   276→   276→  }
   277→   277→
   278→   278→  recordPrediction({
   279→   279→    ticker,
   280→   280→    priceAtPrediction: 0, // Will be updated when we have price data
   281→   281→    technicalScores: result.indicatorScores,
   282→   282→    technicalDirection: techDir,
   283→   283→    technicalScore: result.score,
   284→   284→    fundamentalScores: fundScores,
   285→   285→    fundamentalDirection: fundDir,
   286→   286→    fundamentalScore: fundTotalScore,
   287→   287→    totalScore: result.totalScore ?? result.score,
   288→   288→    direction: result.direction,
   289→   289→    confidence: result.confidence,
   290→   290→  });
   291→   291→}

   293→   293→// ─── 3. EVALUATE PREDICTIONS ────────────────────────────────
   294→   294→
   295→   295→export async function evaluatePredictions(): Promise<{
   296→   296→  evaluated: number;
   297→   297→  correct: number;
   298→   298→  newLessons: number;
   299→   299→  indicatorAccuracyUpdated: boolean;
   300→   300→}> {
   301→   301→  ensureLoaded();
   302→   302→
   303→   303→  // Find unevaluated records that are old enough
   304→   304→  const now = Date.now();
   305→   305→  const minAge = EVALUATION_MIN_DAYS * 24 * 60 * 60 * 1000;
   306→   306→  const maxAge = EVALUATION_MAX_DAYS * 24 * 60 * 60 * 1000;
   307→   307→
   308→   308→  const toEvaluate = records.filter(r => {
   309→   309→    if (r.evaluatedAt) return false;
   310→   310→    if (r.priceAtPrediction <= 0) return false;
   311→   311→    const age = now - new Date(r.timestamp).getTime();
   312→   312→    return age >= minAge && age <= maxAge;
   313→   313→  });
   314→   314→
   315→   315→  if (toEvaluate.length === 0) {
   316→   316→    return { evaluated: 0, correct: 0, newLessons: 0, indicatorAccuracyUpdated: false };
   317→   317→  }
   318→   318→
   319→   319→  // Get unique tickers
   320→   320→  const tickers = [...new Set(toEvaluate.map(r => r.ticker))];
   321→   321→  console.log(`[LEARN] Evaluating ${toEvaluate.length} predictions for ${tickers.length} tickers...`);
   322→   322→
   323→   323→  // Fetch current prices
   324→   324→  const actualPrices: Record<string, number> = {};
   325→   325→  for (let i = 0; i < tickers.length; i += 6) {
   326→   326→    const batch = tickers.slice(i, i + 6);
   327→   327→    const promises = batch.map(async (t) => {
   328→   328→      try {
   329→   329→        // Fetch 1-day data to get latest close
   330→   330→        const data = await fetchHistoricalData(t, '5d');
   331→   331→        if (data && data.length > 0) {
   332→   332→          return { ticker: t, price: data[data.length - 1].close };
   333→   333→        }
   334→   334→      } catch {
   335→   335→        // ignore
   336→   336→      }
   337→   337→      return null;
   338→   338→    });
   339→   339→    const results = await Promise.allSettled(promises);
   340→   340→    for (const r of results) {
   341→   341→      if (r.status === 'fulfilled' && r.value) {
   342→   342→        actualPrices[r.value.ticker] = r.value.price;
   343→   343→      }
   344→   344→    }
   345→   345→    if (i + 6 < tickers.length) {
   346→   346→      await new Promise(resolve => setTimeout(resolve, 500));
   347→   347→    }
   348→   348→  }
   349→   349→
   350→   350→  // Evaluate each prediction
   351→   351→  let evaluated = 0;
   352→   352→  let correct = 0;
   353→   353→  let newLessons = 0;
   354→   354→
   355→   355→  for (const record of toEvaluate) {
   356→   356→    const actual = actualPrices[record.ticker];
   357→   357→    if (!actual) continue;
   358→   358→
   359→   359→    const change = ((actual - record.priceAtPrediction) / record.priceAtPrediction) * 100;
   360→   360→    record.actualPrice = actual;
   361→   361→    record.actualChangePercent = Math.round(change * 100) / 100;
   362→   362→    record.evaluatedAt = new Date().toISOString();
   363→   363→
   364→   364→    // Was the overall prediction correct?
   365→   365→    const predictedUp = record.direction === 'STRONG_BUY' || record.direction === 'BUY' ||
   366→   366→                       record.totalScore > 15;
   367→   367→    const predictedDown = record.direction === 'STRONG_SELL' || record.direction === 'SELL' ||
   368→   368→                         record.totalScore < -15;
   369→   369→    const actualUp = change > 0.5;
   370→   370→    const actualDown = change < -0.5;
   371→   371→
   372→   372→    record.wasCorrect = (predictedUp && actualUp) || (predictedDown && actualDown) ||
   373→   373→                        (!predictedUp && !predictedDown && Math.abs(change) <= 0.5);
   374→   374→
   375→   375→    if (record.wasCorrect) correct++;
   376→   376→
   377→   377→    // Per-indicator correctness
   378→   378→    record.perIndicatorCorrect = {};
   379→   379→
   380→   380→    // Technical indicators
   381→   381→    for (const [indicator, score] of Object.entries(record.technicalScores)) {
   382→   382→      const predictedBullish = score > 10;
   383→   383→      const predictedBearish = score < -10;
   384→   384→      const indicatorCorrect = (predictedBullish && actualUp) || (predictedBearish && actualDown) ||
   385→   385→                               (!predictedBullish && !predictedBearish);
   386→   386→      record.perIndicatorCorrect[indicator] = indicatorCorrect;
   387→   387→    }
   388→   388→
   389→   389→    // Fundamental factors
   390→   390→    if (record.fundamentalScores) {
   391→   391→      for (const [factor, score] of Object.entries(record.fundamentalScores)) {
   392→   392→        const predictedBullish = score > 10;
   393→   393→        const predictedBearish = score < -10;
   394→   394→        const factorCorrect = (predictedBullish && actualUp) || (predictedBearish && actualDown) ||
   395→   395→                              (!predictedBullish && !predictedBearish);
   396→   396→        record.perIndicatorCorrect[`fund_${factor}`] = factorCorrect;
   397→   397→      }
   398→   398→    }
   399→   399→
   400→   400→    evaluated++;
   401→   401→  }
   402→   402→
   403→   403→  // Recalculate indicator accuracies
   404→   404→  recalculateIndicatorAccuracies();
   405→   405→
   406→   406→  // Extract lessons
   407→   407→  newLessons = extractLessons();
   408→   408→
   409→   409→  // Update snapshot
   410→   410→  snapshot.totalEvaluated = records.filter(r => r.evaluatedAt).length;
   411→   411→  snapshot.totalCorrect = records.filter(r => r.wasCorrect).length;
   412→   412→  snapshot.overallAccuracy = snapshot.totalEvaluated > 0
   413→   413→    ? snapshot.totalCorrect / snapshot.totalEvaluated
   414→   414→    : 0;
   415→   415→  snapshot.lastEvaluatedAt = new Date().toISOString();
   416→   416→  snapshot.lastUpdatedAt = new Date().toISOString();
   417→   417→
   418→   418→  // Calculate weight multipliers
   419→   419→  calculateWeightMultipliers();
   420→   420→
   421→   421→  // Save
   422→   422→  saveToDisk();
   423→   423→
   424→   424→  console.log(`[LEARN] Evaluation complete: ${evaluated} evaluated, ${correct} correct (${((correct/evaluated)*100).toFixed(1)}%), ${newLessons} lessons`);
   425→   425→
   426→   426→  return { evaluated, correct, newLessons, indicatorAccuracyUpdated: true };
   427→   427→}

   429→   429→// ─── 4. RECALCULATE INDICATOR ACCURACIES ────────────────────
   430→   430→
   431→   431→function recalculateIndicatorAccuracies(): void {
   432→   432→  const evaluated = records.filter(r => r.evaluatedAt && r.perIndicatorCorrect);
   433→   433→  if (evaluated.length < 5) {
   434→   434→    console.log(`[LEARN] Not enough evaluated records (${evaluated.length}) to calculate accuracies`);
   435→   435→    return;
   436→   436→  }
   437→   437→
   438→   438→  // Technical indicators
   439→   439→  const techIndicators = [
   440→   440→    'rsi', 'macdHistogram', 'bollinger', 'maTrend', 'stochastic',
   441→   441→    'adx', 'atr', 'roc', 'obv', 'volumeConfirm',
   442→   442→    'macdCrossover', 'priceChannel', 'divergence', 'vwap', 'pattern',
   443→   443→  ];
   444→   444→
   445→   445→  for (const ind of techIndicators) {
   446→   446→    const relevantRecords = evaluated.filter(r => r.technicalScores[ind] !== undefined && r.perIndicatorCorrect?.[ind] !== undefined);
   447→   447→    if (relevantRecords.length < 3) continue;
   448→   448→
   449→   449→    const correctCount = relevantRecords.filter(r => r.perIndicatorCorrect?.[ind]).length;
   450→   450→    const correctRecords = relevantRecords.filter(r => r.perIndicatorCorrect?.[ind]);
   451→   451→    const wrongRecords = relevantRecords.filter(r => !r.perIndicatorCorrect?.[ind]);
   452→   452→
   453→   453→    const avgScoreCorrect = correctRecords.length > 0
   454→   454→      ? correctRecords.reduce((s, r) => s + (r.technicalScores[ind] || 0), 0) / correctRecords.length
   455→   455→      : 0;
   456→   456→    const avgScoreWrong = wrongRecords.length > 0
   457→   457→      ? wrongRecords.reduce((s, r) => s + (r.technicalScores[ind] || 0), 0) / wrongRecords.length
   458→   458→      : 0;
   459→   459→
   460→   460→    snapshot.indicatorAccuracies[ind] = {
   461→   461→      indicator: ind,
   462→   462→      totalPredictions: relevantRecords.length,
   463→   463→      correctPredictions: correctCount,
   464→   464→      accuracy: correctCount / relevantRecords.length,
   465→   465→      avgScoreWhenCorrect: Math.round(avgScoreCorrect * 100) / 100,
   466→   466→      avgScoreWhenWrong: Math.round(avgScoreWrong * 100) / 100,
   467→   467→      weight: 0, // Will be calculated in calculateWeightMultipliers
   468→   468→      type: 'technical',
   469→   469→      description: INDICATOR_NAMES[ind] || ind,
   470→   470→    };
   471→   471→  }

   473→   473→  // Fundamental factors
   474→   474→  const fundFactors = ['valuation', 'growth', 'profitability', 'financialHealth', 'analystConsensus'];
   475→   475→  for (const factor of fundFactors) {
   476→   476→    const key = `fund_${factor}`;
   477→   477→    const relevantRecords = evaluated.filter(r => r.fundamentalScores?.[factor] !== undefined && r.perIndicatorCorrect?.[key] !== undefined);
   478→   478→    if (relevantRecords.length < 3) continue;
   479→   479→
   480→   480→    const correctCount = relevantRecords.filter(r => r.perIndicatorCorrect?.[key]).length;
   481→   481→    const correctRecords = relevantRecords.filter(r => r.perIndicatorCorrect?.[key]);
   482→   482→    const wrongRecords = relevantRecords.filter(r => !r.perIndicatorCorrect?.[key]);
   483→   483→
   484→   484→    const avgScoreCorrect = correctRecords.length > 0
   485→   485→      ? correctRecords.reduce((s, r) => s + (r.fundamentalScores?.[factor] || 0), 0) / correctRecords.length
   486→   486→      : 0;
   487→   487→    const avgScoreWrong = wrongRecords.length > 0
   488→   488→      ? wrongRecords.reduce((s, r) => s + (r.fundamentalScores?.[factor] || 0), 0) / wrongRecords.length
   489→   489→      : 0;
   490→   490→
   491→   491→    snapshot.indicatorAccuracies[key] = {
   492→   492→      indicator: factor,
   493→   493→      totalPredictions: relevantRecords.length,
   494→   494→      correctPredictions: correctCount,
   495→   495→      accuracy: correctCount / relevantRecords.length,
   496→   496→      avgScoreWhenCorrect: Math.round(avgScoreCorrect * 100) / 100,
   497→   497→      avgScoreWhenWrong: Math.round(avgScoreWrong * 100) / 100,
   498→   498→      weight: 0,
   499→   499→      type: 'fundamental',
   500→   500→      description: INDICATOR_NAMES[factor] || factor,
   501→   501→    };
   502→   502→  }
   503→   503→
   504→   504→  // Per-ticker accuracy
   505→   505→  const tickerGroups: Record<string, { total: number; correct: number }> = {};
   506→   506→  for (const r of evaluated) {
   507→   507→    if (!tickerGroups[r.ticker]) tickerGroups[r.ticker] = { total: 0, correct: 0 };
   508→   508→    tickerGroups[r.ticker].total++;
   509→   509→    if (r.wasCorrect) tickerGroups[r.ticker].correct++;
   510→   510→  }
   511→   511→  for (const [ticker, stats] of Object.entries(tickerGroups)) {
   512→   512→    snapshot.tickerAccuracy[ticker] = {
   513→   513→      total: stats.total,
   514→   514→      correct: stats.correct,
   515→   515→      accuracy: stats.total > 0 ? stats.correct / stats.total : 0,
   516→   516→    };
   517→   517→  }
   518→   518→}

   520→   520→// ─── 5. CALCULATE WEIGHT MULTIPLIERS ───────────────────────
   521→   521→
   522→   522→function calculateWeightMultipliers(): void {
   523→   523→  const accuracies = snapshot.indicatorAccuracies;
   524→   524→  const entries = Object.values(accuracies);
   525→   525→
   526→   526→  if (entries.length === 0) return;
   527→   527→
   528→   528→  // Calculate multipliers based on accuracy
   529→   529→  // Formula: multiplier = 0.5 + accuracy (so range is 0.5 to 1.5)
   530→   530→  // If accuracy < 0.4, reduce weight (multiplier < 0.9)
   531→   531→  // If accuracy > 0.6, increase weight (multiplier > 1.1)
   532→   532→  // If accuracy = 0.5, keep same (multiplier = 1.0)
   533→   533→
   534→   534→  for (const entry of entries) {
   535→   535→    const acc = entry.accuracy;
   536→   536→    // Sigmoid-like adjustment centered at 0.5
   537→   537→    const multiplier = 0.5 + acc;
   538→   538→    // Clamp between 0.3 (very unreliable) and 1.7 (very reliable)
   539→   539→    entry.weight = Math.round(Math.max(0.3, Math.min(1.7, multiplier)) * 100) / 100;
   540→   540→    snapshot.weightMultipliers[entry.indicator] = entry.weight;
   541→   541→  }
   542→   542→
   543→   543→  // Normalize weights so average is 1.0
   544→   544→  const avgMultiplier = entries.reduce((s, e) => s + e.weight, 0) / entries.length;
   545→   545→  if (avgMultiplier > 0) {
   546→   546→    for (const entry of entries) {
   547→   547→      entry.weight = Math.round((entry.weight / avgMultiplier) * 100) / 100;
   548→   548→      snapshot.weightMultipliers[entry.indicator] = entry.weight;
   549→   549→    }
   550→   550→  }
   551→   551→
   552→   552→  console.log(`[LEARN] Weight multipliers calculated for ${entries.length} indicators (avg accuracy: ${(entries.reduce((s,e) => s + e.accuracy, 0) / entries.length * 100).toFixed(1)}%)`);
   553→   553→}

   573→   573→    priceChannel: 0.05,
   574→   574→    divergence: 0.06,

   576→   576→    pattern: 0.05,

   578→   578→
   579→   579→  // If no learning data yet, return base weights

   582→   582→  }
   583→   583→
   584→   584→  // Apply multipliers
   585→   585→  const adjusted: Record<string, number> = {};
   586→   586→  for (const [key, baseWeight] of Object.entries(baseWeights)) {
   587→   587→    const multiplier = snapshot.weightMultipliers[key] ?? 1.0;
   588→   588→    adjusted[key] = baseWeight * multiplier;
   589→   589→  }
   590→   590→
   591→   591→  // Normalize so total = 1.0
   592→   592→  const total = Object.values(adjusted).reduce((s, w) => s + w, 0);
   593→   593→  if (total > 0) {
   594→   594→    for (const key of Object.keys(adjusted)) {
   595→   595→      adjusted[key] = Math.round((adjusted[key] / total) * 1000) / 1000;
   596→   596→    }
   597→   597→  }
   598→   598→
   599→   599→  return adjusted;
   600→   600→}

   637→   637→// ─── 8. EXTRACT LESSONS ────────────────────────────────────
   638→   638→
   639→   639→function extractLessons(): number {
   640→   640→  const recentEvaluated = records
   641→   641→    .filter(r => r.evaluatedAt && !r.wasCorrect && Math.abs(r.actualChangePercent || 0) > 2)
   642→   642→    .slice(-20);
   643→   643→
   644→   644→  let newLessons = 0;
   645→   645→
   646→   646→  for (const record of recentEvaluated) {
   647→   647→    if (!record.perIndicatorCorrect || !record.actualChangePercent) continue;
   648→   648→
   649→   649→    // Find which indicators were most wrong
   650→   650→    const wrongIndicators: Array<{ name: string; score: number; wasPredicted: string; actual: string }> = [];
   651→   651→
   652→   652→    for (const [ind, correct] of Object.entries(record.perIndicatorCorrect)) {
   653→   653→      if (correct) continue;
   654→   654→
   655→   655→      const isFund = ind.startsWith('fund_');
   656→   656→      const indKey = isFund ? ind.replace('fund_', '') : ind;
   657→   657→      const scores = isFund ? record.fundamentalScores : record.technicalScores;
   658→   658→      const score = scores?.[indKey] ?? 0;
   659→   659→
   660→   660→      const predicted = score > 10 ? 'LARTË' : score < -10 ? 'POSHTË' : 'NEUTRAL';
   661→   661→      const actual = (record.actualChangePercent || 0) > 0 ? 'LARTË' : 'POSHTË';
   662→   662→
   663→   663→      wrongIndicators.push({
   664→   664→        name: INDICATOR_NAMES[indKey] || indKey,
   665→   665→        score: Math.round(score * 10) / 10,
   666→   666→        wasPredicted: predicted,
   667→   667→        actual,
   668→   668→      });
   669→   669→    }
   670→   670→
   671→   671→    if (wrongIndicators.length === 0) continue;
   672→   672→
   673→   673→    // Sort by absolute score (most confident wrong indicators first)
   674→   674→    wrongIndicators.sort((a, b) => Math.abs(b.score) - Math.abs(a.score));
   675→   675→
   676→   676→    // Find if similar lesson exists
   677→   677→    const topWrong = wrongIndicators[0];
   678→   678→    const category = record.actualChangePercent < -3 ? 'wrong_direction' :
   679→   679→                     record.actualChangePercent > 3 ? 'volatility' : 'timing';
   680→   680→
   681→   681→    const lessonMessage = `${record.ticker}: ${topWrong.name} parashikoi ${topWrong.wasPredicted} (pikë: ${topWrong.score}) por çmimi shkoi ${topWrong.actual} me ${Math.abs(record.actualChangePercent).toFixed(1)}%. ${wrongIndicators.length} indikatorë gabuan.`;
   682→   682→
   683→   683→    const severity = Math.min(Math.ceil(Math.abs(record.actualChangePercent) / 3), 5);
   684→   684→
   685→   685→    // Check for duplicate
   686→   686→    const isDuplicate = snapshot.lessons.some(l =>
   687→   687→      l.category === category &&
   688→   688→      l.message.includes(record.ticker) &&
   689→   689→      (Date.now() - new Date(l.createdAt).getTime()) < 7 * 24 * 60 * 60 * 1000
   690→   690→    );
   691→   691→
   692→   692→    if (!isDuplicate) {
   693→   693→      snapshot.lessons.unshift({
   694→   694→        id: `lesson_${Date.now()}_${record.ticker}`,
   695→   695→        category,
   696→   696→        message: lessonMessage,
   697→   697→        severity,
   698→   698→        createdAt: new Date().toISOString(),
   699→   699→      });
   700→   700→      newLessons++;
   701→   701→
   702→   702→      // Keep max 50 lessons
   703→   703→      if (snapshot.lessons.length > 50) {
   704→   704→        snapshot.lessons = snapshot.lessons.slice(0, 50);
   705→   705→      }
   706→   706→    }
   707→   707→  }
   708→   708→
   709→   709→  return newLessons;
   710→   710→}
