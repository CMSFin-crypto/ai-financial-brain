     7→     7→import { Badge } from '@/components/ui/badge';

     9→     9→import {

    19→    19→} from 'recharts';
    20→    20→import {
    21→    21→  BarChart3,
    22→    22→  Activity,
    23→    23→  Search,

    32→    32→} from 'lucide-react';

    63→    63→  confidence: number;

    93→    93→  candlestickData: CandleData[];
    94→    94→  summary: string;

    97→    97→
    98→    98→export function TechnicalAnalysis() {
    99→    99→  const [ticker, setTicker] = useState('');
   100→   100→  const [searchQuery, setSearchQuery] = useState('');
   101→   101→  const [isLoading, setIsLoading] = useState(false);
   102→   102→  const [analysis, setAnalysis] = useState<TechnicalAnalysisResult | null>(null);
   103→   103→  const [error, setError] = useState<string | null>(null);
   104→   104→  const [timeframe, setTimeframe] = useState('1mo');
   105→   105→
   106→   106→  const timeframes = [
   107→   107→    { value: '1d', label: '1D' },
   108→   108→    { value: '5d', label: '5D' },
   109→   109→    { value: '1mo', label: '1Mo' },
   110→   110→    { value: '3mo', label: '3Mo' },
   111→   111→    { value: '6mo', label: '6Mo' },
   112→   112→    { value: '1y', label: '1V' },
   113→   113→    { value: '5y', label: '5V' },
   114→   114→  ];
   115→   115→
   116→   116→  const runAnalysisForTicker = async (tickerSymbol?: string) => {
   117→   117→    const sym = (tickerSymbol || ticker).trim().toUpperCase();
   118→   118→    if (!sym) return;
   119→   119→    setTicker(sym);
   120→   120→    setIsLoading(true);
   121→   121→    setError(null);
   122→   122→    setAnalysis(null);
   123→   123→
   124→   124→    try {
   125→   125→      const res = await fetch('/api/technical-analysis', {
   126→   126→        method: 'POST',
   127→   127→        headers: { 'Content-Type': 'application/json' },
   128→   128→        body: JSON.stringify({ ticker: sym, range: timeframe }),
   129→   129→      });
   130→   130→      const data = await res.json();
   131→   131→
   132→   132→      if (!res.ok) {
   133→   133→        setError(data.error || 'Analiza dështoi');
   134→   134→        return;
   135→   135→      }
   136→   136→      setAnalysis(data.analysis);
   137→   137→    } catch {
   138→   138→      setError('Gabim rrjeti. Provo përsëri.');
   139→   139→    } finally {
   140→   140→      setIsLoading(false);
   141→   141→    }
   142→   142→  };
   143→   143→
   144→   144→  const runAnalysis = () => runAnalysisForTicker();
   145→   145→
   146→   146→  const getSignalIcon = (signal: string) => {
   147→   147→    const s = signal.toLowerCase();
   148→   148→    if (s.includes('bullish') || s.includes('buy') || s === 'overbought')
   149→   149→      return <TrendingUp className="w-3.5 h-3.5 text-emerald-400" />;
   150→   150→    if (s.includes('bearish') || s.includes('sell') || s === 'oversold')
   151→   151→      return <TrendingDown className="w-3.5 h-3.5 text-red-400" />;
   152→   152→    return <Minus className="w-3.5 h-3.5 text-amber-400" />;
   153→   153→  };
   154→   154→
   155→   155→  const getSignalColor = (signal: string) => {
   156→   156→    const s = signal.toLowerCase();
   157→   157→    if (s.includes('bullish') || s.includes('buy')) return 'text-emerald-400';
   158→   158→    if (s.includes('bearish') || s.includes('sell')) return 'text-red-400';
   159→   159→    return 'text-amber-400';
   160→   160→  };
   161→   161→
   162→   162→  const getSignalBg = (signal: string) => {
   163→   163→    const s = signal.toLowerCase();
   164→   164→    if (s.includes('bullish') || s.includes('buy')) return 'bg-emerald-500';
   165→   165→    if (s.includes('bearish') || s.includes('sell')) return 'bg-red-500';
   166→   166→    return 'bg-amber-500';
   167→   167→  };
   168→   168→
   169→   169→  const price = analysis?.priceAnalysis?.currentPrice;
   170→   170→  const priceChange = analysis?.priceAnalysis?.priceChange;
   171→   171→
   172→   172→  return (
   173→   173→    <div className="space-y-4">
   174→   174→      {/* Timeframe Selector */}
   175→   175→      <div className="flex items-center gap-1.5">
   176→   176→        {timeframes.map((tf) => (
   177→   177→          <button
   178→   178→            key={tf.value}
   179→   179→            onClick={() => setTimeframe(tf.value)}
   180→   180→            className={`text-xs px-2.5 py-1 rounded-md transition-colors font-medium ${
   181→   181→              timeframe === tf.value
   182→   182→                ? 'bg-emerald-600 text-white'
   183→   183→                : 'bg-muted/50 text-muted-foreground hover:bg-muted'
   184→   184→            }`}
   185→   185→          >
   186→   186→            {tf.label}
   187→   187→          </button>
   188→   188→        ))}
   189→   189→      </div>
   190→   190→
   191→   191→      {/* Search Bar */}
   192→   192→      <div className="flex gap-2">
   193→   193→        <StockSearch
   194→   194→          onSelect={(t) => runAnalysisForTicker(t)}
   195→   195→          onQueryChange={(q) => setSearchQuery(q)}
   196→   196→          placeholder="Kërko ticker-in... AAPL, VRT, GLW"
   197→   197→          className="flex-1"
   198→   198→          inputClassName="h-10 text-sm"
   199→   199→        />
   200→   200→        <Button
   201→   201→          onClick={() => runAnalysisForTicker(searchQuery || ticker)}
   202→   202→          disabled={isLoading || !(searchQuery.trim() || ticker.trim())}
   203→   203→          className="bg-emerald-600 hover:bg-emerald-700 text-white"
   204→   204→        >
   205→   205→          {isLoading ? <Loader2 className="w-4 h-4 animate-spin" /> : <BarChart3 className="w-4 h-4 mr-1.5" />}
   206→   206→          Analizo
   207→   207→        </Button>
   208→   208→      </div>
   209→   209→
   210→   210→      {/* Error */}
   211→   211→      {error && (
   212→   212→        <div className="text-sm text-red-500 bg-red-500/10 border border-red-500/20 rounded-lg p-3">
   213→   213→          {error}
   214→   214→        </div>
   215→   215→      )}
   216→   216→
   217→   217→      {/* Loading */}
   218→   218→      {isLoading && (
   219→   219→        <div className="space-y-4">
   220→   220→          <Skeleton className="h-[120px] rounded-xl" />
   221→   221→          <Skeleton className="h-[300px] rounded-xl" />
   222→   222→          <div className="grid grid-cols-2 md:grid-cols-3 gap-3">
   223→   223→            <Skeleton className="h-[180px] rounded-xl" />
   224→   224→            <Skeleton className="h-[180px] rounded-xl" />
   225→   225→            <Skeleton className="h-[180px] rounded-xl" />
   226→   226→          </div>
   227→   227→        </div>
   228→   228→      )}
   229→   229→
   230→   230→      {/* Results */}
   231→   231→      {analysis && !isLoading && (
   232→   232→        <div className="space-y-4">
   233→   233→          {/* ═══ HEADER — Ticker, Price, Signal ═══ */}
   234→   234→          <Card className="border-border/50 bg-card/50">
   235→   235→            <CardContent className="pt-5">
   236→   236→              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   237→   237→                <div>
   238→   238→                  <div className="flex items-center gap-2.5">
   239→   239→                    <h3 className="text-2xl font-bold">{analysis.ticker}</h3>
   240→   240→                    <Badge className={`${getSignalBg(analysis.overallSignal)} text-white font-semibold`}>
   241→   241→                      {analysis.overallSignal}
   242→   242→                    </Badge>
   243→   243→                    {analysis.isDemo && (
   244→   244→                      <Badge variant="outline" className="text-[9px] text-muted-foreground border-muted">
   245→   245→                        <AlertTriangle className="w-2.5 h-2.5 mr-0.5" /> Demo
   246→   246→                      </Badge>
   247→   247→                    )}
   248→   248→                  </div>
   249→   249→                  <p className="text-sm text-muted-foreground mt-0.5">{analysis.company}</p>
   250→   250→                  {analysis.sector && (
   251→   251→                    <p className="text-[11px] text-muted-foreground">{analysis.sector}</p>
   252→   252→                  )}
   253→   253→                </div>
   254→   254→
   255→   255→                {/* PRICE DISPLAY — Prominent */}
   256→   256→                <div className="flex items-center gap-5">
   257→   257→                  <div className="text-right">
   258→   258→                    <div className="flex items-center gap-1.5 justify-end">
   259→   259→                      <DollarSign className="w-4 h-4 text-muted-foreground" />
   260→   260→                      <p className="text-3xl font-bold tabular-nums">
   261→   261→                        {price ? price.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 }) : '—'}
   262→   262→                      </p>
   263→   263→                    </div>
   264→   264→                    {priceChange !== undefined && priceChange !== 0 && (
   265→   265→                      <p className={`text-sm font-medium tabular-nums ${priceChange >= 0 ? 'text-emerald-500' : 'text-red-500'}`}>
   266→   266→                        {priceChange >= 0 ? '▲' : '▼'} {priceChange >= 0 ? '+' : ''}{priceChange.toFixed(2)}%
   267→   267→                      </p>
   268→   268→                    )}
   269→   269→                  </div>
   270→   270→                  <div className="text-right">
   271→   271→                    <p className="text-[10px] text-muted-foreground uppercase tracking-wider">Besueshmëria</p>
   272→   272→                    <p className="text-2xl font-bold">{analysis.confidence}<span className="text-sm text-muted-foreground">%</span></p>
   273→   273→                  </div>
   274→   274→                  <div className="text-right">
   275→   275→                    <p className="text-[10px] text-muted-foreground uppercase tracking-wider">Trendi</p>
   276→   276→                    <div className="flex items-center gap-1 justify-end">
   277→   277→                      {getSignalIcon(analysis.priceAnalysis?.trend || '')}
   278→   278→                      <span className="text-sm font-medium capitalize">{analysis.priceAnalysis?.trend}</span>
   279→   279→                    </div>
   280→   280→                    <p className="text-[10px] text-muted-foreground capitalize">
   281→   281→                      {analysis.priceAnalysis?.trendStrength}
   282→   282→                    </p>
   283→   283→                  </div>
   284→   284→                </div>
   285→   285→              </div>
   286→   286→            </CardContent>
   287→   287→          </Card>
   288→   288→
   289→   289→          {/* ═══ CHART — Candlestick + Volume ═══ */}
   290→   290→          {analysis.candlestickData && analysis.candlestickData.length > 0 && (
   291→   291→            <Card className="border-border/50 bg-card/50">
   292→   292→              <CardHeader className="pb-2">
   293→   293→                <CardTitle className="text-sm font-medium flex items-center gap-2">
   294→   294→                  <Activity className="w-4 h-4 text-emerald-500" />
   295→   295→                  Candlestick Chart ({timeframe}) + Volumi
   296→   296→                  {analysis.isRealChart && (
   297→   297→                    <Badge variant="outline" className="text-[9px] border-emerald-500/30 text-emerald-400">Real Data</Badge>
   298→   298→                  )}
   299→   299→                </CardTitle>
   300→   300→              </CardHeader>
   301→   301→              <CardContent>
   302→   302→                <div className="h-[360px]">
   303→   303→                  <ResponsiveContainer width="100%" height="100%">
   304→   304→                    <ComposedChart data={analysis.candlestickData} margin={{ top: 10, right: 60, left: 0, bottom: 0 }}>
   305→   305→                      <CartesianGrid strokeDasharray="3 3" stroke="hsl(var(--border))" />
   306→   306→                      <XAxis dataKey="date" tick={{ fontSize: 9 }} tickFormatter={(v: string) => v.substring(5)} />
   307→   307→                      <YAxis yAxisId="price" domain={['auto', 'auto']} tick={{ fontSize: 10 }} />
   308→   308→                      <YAxis yAxisId="volume" orientation="right" tick={{ fontSize: 9 }} />
   309→   309→                      <Tooltip
   310→   310→                        contentStyle={{
   311→   311→                          background: 'hsl(var(--card))',
   312→   312→                          border: '1px solid hsl(var(--border))',
   313→   313→                          borderRadius: '8px',
   314→   314→                          fontSize: '11px',
   315→   315→                        }}
   316→   316→                        formatter={(value: number, name: string) => {
   317→   317→                          if (name === 'Volumi') return [`${(value / 1e6).toFixed(1)}M`, name];
   318→   318→                          return [`$${value.toFixed(2)}`, name];
   319→   319→                        }}
   320→   320→                        content={({ payload, label }) => {
   321→   321→                          if (!payload || payload.length === 0) return null;
   322→   322→                          const d = payload[0]?.payload;
   323→   323→                          if (!d) return null;
   324→   324→                          const isUp = d.close >= d.open;
   325→   325→                          return (
   326→   326→                            <div className="bg-card border border-border rounded-lg p-2.5 shadow-lg text-[11px]">
   327→   327→                              <p className="font-semibold mb-1">{label}</p>
   328→   328→                              <div className="grid grid-cols-2 gap-x-4 gap-y-0.5">
   329→   329→                                <span className="text-muted-foreground">Hapurje:</span>
   330→   330→                                <span className="text-right font-mono">${d.open?.toFixed(2)}</span>
   331→   331→                                <span className="text-muted-foreground">Mbyllje:</span>
   332→   332→                                <span className={`text-right font-mono ${isUp ? 'text-emerald-500' : 'text-red-500'}`}>${d.close?.toFixed(2)}</span>
   333→   333→                                <span className="text-muted-foreground">High:</span>
   334→   334→                                <span className="text-right font-mono text-emerald-400">${d.high?.toFixed(2)}</span>
   335→   335→                                <span className="text-muted-foreground">Low:</span>
   336→   336→                                <span className="text-right font-mono text-red-400">${d.low?.toFixed(2)}</span>
   337→   337→                                <span className="text-muted-foreground">Volumi:</span>
   338→   338→                                <span className="text-right font-mono">{((d.volume || 0) / 1e6).toFixed(1)}M</span>
   339→   339→                              </div>
   340→   340→                            </div>
   341→   341→                          );
   342→   342→                        }}
   343→   343→                      />
   344→   344→                      <Legend wrapperStyle={{ fontSize: '11px' }} />
   345→   345→                      <Customized component={(props: any) => {
   346→   346→                        const { formattedGraphicalItems, xAxisMap, yAxisMap, offset } = props;
   347→   347→                        if (!formattedGraphicalItems || !xAxisMap || !yAxisMap) return null;
   348→   348→
   349→   349→                        const xAxis = Object.values(xAxisMap)[0] as any;
   350→   350→                        const priceAxis = Object.values(yAxisMap)[0] as any;
   351→   351→                        const volumeAxis = Object.values(yAxisMap)[1] as any;
   352→   352→                        if (!xAxis || !priceAxis || !volumeAxis) return null;
   353→   353→
   354→   354→                        const data = analysis.candlestickData;
   355→   355→                        const bandwidth = Math.max(
   356→   356→                          (xAxis.bandSize || ((offset.width - 40) / data.length)) * 0.6,
   357→   357→                          2
   358→   358→                        );
   359→   359→                        const barWidth = Math.max(bandwidth * 0.65, 1.5);
   360→   360→                        const scaleX = xAxis.scale || xAxis.xScale;
   361→   361→                        if (!scaleX) return null;
   362→   362→
   363→   363→                        const priceScale = priceAxis.scale;
   364→   364→                        const volumeScale = volumeAxis.scale;
   365→   365→
   366→   366→                        return (
   367→   367→                          <g>
   368→   368→                            {data.map((d, i) => {
   369→   369→                              const x = scaleX(d.date) - bandwidth / 2;
   370→   370→                              const isUp = d.close >= d.open;
   371→   371→
   372→   372→                              // Candlestick body
   373→   373→                              const bodyTop = priceScale(Math.max(d.open, d.close));
   374→   374→                              const bodyBottom = priceScale(Math.min(d.open, d.close));
   375→   375→                              const bodyHeight = Math.max(Math.abs(bodyBottom - bodyTop), 1);
   376→   376→
   377→   377→                              // Wick (high to low)
   378→   378→                              const wickTop = priceScale(d.high);
   379→   379→                              const wickBottom = priceScale(d.low);
   380→   380→
   381→   381→                              // Volume bar
   382→   382→                              const volHeight = Math.max((volumeScale(d.volume) || 0), 0);
   383→   383→                              const volBottom = volumeScale(0) || offset.top + offset.height;
   384→   384→                              const volColor = isUp ? 'rgba(16, 185, 129, 0.25)' : 'rgba(239, 68, 68, 0.25)';
   385→   385→
   386→   386→                              return (
   387→   387→                                <g key={`candle-${i}`}>
   388→   388→                                  {/* Volume bar at bottom */}
   389→   389→                                  <rect
   390→   390→                                    x={x}
   391→   391→                                    y={volBottom - volHeight}
   392→   392→                                    width={barWidth}
   393→   393→                                    height={volHeight}
   394→   394→                                    fill={volColor}
   395→   395→                                    rx={1}
   396→   396→                                  />
   397→   397→                                  {/* Wick */}
   398→   398→                                  <line
   399→   399→                                    x1={x + bandwidth / 2}
   400→   400→                                    y1={wickTop}
   401→   401→                                    x2={x + bandwidth / 2}
   402→   402→                                    y2={wickBottom}
   403→   403→                                    stroke={isUp ? '#10b981' : '#ef4444'}
   404→   404→                                    strokeWidth={1}
   405→   405→                                  />
   406→   406→                                  {/* Body */}
   407→   407→                                  <rect
   408→   408→                                    x={x + (bandwidth - barWidth) / 2}
   409→   409→                                    y={bodyTop}
   410→   410→                                    width={barWidth}
   411→   411→                                    height={bodyHeight}
   412→   412→                                    fill={isUp ? '#10b981' : '#ef4444'}
   413→   413→                                    stroke={isUp ? '#10b981' : '#ef4444'}
   414→   414→                                    strokeWidth={0.5}
   415→   415→                                    rx={1}
   416→   416→                                  />
   417→   417→                                </g>
   418→   418→                              );
   419→   419→                            })}
   420→   420→                          </g>
   421→   421→                        );
   422→   422→                      }} />
   423→   423→                    </ComposedChart>
   424→   424→                  </ResponsiveContainer>
   425→   425→                </div>
   426→   426→                {/* Legend explanation */}
   427→   427→                <div className="flex items-center justify-center gap-4 mt-2">
   428→   428→                  <div className="flex items-center gap-1.5">
   429→   429→                    <div className="w-3 h-3 rounded-sm bg-emerald-500" />
   430→   430→                    <span className="text-[10px] text-muted-foreground">Bullish (Çmimi u rrit)</span>
   431→   431→                  </div>
   432→   432→                  <div className="flex items-center gap-1.5">
   433→   433→                    <div className="w-3 h-3 rounded-sm bg-red-500" />
   434→   434→                    <span className="text-[10px] text-muted-foreground">Bearish (Çmimi u ul)</span>
   435→   435→                  </div>
   436→   436→                  <div className="flex items-center gap-1.5">
   437→   437→                    <div className="w-3 h-3 rounded-sm bg-muted/40" />
   438→   438→                    <span className="text-[10px] text-muted-foreground">Volumi</span>
