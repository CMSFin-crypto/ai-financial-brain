   150→
   151→  useEffect(() => {
   152→    const handleClick = (e: MouseEvent) => {
   153→      if (popupRef.current && !popupRef.current.contains(e.target as Node)) {
   154→        onClose();
   155→      }
   156→    };
   157→    const handleEsc = (e: KeyboardEvent) => {
   158→      if (e.key === 'Escape') onClose();

   162→      document.addEventListener('keydown', handleEsc);

   165→      document.removeEventListener('mousedown', handleClick);

   282→// Simple seeded pseudo-random for consistent sparklines
   283→function seededRandom(seed: number): () => number {
   284→  let s = seed;
   285→  return () => {
   286→    s = (s * 16807 + 0) % 2147483647;
   287→    return (s - 1) / 2147483646;
   288→  };
   289→}

   291→function generateSparklineData(priceChange: number, ticker: string): Array<{ value: number }> {
   292→  const points = 12;
   293→  const data: Array<{ value: number }> = [];
   294→  const magnitude = Math.min(Math.abs(priceChange), 10);
   295→  // Create a seed from ticker hash + priceChange for consistency
   296→  const hash = ticker.split('').reduce((acc, c) => acc + c.charCodeAt(0), 0);
   297→  const rand = seededRandom(hash * 1000 + Math.round(priceChange * 100));
   298→  for (let i = 0; i < points; i++) {
   299→    const progress = i / (points - 1);
   300→    const trend = priceChange >= 0 ? progress * magnitude : -progress * magnitude;
   301→    const noise = (rand() - 0.5) * magnitude * 0.4;
   302→    data.push({ value: 50 + trend + noise });
   303→  }
   304→  return data;
   305→}
