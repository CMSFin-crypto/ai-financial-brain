   495→interface SearchResult { ticker: string; company: string; exchange: string; price: number; change: number; volume: number; }

   497→// ── Single Stock Search Component ──
   498→function StockSearchBox() {
   499→  const [query, setQuery] = useState('');
   500→  const [results, setResults] = useState<SearchResult[]>([]);
   501→  const [showDropdown, setShowDropdown] = useState(false);
   502→  const [analyzing, setAnalyzing] = useState<string | null>(null);
   503→  const [analyzedStock, setAnalyzedStock] = useState<FunnelStock | null>(null);
   504→  const [analyzeError, setAnalyzeError] = useState<string | null>(null);
   505→  const [regimeOk, setRegimeOk] = useState(true);
   506→  const [funnelPhases, setFunnelPhases] = useState({ passedLiquidity: 0, passedTrend: 0, passedSetup: 0, passedRisk: 0 });
   507→  const timerRef = useRef<ReturnType<typeof setTimeout>>();
   508→  const inputRef = useRef<HTMLInputElement>(null);
   509→
   510→  // Search with debounce
   511→  const handleInput = (val: string) => {
   512→    setQuery(val);
   513→    if (timerRef.current) clearTimeout(timerRef.current);
   514→    if (val.length < 2) { setResults([]); setShowDropdown(false); return; }
   515→    timerRef.current = setTimeout(async () => {
   516→      try {
   517→        const res = await fetch(`/api/ticker-search?q=${encodeURIComponent(val)}`);
   518→        const json = await res.json();
   519→        setResults(json.results || []);
   520→        setShowDropdown(true);
   521→      } catch { setResults([]); }
   522→    }, 350);
   523→  };
   524→
   525→  const selectTicker = async (ticker: string) => {
   526→    setShowDropdown(false); setQuery(ticker);
   527→    setAnalyzedStock(null); setAnalyzeError(null);
   528→    setAnalyzing(ticker);
   529→    try {
   530→      const res = await fetch(`/api/ibkr-analyze/${ticker}?_t=${Date.now()}`, { cache: 'no-store' });
   531→      const json = await res.json();
   532→      setRegimeOk(json.regimeOk);
   533→      setFunnelPhases(json.funnel || { passedLiquidity: 0, passedTrend: 0, passedSetup: 0, passedRisk: 0 });
   534→      if (json.error && !json.stock) {
   535→        setAnalyzeError(json.error);
   536→      } else if (json.stock) {
   537→        setAnalyzedStock(json.stock);
   538→      }
   539→    } catch { setAnalyzeError('Gabim rrjeti'); }
   540→    setAnalyzing(null);
   541→  };
   542→
   543→  return (
   544→    <div className="space-y-3">
   545→      <div className="relative">
   546→        <div className="flex items-center gap-2">
   547→          <div className="relative flex-1">
   548→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground/50" />
   549→            <input
   550→              ref={inputRef}
   551→              value={query}
   552→              onChange={e => handleInput(e.target.value)}
   553→              onFocus={() => results.length > 0 && setShowDropdown(true)}
   554→              placeholder="Kerko nje aksion (p.sh. AAPL, TSLA, COIN...)"
   555→              className="w-full pl-9 pr-9 py-2.5 rounded-lg bg-muted/10 border border-border/50 text-[13px] text-foreground placeholder:text-muted-foreground/40 focus:outline-none focus:border-blue-500/50 focus:ring-1 focus:ring-blue-500/20 transition-all"
   556→            />
   557→            {query && (
   558→              <button onClick={() => { setQuery(''); setResults([]); setShowDropdown(false); setAnalyzedStock(null); setAnalyzeError(null); }} className="absolute right-3 top-1/2 -translate-y-1/2"><X className="w-3.5 h-3.5 text-muted-foreground/50 hover:text-muted-foreground" /></button>
   559→            )}
