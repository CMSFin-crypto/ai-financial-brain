    83→function clamp(value: number, min: number, max: number): number {
    84→  return Math.max(min, Math.min(max, value));

    87→function mapRange(
    88→  value: number,
    89→  inMin: number,
    90→  inMax: number,
    91→  outMin: number,
    92→  outMax: number,
    93→): number {
    94→  if (inMax === inMin) return (outMin + outMax) / 2;
    95→  const result = outMin + ((value - inMin) / (inMax - inMin)) * (outMax - outMin);
    96→  return clamp(result, Math.min(outMin, outMax), Math.max(outMin, outMax));
    97→}

    99→/** Simple Moving Average — full-length array with NaN padding */
   100→function sma(data: number[], period: number): number[] {
   101→  const result: number[] = new Array(data.length).fill(NaN);
   102→  if (data.length < period) return result;
   103→
   104→  let sum = 0;
   105→  for (let i = 0; i < period; i++) sum += data[i];
   106→  result[period - 1] = sum / period;
   107→
   108→  for (let i = period; i < data.length; i++) {
   109→    sum += data[i] - data[i - period];
   110→    result[i] = sum / period;
   111→  }
   112→  return result;
   113→}

   115→/** Exponential Moving Average — full-length array with NaN padding */
   116→function ema(data: number[], period: number): number[] {
   117→  const result: number[] = new Array(data.length).fill(NaN);
   118→  if (data.length < period) return result;
   119→
   120→  const multiplier = 2 / (period + 1);
   121→
   122→  let sum = 0;
   123→  for (let i = 0; i < period; i++) sum += data[i];
   124→  result[period - 1] = sum / period;
   125→
   126→  for (let i = period; i < data.length; i++) {
   127→    result[i] = (data[i] - result[i - 1]) * multiplier + result[i - 1];
   128→  }
   129→  return result;
   130→}

   132→/** Rolling standard deviation */
   133→function rollingStdDev(data: number[], period: number): number[] {
   134→  const result: number[] = new Array(data.length).fill(NaN);
   135→  if (data.length < period) return result;
   136→
   137→  for (let i = period - 1; i < data.length; i++) {
   138→    const slice = data.slice(i - period + 1, i + 1);
   139→    const mean = slice.reduce((a, b) => a + b, 0) / period;
   140→    const variance = slice.reduce((a, b) => a + (b - mean) ** 2, 0) / period;
   141→    result[i] = Math.sqrt(variance);
   142→  }
   143→  return result;
   144→}

   146→/** Simple linear regression → { slope, intercept } */
   147→function linearRegression(values: number[]): { slope: number; intercept: number } {
   148→  const n = values.length;
   149→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0 };
   150→
   151→  let sumX = 0;
   152→  let sumY = 0;
   153→  let sumXY = 0;
   154→  let sumX2 = 0;
   155→  for (let i = 0; i < n; i++) {
   156→    sumX += i;
   157→    sumY += values[i];
   158→    sumXY += i * values[i];
   159→    sumX2 += i * i;
   160→  }
   161→
   162→  const denom = n * sumX2 - sumX * sumX;
   163→  if (denom === 0) return { slope: 0, intercept: sumY / n };
   164→
   165→  return {
   166→    slope: (n * sumXY - sumX * sumY) / denom,
   167→    intercept: (sumY - ((n * sumXY - sumX * sumY) / denom) * sumX) / n,
   168→  };
   169→}

   171→/** Percentile of a numeric array (0–100) */
   172→function percentile(values: number[], p: number): number {
   173→  const sorted = [...values].filter((v) => !isNaN(v)).sort((a, b) => a - b);
   174→  if (sorted.length === 0) return 0;
   175→  const idx = (p / 100) * (sorted.length - 1);
   176→  const lo = Math.floor(idx);
   177→  const hi = Math.ceil(idx);
   178→  if (lo === hi) return sorted[lo];
   179→  return sorted[lo] + (idx - lo) * (sorted[hi] - sorted[lo]);
   180→}

   182→/** Last non-NaN numeric value from an array */
   183→function lastValid(arr: number[]): number | undefined {
   184→  for (let i = arr.length - 1; i >= 0; i--) {
   185→    if (!isNaN(arr[i])) return arr[i];
   186→  }
   187→  return undefined;
   188→}

   192→/** RSI using Wilder's smoothing */
   193→function calculateRSI(closes: number[], period: number = 14): number[] {
   194→  const result: number[] = new Array(closes.length).fill(NaN);
   195→  if (closes.length < period + 1) return result;
   196→
   197→  let avgGain = 0;
   198→  let avgLoss = 0;
   199→
   200→  for (let i = 1; i <= period; i++) {
   201→    const change = closes[i] - closes[i - 1];
   202→    if (change > 0) avgGain += change;
   203→    else avgLoss += Math.abs(change);
   204→  }
   205→  avgGain /= period;
   206→  avgLoss /= period;
   207→
   208→  result[period] = avgLoss === 0 ? 100 : 100 - 100 / (1 + avgGain / avgLoss);
   209→
   210→  for (let i = period + 1; i < closes.length; i++) {
   211→    const change = closes[i] - closes[i - 1];
   212→    const gain = change > 0 ? change : 0;
   213→    const loss = change < 0 ? Math.abs(change) : 0;
   214→
   215→    avgGain = (avgGain * (period - 1) + gain) / period;
   216→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   217→
   218→    const rs = avgLoss === 0 ? Infinity : avgGain / avgLoss;
   219→    result[i] = clamp(100 - 100 / (1 + rs), 0, 100);
   220→  }
   221→  return result;
   222→}

   224→/** MACD: { macdLine, signalLine, histogram } */
   225→function calculateMACD(
   226→  closes: number[],
   227→): { macdLine: number[]; signalLine: number[]; histogram: number[] } {
   228→  const ema12 = ema(closes, 12);
   229→  const ema26 = ema(closes, 26);
   230→
   231→  const macdLine: number[] = new Array(closes.length).fill(NaN);
   232→  for (let i = 0; i < closes.length; i++) {
   233→    if (!isNaN(ema12[i]) && !isNaN(ema26[i])) {
   234→      macdLine[i] = ema12[i] - ema26[i];
   235→    }
   236→  }
   237→
   238→  // Build valid-only MACD array for signal-line EMA
   239→  const validMacd: number[] = [];
   240→  for (let i = 0; i < macdLine.length; i++) {
   241→    if (!isNaN(macdLine[i])) validMacd.push(macdLine[i]);
   242→  }
   243→
   244→  const sigEma = ema(validMacd, 9);
   245→
   246→  const signalLine: number[] = new Array(closes.length).fill(NaN);
   247→  const histogram: number[] = new Array(closes.length).fill(NaN);
   248→
   249→  let vi = 0;
   250→  for (let i = 0; i < closes.length; i++) {
   251→    if (!isNaN(macdLine[i])) {
   252→      if (vi < sigEma.length && !isNaN(sigEma[vi])) {
   253→        signalLine[i] = sigEma[vi];
   254→        histogram[i] = macdLine[i] - sigEma[vi];
   255→      }
   256→      vi++;
   257→    }
   258→  }
   259→
   260→  return { macdLine, signalLine, histogram };
   261→}

   263→/** Bollinger Bands: { upper, middle, lower, bandwidth } */
   264→function calculateBollingerBands(
   265→  closes: number[],
   266→  period: number = 20,
   267→  stdDevMult: number = 2,
   268→): { upper: number[]; middle: number[]; lower: number[]; bandwidth: number[] } {
   269→  const middle = sma(closes, period);
   270→  const sd = rollingStdDev(closes, period);
   271→
   272→  const upper: number[] = new Array(closes.length).fill(NaN);
   273→  const lower: number[] = new Array(closes.length).fill(NaN);
   274→  const bandwidth: number[] = new Array(closes.length).fill(NaN);
   275→
   276→  for (let i = 0; i < closes.length; i++) {
   277→    if (!isNaN(middle[i]) && !isNaN(sd[i])) {
   278→      upper[i] = middle[i] + stdDevMult * sd[i];
   279→      lower[i] = middle[i] - stdDevMult * sd[i];
   280→      bandwidth[i] = middle[i] === 0 ? 0 : ((upper[i] - lower[i]) / middle[i]) * 100;
   281→    }
   282→  }
   283→
   284→  return { upper, middle, lower, bandwidth };
   285→}

   287→/** Stochastic Oscillator: { k, d } */
   288→function calculateStochastic(
   289→  highs: number[],
   290→  lows: number[],
   291→  closes: number[],
   292→  kPeriod: number = 14,
   293→  dPeriod: number = 3,
   294→): { k: number[]; d: number[] } {
   295→  const k: number[] = new Array(closes.length).fill(NaN);
   296→
   297→  for (let i = kPeriod - 1; i < closes.length; i++) {
   298→    let lowest = lows[i];
   299→    let highest = highs[i];
   300→    for (let j = i - kPeriod + 1; j < i; j++) {
   301→      if (lows[j] < lowest) lowest = lows[j];
   302→      if (highs[j] > highest) highest = highs[j];
   303→    }
   304→    const range = highest - lowest;
   305→    k[i] = range === 0 ? 50 : ((closes[i] - lowest) / range) * 100;
   306→  }
   307→
   308→  const d = sma(k, dPeriod);
   309→  return { k, d };
   310→}

   312→/** ADX with +DI and -DI */
   313→function calculateADX(
   314→  highs: number[],
   315→  lows: number[],
   316→  closes: number[],
   317→  period: number = 14,
   318→): { adx: number[]; plusDI: number[]; minusDI: number[] } {
   319→  const adx: number[] = new Array(closes.length).fill(NaN);
   320→  const plusDI: number[] = new Array(closes.length).fill(NaN);
   321→  const minusDI: number[] = new Array(closes.length).fill(NaN);
   322→
   323→  if (closes.length < period * 2) return { adx, plusDI, minusDI };
   324→
   325→  // Compute raw TR, +DM, -DM
   326→  const trRaw: number[] = new Array(closes.length).fill(0);
   327→  const plusDMRaw: number[] = new Array(closes.length).fill(0);
   328→  const minusDMRaw: number[] = new Array(closes.length).fill(0);
   329→
   330→  for (let i = 1; i < closes.length; i++) {
   331→    trRaw[i] = Math.max(
   332→      highs[i] - lows[i],
   333→      Math.abs(highs[i] - closes[i - 1]),
   334→      Math.abs(lows[i] - closes[i - 1]),
   335→    );
   336→
   337→    const upMove = highs[i] - highs[i - 1];
   338→    const downMove = lows[i - 1] - lows[i];
   339→
   340→    plusDMRaw[i] = upMove > downMove && upMove > 0 ? upMove : 0;
   341→    minusDMRaw[i] = downMove > upMove && downMove > 0 ? downMove : 0;
   342→  }
   343→
   344→  // Wilder's smoothing for first period
   345→  let smoothTR = 0;
   346→  let smoothPlusDM = 0;
   347→  let smoothMinusDM = 0;
   348→  for (let i = 1; i <= period; i++) {
   349→    smoothTR += trRaw[i];
   350→    smoothPlusDM += plusDMRaw[i];
   351→    smoothMinusDM += minusDMRaw[i];
   352→  }
   353→
   354→  const dxValues: number[] = [];
   355→
   356→  const computeDI = (idx: number) => {
   357→    if (smoothTR === 0) {
   358→      plusDI[idx] = 0;
   359→      minusDI[idx] = 0;
   360→      dxValues.push(0);
   361→      return;
   362→    }
   363→    plusDI[idx] = (100 * smoothPlusDM) / smoothTR;
   364→    minusDI[idx] = (100 * smoothMinusDM) / smoothTR;
   365→    const diSum = plusDI[idx] + minusDI[idx];
   366→    dxValues.push(diSum === 0 ? 0 : (100 * Math.abs(plusDI[idx] - minusDI[idx])) / diSum);
   367→  };
   368→
   369→  computeDI(period);
   370→
   371→  for (let i = period + 1; i < closes.length; i++) {
   372→    smoothTR = smoothTR - smoothTR / period + trRaw[i];
   373→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMRaw[i];
   374→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMRaw[i];
   375→    computeDI(i);
   376→  }
   377→
   378→  // ADX = Wilder's smoothing of DX
   379→  if (dxValues.length >= period) {
   380→    let adxVal =
   381→      dxValues.slice(0, period).reduce((a, b) => a + b, 0) / period;
   382→    const adxStartIdx = period - 1; // index in closes where first ADX goes
   383→    const baseIdx = period; // index in closes where first DX was computed
   384→    adx[baseIdx + adxStartIdx] = adxVal;
   385→
   386→    for (let i = period; i < dxValues.length; i++) {
   387→      adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   388→      adx[baseIdx + i] = adxVal;
   389→    }
   390→  }
   391→
   392→  return { adx, plusDI, minusDI };
   393→}

   395→/** Average True Range */
   396→function calculateATR(
   397→  highs: number[],
   398→  lows: number[],
   399→  closes: number[],
   400→  period: number = 14,
   401→): number[] {
   402→  const result: number[] = new Array(closes.length).fill(NaN);
   403→  if (closes.length < period + 1) return result;
   404→
   405→  const tr: number[] = [highs[0] - lows[0]];
   406→  for (let i = 1; i < closes.length; i++) {
   407→    tr.push(
   408→      Math.max(
   409→        highs[i] - lows[i],
   410→        Math.abs(highs[i] - closes[i - 1]),
   411→        Math.abs(lows[i] - closes[i - 1]),
   412→      ),
   413→    );
   414→  }
   415→
   416→  // First ATR = SMA of first `period` TRs (TR starts at index 0, but Wilder uses first `period` starting from index 1)
   417→  let atr = 0;
   418→  for (let i = 1; i <= period; i++) atr += tr[i];
   419→  atr /= period;
   420→  result[period] = atr;
   421→
   422→  for (let i = period + 1; i < closes.length; i++) {
   423→    atr = (atr * (period - 1) + tr[i]) / period;
   424→    result[i] = atr;
   425→  }
   426→  return result;
   427→}

   429→/** On-Balance Volume */
   430→function calculateOBV(closes: number[], volumes: number[]): number[] {
   431→  const obv: number[] = new Array(closes.length).fill(0);
   432→  for (let i = 1; i < closes.length; i++) {
   433→    if (closes[i] > closes[i - 1]) obv[i] = obv[i - 1] + volumes[i];
   434→    else if (closes[i] < closes[i - 1]) obv[i] = obv[i - 1] - volumes[i];
   435→    else obv[i] = obv[i - 1];
   436→  }
   437→  return obv;
   438→}

   443→function scoreRSI(rsiValues: number[]): IndicatorScore {
   444→  const rsiVal = lastValid(rsiValues);
   445→  if (rsiVal === undefined) return noData('RSI');
   446→
   447→  let score: number;
   448→  let name: string;
   449→  let description: string;
   450→
   451→  if (rsiVal < 20) {
   452→    score = mapRange(20 - rsiVal, 0, 20, 70, 100);
   453→    name = 'RSI Mbushje e Shitjes';
   454→    description = `RSI në ${rsiVal.toFixed(1)} — kushtet e thella të mbushjes së shitjes, sinjal i fortë për blerje`;
   455→  } else if (rsiVal < 30) {
   456→    score = mapRange(30 - rsiVal, 0, 10, 50, 70);
   457→    name = 'RSI Në Zonën e Blerjes';
   458→    description = `RSI në ${rsiVal.toFixed(1)} — në zonën e blerjes, momentumi po kthehet pozitiv`;
   459→  } else if (rsiVal < 45) {
   460→    score = mapRange(45 - rsiVal, 0, 15, 40, 50);
   461→    name = 'RSI Lehtësisht Bulliz';
   462→    description = `RSI në ${rsiVal.toFixed(1)} — lehtësisht në favor të blerjes`;
   463→  } else if (rsiVal <= 55) {
   464→    score = 50 - (rsiVal - 50);
   465→    name = 'RSI Neutral';
   466→    description = `RSI në ${rsiVal.toFixed(1)} — zonë neutrale, asnjë sinjal i qartë`;
   467→  } else if (rsiVal < 70) {
   468→    score = mapRange(rsiVal - 55, 0, 15, -10, -40);
   469→    name = 'RSI Lehtësisht Bearish';
   470→    description = `RSI në ${rsiVal.toFixed(1)} — duke hyrë në zonën e shitjes`;
   471→  } else if (rsiVal < 80) {
   472→    score = mapRange(rsiVal - 70, 0, 10, -50, -70);
   473→    name = 'RSI Në Zonën e Shitjes';
   474→    description = `RSI në ${rsiVal.toFixed(1)} — kushtet e mbushjes së blerjes, sinjal i shitjes`;
   475→  } else {
   476→    score = mapRange(rsiVal - 80, 0, 20, -70, -100);
   477→    name = 'RSI Mbushje e Blërjes';
   478→    description = `RSI në ${rsiVal.toFixed(1)} — kushtet e thella të mbushjes së blerjes, sinjal i fortë për shitje`;
   479→  }
   480→
   481→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: Math.abs(score) > 60 ? 'HIGH' : 'MEDIUM', description };
   482→}

   484→function scoreMACDHistogram(histogram: number[]): IndicatorScore {
   485→  const n = histogram.length;
   486→  const h0 = lastValid(histogram);
   487→  if (h0 === undefined) return noData('MACD Histogram');
   488→
   489→  // Get last 3 valid histogram values
   490→  const recent: number[] = [];
   491→  for (let i = n - 1; i >= 0 && recent.length < 3; i--) {
   492→    if (!isNaN(histogram[i])) recent.unshift(histogram[i]);
   493→  }
   494→
   495→  let score: number;
   496→  let name: string;
   497→  let description: string;
   498→
   499→  // Use max absolute value across recent bars as a scale reference
   500→  const maxAbs = Math.max(...recent.map(Math.abs), 0.001);
   501→
   502→  if (recent.length < 2) {
   503→    score = h0 > 0 ? 50 : h0 < 0 ? -50 : 0;
   504→  } else {
   505→    const prev = recent[recent.length - 2];
   506→    const increasing = h0 > prev;
   507→
   508→    if (h0 > 0 && increasing) {
   509→      score = clamp(50 + 50 * (h0 / maxAbs), 50, 100);
   510→      name = 'MACD Histogram Pozitiv në Rritje';
   511→      description = 'Histogrami MACD pozitiv dhe në rritje, momentumi i blerjes po forcohet';
   512→    } else if (h0 > 0) {
   513→      score = clamp(40 + 20 * (h0 / maxAbs), 40, 60);
   514→      name = 'MACD Histogram Pozitiv';
   515→      description = 'Histogrami MACD pozitiv por në ulje, momentumi i blerjes po dobësohet';
   516→    } else if (h0 < 0 && !increasing) {
   517→      score = clamp(-50 - 50 * (Math.abs(h0) / maxAbs), -100, -50);
   518→      name = 'MACD Histogram Negativ në Rritje';
   519→      description = 'Histogrami MACD negativ dhe në rritje, momentumi i shitjes po forcohet';
   520→    } else {
   521→      score = clamp(-40 - 20 * (Math.abs(h0) / maxAbs), -60, -40);
   522→      name = 'MACD Histogram Negativ';
   523→      description = 'Histogrami MACD negativ por në ulje, shitjet po dobësohen';
   524→    }
   525→  }
   526→
   527→  if (!name) {
   528→    name = 'MACD Histogram Neutral';

   536→function scoreBollinger(
   537→  closes: number[],
   538→  bb: { upper: number[]; middle: number[]; lower: number[]; bandwidth: number[] },
   539→): IndicatorScore {
   540→  const last = closes.length - 1;
   541→  const price = closes[last];
   542→  const upper = bb.upper[last];
   543→  const lower = bb.lower[last];
   544→  const bw = bb.bandwidth[last];
   545→
   546→  if (isNaN(upper) || isNaN(lower) || isNaN(bw) || upper === lower) {
   547→    return noData('Bollinger Bands');
   548→  }
   549→
   550→  const position = (price - lower) / (upper - lower); // 0 to 1 (may go outside)
   551→
   552→  // Detect squeeze: bandwidth narrowing over last 5 valid values
   553→  let squeeze = false;
   554→  const recentBW: number[] = [];
   555→  for (let i = last; i >= 0 && recentBW.length < 5; i--) {
   556→    if (!isNaN(bb.bandwidth[i])) recentBW.unshift(bb.bandwidth[i]);
   557→  }
   558→  if (recentBW.length >= 4) {
   559→    squeeze = recentBW[recentBW.length - 1] < recentBW[0] * 0.85;
   560→  }
   561→
   562→  let score: number;
   563→  let name: string;
   564→  let description: string;
   565→
   566→  if (position < 0.1) {
   567→    score = mapRange(position, 0, 0.1, 100, 70);
   568→    name = squeeze ? 'Bollinger Squeeze & Mbushje' : 'Bollinger Nën Band';
   569→    description = squeeze
   570→      ? 'Çmimi nën Bollinger Band të ulët me ngushtim, shpërthim i madh pozitiv i pritur'
   571→      : 'Çmimi nën Bollinger Band të ulët, potencial i lartë për kthim lart';
   572→  } else if (position < 0.3) {
   573→    score = mapRange(position, 0.1, 0.3, 70, 50);
   574→    name = 'Bollinger Pozicion i Ulët';
   575→    description = 'Çmimi në pjesën e poshtme të Bollinger Bands, lehtësisht bulliz';
   576→  } else if (position < 0.7) {
   577→    score = mapRange(position, 0.3, 0.7, 50, -50);
   578→    name = 'Bollinger Neutral';
   579→    description = 'Çmimi në mes të Bollinger Bands, asnjë sinjal i qartë';
   580→  } else if (position < 0.9) {
   581→    score = mapRange(position, 0.7, 0.9, -50, -70);
   582→    name = 'Bollinger Pozicion i Lartë';
   583→    description = 'Çmimi në pjesën e sipërme të Bollinger Bands, lehtësisht bearish';
   584→  } else {
   585→    score = mapRange(position, 0.9, 1.2, -70, -100);
   586→    name = squeeze ? 'Bollinger Squeeze & Mbushje' : 'Bollinger Mbi Band';
   587→    description = squeeze
   588→      ? 'Çmimi mbi Bollinger Band të sipërme me ngushtim, shpërthim i madh negativ i pritur'
   589→      : 'Çmimi mbi Bollinger Band të sipërme, rreziku i kthimit poshtë';
   590→  }
   591→
   592→  // Squeeze amplification
   593→  if (squeeze) {
   594→    score = clamp(score * 1.1, -100, 100);
   595→  }
   596→
   597→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: Math.abs(score) > 60 ? 'HIGH' : 'MEDIUM', description };
   598→}

   786→function scoreROC(closes: number[]): IndicatorScore {
   787→  const last = closes.length - 1;
   788→  if (last < 20) return noData('ROC');
   789→
   790→  const roc5 = ((closes[last] - closes[last - 5]) / closes[last - 5]) * 100;
   791→  const roc10 = ((closes[last] - closes[last - 10]) / closes[last - 10]) * 100;
   792→  const roc20 = ((closes[last] - closes[last - 20]) / closes[last - 20]) * 100;
   793→
   794→  // Weighted combination
   795→  let rawScore = roc5 * 0.5 + roc10 * 0.3 + roc20 * 0.2;
   796→
   797→  // Extreme positive → mean reversion penalty
   798→  if (roc5 > 10) {
   799→    rawScore -= (roc5 - 10) * 2;
   800→  }
   801→  // Extreme negative → mean reversion penalty
   802→  if (roc5 < -10) {
   803→    rawScore += (-10 - roc5) * 2;
   804→  }
   805→
   806→  const score = clamp(rawScore * 5, -100, 100); // Scale percentage to score
   807→
   808→  let name: string;
   809→  let description: string;
   810→
   811→  if (roc5 > 5) {
   812→    name = 'Momentum i Fortë Pozitiv';
   813→    description = `ROC 5-ditor ${roc5.toFixed(2)}%, momentumi i shkurtër shumë pozitiv`;
   814→  } else if (roc5 < -5) {
   815→    name = 'Momentum i Fortë Negativ';
   816→    description = `ROC 5-ditor ${roc5.toFixed(2)}%, momentumi i shkurtër shumë negativ`;
   817→  } else if (rawScore > 1) {
   818→    name = 'Momentum Pozitiv';
   819→    description = `ROC i përgjithshëm pozitiv (5d: ${roc5.toFixed(2)}%, 10d: ${roc10.toFixed(2)}%, 20d: ${roc20.toFixed(2)}%)`;
   820→  } else if (rawScore < -1) {
   821→    name = 'Momentum Negativ';
   822→    description = `ROC i përgjithshëm negativ (5d: ${roc5.toFixed(2)}%, 10d: ${roc10.toFixed(2)}%, 20d: ${roc20.toFixed(2)}%)`;
   823→  } else {
   824→    name = 'Momentum Neutral';
   825→    description = 'Shkalla e ndryshimit afatshkurtër afër zero';
   826→  }
   827→
   828→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: Math.abs(score) > 40 ? 'MEDIUM' : 'LOW', description };
   829→}

   906→function scoreMACDCrossover(
   907→  macdLine: number[],
   908→  signalLine: number[],
   909→): IndicatorScore {
   910→  const n = macdLine.length;
   911→  let score = 0;
   912→  let name = 'Asnjë Kryqëzim MACD';
   913→  let description = 'Asnjë kryqëzim i fundit midis MACD dhe vijës së sinjalit';
   914→  let found = false;
   915→
   916→  // Check last 5 valid bars for crossover
   917→  const recentIndices: number[] = [];
   918→  for (let i = n - 1; i >= 0 && recentIndices.length < 6; i--) {
   919→    if (!isNaN(macdLine[i]) && !isNaN(signalLine[i])) recentIndices.unshift(i);
   920→  }
   921→
   922→  for (let j = 1; j < recentIndices.length; j++) {
   923→    const prevIdx = recentIndices[j - 1];
   924→    const currIdx = recentIndices[j];
   925→
   926→    const prevDiff = macdLine[prevIdx] - signalLine[prevIdx];
   927→    const currDiff = macdLine[currIdx] - signalLine[currIdx];
   928→
   929→    if (prevDiff <= 0 && currDiff > 0) {
   930→      // Bullish crossover
   931→      const recency = j / recentIndices.length; // 1 = most recent
   932→      score = mapRange(recency, 0, 1, 65, 90);
   933→      name = 'Kryqëzim Bulliz MACD';
   934→      description = 'MACD kaloi sipër vijës së sinjalit, sinjal i blerjes';
   935→      found = true;
   936→      break;
   937→    } else if (prevDiff >= 0 && currDiff < 0) {
   938→      // Bearish crossover
   939→      const recency = j / recentIndices.length;
   940→      score = mapRange(recency, 0, 1, -65, -90);
   941→      name = 'Kryqëzim Bearish MACD';
   942→      description = 'MACD kaloi poshtë vijës së sinjalit, sinjal i shitjes';
   943→      found = true;
   944→      break;
   945→    }
   946→  }
   947→
   948→  if (!found) score = 0;
   949→
   950→  return { score, name, signal: found ? (score > 0 ? 'BULLISH' : 'BEARISH') : 'NEUTRAL', impact: found ? 'HIGH' : 'LOW', description };
   951→}

   953→function scorePriceChannel(
   954→  highs: number[],
   955→  lows: number[],
   956→  closes: number[],
   957→): IndicatorScore {
   958→  const last = closes.length - 1;
   959→  if (last < 20) return noData('Price Channel');
   960→
   961→  let ch20High = -Infinity;
   962→  let ch20Low = Infinity;
   963→  for (let i = last - 19; i < last; i++) {
   964→    if (highs[i] > ch20High) ch20High = highs[i];
   965→    if (lows[i] < ch20Low) ch20Low = lows[i];
   966→  }
   967→
   968→  const price = closes[last];
   969→  const range = ch20High - ch20Low;
   970→  if (range <= 0) return noData('Price Channel');
   971→
   972→  let score: number;
   973→  let name: string;
   974→  let description: string;
   975→
   976→  if (price > ch20High) {
   977→    // Breakout
   978→    score = mapRange((price - ch20High) / range, 0, 0.1, 70, 95);
   979→    name = 'Shpërthim mbi Kanalin 20-Ditor';
   980→    description = 'Çmimi theu rezistencën e 20 ditëve, shpërthim i potencialit bulliz';
   981→  } else if (price < ch20Low) {
   982→    // Breakdown
   983→    score = mapRange((ch20Low - price) / range, 0, 0.1, -70, -95);
   984→    name = 'Thyesje nën Kanalin 20-Ditor';
   985→    description = 'Çmimi thehu mbështetjen e 20 ditëve, rrezik i thellë uljeje';
   986→  } else {
   987→    const position = (price - ch20Low) / range;
   988→    if (position > 0.7) {
   989→      score = mapRange(position, 0.7, 1, 50, 75);
   990→      name = 'Çmimi Pranë Rezistencës';
   991→      description = 'Çmimi pranë lartësisë 20-ditore, momentum pozitiv';
   992→    } else if (position < 0.3) {
   993→      score = mapRange(position, 0, 0.3, -75, -50);
   994→      name = 'Çmimi Pranë Mbështetjes';
   995→      description = 'Çmimi pranë ulëtësisë 20-ditore, rreziku i uljes';
   996→    } else {
   997→      score = mapRange(position - 0.5, -0.2, 0.2, -30, 30);
   998→      name = 'Price Channel Neutral';
   999→      description = 'Çmimi në mes të kanalit 20-ditor';
  1000→    }
  1001→  }
  1002→
  1003→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: Math.abs(score) > 60 ? 'MEDIUM' : 'LOW', description };
  1004→}

  1006→function scoreDivergence(
  1007→  closes: number[],
  1008→  rsiValues: number[],
  1009→): IndicatorScore {
  1010→  const last = closes.length - 1;
  1011→  if (last < 30) return noData('Divergjencë');
  1012→
  1013→  // Find price lows and RSI lows in last 30 bars
  1014→  const lookback = Math.min(30, last);
  1015→  const priceSlice = closes.slice(last - lookback, last + 1);
  1016→  const rsiSlice = rsiValues.slice(last - lookback, last + 1);
  1017→
  1018→  // Find local minima (price lows)
  1019→  const priceLows: { idx: number; val: number }[] = [];
  1020→  const rsiAtPriceLows: number[] = [];
  1021→  for (let i = 1; i < priceSlice.length - 1; i++) {
  1022→    if (priceSlice[i] < priceSlice[i - 1] && priceSlice[i] < priceSlice[i + 1] && !isNaN(rsiSlice[i])) {
  1023→      priceLows.push({ idx: i, val: priceSlice[i] });
  1024→      rsiAtPriceLows.push(rsiSlice[i]);
  1025→    }
  1026→  }
  1027→
  1028→  // Find local maxima (price highs)
  1029→  const priceHighs: { idx: number; val: number }[] = [];
  1030→  const rsiAtPriceHighs: number[] = [];
  1031→  for (let i = 1; i < priceSlice.length - 1; i++) {
  1032→    if (priceSlice[i] > priceSlice[i - 1] && priceSlice[i] > priceSlice[i + 1] && !isNaN(rsiSlice[i])) {
  1033→      priceHighs.push({ idx: i, val: priceSlice[i] });
  1034→      rsiAtPriceHighs.push(rsiSlice[i]);
  1035→    }
  1036→  }
  1037→
  1038→  // Bullish divergence: price makes lower low but RSI makes higher low
  1039→  let bullishDiv = false;
  1040→  let bearishDiv = false;
  1041→
  1042→  if (priceLows.length >= 2) {
  1043→    const lastTwo = priceLows.slice(-2);
  1044→    if (lastTwo[1].val < lastTwo[0].val) {
  1045→      const rsiLastTwo = rsiAtPriceLows.slice(-2);
  1046→      if (rsiLastTwo.length >= 2 && rsiLastTwo[1] > rsiLastTwo[0]) {
  1047→        bullishDiv = true;
  1048→      }
  1049→    }
  1050→  }
  1051→
  1052→  // Bearish divergence: price makes higher high but RSI makes lower high
  1053→  if (priceHighs.length >= 2) {
  1054→    const lastTwo = priceHighs.slice(-2);
  1055→    if (lastTwo[1].val > lastTwo[0].val) {
  1056→      const rsiLastTwo = rsiAtPriceHighs.slice(-2);
  1057→      if (rsiLastTwo.length >= 2 && rsiLastTwo[1] < rsiLastTwo[0]) {
  1058→        bearishDiv = true;
  1059→      }
  1060→    }
  1061→  }
  1062→
  1063→  if (bullishDiv) {
  1064→    return {
  1065→      score: 75,
  1066→      name: 'Divergjencë Bulliz',
  1067→      signal: 'BULLISH',
  1068→      impact: 'HIGH',
  1069→      description: 'Çmimi bën ultë të ulëta por RSI bën ultë më të larta, sinjal i forttë i blerjes',
  1070→    };
  1071→  }
  1072→  if (bearishDiv) {
  1073→    return {
  1074→      score: -75,
  1075→      name: 'Divergjencë Bearish',
  1076→      signal: 'BEARISH',
  1077→      impact: 'HIGH',
  1078→      description: 'Çmimi bën maje më të larta por RSI bën maje më të ulëta, sinjal i fortë i shitjes',
  1079→    };
  1080→  }
  1081→
  1082→  return {
  1083→    score: 0,
  1084→    name: 'Asnjë Divergjencë',
  1085→    signal: 'NEUTRAL',
  1086→    impact: 'LOW',
  1087→    description: 'Asnjë divergjencë e dukshme midis çmimit dhe RSI-së',
  1088→  };
  1089→}

  1091→function scoreVWAP(closes: number[], highs: number[], lows: number[], volumes: number[]): IndicatorScore {
  1092→  const last = closes.length - 1;
  1093→  if (last < 20) return noData('VWAP');
  1094→
  1095→  // Approximate VWAP over 20 days: cumulative (H+L+C)/3 * volume / cumulative volume
  1096→  let cumVP = 0;
  1097→  let cumVol = 0;
  1098→  for (let i = last - 19; i <= last; i++) {
  1099→    const tp = (highs[i] + lows[i] + closes[i]) / 3;
  1100→    cumVP += tp * volumes[i];
  1101→    cumVol += volumes[i];
  1102→  }
  1103→
  1104→  const vwap = cumVol === 0 ? closes[last] : cumVP / cumVol;
  1105→  const price = closes[last];
  1106→  const deviation = ((price - vwap) / vwap) * 100; // percentage deviation
  1107→
  1108→  let score: number;
  1109→  let name: string;
  1110→  let description: string;
  1111→
  1112→  if (price > vwap) {
  1113→    score = clamp(mapRange(deviation, 0, 5, 50, 65), 50, 65);
  1114→    name = 'Çmimi Sipër VWAP';
  1115→    description = `Çmimi ${deviation.toFixed(2)}% mbi VWAP, blerësit në kontroll`;
  1116→  } else {
  1117→    score = clamp(mapRange(-deviation, 0, 5, 50, 65), -65, -50);
  1118→    name = 'Çmimi Poshtë VWAP';
  1119→    description = `Çmimi ${(-deviation).toFixed(2)}% poshtë VWAP, shitësit në kontroll`;
  1120→  }
  1121→
  1122→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: 'LOW', description };
  1123→}

  1125→function scorePattern(
  1126→  opens: number[],
  1127→  highs: number[],
  1128→  lows: number[],
  1129→  closes: number[],
  1130→  volumes: number[],
  1131→): IndicatorScore {
  1132→  const last = closes.length - 1;
  1133→  if (last < 10) return noData('Pattern');
  1134→
  1135→  const len = Math.min(10, last + 1);
  1136→  const o = opens.slice(last - len + 1);
  1137→  const h = highs.slice(last - len + 1);
  1138→  const l = lows.slice(last - len + 1);
  1139→  const c = closes.slice(last - len + 1);
  1140→  const v = volumes.slice(last - len + 1);
  1141→
  1142→  let bestScore = 0;
  1143→  let bestName = 'Asnjë Model';
  1144→  let bestDescription = 'Nuk u zbuluan modele qirinjësh në 10 ditët e fundit';
  1145→  let bestSignal: SignalType = 'NEUTRAL';
  1146→  let bestImpact: ImpactLevel = 'LOW';
  1147→
  1148→  const n = o.length;
  1149→
  1150→  // 1. Three consecutive green candles with increasing volume
  1151→  if (n >= 3) {
  1152→    const g1 = c[n - 3] > o[n - 3];
  1153→    const g2 = c[n - 2] > o[n - 2];
  1154→    const g3 = c[n - 1] > o[n - 1];
  1155→    if (g1 && g2 && g3 && v[n - 1] > v[n - 2] && v[n - 2] > v[n - 3]) {
  1156→      bestScore = 75;
  1157→      bestName = 'Tre Qirinj Jeshilë';
  1158→      bestSignal = 'BULLISH';
  1159→      bestImpact = 'HIGH';
  1160→      bestDescription = 'Tre qirinj jeshilë radhazi me volum në rritje, prirje e fortë e blerjes';
  1161→    }
  1162→  }
  1163→
  1164→  // 2. Three consecutive red candles with increasing volume
  1165→  if (n >= 3) {
  1166→    const r1 = c[n - 3] < o[n - 3];
  1167→    const r2 = c[n - 2] < o[n - 2];
  1168→    const r3 = c[n - 1] < o[n - 1];
  1169→    if (r1 && r2 && r3 && v[n - 1] > v[n - 2] && v[n - 2] > v[n - 3]) {
  1170→      if (Math.abs(-75) > Math.abs(bestScore)) {
  1171→        bestScore = -75;
  1172→        bestName = 'Tre Qirinj të Kuq';
  1173→        bestSignal = 'BEARISH';
  1174→        bestImpact = 'HIGH';
  1175→        bestDescription = 'Tre qirinj të kuq radhazi me volum në rritje, prirje e fortë e shitjes';
  1176→      }
  1177→    }
  1178→  }
  1179→
  1180→  // 3. Hammer: small body, long lower shadow (at least 2x body), body in upper 1/3
  1181→  {
  1182→    const body = Math.abs(c[n - 1] - o[n - 1]);
  1183→    const totalRange = h[n - 1] - l[n - 1];
  1184→    const lowerShadow = Math.min(o[n - 1], c[n - 1]) - l[n - 1];
  1185→    const upperShadow = h[n - 1] - Math.max(o[n - 1], c[n - 1]);
  1186→
  1187→    if (totalRange > 0 && body > 0 && lowerShadow >= 2 * body && upperShadow <= body * 0.3) {
  1188→      // Check if at bottom of trend (price was declining)
  1189→      const declining = c[n - 2] < o[n - 2] || c[n - 3] < o[n - 3];
  1190→      if (declining) {
  1191→        const score = 65;
  1192→        if (score > Math.abs(bestScore)) {
  1193→          bestScore = score;
  1194→          bestName = 'Modeli Çekiç';
  1195→          bestSignal = 'BULLISH';
  1196→          bestImpact = 'MEDIUM';
  1197→          bestDescription = 'Modeli çekiç pas një prirjeje të ulët, sinjal i shmangies pozitive';
  1198→        }
  1199→      }
  1200→    }
  1201→  }
  1202→
  1203→  // 4. Shooting star: small body, long upper shadow (at least 2x body), body in lower 1/3
  1204→  {
  1205→    const body = Math.abs(c[n - 1] - o[n - 1]);
  1206→    const totalRange = h[n - 1] - l[n - 1];
  1207→    const lowerShadow = Math.min(o[n - 1], c[n - 1]) - l[n - 1];
  1208→    const upperShadow = h[n - 1] - Math.max(o[n - 1], c[n - 1]);
  1209→
  1210→    if (totalRange > 0 && body > 0 && upperShadow >= 2 * body && lowerShadow <= body * 0.3) {
  1211→      const advancing = c[n - 2] > o[n - 2] || c[n - 3] > o[n - 3];
  1212→      if (advancing) {
  1213→        const score = -65;
  1214→        if (Math.abs(score) > Math.abs(bestScore)) {
  1215→          bestScore = score;
  1216→          bestName = 'Yllë e Qitjes';
  1217→          bestSignal = 'BEARISH';
  1218→          bestImpact = 'MEDIUM';
  1219→          bestDescription = 'Yllë e qitjes pas një prirjeje të lartë, sinjal i shmangies negative';
  1220→        }
  1221→      }
  1222→    }
  1223→  }
  1224→
  1225→  // 5. Bullish engulfing: red candle followed by larger green candle
  1226→  if (n >= 2) {
  1227→    const prevRed = c[n - 2] < o[n - 2];
  1228→    const currGreen = c[n - 1] > o[n - 1];
  1229→    const prevBody = Math.abs(c[n - 2] - o[n - 2]);
  1230→    const currBody = Math.abs(c[n - 1] - o[n - 1]);
  1231→    const engulfsBody = c[n - 1] > o[n - 2] && o[n - 1] < c[n - 2];
  1232→
  1233→    if (prevRed && currGreen && engulfsBody && currBody > prevBody) {
  1234→      const score = 70;
  1235→      if (Math.abs(score) > Math.abs(bestScore)) {
  1236→        bestScore = score;
  1237→        bestName = 'Mbështjellje Bulliz';
  1238→        bestSignal = 'BULLISH';
  1239→        bestImpact = 'HIGH';
  1240→        bestDescription = 'Modeli i mbështjelljes bullize, sinjal i fortë për ndryshim të drejtimit';
  1241→      }
  1242→    }
  1243→  }
  1244→
  1245→  // 6. Bearish engulfing: green candle followed by larger red candle
  1246→  if (n >= 2) {
  1247→    const prevGreen = c[n - 2] > o[n - 2];
  1248→    const currRed = c[n - 1] < o[n - 1];
  1249→    const prevBody = Math.abs(c[n - 2] - o[n - 2]);
  1250→    const currBody = Math.abs(c[n - 1] - o[n - 1]);
  1251→    const engulfsBody = o[n - 1] > c[n - 2] && c[n - 1] < o[n - 2];
  1252→
  1253→    if (prevGreen && currRed && engulfsBody && currBody > prevBody) {
  1254→      const score = -70;
  1255→      if (Math.abs(score) > Math.abs(bestScore)) {
  1256→        bestScore = score;
  1257→        bestName = 'Mbështjellje Bearish';
  1258→        bestSignal = 'BEARISH';
  1259→        bestImpact = 'HIGH';
  1260→        bestDescription = 'Modeli i mbështjelljes bearish, sinjal i fortë për ndryshim të drejtimit';
  1261→      }
  1262→    }
  1263→  }
  1264→
  1265→  return { score: bestScore, name: bestName, signal: bestSignal, impact: bestImpact, description: bestDescription };
  1266→}

  1282→export function predictStock(symbol: string, data: PricePoint[]): PredictionResult {

  1284→    symbol,
  1285→    score: 0,
  1286→    direction: 'NEUTRAL',
  1287→    confidence: 0,
  1288→    shortTerm: { prediction: 'SIDEWAYS', probability: 50, expectedMove: 0 },
  1289→    mediumTerm: { prediction: 'SIDEWAYS', probability: 50, expectedMove: 0 },
  1290→    riskLevel: 'HIGH',
  1291→    volatility: 'MEDIUM',
  1292→    keyFactors: [
  1293→      {
  1294→        name: 'Të Dhëna',
  1295→        signal: 'NEUTRAL' as SignalType,
  1296→        impact: 'HIGH' as ImpactLevel,
  1297→        score: 0,
  1298→        description: 'Të dhëna të pamjaftueshme për analizë teknike',
  1299→      },
  1300→    ],
  1301→    indicatorScores: {},
  1302→    timestamp: new Date().toISOString(),
  1303→  });
  1304→
  1305→  if (!data || data.length < MIN_DATA_POINTS) return emptyResult();
  1306→
  1307→  // ---------- Extract price arrays ----------
  1308→  const closes = data.map((d) => d.close);
  1309→  const highs = data.map((d) => d.high);
  1310→  const lows = data.map((d) => d.low);
  1311→  const volumes = data.map((d) => d.volume);
  1312→  const opens = data.map((d) => d.open);
  1313→
  1314→  // Validate
  1315→  if (closes.some((c) => c <= 0) || volumes.some((v) => v < 0)) return emptyResult();
  1316→
  1317→  // ---------- Calculate raw indicators ----------
  1318→  const rsiValues = calculateRSI(closes, 14);
  1319→  const macd = calculateMACD(closes);
  1320→  const bb = calculateBollingerBands(closes, 20, 2);
  1321→  const sma20 = sma(closes, 20);
  1322→  const sma50 = sma(closes, 50);
  1323→  const sma200 = sma(closes, 200);
  1324→  const stoch = calculateStochastic(highs, lows, closes, 14, 3);
  1325→  const adxResult = calculateADX(highs, lows, closes, 14);
  1326→  const atrValues = calculateATR(highs, lows, closes, 14);
  1327→  const atr20Values = calculateATR(highs, lows, closes, 20);
  1328→

  1330→  const scores: Record<string, IndicatorScore> = {};
  1331→
  1332→  scores.rsi = scoreRSI(rsiValues);
  1333→  scores.macdHistogram = scoreMACDHistogram(macd.histogram);
  1334→  scores.bollinger = scoreBollinger(closes, bb);
  1335→  scores.maTrend = scoreMATrend(closes, sma20, sma50, sma200);
  1336→  scores.stochastic = scoreStochastic(stoch.k, stoch.d);
  1337→  scores.adx = scoreADX(adxResult.adx, adxResult.plusDI, adxResult.minusDI);
  1338→  scores.atr = scoreATR(atrValues, closes);
  1339→  scores.roc = scoreROC(closes);
  1340→  scores.obv = scoreOBV(closes, volumes);
  1341→  scores.volumeConfirm = scoreVolumeConfirmation(closes, volumes);
  1342→  scores.macdCrossover = scoreMACDCrossover(macd.macdLine, macd.signalLine);
  1343→  scores.priceChannel = scorePriceChannel(highs, lows, closes);
  1344→  scores.divergence = scoreDivergence(closes, rsiValues);
  1345→  scores.vwap = scoreVWAP(closes, highs, lows, volumes);
  1346→  scores.pattern = scorePattern(opens, highs, lows, closes, volumes);
  1347→
  1348→  // ---------- Weighted final score ----------
  1349→  let finalScore = 0;
  1350→  const indicatorScores: Record<string, number> = {};
  1351→
  1352→  for (const [key, weight] of Object.entries(WEIGHTS)) {
  1353→    const s = scores[key]?.score ?? 0;
  1354→    finalScore += s * weight;
  1355→    indicatorScores[key] = Math.round(s * 100) / 100;
  1356→  }
  1357→  finalScore = clamp(Math.round(finalScore * 100) / 100, -100, 100);
  1358→
  1359→  // ---------- Confidence ----------
  1360→  const allIndicatorScores = Object.values(scores).map((s) => s.score);
  1361→  const mean = allIndicatorScores.reduce((a, b) => a + b, 0) / allIndicatorScores.length;
  1362→  const variance = allIndicatorScores.reduce((a, b) => a + (b - mean) ** 2, 0) / allIndicatorScores.length;
  1363→  const stdDev = Math.sqrt(variance);
  1364→  const confidence = clamp(Math.round((100 - stdDev * 1.5) * 100) / 100, 0, 100);
  1365→
  1366→  // ---------- Direction ----------
  1367→  const direction: Direction =
  1368→    finalScore > 60 ? 'STRONG_BUY' : finalScore > 25 ? 'BUY' : finalScore > -25 ? 'NEUTRAL' : finalScore > -60 ? 'SELL' : 'STRONG_SELL';
  1369→
  1370→  // ---------- Short-term prediction (1-3 days) ----------
  1371→  const shortTermWeights = { rsi: 0.25, stochastic: 0.20, macdHistogram: 0.25, volumeConfirm: 0.15, pattern: 0.15 };
  1372→  let shortScore = 0;
  1373→  for (const [k, w] of Object.entries(shortTermWeights)) {
  1374→    shortScore += (scores[k]?.score ?? 0) * w;
  1375→  }
  1376→  const shortPrediction: TrendPrediction = shortScore > 20 ? 'UP' : shortScore < -20 ? 'DOWN' : 'SIDEWAYS';
  1377→  const shortProbability = clamp(Math.round(Math.abs(shortScore) * 0.8 + 20), 0, 100);
  1378→  const currentATR = lastValid(atrValues) ?? 0;
  1379→  const lastClose = closes[closes.length - 1];
  1380→  const shortExpectedMove = lastClose > 0 ? Math.round(((currentATR / lastClose) * 100) * 100) / 100 : 0;
  1381→
  1382→  // ---------- Medium-term prediction (1-2 weeks) ----------
  1383→  const medTermWeights = { maTrend: 0.30, macdHistogram: 0.20, bollinger: 0.20, obv: 0.15, roc: 0.15 };
  1384→  let medScore = 0;
  1385→  for (const [k, w] of Object.entries(medTermWeights)) {
  1386→    medScore += (scores[k]?.score ?? 0) * w;
  1387→  }
  1388→  const medPrediction: TrendPrediction = medScore > 20 ? 'UP' : medScore < -20 ? 'DOWN' : 'SIDEWAYS';
  1389→  const medProbability = clamp(Math.round(Math.abs(medScore) * 0.8 + 20), 0, 100);
  1390→  const currentATR20 = lastValid(atr20Values) ?? currentATR;
  1391→  const medExpectedMove = lastClose > 0 ? Math.round(((currentATR20 / lastClose) * 100) * 100) / 100 : 0;
  1392→
  1393→  // ---------- Risk level & volatility ----------
  1394→  const atrHistory: number[] = [];
  1395→  const atrLast = atrValues.length - 1;
  1396→  for (let i = atrLast; i >= 0 && atrHistory.length < 60; i--) {
  1397→    if (!isNaN(atrValues[i]) && atrValues[i] > 0) atrHistory.unshift(atrValues[i]);
  1398→  }
  1399→
  1400→  const p30 = atrHistory.length > 5 ? percentile(atrHistory, 30) : 0;
  1401→  const p70 = atrHistory.length > 5 ? percentile(atrHistory, 70) : Infinity;
  1402→
  1403→  const atrHigh = currentATR > p70;
  1404→  const atrLow = currentATR < p30;
  1405→
  1406→  const volatility: RiskOrVolatility = atrHigh ? 'HIGH' : atrLow ? 'LOW' : 'MEDIUM';
  1407→
  1408→  const riskLevel: RiskOrVolatility =
  1409→    atrHigh || confidence < 30 ? 'HIGH' : atrLow && confidence > 70 ? 'LOW' : 'MEDIUM';
  1410→
  1411→  // ---------- Top 5 key factors ----------
  1412→  const sortedFactors = Object.values(scores)
  1413→    .sort((a, b) => Math.abs(b.score) - Math.abs(a.score))
  1414→    .slice(0, 5);
  1415→
  1416→  const keyFactors: KeyFactor[] = sortedFactors.map((f) => ({
  1417→    name: f.name,
  1418→    signal: f.signal,
  1419→    impact: f.impact,
  1420→    score: Math.round(f.score * 100) / 100,
  1421→    description: f.description,
  1422→  }));
  1423→
  1424→  // ---------- Build result ----------
  1425→  return {
  1426→    symbol,
  1427→    score: finalScore,
  1428→    direction,
  1429→    confidence,
  1430→    shortTerm: {
  1431→      prediction: shortPrediction,
  1432→      probability: shortProbability,
  1433→      expectedMove: shortExpectedMove,
  1434→    },
  1435→    mediumTerm: {
  1436→      prediction: medPrediction,
  1437→      probability: medProbability,
  1438→      expectedMove: medExpectedMove,
  1439→    },
  1440→    riskLevel,
  1441→    volatility,
  1442→    keyFactors,
  1443→    indicatorScores,
  1444→    timestamp: new Date().toISOString(),
  1445→  };
  1446→}
  1447→
  1448→// ===================== RANK FUNCTION ========================

  1450→export function rankStocks(results: PredictionResult[]): PredictionResult[] {
