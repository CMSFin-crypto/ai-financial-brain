   110→function clamp(value: number, min: number, max: number): number {
   111→  return Math.max(min, Math.min(max, value));
   112→}

   114→function mapRange(
   115→  value: number,
   116→  inMin: number,
   117→  inMax: number,
   118→  outMin: number,
   119→  outMax: number,
   120→): number {
   121→  if (inMax === inMin) return (outMin + outMax) / 2;
   122→  const result = outMin + ((value - inMin) / (inMax - inMin)) * (outMax - outMin);
   123→  return clamp(result, Math.min(outMin, outMax), Math.max(outMin, outMax));
   124→}

   126→/** Simple Moving Average — full-length array with NaN padding */
   127→function sma(data: number[], period: number): number[] {
   128→  const result: number[] = new Array(data.length).fill(NaN);
   129→  if (data.length < period) return result;
   130→
   131→  let sum = 0;
   132→  for (let i = 0; i < period; i++) sum += data[i];
   133→  result[period - 1] = sum / period;
   134→
   135→  for (let i = period; i < data.length; i++) {
   136→    sum += data[i] - data[i - period];
   137→    result[i] = sum / period;
   138→  }
   139→  return result;
   140→}

   142→/** Exponential Moving Average — full-length array with NaN padding */
   143→function ema(data: number[], period: number): number[] {
   144→  const result: number[] = new Array(data.length).fill(NaN);
   145→  if (data.length < period) return result;
   146→
   147→  const multiplier = 2 / (period + 1);
   148→
   149→  let sum = 0;
   150→  for (let i = 0; i < period; i++) sum += data[i];
   151→  result[period - 1] = sum / period;
   152→
   153→  for (let i = period; i < data.length; i++) {
   154→    result[i] = (data[i] - result[i - 1]) * multiplier + result[i - 1];
   155→  }
   156→  return result;
   157→}

   159→/** Rolling standard deviation */
   160→function rollingStdDev(data: number[], period: number): number[] {
   161→  const result: number[] = new Array(data.length).fill(NaN);
   162→  if (data.length < period) return result;
   163→
   164→  for (let i = period - 1; i < data.length; i++) {
   165→    const slice = data.slice(i - period + 1, i + 1);
   166→    const mean = slice.reduce((a, b) => a + b, 0) / period;
   167→    const variance = slice.reduce((a, b) => a + (b - mean) ** 2, 0) / period;
   168→    result[i] = Math.sqrt(variance);
   169→  }
   170→  return result;
   171→}

   173→/** Simple linear regression → { slope, intercept } */
   174→function linearRegression(values: number[]): { slope: number; intercept: number } {
   175→  const n = values.length;
   176→  if (n < 2) return { slope: 0, intercept: values[0] ?? 0 };
   177→
   178→  let sumX = 0;
   179→  let sumY = 0;
   180→  let sumXY = 0;
   181→  let sumX2 = 0;
   182→  for (let i = 0; i < n; i++) {
   183→    sumX += i;
   184→    sumY += values[i];
   185→    sumXY += i * values[i];
   186→    sumX2 += i * i;
   187→  }
   188→
   189→  const denom = n * sumX2 - sumX * sumX;
   190→  if (denom === 0) return { slope: 0, intercept: sumY / n };
   191→
   192→  return {
   193→    slope: (n * sumXY - sumX * sumY) / denom,
   194→    intercept: (sumY - ((n * sumXY - sumX * sumY) / denom) * sumX) / n,
   195→  };
   196→}

   198→/** Percentile of a numeric array (0–100) */
   199→function percentile(values: number[], p: number): number {
   200→  const sorted = [...values].filter((v) => !isNaN(v)).sort((a, b) => a - b);
   201→  if (sorted.length === 0) return 0;
   202→  const idx = (p / 100) * (sorted.length - 1);
   203→  const lo = Math.floor(idx);
   204→  const hi = Math.ceil(idx);
   205→  if (lo === hi) return sorted[lo];
   206→  return sorted[lo] + (idx - lo) * (sorted[hi] - sorted[lo]);
   207→}

   209→/** Last non-NaN numeric value from an array */
   210→function lastValid(arr: number[]): number | undefined {
   211→  for (let i = arr.length - 1; i >= 0; i--) {
   212→    if (!isNaN(arr[i])) return arr[i];
   213→  }
   214→  return undefined;
   215→}

   219→/** RSI using Wilder's smoothing */
   220→function calculateRSI(closes: number[], period: number = 14): number[] {
   221→  const result: number[] = new Array(closes.length).fill(NaN);
   222→  if (closes.length < period + 1) return result;
   223→
   224→  let avgGain = 0;
   225→  let avgLoss = 0;
   226→
   227→  for (let i = 1; i <= period; i++) {
   228→    const change = closes[i] - closes[i - 1];
   229→    if (change > 0) avgGain += change;
   230→    else avgLoss += Math.abs(change);
   231→  }
   232→  avgGain /= period;
   233→  avgLoss /= period;
   234→
   235→  result[period] = avgLoss === 0 ? 100 : 100 - 100 / (1 + avgGain / avgLoss);
   236→
   237→  for (let i = period + 1; i < closes.length; i++) {
   238→    const change = closes[i] - closes[i - 1];
   239→    const gain = change > 0 ? change : 0;
   240→    const loss = change < 0 ? Math.abs(change) : 0;
   241→
   242→    avgGain = (avgGain * (period - 1) + gain) / period;
   243→    avgLoss = (avgLoss * (period - 1) + loss) / period;
   244→
   245→    const rs = avgLoss === 0 ? Infinity : avgGain / avgLoss;
   246→    result[i] = clamp(100 - 100 / (1 + rs), 0, 100);
   247→  }
   248→  return result;
   249→}

   251→/** MACD: { macdLine, signalLine, histogram } */
   252→function calculateMACD(
   253→  closes: number[],
   254→): { macdLine: number[]; signalLine: number[]; histogram: number[] } {
   255→  const ema12 = ema(closes, 12);
   256→  const ema26 = ema(closes, 26);
   257→
   258→  const macdLine: number[] = new Array(closes.length).fill(NaN);
   259→  for (let i = 0; i < closes.length; i++) {
   260→    if (!isNaN(ema12[i]) && !isNaN(ema26[i])) {
   261→      macdLine[i] = ema12[i] - ema26[i];
   262→    }
   263→  }
   264→
   265→  // Build valid-only MACD array for signal-line EMA
   266→  const validMacd: number[] = [];
   267→  for (let i = 0; i < macdLine.length; i++) {
   268→    if (!isNaN(macdLine[i])) validMacd.push(macdLine[i]);
   269→  }
   270→
   271→  const sigEma = ema(validMacd, 9);
   272→
   273→  const signalLine: number[] = new Array(closes.length).fill(NaN);
   274→  const histogram: number[] = new Array(closes.length).fill(NaN);
   275→
   276→  let vi = 0;
   277→  for (let i = 0; i < closes.length; i++) {
   278→    if (!isNaN(macdLine[i])) {
   279→      if (vi < sigEma.length && !isNaN(sigEma[vi])) {
   280→        signalLine[i] = sigEma[vi];
   281→        histogram[i] = macdLine[i] - sigEma[vi];
   282→      }
   283→      vi++;
   284→    }
   285→  }
   286→
   287→  return { macdLine, signalLine, histogram };
   288→}

   290→/** Bollinger Bands: { upper, middle, lower, bandwidth } */
   291→function calculateBollingerBands(
   292→  closes: number[],
   293→  period: number = 20,
   294→  stdDevMult: number = 2,
   295→): { upper: number[]; middle: number[]; lower: number[]; bandwidth: number[] } {
   296→  const middle = sma(closes, period);
   297→  const sd = rollingStdDev(closes, period);
   298→
   299→  const upper: number[] = new Array(closes.length).fill(NaN);
   300→  const lower: number[] = new Array(closes.length).fill(NaN);
   301→  const bandwidth: number[] = new Array(closes.length).fill(NaN);
   302→
   303→  for (let i = 0; i < closes.length; i++) {
   304→    if (!isNaN(middle[i]) && !isNaN(sd[i])) {
   305→      upper[i] = middle[i] + stdDevMult * sd[i];
   306→      lower[i] = middle[i] - stdDevMult * sd[i];
   307→      bandwidth[i] = middle[i] === 0 ? 0 : ((upper[i] - lower[i]) / middle[i]) * 100;
   308→    }
   309→  }
   310→
   311→  return { upper, middle, lower, bandwidth };
   312→}

   314→/** Stochastic Oscillator: { k, d } */
   315→function calculateStochastic(
   316→  highs: number[],
   317→  lows: number[],
   318→  closes: number[],
   319→  kPeriod: number = 14,
   320→  dPeriod: number = 3,
   321→): { k: number[]; d: number[] } {
   322→  const k: number[] = new Array(closes.length).fill(NaN);
   323→
   324→  for (let i = kPeriod - 1; i < closes.length; i++) {
   325→    let lowest = lows[i];
   326→    let highest = highs[i];
   327→    for (let j = i - kPeriod + 1; j < i; j++) {
   328→      if (lows[j] < lowest) lowest = lows[j];
   329→      if (highs[j] > highest) highest = highs[j];
   330→    }
   331→    const range = highest - lowest;
   332→    k[i] = range === 0 ? 50 : ((closes[i] - lowest) / range) * 100;
   333→  }
   334→
   335→  const d = sma(k, dPeriod);
   336→  return { k, d };
   337→}

   339→/** ADX with +DI and -DI */
   340→function calculateADX(
   341→  highs: number[],
   342→  lows: number[],
   343→  closes: number[],
   344→  period: number = 14,
   345→): { adx: number[]; plusDI: number[]; minusDI: number[] } {
   346→  const adx: number[] = new Array(closes.length).fill(NaN);
   347→  const plusDI: number[] = new Array(closes.length).fill(NaN);
   348→  const minusDI: number[] = new Array(closes.length).fill(NaN);
   349→
   350→  if (closes.length < period * 2) return { adx, plusDI, minusDI };
   351→
   352→  // Compute raw TR, +DM, -DM
   353→  const trRaw: number[] = new Array(closes.length).fill(0);
   354→  const plusDMRaw: number[] = new Array(closes.length).fill(0);
   355→  const minusDMRaw: number[] = new Array(closes.length).fill(0);
   356→
   357→  for (let i = 1; i < closes.length; i++) {
   358→    trRaw[i] = Math.max(
   359→      highs[i] - lows[i],
   360→      Math.abs(highs[i] - closes[i - 1]),
   361→      Math.abs(lows[i] - closes[i - 1]),
   362→    );
   363→
   364→    const upMove = highs[i] - highs[i - 1];
   365→    const downMove = lows[i - 1] - lows[i];
   366→
   367→    plusDMRaw[i] = upMove > downMove && upMove > 0 ? upMove : 0;
   368→    minusDMRaw[i] = downMove > upMove && downMove > 0 ? downMove : 0;
   369→  }
   370→
   371→  // Wilder's smoothing for first period
   372→  let smoothTR = 0;
   373→  let smoothPlusDM = 0;
   374→  let smoothMinusDM = 0;
   375→  for (let i = 1; i <= period; i++) {
   376→    smoothTR += trRaw[i];
   377→    smoothPlusDM += plusDMRaw[i];
   378→    smoothMinusDM += minusDMRaw[i];
   379→  }
   380→
   381→  const dxValues: number[] = [];
   382→
   383→  const computeDI = (idx: number) => {
   384→    if (smoothTR === 0) {
   385→      plusDI[idx] = 0;
   386→      minusDI[idx] = 0;
   387→      dxValues.push(0);
   388→      return;
   389→    }
   390→    plusDI[idx] = (100 * smoothPlusDM) / smoothTR;
   391→    minusDI[idx] = (100 * smoothMinusDM) / smoothTR;
   392→    const diSum = plusDI[idx] + minusDI[idx];
   393→    dxValues.push(diSum === 0 ? 0 : (100 * Math.abs(plusDI[idx] - minusDI[idx])) / diSum);
   394→  };
   395→
   396→  computeDI(period);
   397→
   398→  for (let i = period + 1; i < closes.length; i++) {
   399→    smoothTR = smoothTR - smoothTR / period + trRaw[i];
   400→    smoothPlusDM = smoothPlusDM - smoothPlusDM / period + plusDMRaw[i];
   401→    smoothMinusDM = smoothMinusDM - smoothMinusDM / period + minusDMRaw[i];
   402→    computeDI(i);
   403→  }
   404→
   405→  // ADX = Wilder's smoothing of DX
   406→  if (dxValues.length >= period) {
   407→    let adxVal =
   408→      dxValues.slice(0, period).reduce((a, b) => a + b, 0) / period;
   409→    const adxStartIdx = period - 1; // index in closes where first ADX goes
   410→    const baseIdx = period; // index in closes where first DX was computed
   411→    adx[baseIdx + adxStartIdx] = adxVal;
   412→
   413→    for (let i = period; i < dxValues.length; i++) {
   414→      adxVal = (adxVal * (period - 1) + dxValues[i]) / period;
   415→      adx[baseIdx + i] = adxVal;
   416→    }
   417→  }
   418→
   419→  return { adx, plusDI, minusDI };
   420→}

   422→/** Average True Range */
   423→function calculateATR(
   424→  highs: number[],
   425→  lows: number[],
   426→  closes: number[],
   427→  period: number = 14,
   428→): number[] {
   429→  const result: number[] = new Array(closes.length).fill(NaN);
   430→  if (closes.length < period + 1) return result;
   431→
   432→  const tr: number[] = [highs[0] - lows[0]];
   433→  for (let i = 1; i < closes.length; i++) {
   434→    tr.push(
   435→      Math.max(
   436→        highs[i] - lows[i],
   437→        Math.abs(highs[i] - closes[i - 1]),
   438→        Math.abs(lows[i] - closes[i - 1]),
   439→      ),
   440→    );
   441→  }
   442→
   443→  // First ATR = SMA of first `period` TRs (TR starts at index 0, but Wilder uses first `period` starting from index 1)
   444→  let atr = 0;
   445→  for (let i = 1; i <= period; i++) atr += tr[i];
   446→  atr /= period;
   447→  result[period] = atr;
   448→
   449→  for (let i = period + 1; i < closes.length; i++) {
   450→    atr = (atr * (period - 1) + tr[i]) / period;
   451→    result[i] = atr;
   452→  }
   453→  return result;
   454→}

   456→/** On-Balance Volume */
   457→function calculateOBV(closes: number[], volumes: number[]): number[] {
   458→  const obv: number[] = new Array(closes.length).fill(0);
   459→  for (let i = 1; i < closes.length; i++) {
   460→    if (closes[i] > closes[i - 1]) obv[i] = obv[i - 1] + volumes[i];
   461→    else if (closes[i] < closes[i - 1]) obv[i] = obv[i - 1] - volumes[i];
   462→    else obv[i] = obv[i - 1];
   463→  }
   464→  return obv;
   465→}

   470→function scoreRSI(rsiValues: number[]): IndicatorScore {
   471→  const rsiVal = lastValid(rsiValues);
   472→  if (rsiVal === undefined) return noData('RSI');
   473→
   474→  let score: number;
   475→  let name: string;
   476→  let description: string;
   477→
   478→  if (rsiVal < 20) {
   479→    score = mapRange(20 - rsiVal, 0, 20, 70, 100);
   480→    name = 'RSI Mbushje e Shitjes';
   481→    description = `RSI në ${rsiVal.toFixed(1)} — kushtet e thella të mbushjes së shitjes, sinjal i fortë për blerje`;
   482→  } else if (rsiVal < 30) {
   483→    score = mapRange(30 - rsiVal, 0, 10, 50, 70);
   484→    name = 'RSI Në Zonën e Blerjes';
   485→    description = `RSI në ${rsiVal.toFixed(1)} — në zonën e blerjes, momentumi po kthehet pozitiv`;
   486→  } else if (rsiVal < 45) {
   487→    score = mapRange(45 - rsiVal, 0, 15, 40, 50);
   488→    name = 'RSI Lehtësisht Bulliz';
   489→    description = `RSI në ${rsiVal.toFixed(1)} — lehtësisht në favor të blerjes`;
   490→  } else if (rsiVal <= 55) {
   491→    score = 50 - (rsiVal - 50);
   492→    name = 'RSI Neutral';
   493→    description = `RSI në ${rsiVal.toFixed(1)} — zonë neutrale, asnjë sinjal i qartë`;
   494→  } else if (rsiVal < 70) {
   495→    score = mapRange(rsiVal - 55, 0, 15, -10, -40);
   496→    name = 'RSI Lehtësisht Bearish';
   497→    description = `RSI në ${rsiVal.toFixed(1)} — duke hyrë në zonën e shitjes`;
   498→  } else if (rsiVal < 80) {
   499→    score = mapRange(rsiVal - 70, 0, 10, -50, -70);
   500→    name = 'RSI Në Zonën e Shitjes';
   501→    description = `RSI në ${rsiVal.toFixed(1)} — kushtet e mbushjes së blerjes, sinjal i shitjes`;
   502→  } else {
   503→    score = mapRange(rsiVal - 80, 0, 20, -70, -100);
   504→    name = 'RSI Mbushje e Blërjes';
   505→    description = `RSI në ${rsiVal.toFixed(1)} — kushtet e thella të mbushjes së blerjes, sinjal i fortë për shitje`;
   506→  }
   507→
   508→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: Math.abs(score) > 60 ? 'HIGH' : 'MEDIUM', description };
   509→}

   511→function scoreMACDHistogram(histogram: number[]): IndicatorScore {
   512→  const n = histogram.length;
   513→  const h0 = lastValid(histogram);
   514→  if (h0 === undefined) return noData('MACD Histogram');
   515→
   516→  // Get last 3 valid histogram values
   517→  const recent: number[] = [];
   518→  for (let i = n - 1; i >= 0 && recent.length < 3; i--) {
   519→    if (!isNaN(histogram[i])) recent.unshift(histogram[i]);
   520→  }
   521→
   522→  let score: number;
   523→  let name: string = '';
   524→  let description: string = '';
   525→
   526→  // Use max absolute value across recent bars as a scale reference
   527→  const maxAbs = Math.max(...recent.map(Math.abs), 0.001);
   528→
   529→  if (recent.length < 2) {
   530→    score = h0 > 0 ? 50 : h0 < 0 ? -50 : 0;
   531→  } else {
   532→    const prev = recent[recent.length - 2];
   533→    const increasing = h0 > prev;
   534→
   535→    if (h0 > 0 && increasing) {
   536→      score = clamp(50 + 50 * (h0 / maxAbs), 50, 100);
   537→      name = 'MACD Histogram Pozitiv në Rritje';
   538→      description = 'Histogrami MACD pozitiv dhe në rritje, momentumi i blerjes po forcohet';
   539→    } else if (h0 > 0) {
   540→      score = clamp(40 + 20 * (h0 / maxAbs), 40, 60);

   563→function scoreBollinger(
   564→  closes: number[],
   565→  bb: { upper: number[]; middle: number[]; lower: number[]; bandwidth: number[] },
   566→): IndicatorScore {
   567→  const last = closes.length - 1;
   568→  const price = closes[last];
   569→  const upper = bb.upper[last];
   570→  const lower = bb.lower[last];
   571→  const bw = bb.bandwidth[last];
   572→
   573→  if (isNaN(upper) || isNaN(lower) || isNaN(bw) || upper === lower) {
   574→    return noData('Bollinger Bands');
   575→  }
   576→
   577→  const position = (price - lower) / (upper - lower); // 0 to 1 (may go outside)
   578→
   579→  // Detect squeeze: bandwidth narrowing over last 5 valid values
   580→  let squeeze = false;
   581→  const recentBW: number[] = [];
   582→  for (let i = last; i >= 0 && recentBW.length < 5; i--) {
   583→    if (!isNaN(bb.bandwidth[i])) recentBW.unshift(bb.bandwidth[i]);
   584→  }
   585→  if (recentBW.length >= 4) {
   586→    squeeze = recentBW[recentBW.length - 1] < recentBW[0] * 0.85;
   587→  }
   588→
   589→  let score: number;
   590→  let name: string;
   591→  let description: string;
   592→
   593→  if (position < 0.1) {
   594→    score = mapRange(position, 0, 0.1, 100, 70);
   595→    name = squeeze ? 'Bollinger Squeeze & Mbushje' : 'Bollinger Nën Band';
   596→    description = squeeze
   597→      ? 'Çmimi nën Bollinger Band të ulët me ngushtim, shpërthim i madh pozitiv i pritur'
   598→      : 'Çmimi nën Bollinger Band të ulët, potencial i lartë për kthim lart';
   599→  } else if (position < 0.3) {
   600→    score = mapRange(position, 0.1, 0.3, 70, 50);
   601→    name = 'Bollinger Pozicion i Ulët';
   602→    description = 'Çmimi në pjesën e poshtme të Bollinger Bands, lehtësisht bulliz';
   603→  } else if (position < 0.7) {
   604→    score = mapRange(position, 0.3, 0.7, 50, -50);
   605→    name = 'Bollinger Neutral';
   606→    description = 'Çmimi në mes të Bollinger Bands, asnjë sinjal i qartë';
   607→  } else if (position < 0.9) {
   608→    score = mapRange(position, 0.7, 0.9, -50, -70);
   609→    name = 'Bollinger Pozicion i Lartë';
   610→    description = 'Çmimi në pjesën e sipërme të Bollinger Bands, lehtësisht bearish';
   611→  } else {
   612→    score = mapRange(position, 0.9, 1.2, -70, -100);
   613→    name = squeeze ? 'Bollinger Squeeze & Mbushje' : 'Bollinger Mbi Band';
   614→    description = squeeze
   615→      ? 'Çmimi mbi Bollinger Band të sipërme me ngushtim, shpërthim i madh negativ i pritur'
   616→      : 'Çmimi mbi Bollinger Band të sipërme, rreziku i kthimit poshtë';
   617→  }
   618→
   619→  // Squeeze amplification
   620→  if (squeeze) {
   621→    score = clamp(score * 1.1, -100, 100);
   622→  }
   623→
   624→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: Math.abs(score) > 60 ? 'HIGH' : 'MEDIUM', description };
   625→}

   813→function scoreROC(closes: number[]): IndicatorScore {
   814→  const last = closes.length - 1;
   815→  if (last < 20) return noData('ROC');
   816→
   817→  const roc5 = ((closes[last] - closes[last - 5]) / closes[last - 5]) * 100;
   818→  const roc10 = ((closes[last] - closes[last - 10]) / closes[last - 10]) * 100;
   819→  const roc20 = ((closes[last] - closes[last - 20]) / closes[last - 20]) * 100;
   820→
   821→  // Weighted combination
   822→  let rawScore = roc5 * 0.5 + roc10 * 0.3 + roc20 * 0.2;
   823→
   824→  // Extreme positive → mean reversion penalty
   825→  if (roc5 > 10) {
   826→    rawScore -= (roc5 - 10) * 2;
   827→  }
   828→  // Extreme negative → mean reversion penalty
   829→  if (roc5 < -10) {
   830→    rawScore += (-10 - roc5) * 2;
   831→  }
   832→
   833→  const score = clamp(rawScore * 5, -100, 100); // Scale percentage to score
   834→
   835→  let name: string;
   836→  let description: string;
   837→
   838→  if (roc5 > 5) {
   839→    name = 'Momentum i Fortë Pozitiv';
   840→    description = `ROC 5-ditor ${roc5.toFixed(2)}%, momentumi i shkurtër shumë pozitiv`;
   841→  } else if (roc5 < -5) {
   842→    name = 'Momentum i Fortë Negativ';
   843→    description = `ROC 5-ditor ${roc5.toFixed(2)}%, momentumi i shkurtër shumë negativ`;
   844→  } else if (rawScore > 1) {
   845→    name = 'Momentum Pozitiv';
   846→    description = `ROC i përgjithshëm pozitiv (5d: ${roc5.toFixed(2)}%, 10d: ${roc10.toFixed(2)}%, 20d: ${roc20.toFixed(2)}%)`;
   847→  } else if (rawScore < -1) {
   848→    name = 'Momentum Negativ';
   849→    description = `ROC i përgjithshëm negativ (5d: ${roc5.toFixed(2)}%, 10d: ${roc10.toFixed(2)}%, 20d: ${roc20.toFixed(2)}%)`;
   850→  } else {
   851→    name = 'Momentum Neutral';
   852→    description = 'Shkalla e ndryshimit afatshkurtër afër zero';
   853→  }
   854→
   855→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: Math.abs(score) > 40 ? 'MEDIUM' : 'LOW', description };
   856→}

   933→function scoreMACDCrossover(
   934→  macdLine: number[],
   935→  signalLine: number[],
   936→): IndicatorScore {
   937→  const n = macdLine.length;
   938→  let score = 0;
   939→  let name = 'Asnjë Kryqëzim MACD';
   940→  let description = 'Asnjë kryqëzim i fundit midis MACD dhe vijës së sinjalit';
   941→  let found = false;
   942→
   943→  // Check last 5 valid bars for crossover
   944→  const recentIndices: number[] = [];
   945→  for (let i = n - 1; i >= 0 && recentIndices.length < 6; i--) {
   946→    if (!isNaN(macdLine[i]) && !isNaN(signalLine[i])) recentIndices.unshift(i);
   947→  }
   948→
   949→  for (let j = 1; j < recentIndices.length; j++) {
   950→    const prevIdx = recentIndices[j - 1];
   951→    const currIdx = recentIndices[j];
   952→
   953→    const prevDiff = macdLine[prevIdx] - signalLine[prevIdx];
   954→    const currDiff = macdLine[currIdx] - signalLine[currIdx];
   955→
   956→    if (prevDiff <= 0 && currDiff > 0) {
   957→      // Bullish crossover
   958→      const recency = j / recentIndices.length; // 1 = most recent
   959→      score = mapRange(recency, 0, 1, 65, 90);
   960→      name = 'Kryqëzim Bulliz MACD';
   961→      description = 'MACD kaloi sipër vijës së sinjalit, sinjal i blerjes';
   962→      found = true;
   963→      break;
   964→    } else if (prevDiff >= 0 && currDiff < 0) {
   965→      // Bearish crossover
   966→      const recency = j / recentIndices.length;
   967→      score = mapRange(recency, 0, 1, -65, -90);
   968→      name = 'Kryqëzim Bearish MACD';
   969→      description = 'MACD kaloi poshtë vijës së sinjalit, sinjal i shitjes';
   970→      found = true;
   971→      break;
   972→    }
   973→  }
   974→
   975→  if (!found) score = 0;
   976→
   977→  return { score, name, signal: found ? (score > 0 ? 'BULLISH' : 'BEARISH') : 'NEUTRAL', impact: found ? 'HIGH' : 'LOW', description };

   980→function scorePriceChannel(
   981→  highs: number[],
   982→  lows: number[],
   983→  closes: number[],
   984→): IndicatorScore {
   985→  const last = closes.length - 1;
   986→  if (last < 20) return noData('Price Channel');
   987→
   988→  let ch20High = -Infinity;
   989→  let ch20Low = Infinity;
   990→  for (let i = last - 19; i < last; i++) {
   991→    if (highs[i] > ch20High) ch20High = highs[i];
   992→    if (lows[i] < ch20Low) ch20Low = lows[i];
   993→  }
   994→
   995→  const price = closes[last];
   996→  const range = ch20High - ch20Low;
   997→  if (range <= 0) return noData('Price Channel');
   998→
   999→  let score: number;
  1000→  let name: string;
  1001→  let description: string;
  1002→
  1003→  if (price > ch20High) {
  1004→    // Breakout
  1005→    score = mapRange((price - ch20High) / range, 0, 0.1, 70, 95);
  1006→    name = 'Shpërthim mbi Kanalin 20-Ditor';
  1007→    description = 'Çmimi theu rezistencën e 20 ditëve, shpërthim i potencialit bulliz';
  1008→  } else if (price < ch20Low) {
  1009→    // Breakdown
  1010→    score = mapRange((ch20Low - price) / range, 0, 0.1, -70, -95);
  1011→    name = 'Thyesje nën Kanalin 20-Ditor';
  1012→    description = 'Çmimi thehu mbështetjen e 20 ditëve, rrezik i thellë uljeje';
  1013→  } else {
  1014→    const position = (price - ch20Low) / range;
  1015→    if (position > 0.7) {
  1016→      score = mapRange(position, 0.7, 1, 50, 75);
  1017→      name = 'Çmimi Pranë Rezistencës';
  1018→      description = 'Çmimi pranë lartësisë 20-ditore, momentum pozitiv';
  1019→    } else if (position < 0.3) {
  1020→      score = mapRange(position, 0, 0.3, -75, -50);
  1021→      name = 'Çmimi Pranë Mbështetjes';
  1022→      description = 'Çmimi pranë ulëtësisë 20-ditore, rreziku i uljes';
  1023→    } else {
  1024→      score = mapRange(position - 0.5, -0.2, 0.2, -30, 30);
  1025→      name = 'Price Channel Neutral';
  1026→      description = 'Çmimi në mes të kanalit 20-ditor';
  1027→    }
  1028→  }
  1029→
  1030→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: Math.abs(score) > 60 ? 'MEDIUM' : 'LOW', description };
  1031→}

  1033→function scoreDivergence(
  1034→  closes: number[],
  1035→  rsiValues: number[],
  1036→): IndicatorScore {
  1037→  const last = closes.length - 1;
  1038→  if (last < 30) return noData('Divergjencë');
  1039→
  1040→  // Find price lows and RSI lows in last 30 bars
  1041→  const lookback = Math.min(30, last);
  1042→  const priceSlice = closes.slice(last - lookback, last + 1);
  1043→  const rsiSlice = rsiValues.slice(last - lookback, last + 1);
  1044→
  1045→  // Find local minima (price lows)
  1046→  const priceLows: { idx: number; val: number }[] = [];
  1047→  const rsiAtPriceLows: number[] = [];
  1048→  for (let i = 1; i < priceSlice.length - 1; i++) {
  1049→    if (priceSlice[i] < priceSlice[i - 1] && priceSlice[i] < priceSlice[i + 1] && !isNaN(rsiSlice[i])) {
  1050→      priceLows.push({ idx: i, val: priceSlice[i] });
  1051→      rsiAtPriceLows.push(rsiSlice[i]);
  1052→    }
  1053→  }
  1054→
  1055→  // Find local maxima (price highs)
  1056→  const priceHighs: { idx: number; val: number }[] = [];
  1057→  const rsiAtPriceHighs: number[] = [];
  1058→  for (let i = 1; i < priceSlice.length - 1; i++) {
  1059→    if (priceSlice[i] > priceSlice[i - 1] && priceSlice[i] > priceSlice[i + 1] && !isNaN(rsiSlice[i])) {
  1060→      priceHighs.push({ idx: i, val: priceSlice[i] });
  1061→      rsiAtPriceHighs.push(rsiSlice[i]);
  1062→    }
  1063→  }
  1064→
  1065→  // Bullish divergence: price makes lower low but RSI makes higher low
  1066→  let bullishDiv = false;
  1067→  let bearishDiv = false;
  1068→
  1069→  if (priceLows.length >= 2) {
  1070→    const lastTwo = priceLows.slice(-2);
  1071→    if (lastTwo[1].val < lastTwo[0].val) {
  1072→      const rsiLastTwo = rsiAtPriceLows.slice(-2);
  1073→      if (rsiLastTwo.length >= 2 && rsiLastTwo[1] > rsiLastTwo[0]) {
  1074→        bullishDiv = true;
  1075→      }
  1076→    }
  1077→  }
  1078→
  1079→  // Bearish divergence: price makes higher high but RSI makes lower high
  1080→  if (priceHighs.length >= 2) {
  1081→    const lastTwo = priceHighs.slice(-2);
  1082→    if (lastTwo[1].val > lastTwo[0].val) {
  1083→      const rsiLastTwo = rsiAtPriceHighs.slice(-2);
  1084→      if (rsiLastTwo.length >= 2 && rsiLastTwo[1] < rsiLastTwo[0]) {
  1085→        bearishDiv = true;
  1086→      }
  1087→    }
  1088→  }
  1089→
  1090→  if (bullishDiv) {
  1091→    return {
  1092→      score: 75,
  1093→      name: 'Divergjencë Bulliz',
  1094→      signal: 'BULLISH',
  1095→      impact: 'HIGH',
  1096→      description: 'Çmimi bën ultë të ulëta por RSI bën ultë më të larta, sinjal i forttë i blerjes',
  1097→    };
  1098→  }
  1099→  if (bearishDiv) {
  1100→    return {
  1101→      score: -75,
  1102→      name: 'Divergjencë Bearish',
  1103→      signal: 'BEARISH',
  1104→      impact: 'HIGH',
  1105→      description: 'Çmimi bën maje më të larta por RSI bën maje më të ulëta, sinjal i fortë i shitjes',
  1106→    };
  1107→  }
  1108→
  1109→  return {
  1110→    score: 0,
  1111→    name: 'Asnjë Divergjencë',
  1112→    signal: 'NEUTRAL',
  1113→    impact: 'LOW',
  1114→    description: 'Asnjë divergjencë e dukshme midis çmimit dhe RSI-së',
  1115→  };
  1116→}

  1118→function scoreVWAP(closes: number[], highs: number[], lows: number[], volumes: number[]): IndicatorScore {
  1119→  const last = closes.length - 1;
  1120→  if (last < 20) return noData('VWAP');
  1121→
  1122→  // Approximate VWAP over 20 days: cumulative (H+L+C)/3 * volume / cumulative volume
  1123→  let cumVP = 0;
  1124→  let cumVol = 0;
  1125→  for (let i = last - 19; i <= last; i++) {
  1126→    const tp = (highs[i] + lows[i] + closes[i]) / 3;
  1127→    cumVP += tp * volumes[i];
  1128→    cumVol += volumes[i];
  1129→  }
  1130→
  1131→  const vwap = cumVol === 0 ? closes[last] : cumVP / cumVol;
  1132→  const price = closes[last];
  1133→  const deviation = ((price - vwap) / vwap) * 100; // percentage deviation
  1134→
  1135→  let score: number;
  1136→  let name: string;
  1137→  let description: string;
  1138→
  1139→  if (price > vwap) {
  1140→    score = clamp(mapRange(deviation, 0, 5, 50, 65), 50, 65);
  1141→    name = 'Çmimi Sipër VWAP';
  1142→    description = `Çmimi ${deviation.toFixed(2)}% mbi VWAP, blerësit në kontroll`;
  1143→  } else {
  1144→    score = clamp(mapRange(-deviation, 0, 5, 50, 65), -65, -50);
  1145→    name = 'Çmimi Poshtë VWAP';
  1146→    description = `Çmimi ${(-deviation).toFixed(2)}% poshtë VWAP, shitësit në kontroll`;
  1147→  }
  1148→
  1149→  return { score, name, signal: score >= 0 ? 'BULLISH' : 'BEARISH', impact: 'LOW', description };
  1150→}

  1152→function scorePattern(
  1153→  opens: number[],
  1154→  highs: number[],
  1155→  lows: number[],
  1156→  closes: number[],
  1157→  volumes: number[],
  1158→): IndicatorScore {
  1159→  const last = closes.length - 1;
  1160→  if (last < 10) return noData('Pattern');
  1161→
  1162→  const len = Math.min(10, last + 1);
  1163→  const o = opens.slice(last - len + 1);
  1164→  const h = highs.slice(last - len + 1);
  1165→  const l = lows.slice(last - len + 1);
  1166→  const c = closes.slice(last - len + 1);
  1167→  const v = volumes.slice(last - len + 1);
  1168→
  1169→  let bestScore = 0;
  1170→  let bestName = 'Asnjë Model';
  1171→  let bestDescription = 'Nuk u zbuluan modele qirinjësh në 10 ditët e fundit';
  1172→  let bestSignal: SignalType = 'NEUTRAL';
  1173→  let bestImpact: ImpactLevel = 'LOW';
  1174→
  1175→  const n = o.length;
  1176→
  1177→  // 1. Three consecutive green candles with increasing volume
  1178→  if (n >= 3) {
  1179→    const g1 = c[n - 3] > o[n - 3];
  1180→    const g2 = c[n - 2] > o[n - 2];
  1181→    const g3 = c[n - 1] > o[n - 1];
  1182→    if (g1 && g2 && g3 && v[n - 1] > v[n - 2] && v[n - 2] > v[n - 3]) {
  1183→      bestScore = 75;
  1184→      bestName = 'Tre Qirinj Jeshilë';
  1185→      bestSignal = 'BULLISH';
  1186→      bestImpact = 'HIGH';
  1187→      bestDescription = 'Tre qirinj jeshilë radhazi me volum në rritje, prirje e fortë e blerjes';
  1188→    }
  1189→  }
  1190→
  1191→  // 2. Three consecutive red candles with increasing volume
  1192→  if (n >= 3) {
  1193→    const r1 = c[n - 3] < o[n - 3];
  1194→    const r2 = c[n - 2] < o[n - 2];
  1195→    const r3 = c[n - 1] < o[n - 1];
  1196→    if (r1 && r2 && r3 && v[n - 1] > v[n - 2] && v[n - 2] > v[n - 3]) {
  1197→      if (Math.abs(-75) > Math.abs(bestScore)) {
  1198→        bestScore = -75;
  1199→        bestName = 'Tre Qirinj të Kuq';
  1200→        bestSignal = 'BEARISH';
  1201→        bestImpact = 'HIGH';
  1202→        bestDescription = 'Tre qirinj të kuq radhazi me volum në rritje, prirje e fortë e shitjes';
  1203→      }
  1204→    }
  1205→  }
  1206→
  1207→  // 3. Hammer: small body, long lower shadow (at least 2x body), body in upper 1/3
  1208→  {
  1209→    const body = Math.abs(c[n - 1] - o[n - 1]);
  1210→    const totalRange = h[n - 1] - l[n - 1];
  1211→    const lowerShadow = Math.min(o[n - 1], c[n - 1]) - l[n - 1];
  1212→    const upperShadow = h[n - 1] - Math.max(o[n - 1], c[n - 1]);
  1213→
  1214→    if (totalRange > 0 && body > 0 && lowerShadow >= 2 * body && upperShadow <= body * 0.3) {
  1215→      // Check if at bottom of trend (price was declining)
  1216→      const declining = c[n - 2] < o[n - 2] || c[n - 3] < o[n - 3];
  1217→      if (declining) {
  1218→        const score = 65;
  1219→        if (score > Math.abs(bestScore)) {
  1220→          bestScore = score;
  1221→          bestName = 'Modeli Çekiç';
  1222→          bestSignal = 'BULLISH';
  1223→          bestImpact = 'MEDIUM';
  1224→          bestDescription = 'Modeli çekiç pas një prirjeje të ulët, sinjal i shmangies pozitive';
  1225→        }
  1226→      }
  1227→    }
  1228→  }
  1229→
  1230→  // 4. Shooting star: small body, long upper shadow (at least 2x body), body in lower 1/3
  1231→  {
  1232→    const body = Math.abs(c[n - 1] - o[n - 1]);
  1233→    const totalRange = h[n - 1] - l[n - 1];
  1234→    const lowerShadow = Math.min(o[n - 1], c[n - 1]) - l[n - 1];
  1235→    const upperShadow = h[n - 1] - Math.max(o[n - 1], c[n - 1]);
  1236→
  1237→    if (totalRange > 0 && body > 0 && upperShadow >= 2 * body && lowerShadow <= body * 0.3) {
  1238→      const advancing = c[n - 2] > o[n - 2] || c[n - 3] > o[n - 3];
  1239→      if (advancing) {
  1240→        const score = -65;
  1241→        if (Math.abs(score) > Math.abs(bestScore)) {
  1242→          bestScore = score;
  1243→          bestName = 'Yllë e Qitjes';
  1244→          bestSignal = 'BEARISH';
  1245→          bestImpact = 'MEDIUM';
  1246→          bestDescription = 'Yllë e qitjes pas një prirjeje të lartë, sinjal i shmangies negative';
  1247→        }
  1248→      }
  1249→    }
  1250→  }
  1251→
  1252→  // 5. Bullish engulfing: red candle followed by larger green candle
  1253→  if (n >= 2) {
  1254→    const prevRed = c[n - 2] < o[n - 2];
  1255→    const currGreen = c[n - 1] > o[n - 1];
  1256→    const prevBody = Math.abs(c[n - 2] - o[n - 2]);
  1257→    const currBody = Math.abs(c[n - 1] - o[n - 1]);
  1258→    const engulfsBody = c[n - 1] > o[n - 2] && o[n - 1] < c[n - 2];
  1259→
  1260→    if (prevRed && currGreen && engulfsBody && currBody > prevBody) {
  1261→      const score = 70;
  1262→      if (Math.abs(score) > Math.abs(bestScore)) {
  1263→        bestScore = score;
  1264→        bestName = 'Mbështjellje Bulliz';
  1265→        bestSignal = 'BULLISH';
  1266→        bestImpact = 'HIGH';
  1267→        bestDescription = 'Modeli i mbështjelljes bullize, sinjal i fortë për ndryshim të drejtimit';
  1268→      }
  1269→    }
  1270→  }
  1271→
  1272→  // 6. Bearish engulfing: green candle followed by larger red candle
  1273→  if (n >= 2) {
  1274→    const prevGreen = c[n - 2] > o[n - 2];
  1275→    const currRed = c[n - 1] < o[n - 1];
  1276→    const prevBody = Math.abs(c[n - 2] - o[n - 2]);
  1277→    const currBody = Math.abs(c[n - 1] - o[n - 1]);
  1278→    const engulfsBody = o[n - 1] > c[n - 2] && c[n - 1] < o[n - 2];
  1279→
  1280→    if (prevGreen && currRed && engulfsBody && currBody > prevBody) {
  1281→      const score = -70;
  1282→      if (Math.abs(score) > Math.abs(bestScore)) {
  1283→        bestScore = score;
  1284→        bestName = 'Mbështjellje Bearish';
  1285→        bestSignal = 'BEARISH';
  1286→        bestImpact = 'HIGH';
  1287→        bestDescription = 'Modeli i mbështjelljes bearish, sinjal i fortë për ndryshim të drejtimit';
  1288→      }
  1289→    }
  1290→  }
  1291→
  1292→  return { score: bestScore, name: bestName, signal: bestSignal, impact: bestImpact, description: bestDescription };
  1293→}

  1327→    ],

  1330→    technicalScore: 0,
  1331→    fundamentalData: null,
  1332→    learningData: null,
  1333→    combinedScore: 0,
  1334→  });
  1335→

  1337→    console.warn(`[PREDICT-ENGINE] ${symbol}: Insufficient data (${data?.length ?? 0} < ${MIN_DATA_POINTS})`);
  1338→    return emptyResult();
  1339→  }
  1340→
  1341→  // ---------- Extract price arrays ----------
  1342→  const closes = data.map((d) => d.close);
  1343→  const highs = data.map((d) => d.high);
  1344→  const lows = data.map((d) => d.low);
  1345→  const volumes = data.map((d) => d.volume);
  1346→  const opens = data.map((d) => d.open);
  1347→
  1348→  // Validate
  1349→  const zeroCloseCount = closes.filter(c => c <= 0).length;
  1350→  const negVolCount = volumes.filter(v => v < 0).length;
  1351→  if (zeroCloseCount > 0 || negVolCount > 0) {
  1352→    console.warn(`[PREDICT-ENGINE] ${symbol}: Invalid data — zeroClose: ${zeroCloseCount}, negVol: ${negVolCount}, totalPoints: ${data.length}`);
  1353→    return emptyResult();
  1354→  }
  1355→
  1356→  // ---------- Calculate raw indicators ----------
  1357→  const rsiValues = calculateRSI(closes, 14);
  1358→  const macd = calculateMACD(closes);
  1359→  const bb = calculateBollingerBands(closes, 20, 2);
  1360→  const sma20 = sma(closes, 20);
  1361→  const sma50 = sma(closes, 50);
  1362→  const sma200 = sma(closes, 200);
  1363→  const stoch = calculateStochastic(highs, lows, closes, 14, 3);
  1364→  const adxResult = calculateADX(highs, lows, closes, 14);
  1365→  const atrValues = calculateATR(highs, lows, closes, 14);
  1366→  const atr20Values = calculateATR(highs, lows, closes, 20);
  1367→
  1368→  // ---------- Score each indicator ----------
  1369→  const scores: Record<string, IndicatorScore> = {};
  1370→
  1371→  scores.rsi = scoreRSI(rsiValues);
  1372→  scores.macdHistogram = scoreMACDHistogram(macd.histogram);
  1373→  scores.bollinger = scoreBollinger(closes, bb);
  1374→  scores.maTrend = scoreMATrend(closes, sma20, sma50, sma200);
  1375→  scores.stochastic = scoreStochastic(stoch.k, stoch.d);
  1376→  scores.adx = scoreADX(adxResult.adx, adxResult.plusDI, adxResult.minusDI);
  1377→  scores.atr = scoreATR(atrValues, closes);
  1378→  scores.roc = scoreROC(closes);
  1379→  scores.obv = scoreOBV(closes, volumes);
  1380→  scores.volumeConfirm = scoreVolumeConfirmation(closes, volumes);
  1381→  scores.macdCrossover = scoreMACDCrossover(macd.macdLine, macd.signalLine);
  1382→  scores.priceChannel = scorePriceChannel(highs, lows, closes);
  1383→  scores.divergence = scoreDivergence(closes, rsiValues);
  1384→  scores.vwap = scoreVWAP(closes, highs, lows, volumes);
  1385→  scores.pattern = scorePattern(opens, highs, lows, closes, volumes);
  1386→

  1388→  const scoreEntries = Object.entries(scores);
  1389→  const nonZeroScores = scoreEntries.filter(([, s]) => s.score !== 0);
  1390→  console.log(`[PREDICT-ENGINE] ${symbol}: ${data.length} points, ${nonZeroScores.length}/${scoreEntries.length} indicators non-zero, lastClose=${closes[closes.length - 1]?.toFixed(2)}`);
  1391→
  1392→  // ---------- Weighted final score ----------
  1393→  const weights = getWeights();
  1394→  let finalScore = 0;
  1395→  const indicatorScores: Record<string, number> = {};
  1396→
  1397→  for (const [key, weight] of Object.entries(weights)) {
  1398→    const s = scores[key]?.score ?? 0;
  1399→    finalScore += s * weight;
  1400→    indicatorScores[key] = Math.round(s * 100) / 100;
  1401→  }
  1402→  finalScore = clamp(Math.round(finalScore * 100) / 100, -100, 100);
  1403→
  1404→  // ---------- Confidence ----------
  1405→  const allIndicatorScores = Object.values(scores).map((s) => s.score);
  1406→  const mean = allIndicatorScores.reduce((a, b) => a + b, 0) / allIndicatorScores.length;
  1407→  const variance = allIndicatorScores.reduce((a, b) => a + (b - mean) ** 2, 0) / allIndicatorScores.length;
  1408→  const stdDev = Math.sqrt(variance);
  1409→  const confidence = clamp(Math.round((100 - stdDev * 1.5) * 100) / 100, 0, 100);
  1410→
  1411→  // ---------- Direction ----------
  1412→  const direction: Direction =
  1413→    finalScore > 60 ? 'STRONG_BUY' : finalScore > 25 ? 'BUY' : finalScore > -25 ? 'NEUTRAL' : finalScore > -60 ? 'SELL' : 'STRONG_SELL';
  1414→
  1415→  // ---------- Short-term prediction (1-3 days) ----------
  1416→  const shortTermWeights = { rsi: 0.25, stochastic: 0.20, macdHistogram: 0.25, volumeConfirm: 0.15, pattern: 0.15 };
  1417→  let shortScore = 0;
  1418→  for (const [k, w] of Object.entries(shortTermWeights)) {
  1419→    shortScore += (scores[k]?.score ?? 0) * w;
  1420→  }
  1421→  const shortPrediction: TrendPrediction = shortScore > 20 ? 'UP' : shortScore < -20 ? 'DOWN' : 'SIDEWAYS';
  1422→  const shortProbability = clamp(Math.round(Math.abs(shortScore) * 0.8 + 20), 0, 100);
  1423→  const currentATR = lastValid(atrValues) ?? 0;
  1424→  const lastClose = closes[closes.length - 1];
  1425→  const shortExpectedMove = lastClose > 0 ? Math.round(((currentATR / lastClose) * 100) * 100) / 100 : 0;
  1426→
  1427→  // ---------- Medium-term prediction (1-2 weeks) ----------
  1428→  const medTermWeights = { maTrend: 0.30, macdHistogram: 0.20, bollinger: 0.20, obv: 0.15, roc: 0.15 };
  1429→  let medScore = 0;
  1430→  for (const [k, w] of Object.entries(medTermWeights)) {
  1431→    medScore += (scores[k]?.score ?? 0) * w;
  1432→  }
  1433→  const medPrediction: TrendPrediction = medScore > 20 ? 'UP' : medScore < -20 ? 'DOWN' : 'SIDEWAYS';
  1434→  const medProbability = clamp(Math.round(Math.abs(medScore) * 0.8 + 20), 0, 100);
  1435→  const currentATR20 = lastValid(atr20Values) ?? currentATR;
  1436→  const medExpectedMove = lastClose > 0 ? Math.round(((currentATR20 / lastClose) * 100) * 100) / 100 : 0;
  1437→
  1438→  // ---------- Risk level & volatility ----------
  1439→  const atrHistory: number[] = [];
  1440→  const atrLast = atrValues.length - 1;
  1441→  for (let i = atrLast; i >= 0 && atrHistory.length < 60; i--) {
  1442→    if (!isNaN(atrValues[i]) && atrValues[i] > 0) atrHistory.unshift(atrValues[i]);
  1443→  }
  1444→
  1445→  const p30 = atrHistory.length > 5 ? percentile(atrHistory, 30) : 0;
  1446→  const p70 = atrHistory.length > 5 ? percentile(atrHistory, 70) : Infinity;
  1447→
  1448→  const atrHigh = currentATR > p70;
  1449→  const atrLow = currentATR < p30;
  1450→
  1451→  const volatility: RiskOrVolatility = atrHigh ? 'HIGH' : atrLow ? 'LOW' : 'MEDIUM';
  1452→
  1453→  const riskLevel: RiskOrVolatility =
  1454→    atrHigh || confidence < 30 ? 'HIGH' : atrLow && confidence > 70 ? 'LOW' : 'MEDIUM';
  1455→
  1456→  // ---------- Top 5 key factors ----------
  1457→  const sortedFactors = Object.values(scores)
  1458→    .sort((a, b) => Math.abs(b.score) - Math.abs(a.score))
  1459→    .slice(0, 5);
  1460→
  1461→  const keyFactors: KeyFactor[] = sortedFactors.map((f) => ({
  1462→    name: f.name,
  1463→    signal: f.signal,
  1464→    impact: f.impact,
  1465→    score: Math.round(f.score * 100) / 100,
  1466→    description: f.description,
  1467→  }));
  1468→
  1469→  // ---------- Build result ----------
  1470→  return {
  1471→    symbol,
  1472→    score: finalScore,
  1473→    direction,
  1474→    confidence,
  1475→    shortTerm: {
  1476→      prediction: shortPrediction,
  1477→      probability: shortProbability,
  1478→      expectedMove: shortExpectedMove,
  1479→    },
  1480→    mediumTerm: {
  1481→      prediction: medPrediction,
  1482→      probability: medProbability,
  1483→      expectedMove: medExpectedMove,
  1484→    },
  1485→    riskLevel,
  1486→    volatility,
  1487→    keyFactors,
  1488→    indicatorScores,
  1489→    timestamp: new Date().toISOString(),
  1490→    technicalScore: finalScore,
  1491→    fundamentalData: null,
  1492→    learningData: null,
  1493→    combinedScore: finalScore,
  1494→  };
  1495→}

  1497→// ============= ENHANCED PREDICTION WITH FUNDAMENTALS + LEARNING =============
  1498→
  1499→interface EnhancedPredictionOptions {
  1500→  fundamentalScore?: number | null;
  1501→  fundamentalSummary?: string;
  1502→  fundamentalScores?: Record<string, number>;
  1503→  learningStats?: {
  1504→    totalPredictions: number;
  1505→    directionAccuracy: number;
  1506→    shortTermAccuracy: number;
  1507→    mediumTermAccuracy: number;
  1508→    bestIndicators: string[];
  1509→    worstIndicators: string[];
  1510→    recentAccuracy: number;
  1511→  } | null;
  1512→  /** Weight for fundamentals (0-1, default 0.25) */
  1513→  fundamentalWeight?: number;
  1514→}
  1515→
  1516→/**
  1517→ * Enhanced prediction that combines technical + fundamental scores.
  1518→ * Uses learning weights if available.
  1519→ */
  1520→export function predictStockEnhanced(
  1521→  baseResult: PredictionResult,
  1522→  options: EnhancedPredictionOptions = {},
  1523→): PredictionResult {
  1524→  const fundWeight = options.fundamentalWeight ?? 0.25;
  1525→  const techWeight = 1 - fundWeight;
  1526→  const techScore = baseResult.technicalScore;
  1527→  const fundScore = options.fundamentalScore ?? 0;
  1528→
  1529→  // Adjust fund weight based on learning data
  1530→  let adjustedFundWeight = fundWeight;
  1531→  if (options.learningStats && options.learningStats.totalPredictions > 10) {
  1532→    // If fundamentals have been more accurate, increase their weight
  1533→    const avgAcc = (options.learningStats.directionAccuracy + options.learningStats.recentAccuracy) / 2;
  1534→    if (avgAcc > 55) {
  1535→      adjustedFundWeight = Math.min(fundWeight * 1.3, 0.40);
  1536→    } else if (avgAcc < 45) {
  1537→      adjustedFundWeight = fundWeight * 0.7;
  1538→    }
  1539→  }
  1540→
  1541→  const adjustedTechWeight = 1 - adjustedFundWeight;
  1542→  const combinedScore = clamp(
  1543→    Math.round((techScore * adjustedTechWeight + fundScore * adjustedFundWeight) * 100) / 100,
  1544→    -100, 100
  1545→  );
  1546→
  1547→  // Recalculate direction based on combined score
  1548→  const direction: Direction =
  1549→    combinedScore > 60 ? 'STRONG_BUY' : combinedScore > 25 ? 'BUY' : combinedScore > -25 ? 'NEUTRAL' : combinedScore > -60 ? 'SELL' : 'STRONG_SELL';
  1550→
  1551→  // Add fundamental key factors if available
  1552→  const keyFactors = [...baseResult.keyFactors];
  1553→  if (options.fundamentalSummary) {
  1554→    // Insert fundamental as first factor if significant
  1555→    if (Math.abs(fundScore) > 15) {
  1556→      keyFactors.unshift({
  1557→        name: 'Analiza Fondamentale',
  1558→        signal: fundScore > 0 ? 'BULLISH' as SignalType : 'BEARISH' as SignalType,
  1559→        impact: Math.abs(fundScore) > 40 ? 'HIGH' as ImpactLevel : 'MEDIUM' as ImpactLevel,
  1560→        score: fundScore,
  1561→        description: options.fundamentalSummary,
  1562→      });
  1563→    }
  1564→  }
  1565→
  1566→  // Trim to top 5
  1567→  const topFactors = keyFactors
  1568→    .sort((a, b) => Math.abs(b.score) - Math.abs(a.score))
  1569→    .slice(0, 5);
  1570→
  1571→  // Boost confidence if both technical and fundamental agree
  1572→  let confidence = baseResult.confidence;
  1573→  if (techScore > 15 && fundScore > 15) {
  1574→    confidence = clamp(confidence + 8, 0, 100);
  1575→  } else if (techScore < -15 && fundScore < -15) {
  1576→    confidence = clamp(confidence + 8, 0, 100);
  1577→  } else if ((techScore > 15 && fundScore < -15) || (techScore < -15 && fundScore > 15)) {
  1578→    confidence = clamp(confidence - 10, 0, 100);
  1579→  }
  1580→  confidence = Math.round(confidence * 100) / 100;
  1581→
  1582→  // Update short/medium term predictions with fundamental influence
  1583→  const shortTerm = { ...baseResult.shortTerm };
  1584→  const mediumTerm = { ...baseResult.mediumTerm };
  1585→
  1586→  // Fundamental influence on medium term (stronger effect)
  1587→  if (Math.abs(fundScore) > 20) {
  1588→    const fundBias = fundScore > 0 ? 5 : -5;
  1589→    mediumTerm.probability = clamp(mediumTerm.probability + Math.abs(fundBias), 0, 100);
  1590→    if (fundScore > 40 && mediumTerm.prediction === 'SIDEWAYS') {
  1591→      mediumTerm.prediction = 'UP';
  1592→    } else if (fundScore < -40 && mediumTerm.prediction === 'SIDEWAYS') {
  1593→      mediumTerm.prediction = 'DOWN';
  1594→    }
  1595→  }
  1596→
  1597→  return {
  1598→    ...baseResult,
  1599→    score: combinedScore,
  1600→    direction,
  1601→    confidence,
  1602→    keyFactors: topFactors,
  1603→    fundamentalData: options.fundamentalScore !== null && options.fundamentalScore !== undefined
  1604→      ? {
  1605→          score: fundScore,
  1606→          summary: options.fundamentalSummary || '',
  1607→          scores: options.fundamentalScores || {},
  1608→        }
  1609→      : null,
  1610→    learningData: options.learningStats ?? null,
  1611→    combinedScore,
  1612→    shortTerm,
  1613→    mediumTerm,
  1614→  };
  1615→}

  1620→  return [...results].sort((a, b) => b.score - a.score);
