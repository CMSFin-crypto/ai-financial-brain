    97→// Results are ALWAYS the same for same ticker

   100→function hashString(str: string): number {
   101→  let hash = 0;
   102→  for (let i = 0; i < str.length; i++) {
   103→    const ch = str.charCodeAt(i);
   104→    hash = ((hash << 5) - hash) + ch;
   105→    hash |= 0;
   106→  }
   107→  return Math.abs(hash);
   108→}
   109→
   110→function seededRandom(seed: number): number {
   111→  const x = Math.sin(seed + 1) * 10000;
   112→  return x - Math.floor(x);
   113→}
   114→
   115→// Deterministic random within range, seeded by ticker
   116→function dr(ticker: string, index: number, min: number, max: number): number {
   117→  const seed = hashString(ticker) + index * 7919;
   118→  return min + seededRandom(seed) * (max - min);
   119→}
   120→
   121→function drInt(ticker: string, index: number, min: number, max: number): number {
   122→  return Math.round(dr(ticker, index, min, max));
   123→}

   125→// Generate ~130 days (6 months) of realistic candlestick data, deterministic per ticker
   126→function generateCandlestickData(ticker: string, basePrice: number, trend: 'uptrend' | 'downtrend' | 'sideways') {
   127→  const data: Array<{ date: string; open: number; high: number; low: number; close: number; volume: number }> = [];
   128→  const today = new Date();
   129→
   130→  // Work backwards ~130 trading days (6 months)
   131→  let price = basePrice;
   132→  const days = 130;
   133→
   134→  // First pass: calculate all closes working backwards
   135→  const closes: number[] = [];
   136→  for (let i = days; i >= 0; i--) {
   137→    closes.push(price);
   138→    const drift = trend === 'uptrend' ? 0.002 : trend === 'downtrend' ? -0.0015 : 0;
   139→    const change = dr(ticker, i * 7, -0.015, 0.018) + drift;
   140→    price = price / (1 + change); // reverse to get previous price
   141→  }
   142→
   143→  // Reverse so we go from oldest to newest
   144→  closes.reverse();
   145→
   146→  // Second pass: generate OHLC for each day
   147→  for (let i = 0; i < days; i++) {
   148→    const d = new Date(today);
   149→    d.setDate(d.getDate() - (days - 1 - i));
   150→    // Skip weekends
   151→    if (d.getDay() === 0 || d.getDay() === 6) continue;
   152→
   153→    const dateStr = d.toISOString().split('T')[0];
   154→    const open = closes[i];
   155→    const close = closes[i + 1] || basePrice;
   156→    const spread = Math.abs(close - open);
   157→    const high = Math.max(open, close) + spread * dr(ticker, i * 13 + 1000, 0.1, 0.5);
   158→    const low = Math.min(open, close) - spread * dr(ticker, i * 13 + 2000, 0.1, 0.5);
   159→    const volume = Math.floor(dr(ticker, i * 17 + 3000, 30000000, 80000000));
   160→
   161→    data.push({
   162→      date: dateStr,
   163→      open: +open.toFixed(2),
   164→      high: +high.toFixed(2),
   165→      low: +Math.max(low, 0.01).toFixed(2),
   166→      close: +close.toFixed(2),
   167→      volume,
   168→    });
   169→  }
   170→
   171→  return data;
   172→}
