   244→// ── Bracket Order Block ──
   245→function BracketOrderBlock({ order }: { order: object }) {
   246→  const [copied, setCopied] = useState(false);
   247→  const jsonStr = JSON.stringify(order, null, 2);
   248→  const handleCopy = () => {
   249→    navigator.clipboard.writeText(jsonStr);
   250→    setCopied(true);
   251→    setTimeout(() => setCopied(false), 2000);
   252→  };
   253→  return (
   254→    <div className="mt-2 rounded-lg bg-emerald-500/5 border border-emerald-500/15 p-3">
   255→      <div className="flex items-center justify-between mb-2">
   256→        <div className="flex items-center gap-2">
   257→          <FileText className="w-4 h-4 text-emerald-400" />
   258→          <p className="text-[13px] font-semibold text-emerald-400">IBKR Bracket Order</p>
   259→        </div>
   260→        <button onClick={handleCopy} className="flex items-center gap-1 text-[11px] px-2 py-1 rounded bg-emerald-500/10 text-emerald-400 hover:bg-emerald-500/20 transition-colors">
   261→          {copied ? <Check className="w-3 h-3" /> : <Copy className="w-3 h-3" />}
   262→          {copied ? 'Kopjuar!' : 'Kopjo JSON'}
   263→        </button>
   264→      </div>
   265→      <pre className="text-[11px] text-foreground/70 bg-black/20 rounded p-2.5 overflow-x-auto max-h-40 leading-relaxed">{jsonStr}</pre>
   266→    </div>
   267→  );
   268→}

   711→// ── Single Stock Search Component ──
   712→function StockSearchBox() {
   713→  const [query, setQuery] = useState('');
   714→  const [results, setResults] = useState<SearchResult[]>([]);
   715→  const [showDropdown, setShowDropdown] = useState(false);
   716→  const [analyzing, setAnalyzing] = useState<string | null>(null);
   717→  const [analyzedStock, setAnalyzedStock] = useState<FunnelStock | null>(null);
   718→  const [analyzeError, setAnalyzeError] = useState<string | null>(null);
   719→  const [regimeOk, setRegimeOk] = useState(true);
   720→  const [funnelPhases, setFunnelPhases] = useState({ passedLiquidity: 0, passedTrend: 0, passedSetup: 0, passedRisk: 0 });
   721→  const timerRef = useRef<ReturnType<typeof setTimeout>>(undefined);
   722→  const inputRef = useRef<HTMLInputElement>(null);
   723→
   724→  // Search with debounce
   725→  const handleInput = (val: string) => {
   726→    setQuery(val);
   727→    if (timerRef.current) clearTimeout(timerRef.current);
   728→    if (val.length < 2) { setResults([]); setShowDropdown(false); return; }
   729→    timerRef.current = setTimeout(async () => {
   730→      try {
   731→        const res = await fetch(`/api/ticker-search?q=${encodeURIComponent(val)}`);
   732→        const json = await res.json();
   733→        setResults(json.results || []);
   734→        setShowDropdown(true);
   735→      } catch { setResults([]); }
   736→    }, 350);
   737→  };
   738→
   739→  const selectTicker = async (ticker: string) => {
   740→    setShowDropdown(false); setQuery(ticker);
   741→    setAnalyzedStock(null); setAnalyzeError(null);
   742→    setAnalyzing(ticker);
   743→    try {
   744→      const res = await fetch(`/api/ibkr-analyze/${ticker}?_t=${Date.now()}`, { cache: 'no-store' });
   745→      const json = await res.json();
   746→      setRegimeOk(json.regimeOk);
   747→      setFunnelPhases(json.funnel || { passedLiquidity: 0, passedTrend: 0, passedSetup: 0, passedRisk: 0 });
   748→      if (json.error && !json.stock) {
   749→        setAnalyzeError(json.error);
   750→      } else if (json.stock) {
   751→        setAnalyzedStock(json.stock);
   752→      }
   753→    } catch { setAnalyzeError('Gabim rrjeti'); }
   754→    setAnalyzing(null);
   755→  };
   756→
   757→  return (
   758→    <div className="space-y-3">
   759→      <div className="relative">
   760→        <div className="flex items-center gap-2">
   761→          <div className="relative flex-1">
   762→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground/50" />
   763→            <input
   764→              ref={inputRef}
   765→              value={query}
   766→              onChange={e => handleInput(e.target.value)}
   767→              onFocus={() => results.length > 0 && setShowDropdown(true)}
   768→              placeholder="Kerko nje aksion (p.sh. AAPL, TSLA, COIN...)"
   769→              className="w-full pl-9 pr-9 py-2.5 rounded-lg bg-muted/10 border border-border/50 text-[13px] text-foreground placeholder:text-muted-foreground/40 focus:outline-none focus:border-blue-500/50 focus:ring-1 focus:ring-blue-500/20 transition-all"
   770→            />
   771→            {query && (
   772→              <button onClick={() => { setQuery(''); setResults([]); setShowDropdown(false); setAnalyzedStock(null); setAnalyzeError(null); }} className="absolute right-3 top-1/2 -translate-y-1/2"><X className="w-3.5 h-3.5 text-muted-foreground/50 hover:text-muted-foreground" /></button>
   773→            )}
   774→          </div>
   775→          {query.length >= 1 && !analyzing && (
   776→            <button onClick={() => selectTicker(query.toUpperCase())} className="flex items-center gap-1.5 text-[13px] px-3 py-2.5 rounded-lg bg-violet-500/10 border border-violet-500/30 text-violet-400 hover:bg-violet-500/20 transition-colors whitespace-nowrap">
   777→              <Activity className="w-3.5 h-3.5" /> Analizo
   778→            </button>
   779→          )}
   780→          {analyzing && (
   781→            <div className="flex items-center gap-1.5 text-[13px] px-3 py-2.5 text-violet-400 whitespace-nowrap">
   782→              <Loader2 className="w-3.5 h-3.5 animate-spin" /> Duke analizuar {analyzing}...
   783→            </div>
   784→          )}
   785→        </div>
   786→        {/* Dropdown */}
   787→        {showDropdown && results.length > 0 && (
   788→          <div className="absolute z-50 top-full mt-1 w-full bg-popover border border-border/60 rounded-lg shadow-xl overflow-hidden">
   789→            {results.map(r => (
   790→              <button key={r.ticker} onClick={() => selectTicker(r.ticker)} className="w-full flex items-center gap-3 px-3.5 py-2.5 hover:bg-muted/10 transition-colors text-left">
   791→                <div className="flex-1 min-w-0">
   792→                  <span className="font-bold text-[13px] text-foreground">{r.ticker}</span>
   793→                  <span className="text-[12px] text-muted-foreground ml-2 truncate">{r.company}</span>
   794→                </div>
   795→                <div className="text-right flex-shrink-0">
   796→                  <span className="text-[13px] text-foreground font-medium">${r.price}</span>
   797→                  <span className={`text-[12px] ml-1.5 ${r.change >= 0 ? 'text-emerald-400' : 'text-red-400'}`}>{r.change >= 0 ? '+' : ''}{r.change.toFixed(1)}%</span>
   798→                </div>
   799→              </button>
   800→            ))}
   801→          </div>
   802→        )}
   803→      </div>
   804→
   805→      {/* Analyze Error */}
   806→      {analyzeError && (

   811→          </CardContent>
   812→        </Card>
   813→      )}
   814→
   815→      {/* Analyzed Stock Result */}
   816→      {analyzedStock && (
   817→        <div className="space-y-2">
   818→          {/* Funnel diagnostic for single stock */}
   819→          <div className="flex items-center gap-1.5 flex-wrap text-[11px]">
   820→            <span className={`px-2 py-1 rounded-md ${funnelPhases.passedLiquidity ? 'bg-cyan-500/20 text-cyan-400' : 'bg-red-500/10 text-red-400/60 line-through'}`}>Likuiditet {funnelPhases.passedLiquidity ? '✓' : '✗'}</span>
   821→            <ArrowDown className="w-3 h-3 text-muted-foreground/30" />
   822→            <span className={`px-2 py-1 rounded-md ${funnelPhases.passedTrend ? 'bg-emerald-500/20 text-emerald-400' : 'bg-red-500/10 text-red-400/60 line-through'}`}>Trend + RS {funnelPhases.passedTrend ? '✓' : '✗'}</span>
   823→            <ArrowDown className="w-3 h-3 text-muted-foreground/30" />
   824→            <span className={`px-2 py-1 rounded-md ${funnelPhases.passedSetup ? 'bg-violet-500/20 text-violet-400' : 'bg-red-500/10 text-red-400/60 line-through'}`}>Setup {funnelPhases.passedSetup ? '✓' : '✗'}</span>
   825→            <ArrowDown className="w-3 h-3 text-muted-foreground/30" />
