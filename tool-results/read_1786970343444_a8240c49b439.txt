    32→// Safe parse — returns 0 instead of NaN
    33→function safeNum(v: unknown): number {
    34→  const n = typeof v === 'number' ? v : parseFloat(String(v));
    35→  return isNaN(n) ? 0 : n;
    36→}

    38→// Map Yahoo recommendation key to our rating
    39→function mapRecommendation(key: string): 'STRONG_BUY' | 'BUY' | 'HOLD' | 'SELL' {
    40→  const map: Record<string, 'STRONG_BUY' | 'BUY' | 'HOLD' | 'SELL'> = {
    41→    'strong_buy': 'STRONG_BUY',
    42→    'buy': 'BUY',
    43→    'hold': 'HOLD',
    44→    'sell': 'SELL',
    45→    'strong_sell': 'SELL',
    46→  };
    47→  return map[key] || 'HOLD';
    48→}

    50→// Growth scoring factors (each contributes 0-100 scaled)
    51→// Uses REAL fundamental data when available, falls back to profile
    52→function calculateGrowthScore(s: StockProfile, priceChange: number, fund: YahooFundamentals | null): number {
    53→  let score = 0;
    54→
    55→  // Signal & Trend (from profile — these are set manually based on technical analysis)
    56→  if (s.signal === 'BULLISH') score += 20;
    57→  else if (s.signal === 'NEUTRAL') score += 8;
    58→
    59→  if (s.trend === 'uptrend') score += 15;
    60→  else if (s.trend === 'sideways') score += 5;
    61→
    62→  // Rating — prefer real analyst recommendation
    63→  const rating = fund && fund.recommendationKey ? mapRecommendation(fund.recommendationKey) : s.rating;
    64→  if (rating === 'STRONG_BUY') score += 15;
    65→  else if (rating === 'BUY') score += 12;
    66→  else if (rating === 'HOLD') score += 4;
    67→  else if (rating === 'SELL') score -= 5;
    68→
    69→  // Revenue Growth — prefer real data
    70→  const revNum = fund && fund.revenueGrowth ? fund.revenueGrowth * 100 : safeNum(s.revGrowth);
    71→  if (revNum >= 30) score += 12;
    72→  else if (revNum >= 15) score += 10;
    73→  else if (revNum >= 8) score += 7;
    74→  else if (revNum >= 3) score += 4;
    75→  else if (revNum > 0) score += 2;
    76→  else score -= 3;
    77→
    78→  // EPS Growth — prefer real data
    79→  const epsNum = fund && fund.earningsGrowth ? fund.earningsGrowth * 100 : safeNum(s.epsGrowth);
    80→  if (epsNum >= 40) score += 12;
    81→  else if (epsNum >= 20) score += 10;
    82→  else if (epsNum >= 10) score += 7;
    83→  else if (epsNum >= 5) score += 4;
    84→  else if (epsNum > 0) score += 2;
    85→  else score -= 3;
    86→
    87→  // PEG Ratio — prefer real data
    88→  const peg = fund && fund.pegRatio > 0 ? fund.pegRatio : s.peg;
    89→  if (peg > 0 && peg <= 1.0) score += 8;
    90→  else if (peg > 0 && peg <= 1.5) score += 6;
    91→  else if (peg > 0 && peg <= 2.0) score += 4;
    92→  else if (peg > 0 && peg <= 3.0) score += 2;
    93→  else score -= 2;
    94→
    95→  // Moat (from profile — structural advantage)
    96→  if (s.moat === 'WIDE') score += 8;
    97→  else if (s.moat === 'NARROW') score += 3;
    98→
    99→  // Price momentum
   100→  if (priceChange >= 2) score += 10;
   101→  else if (priceChange >= 1) score += 7;
   102→  else if (priceChange >= 0.3) score += 4;
   103→  else if (priceChange >= 0) score += 2;
   104→  else if (priceChange >= -1) score += 0;
   105→  else score -= 3;
   106→
   107→  // Operating margin — prefer real data
   108→  const opMarginNum = fund && fund.operatingMargins > 0 ? fund.operatingMargins * 100 : safeNum(s.opMargin);
   109→  if (opMarginNum >= 40) score += 5;
   110→  else if (opMarginNum >= 25) score += 4;
   111→  else if (opMarginNum >= 15) score += 2;
   112→
   113→  // Upside from analyst target — prefer real data
   114→  if (fund && fund.currentPrice > 0 && fund.targetMeanPrice > 0) {
   115→    const upside = ((fund.targetMeanPrice - fund.currentPrice) / fund.currentPrice) * 100;
   116→    if (upside >= 30) score += 5;
   117→    else if (upside >= 15) score += 3;
   118→    else if (upside >= 5) score += 1;
   119→  }
   120→
   121→  return Math.max(0, Math.min(100, score));
   122→}

   124→// Risk scoring factors (0-100, higher = more risk of decline)
   125→function calculateRiskScore(s: StockProfile, priceChange: number, fund: YahooFundamentals | null): number {
   126→  let score = 0;
   127→
   128→  if (s.signal === 'BEARISH') score += 20;
   129→  else if (s.signal === 'NEUTRAL') score += 8;
   130→
   131→  if (s.trend === 'downtrend') score += 15;
   132→  else if (s.trend === 'sideways') score += 5;
   133→
   134→  const rating = fund && fund.recommendationKey ? mapRecommendation(fund.recommendationKey) : s.rating;
   135→  if (rating === 'SELL') score += 12;
   136→  else if (rating === 'HOLD') score += 6;
   137→
   138→  // Revenue decline
   139→  const revNum = fund && fund.revenueGrowth ? fund.revenueGrowth * 100 : safeNum(s.revGrowth);
   140→  if (revNum < -5) score += 12;
   141→  else if (revNum < 0) score += 8;
   142→  else if (revNum < 3) score += 3;
   143→
   144→  // EPS decline
   145→  const epsNum = fund && fund.earningsGrowth ? fund.earningsGrowth * 100 : safeNum(s.epsGrowth);
   146→  if (epsNum < -10) score += 10;
   147→  else if (epsNum < 0) score += 6;
   148→  else if (epsNum < 5) score += 2;
   149→
   150→  // Valuation risk — prefer real PE
   151→  const pe = fund && fund.trailingPE > 0 ? fund.trailingPE : s.pe;
   152→  if (pe > 80) score += 8;
   153→  else if (pe > 50) score += 5;
   154→  else if (pe > 35) score += 2;
   155→
   156→  if (s.moat === 'NONE') score += 8;
   157→  else if (s.moat === 'NARROW') score += 4;
   158→
   159→  if (priceChange <= -2) score += 10;
   160→  else if (priceChange <= -1) score += 7;
   161→  else if (priceChange <= -0.3) score += 4;
   162→  else if (priceChange < 0) score += 2;
   163→
   164→  // Debt — prefer real data
   165→  const debtEq = fund && fund.debtToEquity > 0 ? fund.debtToEquity : s.debtEq;
   166→  if (debtEq > 3) score += 8;
   167→  else if (debtEq > 2) score += 5;
   168→  else if (debtEq > 1) score += 2;
   169→
   170→  // Operating margin weakness — prefer real data
   171→  const opMarginNum = fund && fund.operatingMargins > 0 ? fund.operatingMargins * 100 : safeNum(s.opMargin);
   172→  if (opMarginNum < 5) score += 7;
   173→  else if (opMarginNum < 10) score += 4;
   174→  else if (opMarginNum < 15) score += 2;
   175→
   176→  // Negative upside (analysts see decline)
   177→  if (fund && fund.currentPrice > 0 && fund.targetMeanPrice > 0) {
   178→    const upside = ((fund.targetMeanPrice - fund.currentPrice) / fund.currentPrice) * 100;
   179→    if (upside < -15) score += 8;
   180→    else if (upside < -5) score += 4;
   181→  }
   182→
   183→  return Math.max(0, Math.min(100, score));
   184→}

   220→
   221→  return reasons.slice(0, 5);

   227→  if (s.signal === 'BEARISH') reasons.push(`Sinjali teknik: BEARISH me trend ${s.trend}`);

   231→

   235→  if (epsNum < 0) reasons.push(`EPS ne renie: ${epsNum.toFixed(1)}%`);
   236→
   237→  const pe = fund && fund.trailingPE > 0 ? fund.trailingPE : s.pe;
   238→  if (pe > 50) reasons.push(`P/E ${pe.toFixed(1)}x — vleresim shume i larte`);
   239→
   240→  if (s.moat === 'NONE') reasons.push(`Asnje avantazhi konkurrues (Moat: NONE)`);
   241→  if (priceChange < 0) reasons.push(`Momentum negativ: ${priceChange.toFixed(2)}%`);
   242→
   243→  const debtEq = fund && fund.debtToEquity > 0 ? fund.debtToEquity : s.debtEq;
   244→  if (debtEq > 2) reasons.push(`Detyrimet e larta: Debt/Equity ${debtEq.toFixed(1)}x`);
   245→
   246→  if (s.weaknesses.length > 0) reasons.push(`Rreziku: ${s.weaknesses[0]}`);
   247→
   248→  // Negative upside
   249→  if (fund && fund.currentPrice > 0 && fund.targetMeanPrice > 0) {
   250→    const upside = ((fund.targetMeanPrice - fund.currentPrice) / fund.currentPrice) * 100;
   251→    if (upside < 0) reasons.push(`Upside negativ: ${upside.toFixed(1)}% (target $${fund.targetMeanPrice.toFixed(2)} < cmimi $${fund.currentPrice.toFixed(2)})`);
   252→  }
   253→
   254→  if (s.trend === 'downtrend') reasons.push(`Trendi afatgjatshem: DOWNTREND`);
   255→
   256→  return reasons.slice(0, 5);

   263→/**
   264→ * Fetch real prices in small batches to avoid rate limits.
   265→ */
   266→async function fetchPricesInBatches(tickers: string[]): Promise<Record<string, { price: number; change: number }>> {
   267→  const results: Record<string, { price: number; change: number }> = {};
   268→  
   269→  // First try bulk fetch (fastest)
   270→  try {
   271→    const bulk = await getRealPrices(tickers);
   272→    Object.assign(results, bulk);
   273→    console.log(`[TOP-MOVERS] Bulk fetch got ${Object.keys(bulk).length} prices`);
   274→  } catch {
   275→    console.log('[TOP-MOVERS] Bulk fetch failed, trying individual...');
   276→  }
   277→
   278→  // For any missing tickers, try individual fetch
   279→  const missing = tickers.filter(t => !results[t]);
   280→  if (missing.length > 0) {
   281→    const priorityTickers = ['NVDA', 'AAPL', 'MSFT', 'GOOGL', 'AMZN', 'META', 'TSLA', 'AVGO', 'LLY', 'JPM', 'V', 'CRM', 'AMD', 'NFLX', 'PLTR', 'UNH', 'MRVL', 'COIN', 'SMCI', 'SNOW'];
   282→    const toFetch = priorityTickers.filter(t => missing.includes(t)).slice(0, 15);
   283→    
   284→    const BATCH_SIZE = 3;
   285→    for (let i = 0; i < toFetch.length; i += BATCH_SIZE) {
   286→      const batch = toFetch.slice(i, i + BATCH_SIZE);
   287→      await Promise.allSettled(
   288→        batch.map(async (ticker) => {
   289→          try {
   290→            const price = await getRealPrice(ticker);
   291→            if (price && price.price > 0) {
   292→              results[ticker] = { price: price.price, change: price.change };
   293→            }
   294→          } catch {
   295→            // Skip
   296→          }
   297→        })
   298→      );
   299→      if (i + BATCH_SIZE < toFetch.length) {
   300→        await new Promise(r => setTimeout(r, 200));
   301→      }
   302→    }
   303→  }
   304→
   305→  return results;
   306→}

   308→export async function GET() {
   309→  try {
   310→    // Return cached if fresh
   311→    if (cachedResult && Date.now() - cachedResult.fetchedAt < CACHE_TTL) {
   312→      return NextResponse.json({
   313→        ...cachedResult.data,
   314→        cached: true,
   315→      });
   316→    }
   317→
   318→    console.log('[TOP-MOVERS] Fetching real prices AND fundamentals...');
   319→
   320→    const allStocks = getAllStocks();
   321→    const allTickers = Object.keys(allStocks);
   322→    
   323→    // ═══ STEP 1: Fetch real prices ═══
   324→    const realPrices = await fetchPricesInBatches(allTickers);
   325→    const liveCount = Object.keys(realPrices).length;
   326→    console.log(`[TOP-MOVERS] Got ${liveCount}/${allTickers.length} real prices`);
   327→
   328→    // ═══ STEP 2: Fetch real fundamental data — FAST BATCH FIRST ═══
   329→    // Use v7/finance/quote batch endpoint (1-2 requests for ALL stocks)
   330→    let realFundamentals = await getBatchQuotesFast(allTickers);
   331→    let fundCount = Object.keys(realFundamentals).length;
   332→    console.log(`[TOP-MOVERS] Fast batch got ${fundCount}/${allTickers.length} fundamentals`);
   333→
   334→    // Fallback: use individual quoteSummary for any missing stocks (limited to 15)
   335→    if (fundCount < allTickers.length) {
   336→      const missing = allTickers.filter(t => !realFundamentals[t]).slice(0, 15);
   337→      if (missing.length > 0) {
   338→        try {
   339→          const extras = await getRealFundamentalsBatch(missing);
   340→          Object.assign(realFundamentals, extras);
   341→          fundCount = Object.keys(realFundamentals).length;
   342→          console.log(`[TOP-MOVERS] After fallback: ${fundCount}/${allTickers.length} fundamentals`);
   343→        } catch (e) {
   344→          console.log(`[TOP-MOVERS] Fallback fundamentals failed:`, e);
   345→        }
   346→      }
   347→    }
   348→
   349→    // ═══ STEP 3: Score all stocks using real data ═══
   350→    const scored: ScoredStock[] = allTickers.map(ticker => {
   351→      const profile = allStocks[ticker];
   352→      const livePrice = realPrices[ticker];
   353→      const fund = realFundamentals[ticker] || null;
   354→      const isLive = !!livePrice && livePrice.price > 0;
   355→      const hasFundamentals = !!fund && fund.currentPrice > 0;
   356→
   357→      // Use live price if available, otherwise use fund price, otherwise profile price
   358→      let currentPrice = profile.price;
   359→      let priceChange = profile.change;
   360→
   361→      if (isLive) {
   362→        currentPrice = livePrice.price;
   363→        priceChange = livePrice.change;
   364→      } else if (hasFundamentals && fund!.currentPrice > 0) {
   365→        currentPrice = fund!.currentPrice;
   366→        const prev = fund!.previousClose;
   367→        if (prev > 0) {
   368→          priceChange = ((fund!.currentPrice - prev) / prev) * 100;
   369→        }
   370→      }
   371→
   372→      return {
   373→        ticker,
   374→        company: profile.company,
   375→        sector: profile.sector,
   376→        industry: profile.industry,
   377→        currentPrice,
   378→        priceChange,
   379→        isLive,
   380→        hasFundamentals,
   381→        growthScore: calculateGrowthScore(profile, priceChange, fund),
   382→        riskScore: calculateRiskScore(profile, priceChange, fund),
   383→        profile,
   384→        fund,
   385→        growthReasons: generateGrowthReasons(profile, priceChange, fund),
   386→        riskReasons: generateRiskReasons(profile, priceChange, fund),
   387→      };
   388→    });
   389→
   390→    // ═══ STEP 3.5: Swing analysis for top candidates ═══
   391→    // Get top ~15 by growth and ~15 by risk for swing analysis
   392→    const growthCandidates = [...scored]
   393→      .filter(s => s.currentPrice > 0)
   394→      .sort((a, b) => b.growthScore - a.growthScore)
   395→      .slice(0, 20)
   396→      .map(s => s.ticker);
   397→    const riskCandidates = [...scored]
   398→      .filter(s => s.currentPrice > 0)
   399→      .sort((a, b) => b.riskScore - a.riskScore)
   400→      .slice(0, 20)
   401→      .map(s => s.ticker);
   402→
   403→    // Combine unique tickers for swing batch
   404→    const swingTickers = [...new Set([...growthCandidates, ...riskCandidates])];
   405→    console.log(`[TOP-MOVERS] Running swing analysis for ${swingTickers.length} candidates...`);
   406→    const swingResults = await analyzeSwingBatch(swingTickers, realPrices, 5);
   407→    console.log(`[TOP-MOVERS] Swing analysis complete: ${Object.keys(swingResults).length} results`);
   408→
   409→    // Merge swing data into scored stocks
   410→    for (const s of scored) {
   411→      const swing = swingResults[s.ticker];
   412→      if (swing) {
   413→        s.swingData = swing;
   414→        s.swingScore = swing.swingScore;
   415→      } else {
   416→        s.swingData = null;
   417→        s.swingScore = 50; // neutral
   418→      }
   419→    }
   420→
   421→    // ═══ STEP 4: Re-rank with SWING-WEIGHTED scoring ═══
   422→    // For growth: blend 40% fundamental growth + 60% swing technical score
   423→    // This makes the picks much more accurate for actual swing trading
   424→    const growthCandidates2 = [...scored]
   425→      .map(s => ({
   426→        ...s,
   427→        blendedScore: s.growthScore * 0.4 + s.swingScore * 0.6,
   428→      }))
   429→      .sort((a, b) => b.blendedScore - a.blendedScore)
   430→      .slice(0, 5);
   431→
   432→    // For risk: stocks with HIGHEST risk score AND LOWEST swing score (double bearish)
   433→    const riskCandidates2 = [...scored]
   434→      .map(s => ({
   435→        ...s,
   436→        blendedRisk: s.riskScore * 0.5 + (100 - s.swingScore) * 0.5,
   437→      }))
   438→      .sort((a, b) => b.blendedRisk - a.blendedRisk)
   439→      .slice(0, 5);
   440→
   441→    const topGrowth = growthCandidates2;
   442→    const topRisk = riskCandidates2;
   443→
   444→    // Enrich with REAL data for output
   445→    const enriched = (list: ScoredStock[], type: 'growth' | 'risk') =>
   446→      list.map(stock => {
   447→        const fund = stock.fund;
   448→
   449→        // Use real target prices if available
   450→        let targetPrice = stock.profile.targetPrice;
   451→        let highTarget = stock.profile.highTarget;
   452→        let lowTarget = stock.profile.lowTarget;
   453→        let upside = 0;
   454→        let rating: 'STRONG_BUY' | 'BUY' | 'HOLD' | 'SELL' = stock.profile.rating;
   455→        let pe = stock.profile.pe;
   456→        let fwdPE = stock.profile.fwdPE;
   457→        let peg = stock.profile.peg;
   458→        let revGrowth = stock.profile.revGrowth;
   459→        let epsGrowth = stock.profile.epsGrowth;
   460→        let opMargin = stock.profile.opMargin;
   461→        let debtEq = stock.profile.debtEq;
   462→        let buyCount = stock.profile.buyCount;
   463→        let sellCount = stock.profile.sellCount;
   464→
   465→        if (fund) {
   466→          // Override with real data
   467→          pe = fund.trailingPE || stock.profile.pe;
   468→          fwdPE = fund.forwardPE || stock.profile.fwdPE;
   469→          peg = fund.pegRatio || stock.profile.peg;
   470→          debtEq = fund.debtToEquity || stock.profile.debtEq;
   471→
   472→          // Revenue growth: convert decimal to percentage string
   473→          if (fund.revenueGrowth) {
   474→            revGrowth = (fund.revenueGrowth * 100).toFixed(1) + '%';
   475→          }
   476→          // EPS growth: convert decimal to percentage string
   477→          if (fund.earningsGrowth) {
   478→            epsGrowth = (fund.earningsGrowth * 100).toFixed(1) + '%';
   479→          }
   480→          // Operating margin: convert decimal to percentage string
   481→          if (fund.operatingMargins > 0) {
   482→            opMargin = (fund.operatingMargins * 100).toFixed(1) + '%';
   483→          }
   484→
   485→          // Real analyst targets — with PRE-SPLIT validation
   486→          if (fund.targetMeanPrice > 0) {
   487→            const targetRatio = fund.targetMeanPrice / stock.currentPrice;
   488→
   489→            // PRE-SPLIT DETECTION: If target > 3x price or < 0.25x price,
   490→            // it's likely a stale pre-split target. Discard it.
   491→            if (targetRatio > 3.0 || targetRatio < 0.25) {
   492→              console.warn(`[TOP-MOVERS] ${stock.ticker}: SUSPICIOUS target=$${fund.targetMeanPrice.toFixed(2)} vs price=$${stock.currentPrice.toFixed(2)} (ratio=${targetRatio.toFixed(2)}). Likely pre-split data — DISCARDING target.`);
   493→              // Fall through to profile target below
   494→            } else {
   495→              targetPrice = '$' + fund.targetMeanPrice.toFixed(2);
   496→              highTarget = '$' + fund.targetHighPrice.toFixed(2);
   497→              lowTarget = '$' + fund.targetLowPrice.toFixed(2);
   498→              upside = ((fund.targetMeanPrice - stock.currentPrice) / stock.currentPrice) * 100;
   499→            }
   500→          }
   501→
   502→          // If no valid live target, use profile target
   503→          if (upside === 0) {
   504→            const targetNum = safeNum(String(stock.profile.targetPrice).replace(/[^0-9.]/g, ''));
   505→            if (targetNum > 0) {
   506→              const profileRatio = targetNum / stock.currentPrice;
   507→              // Also validate profile targets against current live price
   508→              if (profileRatio <= 3.0 && profileRatio >= 0.25) {
   509→                upside = ((targetNum - stock.currentPrice) / stock.currentPrice * 100);
   510→              } else {
   511→                console.warn(`[TOP-MOVERS] ${stock.ticker}: Profile target also suspicious (ratio=${profileRatio.toFixed(2)}). No valid target.`);
   512→                upside = 0;
   513→              }
   514→            }
   515→          }
   516→
   517→          // SANITY CHECK: Hard cap at 100% — no legitimate upside exceeds this for large caps
   518→          if (Math.abs(upside) > 100) {
   519→            console.warn(`[TOP-MOVERS] ${stock.ticker}: Upside ${upside.toFixed(1)}% exceeds 100% cap — setting to ±100% (price=$${stock.currentPrice.toFixed(2)} target=${targetPrice})`);
   520→            upside = Math.sign(upside) * 100;
   521→          }
   522→
   523→          // Real recommendation
   524→          if (fund.recommendationKey) {
   525→            rating = mapRecommendation(fund.recommendationKey);
   526→          }
   527→
   528→          // Analyst counts
   529→          if (fund.numberOfAnalystOpinions > 0) {
   530→            const total = Math.round(fund.numberOfAnalystOpinions);
   531→            // Estimate buy/sell split from recommendation
   532→            if (rating === 'STRONG_BUY') { buyCount = Math.round(total * 0.75); sellCount = Math.round(total * 0.05); }
   533→            else if (rating === 'BUY') { buyCount = Math.round(total * 0.55); sellCount = Math.round(total * 0.15); }
   534→            else if (rating === 'HOLD') { buyCount = Math.round(total * 0.25); sellCount = Math.round(total * 0.25); }
   535→            else if (rating === 'SELL') { buyCount = Math.round(total * 0.1); sellCount = Math.round(total * 0.55); }
   536→          }
   537→        } else {
   538→          // No real data — calculate upside from profile target
   539→          const targetNum = safeNum(String(stock.profile.targetPrice).replace(/[^0-9.]/g, ''));
   540→          if (targetNum > 0) {
   541→            const profileRatio = targetNum / stock.currentPrice;
   542→            if (profileRatio <= 3.0 && profileRatio >= 0.25) {
   543→              upside = ((targetNum - stock.currentPrice) / stock.currentPrice * 100);
   544→            } else {
   545→              console.warn(`[TOP-MOVERS] ${stock.ticker}: No live data + suspicious profile target ratio=${profileRatio.toFixed(2)}. Setting upside to 0.`);
   546→              upside = 0;
   547→            }
   548→          }
   549→          // SANITY CHECK: Hard cap at 100%
   550→          if (Math.abs(upside) > 100) {
   551→            upside = Math.sign(upside) * 100;
   552→          }
   553→        }
   554→
   555→        // Build swing info
   556→      const swing = stock.swingData;
   557→      const swingInfo = swing ? {
   558→        swingScore: swing.swingScore,
   559→        rsi: swing.rsi,
   560→        macdSignal: swing.macdSignal,
   561→        bollingerPosition: swing.bollingerPosition,
   562→        bollingerSqueeze: swing.bollingerSqueeze,
   563→        sma20vs50: swing.sma20vs50,
   564→        ema9vs21: swing.ema9vs21,
   565→        atrPercent: swing.atrPercent,
   566→        volumeTrend: swing.volumeTrend,
   567→        supportLevel: swing.supportLevel,
   568→        resistanceLevel: swing.resistanceLevel,
   569→        entryPrice: swing.entryPrice,
   570→        stopLoss: swing.stopLoss,
   571→        swingTarget: swing.targetPrice,
   572→        riskRewardRatio: swing.riskRewardRatio,
   573→        swingReasons: swing.swingReasons,
   574→        warningFlags: swing.warningFlags,
   575→      } : null;
   576→
   577→      // For growth: blend reasons (original + swing)
   578→      // For risk: add swing warnings
   579→      const combinedReasons = type === 'growth' && swing
   580→        ? [...swing.swingReasons, ...stock.growthReasons].slice(0, 6)
   581→        : type === 'risk' && swing
   582→          ? [...stock.riskReasons, ...swing.warningFlags].slice(0, 6)
   583→          : (type === 'growth' ? stock.growthReasons : stock.riskReasons);
   584→
   585→      return {
   586→          ticker: stock.ticker,
   587→          company: stock.company,
   588→          sector: stock.sector,
   589→          industry: stock.industry,
   590→          currentPrice: stock.currentPrice,
   591→          priceChange: stock.priceChange,
   592→          isLive: stock.isLive,
   593→          hasFundamentals: stock.hasFundamentals,
   594→          score: type === 'growth' ? (stock as any).blendedScore || stock.growthScore : (stock as any).blendedRisk || stock.riskScore,
   595→          targetPrice,
   596→          highTarget,
   597→          lowTarget,
   598→          upside: parseFloat(upside.toFixed(1)),
   599→          rating,
   600→          signal: stock.profile.signal,
   601→          trend: stock.profile.trend,
   602→          pe,
   603→          fwdPE,
   604→          peg,
   605→          revGrowth,
   606→          epsGrowth,
   607→          opMargin,
   608→          debtEq,
   609→          moat: stock.profile.moat,
   610→          marketCap: stock.profile.marketCap,
   611→          reasons: combinedReasons,
   612→          strengths: stock.profile.strengths,
   613→          weaknesses: stock.profile.weaknesses,
   614→          buyCount,
   615→          sellCount,
   616→          swing: swingInfo,
   617→        };
   618→      });
   619→
   620→    const result = {
   621→      topGrowth: enriched(topGrowth, 'growth'),
   622→      topRisk: enriched(topRisk, 'risk'),
   623→      totalAnalyzed: allTickers.length,
   624→      liveCount,
   625→      totalFetched: allTickers.length,
   626→      fundCount,
   627→      swingAnalyzed: Object.keys(swingResults).length,
   628→      timestamp: new Date().toISOString(),
   629→    };
   630→
   631→    cachedResult = { data: result as unknown as NonNullable<typeof cachedResult>['data'], fetchedAt: Date.now() };
   632→    console.log(`[TOP-MOVERS] Growth: ${topGrowth.map(s => s.ticker).join(',')} | Risk: ${topRisk.map(s => s.ticker).join(',')} | Live: ${liveCount} | Fund: ${fundCount}`);
   633→
   634→    return NextResponse.json({ ...result, cached: false });
   635→  } catch (error) {
   636→    console.error('[TOP-MOVERS] Error:', error);
   637→
   638→    if (cachedResult) {
   639→      return NextResponse.json({ ...cachedResult.data, cached: true, stale: true });
   640→    }
   641→
   642→    return NextResponse.json({ error: 'Gabim ne analizen e tregjit' }, { status: 502 });
   643→  }
