    89→// ═══ Technical indicator computations ═══
    90→function computeSMA(closes: number[], period: number): (number | null)[] {
    91→  const result: (number | null)[] = [];
    92→  for (let i = 0; i < closes.length; i++) {
    93→    if (i < period - 1) { result.push(null); continue; }
    94→    let sum = 0;
    95→    for (let j = i - period + 1; j <= i; j++) sum += closes[j];
    96→    result.push(sum / period);
    97→  }
    98→  return result;
    99→}
   100→
   101→function computeEMA(closes: number[], period: number): (number | null)[] {
   102→  const k = 2 / (period + 1);
   103→  const result: (number | null)[] = [];
   104→  let ema: number | null = null;
   105→  for (let i = 0; i < closes.length; i++) {
   106→    if (i < period - 1) { result.push(null); continue; }
   107→    if (ema === null) {
   108→      let sum = 0; for (let j = i - period + 1; j <= i; j++) sum += closes[j];
   109→      ema = sum / period;
   110→    } else {
   111→      ema = closes[i] * k + ema * (1 - k);
   112→    }
   113→    result.push(ema);
   114→  }
   115→  return result;
   116→}

   118→function computeRSI(closes: number[], period: number = 14): (number | null)[] {
   119→  const result: (number | null)[] = [];
   120→  const gains: number[] = [];
   121→  const losses: number[] = [];
   122→  for (let i = 1; i < closes.length; i++) {
   123→    const diff = closes[i] - closes[i - 1];
   124→    gains.push(diff > 0 ? diff : 0);
   125→    losses.push(diff < 0 ? -diff : 0);
   126→  }
   127→  let avgGain = 0, avgLoss = 0;
   128→  for (let i = 0; i < Math.min(period, gains.length); i++) { avgGain += gains[i]; avgLoss += losses[i]; }
   129→  avgGain /= period; avgLoss /= period;
   130→  for (let i = 0; i < period - 1; i++) result.push(null);
   131→  const rs = avgLoss === 0 ? 100 : avgGain / avgLoss;
   132→  result.push(100 - 100 / (1 + rs));
   133→  for (let i = period; i < gains.length; i++) {
   134→    avgGain = (avgGain * (period - 1) + gains[i]) / period;
   135→    avgLoss = (avgLoss * (period - 1) + losses[i]) / period;
   136→    const rs2 = avgLoss === 0 ? 100 : avgGain / avgLoss;
   137→    result.push(100 - 100 / (1 + rs2));
   138→  }
   139→  return result;
   140→}

   142→function computeMACD(closes: number[]): { macd: (number | null)[]; signal: (number | null)[]; histogram: (number | null)[] } {
   143→  const ema12 = computeEMA(closes, 12);
   144→  const ema26 = computeEMA(closes, 26);
   145→  const macdLine: (number | null)[] = [];
   146→  for (let i = 0; i < closes.length; i++) {
   147→    if (ema12[i] === null || ema26[i] === null) { macdLine.push(null); continue; }
   148→    macdLine.push(ema12[i]! - ema26[i]!);
   149→  }
   150→  // Signal = 9-period EMA of MACD line
   151→  const validMACD = macdLine.filter((v): v is number => v !== null);
   152→  const signalEma = computeEMA(validMACD, 9);
   153→  const signal: (number | null)[] = [];
   154→  let vi = 0;
   155→  for (let i = 0; i < closes.length; i++) {
   156→    if (macdLine[i] === null) { signal.push(null); continue; }
   157→    signal.push(signalEma[vi] ?? null);
   158→    vi++;
   159→  }
   160→  const histogram: (number | null)[] = [];
   161→  for (let i = 0; i < closes.length; i++) {
   162→    if (macdLine[i] === null || signal[i] === null) { histogram.push(null); continue; }
   163→    histogram.push(macdLine[i]! - signal[i]!);
   164→  }
   165→  return { macd: macdLine, signal, histogram };
   166→}

   168→function computeBollingerBands(closes: number[], period: number = 20, mult: number = 2): { upper: (number | null)[]; middle: (number | null)[]; lower: (number | null)[] } {
   169→  const middle = computeSMA(closes, period);
   170→  const upper: (number | null)[] = [];
   171→  const lower: (number | null)[] = [];
   172→  for (let i = 0; i < closes.length; i++) {
   173→    if (middle[i] === null) { upper.push(null); lower.push(null); continue; }
   174→    let sumSq = 0;
   175→    for (let j = i - period + 1; j <= i; j++) sumSq += (closes[j] - middle[i]!) ** 2;
   176→    const std = Math.sqrt(sumSq / period);
   177→    upper.push(middle[i]! + mult * std);
   178→    lower.push(middle[i]! - mult * std);
   179→  }
   180→  return { upper, middle, lower };
   181→}

   183→// ═══ TradingView-Style Technical Chart ═══
   184→function CandlestickChart({ data }: { data: CandleData[] }) {
   185→  const chart = useMemo(() => {
   186→    if (!data || data.length < 3) return null;
   187→
   188→    // TradingView dark palette
   189→    const BG = 'transparent';
   190→    const GRID = '#2a2e39';
   191→    const GRID_OP = 0.4;
   192→    const TXT = '#787b86';
   193→    const BULL = '#26a69a';
   194→    const BEAR = '#ef5350';
   195→    const SMA20_CLR = '#f0b90b';
   196→    const SMA50_CLR = '#2196f3';
   197→    const EMA12_CLR = '#e040fb';
   198→    const BB_CLR = '#7c4dff';
   199→    const RSI_CLR = '#e040fb';
   200→    const MACD_CLR = '#2196f3';
   201→    const SIG_CLR = '#ff6d00';
   202→
   203→    const W = 880;
   204→    const H = 540;
   205→    const R = 60; // right axis width
   206→    const L = 2;  // left margin
   207→    const chartW = W - L - R;
   208→    const n = data.length;
   209→    const barW = chartW / n;
   210→    const candleW = Math.max(barW * 0.6, 2);
   211→
   212→    // Layout: price (with volume overlay), separator, RSI, separator, MACD
   213→    const priceH = 280;
   214→    const volRatio = 0.18; // volume takes bottom 18% of price panel
   215→    const volH = priceH * volRatio;
   216→    const candleH = priceH - volH; // candles use top 82%
   217→    const sep = 1;
   218→    const rsiH = 80;
   219→    const macdH = 85;
   220→    const bottomLabel = 18;
   221→
   222→    const priceTop = 0;
   223→    const volTop = priceTop + candleH;
   224→    const rsiTop = priceTop + priceH + sep;
   225→    const macdTop = rsiTop + rsiH + sep;
   226→    const totalH = macdTop + macdH + bottomLabel;
   227→
   228→    // ═══ Compute indicators ═══
   229→    const closes = data.map(d => d.close);
   230→    const sma20 = computeSMA(closes, Math.min(20, n));
   231→    const sma50 = n >= 50 ? computeSMA(closes, 50) : computeSMA(closes, n);
   232→    const ema12 = computeEMA(closes, Math.min(12, n));
   233→    const bb = computeBollingerBands(closes, Math.min(20, n));
   234→    const rsi = computeRSI(closes, 14);
   235→    const macdData = computeMACD(closes);
   236→
   237→    // ═══ Price Y scale (candles area only, not volume) ═══
   238→    const hiVals = [ ...data.map(d => d.high), ...bb.upper.filter((v): v is number => v !== null), ...sma20.filter((v): v is number => v !== null) ];
   239→    const loVals = [ ...data.map(d => d.low), ...bb.lower.filter((v): v is number => v !== null), ...sma20.filter((v): v is number => v !== null) ];
   240→    const allH = Math.max(...hiVals);
   241→    const allL = Math.min(...loVals);
   242→    const pad = (allH - allL) * 0.06 || 1;
   243→    const pMin = allL - pad;
   244→    const pMax = allH + pad;
   245→    const yP = (v: number) => priceTop + (1 - (v - pMin) / (pMax - pMin)) * candleH;
   246→
   247→    // ═══ Volume Y scale (bottom of price panel) ═══
   248→    const maxVol = Math.max(...data.map(d => d.volume || 0)) || 1;
   249→    const yV = (v: number) => volTop + volH - (v / maxVol) * volH;
   250→
   251→    // ═══ RSI scale 0-100 ═══
   252→    const yR = (v: number) => rsiTop + (1 - v / 100) * rsiH;
   253→
   254→    // ═══ MACD scale ═══
   255→    const macdVals = macdData.macd.filter((v): v is number => v !== null);
   256→    const histVals = macdData.histogram.filter((v): v is number => v !== null);
   257→    const mMax = Math.max(Math.abs(Math.min(...macdVals, ...histVals)), Math.abs(Math.max(...macdVals, ...histVals)), 0.01);
   258→    const yM = (v: number) => macdTop + macdH / 2 - (v / mMax) * (macdH / 2);
   259→
   260→    // ═══ Helpers ═══
   261→    const xOf = (i: number) => L + barW * i + barW / 2;
   262→    const labelN = n <= 10 ? 1 : n <= 20 ? 2 : n <= 30 ? 4 : 5;
   263→
   264→    const line = (vals: (number | null)[], color: string, sw = 1, id?: string) => {
   265→      const pts: string[] = [];
   266→      for (let i = 0; i < vals.length; i++) {
   267→        if (vals[i] !== null) pts.push(`${xOf(i)},${vals[i]}`);
   268→      }
   269→      return pts.length > 1 ? <polyline key={id ?? color} points={pts.join(' ')} fill="none" stroke={color} strokeWidth={sw} strokeLinejoin="round" strokeLinecap="round" /> : null;
   270→    };
   271→
   272→    // Price grid
   273→    const pGrid: { y: number; lbl: string }[] = [];
   274→    for (let i = 0; i <= 5; i++) {
   275→      const v = pMin + (pMax - pMin) * (i / 5);
   276→      pGrid.push({ y: yP(v), lbl: '$' + v.toFixed(1) });
   277→    }
   278→
   279→    // BB fill
   280→    const bbPts: string[] = [];
   281→    for (let i = 0; i < n; i++) if (bb.upper[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.upper[i]!)}`);
   282→    for (let i = n - 1; i >= 0; i--) if (bb.lower[i] !== null) bbPts.push(`${xOf(i)},${yP(bb.lower[i]!)}`);
   283→
   284→    // Last values for legend
   285→    const lastSma20 = sma20[n - 1];
   286→    const lastSma50 = sma50[n - 1];
   287→    const lastEma12 = ema12[n - 1];
   288→    const lastRsi = rsi[n - 1];
   289→    const lastMacd = macdData.macd[n - 1];
   290→    const lastSig = macdData.signal[n - 1];
   291→    const lastClose = data[n - 1].close;
   292→    const lastY = yP(lastClose);
   293→
   294→    return (
   295→      <svg viewBox={`0 0 ${W} ${totalH}`} className="w-full h-full" style={{ background: BG }}>
   296→        <defs>
   297→          <clipPath id="priceClip"><rect x={L} y={priceTop} width={chartW} height={priceH} /></clipPath>
   298→          <clipPath id="rsiClip"><rect x={L} y={rsiTop} width={chartW} height={rsiH} /></clipPath>
   299→          <clipPath id="macdClip"><rect x={L} y={macdTop} width={chartW} height={macdH} /></clipPath>
   300→        </defs>
   301→
   302→        {/* ═══════ PRICE CHART (candles + volume) ═══════ */}
   303→
   304→        {/* Grid lines — dotted, TradingView style */}
   305→        {pGrid.map((g, i) => (
   306→          <g key={"g" + i}>
   307→            <line x1={L} y1={g.y} x2={W - R} y2={g.y} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   308→            <text x={W - R + 6} y={g.y + 3.5} fill={TXT} fontSize={10} fontFamily="Trebuchet MS, Tahoma, sans-serif">{g.lbl}</text>
   309→          </g>
   310→        ))}
   311→
   312→        {/* Vertical grid (time axis) */}
   313→        {data.map((d, i) => {
   314→          if (i % labelN !== 0 || i === 0) return null;
   315→          return <line key={"vg" + i} x1={xOf(i)} y1={priceTop} x2={xOf(i)} y2={priceTop + priceH} stroke={GRID} strokeOpacity={0.25} strokeDasharray="1 3" />;
   316→        })}
   317→
   318→        {/* Volume bars (behind candles, at bottom of price panel) */}
   319→        <g clipPath="url(#priceClip)">
   320→          {data.map((d, i) => {
   321→            const cx = xOf(i);
   322→            const isUp = d.close >= d.open;
   323→            const vy = yV(d.volume || 0);
   324→            return (
   325→              <rect key={"vol" + i} x={cx - candleW / 2} y={vy} width={candleW} height={Math.max(volTop + volH - vy, 0)}
   326→                fill={isUp ? 'rgba(38,166,154,0.25)' : 'rgba(239,83,80,0.25)'} />
   327→            );
   328→          })}
   329→        </g>
   330→
   331→        {/* Bollinger Bands fill */}
   332→        {bbPts.length > 2 && <polygon points={bbPts.join(' ')} fill="rgba(124,77,255,0.06)" stroke="none" />}
   333→
   334→        {/* Bollinger Band lines */}
   335→        {line(bb.upper.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-u')}
   336→        {line(bb.middle.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-m')}
   337→        {line(bb.lower.map(v => v !== null ? yP(v) : null), BB_CLR, 1, 'bb-l')}
   338→
   339→        {/* Moving Averages */}
   340→        {line(sma20.map(v => v !== null ? yP(v) : null), SMA20_CLR, 1.5, 's20')}
   341→        {line(sma50.map(v => v !== null ? yP(v) : null), SMA50_CLR, 1.5, 's50')}
   342→        {line(ema12.map(v => v !== null ? yP(v) : null), EMA12_CLR, 1.2, 'e12')}
   343→
   344→        {/* Candlesticks */}
   345→        <g clipPath="url(#priceClip)">
   346→          {data.map((d, i) => {
   347→            const cx = xOf(i);
   348→            const up = d.close >= d.open;
   349→            const c = up ? BULL : BEAR;
   350→            const bT = yP(Math.max(d.open, d.close));
   351→            const bB = yP(Math.min(d.open, d.close));
   352→            const bH = Math.max(bB - bT, 1);
   353→            return (
   354→              <g key={"k" + i}>
   355→                <line x1={cx} y1={yP(d.high)} x2={cx} y2={yP(d.low)} stroke={c} strokeWidth={1} />
   356→                <rect x={cx - candleW / 2} y={bT} width={candleW} height={bH} fill={c} stroke={c} strokeWidth={0.5} />
   357→              </g>
   358→            );
   359→          })}
   360→        </g>
   361→
   362→        {/* Current price line + tag */}
   363→        <line x1={L} y1={lastY} x2={W - R} y2={lastY} stroke={data[n-1].close >= data[n-1].open ? BULL : BEAR} strokeDasharray="4 2" strokeOpacity={0.7} strokeWidth={1} />
   364→        <rect x={W - R + 1} y={lastY - 10} width={R - 2} height={20} rx={2} fill={data[n-1].close >= data[n-1].open ? BULL : BEAR} />
   365→        <text x={W - R + 8} y={lastY + 4} fill="white" fontSize={10.5} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">{lastClose.toFixed(2)}</text>
   366→
   367→        {/* TV-style top-left indicator legend */}
   368→        {lastSma20 !== null && <text x={L + 8} y={priceTop + 14} fill={SMA20_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">SMA20 {lastSma20.toFixed(2)}</text>}
   369→        {lastSma50 !== null && <text x={L + 8} y={priceTop + 26} fill={SMA50_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">SMA50 {lastSma50.toFixed(2)}</text>}
   370→        {lastEma12 !== null && <text x={L + 8} y={priceTop + 38} fill={EMA12_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">EMA12 {lastEma12.toFixed(2)}</text>}
   371→        <text x={L + 8} y={priceTop + 50} fill={BB_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">BB (20, 2)</text>
   372→
   373→        {/* Volume label */}
   374→        <text x={L + 8} y={volTop + 12} fill={TXT} fontSize={9} fontFamily="Trebuchet MS, sans-serif" opacity={0.6}>Vol</text>
   375→        <text x={W - R + 6} y={volTop + 12} fill={TXT} fontSize={9} fontFamily="Trebuchet MS, sans-serif" opacity={0.5}>{(maxVol / 1e6).toFixed(1)}M</text>
   376→
   377→        {/* ═══════ PANEL SEPARATOR ═══════ */}
   378→        <line x1={L} y1={rsiTop - 1} x2={W - R} y2={rsiTop - 1} stroke={GRID} strokeOpacity={0.5} />
   379→
   380→        {/* ═══════ RSI (14) ═══════ */}
   381→        {/* RSI grid */}
   382→        <line x1={L} y1={yR(70)} x2={W - R} y2={yR(70)} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   383→        <line x1={L} y1={yR(50)} x2={W - R} y2={yR(50)} stroke={GRID} strokeOpacity={GRID_OP * 0.6} strokeDasharray="1 3" />
   384→        <line x1={L} y1={yR(30)} x2={W - R} y2={yR(30)} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   385→        {/* RSI Y labels */}
   386→        <text x={W - R + 6} y={yR(70) + 3.5} fill={TXT} fontSize={9.5} fontFamily="Trebuchet MS, sans-serif">70.00</text>
   387→        <text x={W - R + 6} y={yR(50) + 3.5} fill={TXT} fontSize={9.5} fontFamily="Trebuchet MS, sans-serif" opacity={0.6}>50.00</text>
   388→        <text x={W - R + 6} y={yR(30) + 3.5} fill={TXT} fontSize={9.5} fontFamily="Trebuchet MS, sans-serif">30.00</text>
   389→        {/* RSI line */}
   390→        <g clipPath="url(#rsiClip)">
   391→          {line(rsi.map(v => v !== null ? yR(v) : null), RSI_CLR, 1.5, 'rsi')}
   392→        </g>
   393→        {/* TV-style RSI label top-left */}
   394→        <text x={L + 8} y={rsiTop + 14} fill={RSI_CLR} fontSize={10.5} fontFamily="Trebuchet MS, sans-serif" fontWeight="700">RSI (14)</text>
   395→        {lastRsi !== null && (
   396→          <text x={L + 72} y={rsiTop + 14} fill={lastRsi > 70 ? BEAR : lastRsi < 30 ? BULL : TXT} fontSize={10.5} fontFamily="Trebuchet MS, sans-serif" fontWeight="600">{lastRsi.toFixed(2)}</text>
   397→        )}
   398→
   399→        {/* ═══════ PANEL SEPARATOR ═══════ */}
   400→        <line x1={L} y1={macdTop - 1} x2={W - R} y2={macdTop - 1} stroke={GRID} strokeOpacity={0.5} />
   401→
   402→        {/* ═══════ MACD ═══════ */}
   403→        {/* Zero line */}
   404→        <line x1={L} y1={macdTop + macdH / 2} x2={W - R} y2={macdTop + macdH / 2} stroke={GRID} strokeOpacity={GRID_OP} strokeDasharray="1 3" />
   405→        <text x={W - R + 6} y={macdTop + macdH / 2 + 3.5} fill={TXT} fontSize={9.5} fontFamily="Trebuchet MS, sans-serif" opacity={0.5}>0.00</text>
   406→        {/* Histogram */}
   407→        <g clipPath="url(#macdClip)">
   408→          {macdData.histogram.map((v, i) => {
   409→            if (v === null) return null;
   410→            const cx = xOf(i);
   411→            const pos = v >= 0;
   412→            const barY = pos ? yM(v) : macdTop + macdH / 2;
   413→            const barH = Math.abs(yM(v) - (macdTop + macdH / 2));
   414→            return (
   415→              <rect key={"h" + i} x={cx - candleW * 0.4} y={barY} width={candleW * 0.8} height={Math.max(barH, 0.5)}
   416→                fill={pos ? 'rgba(38,166,154,0.6)' : 'rgba(239,83,80,0.6)'} />
   417→            );
   418→          })}
   419→          {/* MACD & Signal lines */}
   420→          {line(macdData.macd.map(v => v !== null ? yM(v) : null), MACD_CLR, 1.5, 'macd')}
   421→          {line(macdData.signal.map(v => v !== null ? yM(v) : null), SIG_CLR, 1.2, 'sig')}
   422→        </g>
   423→        {/* TV-style MACD label top-left */}
   424→        <text x={L + 8} y={macdTop + 14} fill={MACD_CLR} fontSize={10.5} fontFamily="Trebuchet MS, sans-serif" fontWeight="700">MACD</text>
   425→        {lastMacd !== null && (
   426→          <text x={L + 52} y={macdTop + 14} fill={MACD_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif">{lastMacd.toFixed(2)}</text>
   427→        )}
   428→        {lastSig !== null && (
   429→          <text x={L + 110} y={macdTop + 14} fill={SIG_CLR} fontSize={10} fontFamily="Trebuchet MS, sans-serif">Signal {lastSig.toFixed(2)}</text>
   430→        )}
   431→
   432→        {/* ═══════ X-AXIS DATE LABELS ═══════ */}
   433→        {data.map((d, i) => {
   434→          if (i % labelN !== 0) return null;
   435→          return (
   436→            <text key={"d" + i} x={xOf(i)} y={macdTop + macdH + 14} fill={TXT} fontSize={10} textAnchor="middle" fontFamily="Trebuchet MS, sans-serif">
   437→              {d.date.substring(5)}
   438→            </text>
   439→          );
   440→        })}
   441→      </svg>
   442→    );
   443→  }, [data]);
   444→
   445→  return chart || <div className="flex items-center justify-center h-full text-muted-foreground text-sm">Asnjë të dhënë grafiku</div>;
   446→}
