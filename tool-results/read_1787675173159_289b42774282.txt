   122→// ── Helpers ──
   123→function calcEMA(data: number[], period: number): number[] {
   124→  const r: number[] = new Array(data.length).fill(NaN);
   125→  if (data.length < period) return r;
   126→  let s = 0; for (let i = 0; i < period; i++) s += data[i];
   127→  r[period - 1] = s / period;
   128→  const k = 2 / (period + 1);
   129→  for (let i = period; i < data.length; i++) r[i] = data[i] * k + r[i - 1] * (1 - k);
   130→  return r;
   131→}
   132→
   133→function calcATR(data: HistoricalDataPoint[], period = 14): number {
   134→  if (data.length < period + 1) return 0;
   135→  const trs: number[] = [];
   136→  for (let i = 1; i < data.length; i++) {
   137→    trs.push(Math.max(data[i].high - data[i].low, Math.abs(data[i].high - data[i-1].close), Math.abs(data[i].low - data[i-1].close)));
   138→  }
   139→  return trs.slice(-period).reduce((a, b) => a + b, 0) / period;
   140→}
   141→
   142→function pct(data: number[], days: number): number {
   143→  if (data.length < days + 1) return 0;
   144→  const c = data[data.length - 1], p = data[data.length - 1 - days];
   145→  return p > 0 ? ((c - p) / p) * 100 : 0;
   146→}

   148→// Position sizing: risk-based share calculation
   149→function calcPositionSize(entry: number, stop: number, riskBudgetPct: number, accountEquity: number): { shares: number; positionValue: number; riskDollars: number; riskPct: number } {
   150→  const riskPerShare = entry - stop;
   151→  if (riskPerShare <= 0 || entry <= 0) return { shares: 0, positionValue: 0, riskDollars: 0, riskPct: 0 };
   152→  const riskDollars = accountEquity * (riskBudgetPct / 100);
   153→  const shares = Math.max(0, Math.floor(riskDollars / riskPerShare));
   154→  const positionValue = shares * entry;
   155→  const actualRiskPct = positionValue > 0 ? ((shares * riskPerShare) / accountEquity) * 100 : 0;
   156→  return { shares, positionValue: Math.round(positionValue * 100) / 100, riskDollars: Math.round(riskDollars * 100) / 100, riskPct: Math.round(actualRiskPct * 100) / 100 };
   157→}

   159→// IBKR Bracket Order generator
   160→function generateBracketOrder(symbol: string, action: 'BUY' | 'SELL', entry: number, stop: number, target: number, shares: number): object {
   161→  const cOID = `ibkr_${symbol}_${Date.now()}`;
   162→  const parentOID = `${cOID}_parent`;
   163→  const stopChildOID = `${cOID}_stop`;
   164→  const targetChildOID = `${cOID}_target`;
   165→
   166→  return {
   167→    orders: [
   168→      {
   169→        cOID: parentOID,
   170→        symbol,
   171→        orderType: 'LMT',
   172→        side: action,
   173→        quantity: shares,
   174→        price: Math.round(entry * 100) / 100,
   175→        tif: 'GTC',
   176→        transmit: false,
   177→      },
   178→      {
   179→        cOID: stopChildOID,
   180→        symbol,
   181→        orderType: 'STP',
   182→        side: action === 'BUY' ? 'SELL' : 'BUY',
   183→        quantity: shares,
   184→        price: Math.round(stop * 100) / 100,
   185→        tif: 'GTC',
   186→        parentId: parentOID,
   187→        transmit: false,
   188→      },
   189→      {
   190→        cOID: targetChildOID,
   191→        symbol,
   192→        orderType: 'LMT',
   193→        side: action === 'BUY' ? 'SELL' : 'BUY',
   194→        quantity: shares,
   195→        price: Math.round(target * 100) / 100,
   196→        tif: 'GTC',
   197→        parentId: parentOID,
   198→        transmit: true, // last child transmits the bracket
   199→      },
   200→    ],
   201→    summary: `Bracket: ${action} ${shares} ${symbol} @ $${entry.toFixed(2)} | Stop $${stop.toFixed(2)} | Target $${target.toFixed(2)}`,
   202→  };
   203→}

   303→export async function GET() {

   313→      fetchHistoricalData('QQQ', '1y'),
   314→      ...SECTOR_ETFS.map(etf => fetchHistoricalData(etf, '1y')),
   315→    ]);
   316→    if (!spyData || !qqqData) return NextResponse.json({ error: 'Te dhena SPY/QQQ mungojne' }, { status: 500 });

   318→    // Build sector ETF data map
   319→    const sectorEtfData: Record<string, { closes: number[]; sma50: number[] }> = {};

   321→      const d = sectorDataArr[i];
   322→      if (d) {
   323→        const closes = d.map(dd => dd.close);
   324→        sectorEtfData[etf] = { closes, sma50: calculateSMA(closes, 50) };
   325→      }
   326→    });
   327→
   328→    const spyC = spyData.map(d => d.close), qqqC = qqqData.map(d => d.close);
   329→    const spyS50 = calculateSMA(spyC, 50), spyS200 = calculateSMA(spyC, 200);
   330→    const qqqS50 = calculateSMA(qqqC, 50), qqqS200 = calculateSMA(qqqC, 200);
   331→    const sL = spyC.length - 1, qL = qqqC.length - 1;
   332→    const spyA50 = spyC[sL] > (spyS50[sL]||0), spyA200 = spyC[sL] > (spyS200[sL]||0);
   333→    const qqqA50 = qqqC[qL] > (qqqS50[qL]||0), qqqA200 = qqqC[qL] > (qqqS200[qL]||0);
   334→    const regimeOk = spyA50 && spyA200 && qqqA50 && qqqA200;
   335→    const spyRS60 = pct(spyC, 60);

   338→    const syms = DEDUPED_UNIVERSE.filter(s => !ETF_SET.has(s));

   342→    for (let i = 0; i < syms.length; i += BATCH) {

   344→      const res = await Promise.allSettled(batch.map(async s => ({ s, d: await fetchHistoricalData(s, '1y') })));

   346→      if (i + BATCH < syms.length) await new Promise(r => setTimeout(r, 250));
   347→    }
   348→

   350→
   351→    // ── 2. PHASE 1 — Mechanical filter (liquidity + trend + ADX + stacked MA + event risk) ──
   352→    const phase1: FunnelStock[] = [];
   353→    let passedEventRiskCount = 0;

   357→      if (!data || data.length < 200) continue;

   369→      const passedLiq = price >= 10 && avgVol20 >= 1_000_000 && avgDolVol >= 20_000_000;
   370→
   371→      // Trend checks (existing)
   372→      const sma50 = calculateSMA(closes, 50);
   373→      const sma200 = calculateSMA(closes, 200);
   374→      const above50 = price > (sma50[last] || 0);
   375→      const above200 = price > (sma200[last] || 0);
   376→      const golden = (sma50[last] || 0) > (sma200[last] || 0);
   377→      const rs60 = pct(closes, 60);
   378→      const passedTrend = above50 && golden && rs60 > spyRS60;
   379→
   380→      // NEW: Stacked MA — close > EMA20 > SMA50 > SMA200
   381→      const ema20 = calcEMA(closes, 20);
   382→      const ema20Val = ema20[last] || 0;

   387→      // NEW: ADX 14 > 25 (trend strength)
   388→      const adxArr = calculateADX(highs, lows, closes, 14);
   389→      const adx = adxArr[last] || 0;
   390→      const passedADX = adx > 25;
   391→
   392→      // NEW: Event risk check
   393→      const eventResult = checkMultiEventRisk(sym);
   394→      const hasCriticalEvent = eventResult.hasCriticalEvent || eventResult.compositeRiskScore <= -50;
   395→      const passedEventRisk = !hasCriticalEvent;
   396→      if (passedEventRisk) passedEventRiskCount++;
   397→
   398→      const sector = SECTOR_MAP[sym] || 'Other';
   399→

   401→      phase1.push({
   402→        symbol: sym, price, sector,
   403→        avgVol20d: avgVol20, avgDolVol20d: avgDolVol,
   404→        passedLiquidity: passedLiq, passedTrend,
   405→        passedStackedMA: stackedMA, passedADX, passedEventRisk,

   407→        riskScore: 0, totalScore: 0,
   408→        setup: 'NONE', horizon: '',
   409→        rsi: 0, atr: 0, atrPct: 0, adx: Math.round(adx * 10) / 10,
   410→        volRatio: 0, volDeclining: false, lastDaySpike: false,
   411→        pullbackDays: 0, pullbackPct: 0, distFromEMA10: 0, distFromEMA20: 0,
   412→        aboveSMA50: above50, aboveSMA200: above200, sma50Above200: golden, stackedMA,
   413→        rsVsSPY: 0, rsVsQQQ: 0, rsVsSPY60d: rs60 - spyRS60,
   414→        entry: 0, stop: 0, target1R: 0, target2R: 0, target3R: 0,
   415→        riskPct: 0, rewardRiskRatio: 0, swingLow: 0,
   416→        positionSize: 0, positionValue: 0, riskDollars: 0,
   417→        eventRisk: eventResult.summary, eventRiskSeverity: eventResult.worstEvent.severity,
   418→        catalystStatus: eventResult.catalystStatus,
   419→        daysToEarnings: eventResult.daysToEarnings,
   420→        macroEventWithin24h: eventResult.macroEventWithin24h,
   421→        material8KLast30d: eventResult.material8KLast30d,
   422→        material8KSentiment: eventResult.material8KSentiment,
   423→        positionSizeMultiplier: eventResult.positionSizeMultiplier,
   424→        allowNewEntry: eventResult.allowNewEntry,
   425→        sectorEtf: SECTOR_ETF_MAP[sector] || '',
   426→        rsVsSector20d: 0,
   427→        sectorAboveSma50: false,
   428→        sectorRsStatus: 'INLINE',
   429→        bracketOrder: null,
   430→        decision: 'NO_TRADE', reasons: [], warnings: [],
   431→      });
   432→    }
   433→
   434→    const passedLiquidity = phase1.filter(s => s.passedLiquidity).length;
   435→    const passedTrend = phase1.filter(s => s.passedTrend).length;
   436→
   437→    // ── 3. PHASE 2 — Technical analysis on trend-passed stocks ──
   438→    const phase2: FunnelStock[] = [];
   439→
   440→    for (const stock of phase1) {
   441→      if (!stock.passedTrend) continue;
   442→
   443→      const data = hist[stock.symbol];
   444→      if (!data) continue;
   445→      const closes = data.map(d => d.close), highs = data.map(d => d.high), lows = data.map(d => d.low);
   446→      const vols = data.map(d => d.volume);
   447→      const last = closes.length - 1;
   448→      const price = closes[last];
   449→
   450→      const ema10 = calcEMA(closes, 10);
   451→      const ema20 = calcEMA(closes, 20);
   452→      const rsiArr = calculateRSI(closes, 14);
   453→      const rsi = rsiArr[last] || 50;
   454→      const atr = calcATR(data, 14);
   455→      const atrPct = price > 0 ? (atr / price) * 100 : 0;
   456→
   457→      // ── A) Trend Quality (0-100) — 25% weight ──
   458→      const sma50 = calculateSMA(closes, 50);
   459→      const sma200 = calculateSMA(closes, 200);
   460→      let tScore = 0;
   461→      if (price > (sma50[last]||0)) tScore += 20;
   462→      if (price > (sma200[last]||0)) tScore += 20;
   463→      if ((sma50[last]||0) > (sma200[last]||0)) tScore += 15;
   464→      // NEW: Stacked MA bonus (+15)
   465→      if (stock.stackedMA) tScore += 15;
   466→      // Higher-high structure
   467→      const h20a = Math.max(...highs.slice(-40, -20));
   468→      const h20b = Math.max(...highs.slice(-20));
   469→      if (h20b > h20a) tScore += 15;
   470→      // NEW: ADX bonus (+15 if > 25)
   471→      if (stock.adx > 25) tScore += 15;
   472→      stock.trendScore = Math.min(100, tScore);
   473→
   474→      // ── B) Relative Strength (0-100) — 20% weight ──
   475→      const rsSpy22 = pct(closes, 22) - pct(spyC, 22);
   476→      const rsQqq22 = pct(closes, 22) - pct(qqqC, 22);
   477→      const rsSpy60 = pct(closes, 60) - pct(spyC, 60);
   478→      let rsScore = 50;
   479→      if (rsSpy22 > 0) rsScore += Math.min(25, rsSpy22 * 3);
   480→      else rsScore -= Math.min(25, Math.abs(rsSpy22) * 3);
   481→      if (rsSpy60 > 0) rsScore += Math.min(25, rsSpy60 * 2);
   482→      else rsScore -= Math.min(25, Math.abs(rsSpy60) * 2);
   483→      stock.rsScore = Math.round(Math.max(0, Math.min(100, rsScore)));
   484→      stock.rsVsSPY = Math.round(rsSpy22 * 100) / 100;
   485→      stock.rsVsQQQ = Math.round(rsQqq22 * 100) / 100;
   486→      stock.rsVsSPY60d = Math.round(rsSpy60 * 100) / 100;
   487→
   488→      // ── B2) RS vs Sector ETF (20D) ──
   489→      const sector = stock.sector;
   490→      const sectorEtf = SECTOR_ETF_MAP[sector] || '';
   491→      const etfInfo = sectorEtf ? sectorEtfData[sectorEtf] : null;
   492→      let rsVsSector20d = 0;
   493→      let sectorAboveSma50 = false;
   494→      let sectorRsStatus = 'INLINE' as string;
   495→      if (etfInfo && etfInfo.closes.length >= 21) {
   496→        const stockRet20d = pct(closes, 20);
   497→        const sectorRet20d = pct(etfInfo.closes, 20);
   498→        rsVsSector20d = Math.round((stockRet20d - sectorRet20d) * 100) / 100;
   499→        const etfLast = etfInfo.closes.length - 1;
   500→        sectorAboveSma50 = etfInfo.closes[etfLast] > (etfInfo.sma50[etfLast] || 0);
   501→        if (rsVsSector20d >= 3) sectorRsStatus = 'LEADING';
   502→        else if (rsVsSector20d <= -3) sectorRsStatus = 'LAGGING';
   503→        else sectorRsStatus = 'INLINE';
   504→      }
   505→      stock.sectorEtf = sectorEtf;
   506→      stock.rsVsSector20d = rsVsSector20d;
   507→      stock.sectorAboveSma50 = sectorAboveSma50;
   508→      stock.sectorRsStatus = sectorRsStatus;
   509→
   510→      // Blend RS vs Sector into rsScore: bonus/penalty
   511→      if (sectorRsStatus === 'LEADING' && sectorAboveSma50) stock.rsScore = Math.min(100, stock.rsScore + 8);
   512→      else if (sectorRsStatus === 'LAGGING' && !sectorAboveSma50) stock.rsScore = Math.max(0, stock.rsScore - 8);
   513→
   514→      // ── C) Momentum (0-100) — 15% weight ──
   515→      const mom5 = pct(closes, 5), mom10 = pct(closes, 10), mom22 = pct(closes, 22);
   516→      let mScore = 50;
   517→      if (mom5 > -2) mScore += 10; else mScore -= 10;
   518→      if (mom10 > 0) mScore += 15; else mScore -= 10;
   519→      if (mom22 > 0) mScore += 15; else mScore -= 10;
   520→      if (mom5 < 8) mScore += 10; else mScore -= 15;
   521→      stock.momentumScore = Math.round(Math.max(0, Math.min(100, mScore)));
   522→
   523→      // ── D) Volume Confirmation (0-100) — 15% weight ──
   524→      const avgVol20 = vols.slice(-20).reduce((a,b) => a+b, 0) / 20;
   525→      const recent3 = vols.slice(-3).reduce((a,b) => a+b, 0) / 3;
   526→      const pbVol = vols.slice(-5, -1);
   527→      const priorVol = vols.slice(-10, -5);
   528→      const avgPb = pbVol.length > 0 ? pbVol.reduce((a,b) => a+b, 0)/pbVol.length : 0;
   529→      const avgPrior = priorVol.length > 0 ? priorVol.reduce((a,b) => a+b, 0)/priorVol.length : 0;
   530→      const volDeclining = avgPb < avgPrior * 0.95;
   531→      const lastDaySpike = vols[last] > avgVol20 * 1.1;
   532→      const volRatio = avgVol20 > 0 ? recent3 / avgVol20 : 1;
   533→      let vScore = 50;
   534→      if (volDeclining) vScore += 20;
   535→      if (lastDaySpike) vScore += 15;
   536→      if (volRatio > 0.8 && volRatio < 1.5) vScore += 10;
   537→      if (avgVol20 > 5_000_000) vScore += 5;
   538→      stock.volConfScore = Math.round(Math.max(0, Math.min(100, vScore)));
   539→      stock.volRatio = Math.round(volRatio * 100) / 100;
   540→      stock.volDeclining = volDeclining;
   541→      stock.lastDaySpike = lastDaySpike;
   542→
   543→      // ── E) Setup Quality (0-100) — 10% weight ──
   544→      const ema10Val = ema10[last], ema20Val = ema20[last];
   545→      const dist10 = !isNaN(ema10Val) ? ((price - ema10Val) / ema10Val) * 100 : 99;
   546→      const dist20 = !isNaN(ema20Val) ? ((price - ema20Val) / ema20Val) * 100 : 99;
   547→      stock.distFromEMA10 = Math.round(dist10 * 100) / 100;
   548→      stock.distFromEMA20 = Math.round(dist20 * 100) / 100;
   549→
   550→      // Find pullback
   551→      let highIdx = last;
   552→      for (let i = last; i >= Math.max(0, last - 10); i--) if (closes[i] >= closes[highIdx]) highIdx = i;
   553→      const peakPrice = closes[highIdx];
   554→      const pbPct = peakPrice > 0 ? ((price - peakPrice) / peakPrice) * 100 : 0;
   555→      const swLow = Math.min(...lows.slice(highIdx, last + 1));
   556→      let pbDays = 0;
   557→      for (let i = highIdx + 1; i <= last; i++) if (closes[i] < closes[i-1]) pbDays++;
   558→
   559→      stock.pullbackDays = pbDays;
   560→      stock.pullbackPct = Math.round(pbPct * 100) / 100;
   561→      stock.swingLow = Math.round(swLow * 100) / 100;
   562→
   563→      let setup: 'PULLBACK' | 'BREAKOUT' | 'TREND_CONT' | 'NONE' = 'NONE';
   564→      let sScore = 0;
   565→      const reasons: string[] = [];
   566→
   567→      // Pullback
   568→      if (pbDays >= 2 && pbDays <= 8 && pbPct >= -8 && pbPct <= -0.5) {
   569→        setup = 'PULLBACK';
   570→        sScore += 30;
   571→        if (pbDays >= 3 && pbDays <= 6) { sScore += 20; reasons.push(`Pullback ${pbDays}d (ideal)`); }
   572→        else { sScore += 10; reasons.push(`Pullback ${pbDays}d`); }
   573→        if (Math.abs(dist10) < 3 || Math.abs(dist20) < 3) { sScore += 20; reasons.push('Afer EMA 10/20'); }
   574→        if (volDeclining) { sScore += 15; reasons.push('Volumi ne renie (i mire)'); }
   575→        if (lastDaySpike) { sScore += 15; reasons.push('Volum konfirmim'); }
   576→        if (rsi >= 40 && rsi <= 65) { sScore += 10; reasons.push(`RSI ${rsi.toFixed(0)}`); }
   577→      }
   578→
   579→      // Breakout
   580→      if (setup === 'NONE') {
   581→        const high20 = Math.max(...highs.slice(-20));
   582→        if (price >= high20 * 0.98 && lastDaySpike && rsi >= 45 && rsi <= 70) {
   583→          setup = 'BREAKOUT'; sScore = 55;
   584→          reasons.push('Breakout 20d + volum');
   585→          if (rsi >= 45 && rsi <= 65) { sScore += 15; reasons.push(`RSI ${rsi.toFixed(0)}`); }
   586→        }
   587→      }
   588→
   589→      // Trend continuation
   590→      if (setup === 'NONE' && pbPct > -0.5 && pbPct < 2 && price > (sma50[last]||0)) {
   591→        setup = 'TREND_CONT'; sScore = 40;
   592→        reasons.push('Trend continuation');
   593→        if (rsi >= 50 && rsi <= 65) { sScore += 15; reasons.push(`RSI ${rsi.toFixed(0)}`); }
   594→        if (lastDaySpike) { sScore += 10; reasons.push('Volum konfirmim'); }
   595→      }
   596→
   597→      stock.setup = setup;
   598→      stock.setupScore = Math.min(100, sScore);
   599→      stock.rsi = Math.round(rsi * 10) / 10;
   600→      stock.atr = Math.round(atr * 100) / 100;
   601→      stock.atrPct = Math.round(atrPct * 100) / 100;
   602→      stock.reasons = reasons;
   603→
   604→      // Entry / Stop / Target (BLUEPRINT-ALIGNED)
   605→      const isBreakout = setup === 'BREAKOUT';
   606→      const high20 = Math.max(...highs.slice(-20));
   607→      const entry = isBreakout
   608→        ? Math.round((high20 * 1.002) * 100) / 100
   609→        : Math.round(price * 100) / 100;
   610→
   611→      // NEW: Stop = max(entry - 1.5*ATR, swingLow - 0.2*ATR) — use the tighter (higher) one
   612→      const stopAtr = Math.round((entry - atr * 1.5) * 100) / 100;
   613→      const stopSwing = Math.round((swLow - atr * 0.2) * 100) / 100;
   614→      const stop = Math.max(stopAtr, stopSwing); // tighter stop = higher price
   615→
   616→      const riskPerShare = entry - stop;
   617→      const riskPct = entry > 0 ? (riskPerShare / entry) * 100 : 0;
   618→      const target1R = Math.round((entry + riskPerShare) * 100) / 100;
   619→      const target2R = Math.round((entry + riskPerShare * 2) * 100) / 100;
   620→      const target3R = Math.round((entry + riskPerShare * 3) * 100) / 100;
   621→      const rr = riskPerShare > 0 ? (riskPerShare * 3) / riskPerShare : 0; // based on 3R now
   622→
   623→      stock.entry = entry;
   624→      stock.stop = stop;
   625→      stock.target1R = target1R;
   626→      stock.target2R = target2R;
   627→      stock.target3R = target3R;
   628→      stock.riskPct = Math.round(riskPct * 100) / 100;
   629→      stock.rewardRiskRatio = Math.round(rr * 10) / 10;
   630→
   631→      // NEW: Position sizing (apply catalyst multiplier)
   632→      const multiplier = stock.positionSizeMultiplier > 0 ? stock.positionSizeMultiplier : 1;
   633→      const pos = calcPositionSize(entry, stop, MAX_RISK_PCT * multiplier, ACCOUNT_EQUITY);
   634→      stock.positionSize = pos.shares;
   635→      stock.positionValue = pos.positionValue;
   636→      stock.riskDollars = pos.riskDollars;
   637→
   638→      // Horizon based on ATR
   639→      if (atrPct < 1.5) stock.horizon = '10D';
   640→      else if (atrPct < 3) stock.horizon = '5D';
   641→      else stock.horizon = '5D';
   642→
   643→      // ── F) Risk Quality (0-100) — 5% weight ──
   644→      let rScore = 50;
   645→      if (riskPct <= 3) rScore += 20;
   646→      else if (riskPct <= 5) rScore += 10;
   647→      else if (riskPct > 7) rScore -= 20;
   648→      if (rr >= 2) rScore += 15; // 2R+ is good
   649→      else if (rr >= 1.5) rScore += 5;
   650→      else rScore -= 15;
   651→      if (atrPct < 2) rScore += 10;
   652→      else if (atrPct > 4) rScore -= 10;
   653→      // NEW: Event risk penalty
   654→      if (!stock.passedEventRisk) rScore -= 25;
   655→      stock.riskScore = Math.round(Math.max(0, Math.min(100, rScore)));
   656→
   657→      // ── Total Score ──
   658→      stock.totalScore = Math.round(
   659→        stock.trendScore * 0.25 +
   660→        stock.rsScore * 0.20 +
   661→        stock.momentumScore * 0.15 +
   662→        stock.volConfScore * 0.15 +
   663→        sScore * 0.10 +
   664→        50 * 0.10 + // fundamentals placeholder (neutral)
   665→        stock.riskScore * 0.05
   666→      );
   667→
   668→      // Only keep stocks with valid setups
   669→      if (setup !== 'NONE') phase2.push(stock);
   670→    }
   671→
   672→    const passedSetup = phase2.length;
   673→
   674→    // ── 4. PHASE 3 — Risk gate + Decision + Event risk ──
   675→    const phase3: FunnelStock[] = [];
   676→
   677→    for (const stock of phase2) {
   678→      const warnings: string[] = [...stock.warnings];
   679→      let decision: Decision = 'NO_TRADE';
   680→
   681→      // R:R gate (based on 3R now, so 2.0+ is the threshold)
   682→      if (stock.rewardRiskRatio < 2.0) {
   683→        warnings.push(`R:R ${stock.rewardRiskRatio} — i ulet (duhet 1:2 me 3R target)`);
   684→      }
   685→
   686→      // RSI gate
   687→      if (stock.rsi > 75) {
   688→        warnings.push(`RSI ${stock.rsi} — i mbivleresuar`);
   689→      }
   690→
   691→      // Risk % gate
   692→      if (stock.riskPct > 8) {
   693→        warnings.push(`Rreziku ${stock.riskPct}% — shume i larte`);
   694→      }
   695→
   696→      // ADX gate
   697→      if (stock.adx < 20) {
   698→        warnings.push(`ADX ${stock.adx} — trendi i dobet (duhet > 25)`);
   699→      }
   700→
   701→      // Regime gate
   702→      if (!regimeOk) {
   703→        warnings.push('REGJIMI jo OK — vetem watcher');
   704→      }
   705→
   706→      // NEW: Event risk gate (enhanced with catalyst status)
   707→      if (!stock.passedEventRisk) {
   708→        warnings.push(`EVENT RISK: ${stock.eventRisk}`);
   709→      }
   710→      if (!stock.allowNewEntry) {
   711→        warnings.push(`CATALYST GATE: ${stock.catalystStatus} — mos hap long të ri`);
   712→      } else if (stock.positionSizeMultiplier < 1) {
   713→        warnings.push(`CATALYST GATE: ${stock.catalystStatus} — pozicion ${Math.round(stock.positionSizeMultiplier * 100)}%`);
   714→      }
   715→
   716→      // Stacked MA bonus/penalty
   717→      if (!stock.stackedMA) {
   718→        warnings.push('MA jo te stackuara (close/EMA20/SMA50/SMA200)');
   719→      }
   720→
   721→      // Sector RS gate
   722→      if (stock.sectorRsStatus === 'LAGGING' && !stock.sectorAboveSma50) {
   723→        warnings.push(`SECTOR RS: ${stock.rsVsSector20d}% vs ${stock.sectorEtf} — sektori i dobet`);
   724→      }
   725→
   726→      // Extended (chasing)
   727→      if (stock.pullbackPct > -0.5 && stock.setup === 'PULLBACK') {
   728→        warnings.push('Jo te vertete nje pullback — extended');
   729→      }
   730→
   731→      // Determine decision (catalyst gate + sector RS affects READY)
   732→      const hasRR = stock.rewardRiskRatio >= 2.0;
   733→      const rsiOk = stock.rsi >= 30 && stock.rsi <= 75;
   734→      const riskOk = stock.riskPct <= 8;
   735→      const scoreOk = stock.totalScore >= 45;
   736→      const eventOk = stock.passedEventRisk;
   737→      const catalystOk = stock.allowNewEntry;
   738→      const sectorRsOk = !(stock.sectorRsStatus === 'LAGGING' && !stock.sectorAboveSma50);
   739→
   740→      if (hasRR && rsiOk && riskOk && scoreOk && regimeOk && eventOk && catalystOk && sectorRsOk) {
   741→        decision = 'READY';
   742→      } else if (hasRR && rsiOk && riskOk && scoreOk && (eventOk || !regimeOk)) {
   743→        decision = 'WATCHLIST';
   744→        if (!regimeOk) warnings.push('WATCHLIST: Regjimi nuk lejon long tani');
   745→        if (!eventOk) warnings.push('WATCHLIST: Event risk — prit');
   746→        if (!catalystOk) warnings.push('WATCHLIST: Catalyst gate — prit');
   747→        if (!sectorRsOk) warnings.push('WATCHLIST: Sector RS i dobet — prit');
   748→      } else if (!hasRR || stock.riskPct > 6) {
   749→        if (stock.setupScore >= 50) decision = 'WATCHLIST';
   750→        else decision = 'NO_TRADE';
   751→      } else if (stock.rsi > 72 || stock.pullbackPct > -0.3) {
   752→        decision = 'EXTENDED';
   753→      } else {
   754→        decision = 'WATCHLIST';
   755→      }
   756→
   757→      // Event risk override
   758→      if (!eventOk && decision === 'READY') {
   759→        decision = 'EVENT_RISK';
   760→        warnings.push('EVENT_RISK: Ngjarje kritike — mos hyr');
   761→      }
   762→
   763→      // RSI overbought override
   764→      if (stock.rsi > 70 && stock.totalScore >= 55) {
   765→        decision = 'EVENT_RISK';
   766→        warnings.push('EVENT_RISK: RSI i larte, rrezik kthimi');
   767→      }
   768→
   769→      stock.decision = decision;
   770→      stock.warnings = warnings;
   771→
   772→      // NEW: Generate bracket order for READY trades
   773→      if (decision === 'READY' && stock.positionSize > 0) {
   774→        stock.bracketOrder = generateBracketOrder(
   775→          stock.symbol, 'BUY', stock.entry, stock.stop, stock.target3R, stock.positionSize
   776→        );
   777→      }
   778→
   779→      if (decision === 'READY' || decision === 'WATCHLIST' || decision === 'EVENT_RISK') {
   780→        phase3.push(stock);
   781→      }
   782→    }
   783→
   784→    const passedRisk = phase3.length;
   785→
   786→    // ── 5. PHASE 4 — Sector exposure limit + Top 10 final ──
   787→    const sectorCount: Record<string, number> = {};
   788→    const afterSectorLimit: FunnelStock[] = [];
   789→
   790→    // Sort by totalScore descending first
   791→    phase3.sort((a, b) => b.totalScore - a.totalScore);
   792→
   793→    for (const stock of phase3) {
   794→      const sec = stock.sector;
   795→      if (!sectorCount[sec]) sectorCount[sec] = 0;
   796→      if (sectorCount[sec] < MAX_PER_SECTOR) {
   797→        afterSectorLimit.push(stock);
   798→        sectorCount[sec]++;
   799→      } else {
   800→        stock.warnings.push(`Sector limit: ${sec} ka tashme ${MAX_PER_SECTOR} aksione`);
   801→      }
   802→    }
   803→
   804→    const passedSectorLimit = afterSectorLimit.length;
   805→    const topStocks = afterSectorLimit.slice(0, 10);
   806→    // READY first, then WATCHLIST, then others
   807→    topStocks.sort((a, b) => {
   808→      const order: Record<Decision, number> = { READY: 0, WATCHLIST: 1, EVENT_RISK: 2, EXTENDED: 3, NO_TRADE: 4 };
   809→      const diff = (order[a.decision] || 5) - (order[b.decision] || 5);
   810→      return diff !== 0 ? diff : b.totalScore - a.totalScore;
   811→    });
   812→
   813→    // Build sector exposure summary
   814→    const sectorExposure: Record<string, number> = {};
   815→    for (const s of topStocks) {
   816→      if (!sectorExposure[s.sector]) sectorExposure[s.sector] = 0;
   817→      sectorExposure[s.sector]++;
   818→    }
   819→
   820→    const elapsed = ((Date.now() - t0) / 1000).toFixed(1);
   821→    console.log(`[IBKR v2] ${syms.length} → ${passedLiquidity} → ${passedTrend} → ${passedSetup} → ${passedRisk} → ${passedSectorLimit} → ${topStocks.length} (${elapsed}s)`);
   822→
   823→    return NextResponse.json({
   824→      scannedAt: new Date().toISOString(),
   825→      regimeOk,
   826→      regimeDetail: {
   827→        spy: { above50: spyA50, above200: spyA200 },
   828→        qqq: { above50: qqqA50, above200: qqqA200 },
   829→      },
   830→      funnel: {
   831→        universe: syms.length,
   832→        passedLiquidity,
   833→        passedTrend,
   834→        passedSetup,
   835→        passedRisk,
   836→        passedEventRisk: passedEventRiskCount,
   837→        passedSectorLimit,
   838→        displayed: topStocks.length,
   839→      },
   840→      results: topStocks,
   841→      sectorExposure,
   842→    } satisfies FunnelResponse);
   843→  } catch (err: any) {
   844→    console.error('[IBKR v2] Error:', err);
   845→    return NextResponse.json({ error: err?.message || 'Gabim skaneri' }, { status: 500 });
   846→  }
   847→}
