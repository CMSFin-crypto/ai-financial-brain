export async function getRealPrice(
  ticker: string,
  options: FetchOptions = {}
): Promise<LivePrice | null> {
  const t = ticker.toUpperCase().trim();

  const cached = priceCache.get(t);
  if (!options.forceRefresh && cached && Date.now() - cached.fetchedAt < CACHE_TTL_MS) {
    return cached.data;
  }

  const yfPrice = await fetchFromYahooWithRetry(t);
  if (yfPrice) {
    priceCache.set(t, { data: yfPrice, fetchedAt: Date.now() });
    return yfPrice;
  }

  const avPrice = await fetchFromAlphaVantage(t);
  if (avPrice) {
    priceCache.set(t, { data: avPrice, fetchedAt: Date.now() });
    return avPrice;
  }

  console.warn(`[PRICE] Could not fetch real price for ${ticker} from any source`);
  return null;
}

export async function getRealPrices(
  tickers: string[],
  options: FetchOptions = {}
): Promise<Record<string, LivePrice>> {
  const prices: Record<string, LivePrice> = {};
  const toFetch: string[] = [];

  for (const ticker of tickers) {
    const t = ticker.toUpperCase().trim();
    const cached = priceCache.get(t);
    if (!options.forceRefresh && cached && Date.now() - cached.fetchedAt < CACHE_TTL_MS) {
      prices[t] = cached.data;
    } else {
      toFetch.push(t);
    }
  }

  if (toFetch.length === 0) {
    console.log(`[PRICE] All ${tickers.length} prices from cache`);
    return prices;
  }

  console.log(`[PRICE] Cache hits: ${Object.keys(prices).length}, fetching: ${toFetch.length} tickers...`);

  const BATCH_SIZE = 6;
  const BATCH_DELAY_MS = 800;
  const chunks = chunkArray(toFetch, BATCH_SIZE);

  for (let i = 0; i < chunks.length; i++) {
    const chunk = chunks[i];
    const results = await Promise.allSettled(chunk.map(t => getRealPrice(t, options)));

    chunk.forEach((ticker, j) => {
      const result = results[j];
      if (result.status === 'fulfilled' && result.value) {
        prices[ticker] = result.value;
      }
    });

    if (i < chunks.length - 1) {
      await delay(BATCH_DELAY_MS);
    }
  }

  console.log(`[PRICE] Batch complete: ${Object.keys(prices).length}/${tickers.length} prices obtained`);
  return prices;
}

export async function fetchHistoricalData(
  ticker: string,
  range?: string,
  options: FetchOptions = {}
): Promise<HistoricalDataPoint[] | null> {
  const t = ticker.toUpperCase().trim();
  const r = range || '6mo';
  const interval = r === '1d' ? '5m' : '1d';

  const cacheKey = `${t}_${r}`;
  const cached = chartCache.get(cacheKey);
  if (!options.forceRefresh && cached && Date.now() - cached.fetchedAt < CHART_CACHE_TTL_MS) {
    return cached.data;
  }

  const endpoints = ['https://query1.finance.yahoo.com', 'https://query2.finance.yahoo.com'];

  for (const base of endpoints) {
    try {
      const url = `${base}/v8/finance/chart/${t}?range=${r}&interval=${interval}&includePrePost=false`;
      const res = await fetch(url, {
        signal: AbortSignal.timeout(10000),
        headers: BROWSER_HEADERS,
      });

      if (!res.ok) continue;

      const data = await res.json();
      const result = data?.chart?.result?.[0];
      if (!result) continue;

      const timestamps: number[] = result.timestamp || [];
      const quote = result.indicators?.quote?.[0];
      if (!quote || !timestamps.length) continue;

      const points: HistoricalDataPoint[] = [];
      for (let i = 0; i < timestamps.length; i++) {
        const open = quote.open?.[i];
        const high = quote.high?.[i];
        const low = quote.low?.[i];
        const close = quote.close?.[i];
        const vol = quote.volume?.[i];

        if (close && close > 0 && open && high && low) {
          const d = new Date(timestamps[i] * 1000);
          if (d.getDay() === 0 || d.getDay() === 6) continue;
          points.push({
            date: d.toISOString().split('T')[0],
            open: +open.toFixed(2),
            high: +high.toFixed(2),
            low: +low.toFixed(2),
            close: +close.toFixed(2),
            volume: vol || 0,
          });
        }
      }

      if (points.length >= 15) {
        console.log(`[CHART] ${t}: fetched ${points.length} days of real data from ${base}`);
        chartCache.set(cacheKey, { data: points, fetchedAt: Date.now() });
        return points;
      }
    } catch (err: any) {
      const msg = err?.name === 'TimeoutError' ? 'timeout' : err?.message || 'unknown';
      console.log(`[CHART] ${t}: ${base} failed: ${msg}`);
      continue;
    }
  }

  return null;
}

export async function getBatchQuotesFast(
  tickers: string[],
  options: FetchOptions = {}
): Promise<Record<string, YahooFundamentals>> {
  const cacheKey = [...tickers].sort().join(',');
  const cached = batchQuoteCache.get(cacheKey);
  if (!options.forceRefresh && cached && Date.now() - cached.fetchedAt < BATCH_QUOTE_CACHE_TTL) {
    return cached.data;
  }

  const results: Record<string, YahooFundamentals> = {};
  const localData = loadLocalFundamentals();

  for (const ticker of tickers) {
    const t = ticker.toUpperCase().trim();

    if (!d) continue;

    const fund = localToYahooFundamentals(t, d);
    if (!fund) continue;

    results[t] = fund;
    fundCache.set(t, { data: fund, fetchedAt: Date.now() });
  }

  batchQuoteCache.set(cacheKey, { data: results, fetchedAt: Date.now() });
  console.log(`[BATCH-QUOTE] Total: ${Object.keys(results).length}/${tickers.length} fundamentals obtained (local JSON)`);
  return results;
}

async function loadPreviousScores(): Promise<Record<string, { score: number; predictedAt: Date }>> {
  try {
    const cutoff = new Date(Date.now() - 18 * 60 * 60 * 1000);

    const preds = await prisma.prediction.findMany({
      where: {
        horizonDays: 1,
        modelVersion: 'predict-v3-regime-spillover',
        predictedAt: { lte: cutoff },
      },
      orderBy: { predictedAt: 'desc' },
      select: { symbol: true, rawScore: true, predictedAt: true },
      take: 1000,
    });

    const bySymbol = new Map<string, { score: number; predictedAt: Date }>();
    for (const p of preds) {
      if (p.rawScore != null && !bySymbol.has(p.symbol)) {
        bySymbol.set(p.symbol, { score: p.rawScore, predictedAt: p.predictedAt });
      }
    }

    console.log(`[SCAN] Loaded previous-day scores for ${bySymbol.size} symbols from DB`);
    return Object.fromEntries(bySymbol);
  } catch (err) {
    console.warn('[SCAN] Failed to load previous scores from DB:', err);
    return {};
  }
}

import prisma from '@/lib/prisma';

export async function logExecutionEvent(input: {
  predictionId?: string;
  symbol: string;
  side: 'BUY' | 'SELL';
  orderType: string;
  quantity: number;
  intendedPrice?: number;
  submittedPrice?: number;
  filledPrice?: number;
  spreadAtDecision?: number;
  spreadAtFill?: number;
  submittedAt: Date;
  ackAt?: Date;
  filledAt?: Date;
  status: string;
  rejectReason?: string;
  stopAttached?: boolean;
  venue?: string;
}) {
  const latencyMs = input.ackAt
    ? input.ackAt.getTime() - input.submittedAt.getTime()
    : null;

  let slippageBps: number | null = null;
  if (input.intendedPrice && input.filledPrice) {
    const raw = ((input.filledPrice - input.intendedPrice) / input.intendedPrice) * 10000;
    slippageBps = input.side === 'BUY' ? raw : -raw;
  }

  return prisma.executionEvent.create({
    data: {
      ...input,
      latencyMs,
      slippageBps,
      partialFillPct: input.filledPrice ? 100 : 0,
      stopAttached: input.stopAttached ?? false,
    },
  });
}

import prisma from '@/lib/prisma';

function mean(values: number[]) {
  return values.length ? values.reduce((a, b) => a + b, 0) / values.length : null;
}

function percentile(values: number[], p: number) {
  if (!values.length) return null;
  const sorted = [...values].sort((a, b) => a - b);
  const idx = Math.min(sorted.length - 1, Math.floor((p / 100) * sorted.length));
  return sorted[idx];
}

export async function getExecutionQuality(params?: { symbol?: string; days?: number }) {
  const since = new Date();
  since.setDate(since.getDate() - (params?.days ?? 7));

  const rows = await prisma.executionEvent.findMany({
    where: {
      ...(params?.symbol ? { symbol: params.symbol } : {}),
      createdAt: { gte: since },
    },
    orderBy: { createdAt: 'desc' },
  });

  const total = rows.length;
  const filled = rows.filter(r => r.status === 'FILLED');
  const rejected = rows.filter(r => r.status === 'REJECTED');
  const latencies = rows.flatMap(r => (r.latencyMs != null ? [r.latencyMs] : []));
  const slippages = rows.flatMap(r => (r.slippageBps != null ? [r.slippageBps] : []));

  return {
    sampleSize: total,
    fillRate: total ? filled.length / total : null,
    rejectRate: total ? rejected.length / total : null,
    avgLatencyMs: mean(latencies),
    p95LatencyMs: percentile(latencies, 95),
    avgSlippageBps: mean(slippages),
    p95SlippageBps: percentile(slippages, 95),
    stopAttachedRate: total ? rows.filter(r => r.stopAttached).length / total : null,
    topRejectReasons: Object.entries(
      rejected.reduce<Record<string, number>>((acc, row) => {
        const k = row.rejectReason || 'UNKNOWN';
        acc[k] = (acc[k] || 0) + 1;
        return acc;
      }, {})
    ).sort((a, b) => b[1] - a[1]).slice(0, 5),
  };
}

export type HealthLevel = 'OK' | 'L1_SOFT_PAUSE' | 'L2_SESSION_HALT' | 'L3_SYSTEM_HOLD' | 'L4_HARD_KILL';

type HealthInput = {
  fillRate?: number | null;
  rejectRate?: number | null;
  avgSlippageBps?: number | null;
  p95SlippageBps?: number | null;
  avgLatencyMs?: number | null;
  maxDrawdownPct?: number | null;
  consecutiveLosses?: number;
  stopAttachedRate?: number | null;
  exposurePct?: number | null;
  baselineSlippageBps?: number | null;
  baselineLatencyMs?: number | null;
};

export function assessStrategyHealth(input: HealthInput) {
  const issues: string[] = [];
  let level: HealthLevel = 'OK';

  const setLevel = (next: HealthLevel) => {
    const order = ['OK', 'L1_SOFT_PAUSE', 'L2_SESSION_HALT', 'L3_SYSTEM_HOLD', 'L4_HARD_KILL'];
    if (order.indexOf(next) > order.indexOf(level)) level = next;
  };

  if ((input.fillRate ?? 1) < 0.95) {
    issues.push('Fill rate too low');
    setLevel('L1_SOFT_PAUSE');
  }
  if ((input.rejectRate ?? 0) > 0.02) {
    issues.push('Reject rate too high');
    setLevel('L1_SOFT_PAUSE');
  }
  if (input.baselineSlippageBps != null && (input.avgSlippageBps ?? 0) > input.baselineSlippageBps * 1.2) {
    issues.push('Average slippage above baseline');
    setLevel('L1_SOFT_PAUSE');
  }
  if (input.baselineSlippageBps != null && (input.p95SlippageBps ?? 0) > input.baselineSlippageBps * 2) {
    issues.push('P95 slippage too high');
    setLevel('L2_SESSION_HALT');
  }
  if (input.baselineLatencyMs != null && (input.avgLatencyMs ?? 0) > input.baselineLatencyMs * 2) {
    issues.push('Latency degraded');
    setLevel('L1_SOFT_PAUSE');
  }
  if ((input.maxDrawdownPct ?? 0) > 5) {
    issues.push('Drawdown exceeded');
    setLevel('L2_SESSION_HALT');
  }
  if ((input.consecutiveLosses ?? 0) >= 5) {
    issues.push('Consecutive losses too high');
    setLevel('L2_SESSION_HALT');
  }
  if ((input.stopAttachedRate ?? 1) < 1) {
    issues.push('Stops not consistently attached');
    setLevel('L2_SESSION_HALT');
  }
  if ((input.exposurePct ?? 0) > 0.35) {
    issues.push('Exposure too concentrated');
    setLevel('L2_SESSION_HALT');
  }
  if ((input.rejectRate ?? 0) > 0.1 || (input.maxDrawdownPct ?? 0) > 10) {
    issues.push('Critical instability');
    setLevel('L4_HARD_KILL');
  }

  const action =
    level === 'OK' ? 'CONTINUE'
    : level === 'L1_SOFT_PAUSE' ? 'PAUSE_NEW_ENTRIES'
    : level === 'L2_SESSION_HALT' ? 'HALT_SESSION'
    : level === 'L3_SYSTEM_HOLD' ? 'HOLD_AUTOMATION'
    : 'HARD_KILL';

  return { level, action, issues };
}

): Promise<Record<string, LivePrice>> {
  const prices: Record<string, LivePrice> = {};
  const toFetch: string[] = [];

  for (const ticker of tickers) {
    const t = ticker.toUpperCase().trim();
    const cached = priceCache.get(t);
    if (!options.forceRefresh && cached && Date.now() - cached.fetchedAt < CACHE_TTL_MS) {
      prices[t] = cached.data;
    } else {
      toFetch.push(t);
    }
  }

  if (toFetch.length === 0) {
    console.log(`[PRICE] All ${tickers.length} prices from cache`);
    return prices;
  }

  console.log(`[PRICE] Cache hits: ${Object.keys(prices).length}, fetching: ${toFetch.length} tickers...`);

  const BATCH_SIZE = 6;
  const BATCH_DELAY_MS = 800;
  const chunks = chunkArray(toFetch, BATCH_SIZE);

  for (let i = 0; i < chunks.length; i++) {
    const chunk = chunks[i];
    const results = await Promise.allSettled(chunk.map(t => getRealPrice(t, options)));

    chunk.forEach((ticker, j) => {
      const result = results[j];
      if (result.status === 'fulfilled' && result.value) {
        prices[ticker] = result.value;
      }
    });

    if (i < chunks.length - 1) {
      await delay(BATCH_DELAY_MS);
    }
  }

  console.log(`[PRICE] Batch complete: ${Object.keys(prices).length}/${tickers.length} prices obtained`);
  return prices;
}

export async function fetchHistoricalData(
  ticker: string,
  range?: string,
  options: FetchOptions = {}
): Promise<HistoricalDataPoint[] | null> {
  const t = ticker.toUpperCase().trim();
  const r = range || '6mo';
  const interval = r === '1d' ? '5m' : '1d';

  const cacheKey = `${t}_${r}`;
  const cached = chartCache.get(cacheKey);
  if (!options.forceRefresh && cached && Date.now() - cached.fetchedAt < CHART_CACHE_TTL_MS) {
    return cached.data;
  }

  const endpoints = ['https://query1.finance.yahoo.com', 'https://query2.finance.yahoo.com'];

  for (const base of endpoints) {
    try {
      const url = `${base}/v8/finance/chart/${t}?range=${r}&interval=${interval}&includePrePost=false`;
      const res = await fetch(url, {
        signal: AbortSignal.timeout(10000),
        headers: BROWSER_HEADERS,
      });

      if (!res.ok) continue;

      const data = await res.json();
      const result = data?.chart?.result?.[0];
      if (!result) continue;

      const timestamps: number[] = result.timestamp || [];
      const quote = result.indicators?.quote?.[0];
      if (!quote || !timestamps.length) continue;

      const points: HistoricalDataPoint[] = [];
      for (let i = 0; i < timestamps.length; i++) {
        const open = quote.open?.[i];
        const high = quote.high?.[i];
        const low = quote.low?.[i];
        const close = quote.close?.[i];
        const vol = quote.volume?.[i];

        if (close && close > 0 && open && high && low) {
          const d = new Date(timestamps[i] * 1000);
          if (d.getDay() === 0 || d.getDay() === 6) continue;
          points.push({
            date: d.toISOString().split('T')[0],
            open: +open.toFixed(2),
            high: +high.toFixed(2),
            low: +low.toFixed(2),
            close: +close.toFixed(2),
            volume: vol || 0,
          });
        }
      }

      if (points.length >= 15) {
        console.log(`[CHART] ${t}: fetched ${points.length} days of real data from ${base}`);
        chartCache.set(cacheKey, { data: points, fetchedAt: Date.now() });
        return points;
      }
    } catch (err: any) {
      const msg = err?.name === 'TimeoutError' ? 'timeout' : err?.message || 'unknown';
      console.log(`[CHART] ${t}: ${base} failed: ${msg}`);
      continue;
    }
  }

  return null;
}

export async function getBatchQuotesFast(
  tickers: string[],

async function loadPreviousScores(): Promise<Record<string, { score: number; predictedAt: Date }>> {
  try {
    const cutoff = new Date(Date.now() - 18 * 60 * 60 * 1000);

    const preds = await prisma.prediction.findMany({
      where: {
        horizonDays: 1,
        modelVersion: 'predict-v3-regime-spillover',
        predictedAt: { lte: cutoff },
      },
      orderBy: { predictedAt: 'desc' },
      select: { symbol: true, rawScore: true, predictedAt: true },
      take: 1000,
    });

    const bySymbol = new Map<string, { score: number; predictedAt: Date }>();
    for (const p of preds) {
      if (p.rawScore != null && !bySymbol.has(p.symbol)) {
        bySymbol.set(p.symbol, { score: p.rawScore, predictedAt: p.predictedAt });
      }
    }

    console.log(`[SCAN] Loaded previous-day scores for ${bySymbol.size} symbols from DB`);
    return Object.fromEntries(bySymbol);
  } catch (err) {
    console.warn('[SCAN] Failed to load previous scores from DB:', err);
    return {};
  }
}

import prisma from '@/lib/prisma';

export async function logExecutionEvent(input: {
  predictionId?: string;
  symbol: string;
  side: 'BUY' | 'SELL';
  orderType: string;
  quantity: number;
  intendedPrice?: number;
  submittedPrice?: number;
  filledPrice?: number;
  spreadAtDecision?: number;
  spreadAtFill?: number;
  submittedAt: Date;
  ackAt?: Date;
  filledAt?: Date;
  status: string;
  rejectReason?: string;
  stopAttached?: boolean;
  venue?: string;
}) {
  const latencyMs = input.ackAt
    ? input.ackAt.getTime() - input.submittedAt.getTime()
    : null;

  let slippageBps: number | null = null;
  if (input.intendedPrice && input.filledPrice) {
    const raw = ((input.filledPrice - input.intendedPrice) / input.intendedPrice) * 10000;
    slippageBps = input.side === 'BUY' ? raw : -raw;
  }

  return prisma.executionEvent.create({
    data: {
      ...input,
      latencyMs,
      slippageBps,
      partialFillPct: input.filledPrice ? 100 : 0,
      stopAttached: input.stopAttached ?? false,
    },
  });
}

import prisma from '@/lib/prisma';

function mean(values: number[]) {
  return values.length ? values.reduce((a, b) => a + b, 0) / values.length : null;
}

function percentile(values: number[], p: number) {
  if (!values.length) return null;
  const sorted = [...values].sort((a, b) => a - b);
  const idx = Math.min(sorted.length - 1, Math.floor((p / 100) * sorted.length));
  return sorted[idx];
}

export async function getExecutionQuality(params?: { symbol?: string; days?: number }) {
  const since = new Date();
  since.setDate(since.getDate() - (params?.days ?? 7));

  const rows = await prisma.executionEvent.findMany({
    where: {
      ...(params?.symbol ? { symbol: params.symbol } : {}),
      createdAt: { gte: since },
    },
    orderBy: { createdAt: 'desc' },
  });

  const total = rows.length;
  const filled = rows.filter(r => r.status === 'FILLED');
  const rejected = rows.filter(r => r.status === 'REJECTED');
  const latencies = rows.flatMap(r => (r.latencyMs != null ? [r.latencyMs] : []));
  const slippages = rows.flatMap(r => (r.slippageBps != null ? [r.slippageBps] : []));

  return {
    sampleSize: total,
    fillRate: total ? filled.length / total : null,
    rejectRate: total ? rejected.length / total : null,
    avgLatencyMs: mean(latencies),
    p95LatencyMs: percentile(latencies, 95),
    avgSlippageBps: mean(slippages),
    p95SlippageBps: percentile(slippages, 95),
    stopAttachedRate: total ? rows.filter(r => r.stopAttached).length / total : null,
    topRejectReasons: Object.entries(
      rejected.reduce<Record<string, number>>((acc, row) => {
        const k = row.rejectReason || 'UNKNOWN';
        acc[k] = (acc[k] || 0) + 1;
        return acc;
